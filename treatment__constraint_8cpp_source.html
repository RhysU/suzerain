86void
87treatment::invert_mass_plus_scaled_operator(
88 const complex_t& phi,
89 multi_array::ref<complex_t,4>& state,
90 const lowstorage::method_interface<complex_t>& method,
91 const real_t delta_t,
92 const std::size_t substep_index,
93 multi_array::ref<complex_t,4> *ic0) const
94{
95 // State enters method as coefficients in X and Z directions
96 // State enters method as collocation point values in Y direction
97
98 SUZERAIN_TIMER_SCOPED("constraint::treatment");
99
100 // Shorthand
101 using ndx::e;
102 using ndx::mx;
103 using ndx::my;
104 using ndx::mz;
105 using ndx::rho;
106 const int rho_1 = ndx::rho_(1);
107
108 // Sidesteps assertions when local rank contains no wavespace information
109 const int N = (int) state.shape()[1];
110 if (SUZERAIN_UNLIKELY(0 == N)) return;
111
112 // Incoming state has wall-normal pencils of interleaved state scalars?
113 // Any amount of incoming state is valid so long as there's enough there
114 SUZERAIN_ENSURE(state.strides()[1] == 1 );
115 SUZERAIN_ENSURE(state.strides()[0] == (unsigned) N );
116 SUZERAIN_ENSURE(state.shape() [0] > (unsigned) e );
117 SUZERAIN_ENSURE(state.shape() [0] > (unsigned) mx );
118 SUZERAIN_ENSURE(state.shape() [0] > (unsigned) mx );
119 SUZERAIN_ENSURE(state.shape() [0] > (unsigned) mx );
120 SUZERAIN_ENSURE(state.shape() [0] > (unsigned) rho);
121
122 // See treatment_channel writeup (redux) for information on steps below.
123
124 // Have a tantrum if caller expects us to compute any constraints
125 SUZERAIN_ENSURE(ic0 == NULL);
126
127 // On zero-zero rank, re-use ic0 to wrap cdata for BaseClass invocation
128 if (rank_has_zero_zero_modes) {
129
130 // Prepare RHS data for density, momentum, and total energy constraints
131 //
132 // Notice scaling by Mach^2 to cause mx-, my-, and mz-related forcing
133 // work to have nondimensional energy "units" because we will
134 // directly add the result to the total energy equation.
135 //
136 // Notice ndx::rho constraint ignores other equations BY DESIGN.
137 // This choice is reflected in the later update of out.Crho.
138 cdata.setZero(state.shape()[0]*N, cdata.cols());
139 cdata.col(e ).segment(N*e , N).real() = physical[e ]->shape;
140 cdata.col(mx ).segment(N*e , N).real() = physical[mx ]->shape
141 * inp.u() * (Ma * Ma);
142 cdata.col(mx ).segment(N*mx , N).real() = physical[mx ]->shape;
143 cdata.col(my ).segment(N*e , N).real() = physical[my ]->shape
144 * inp.v() * (Ma * Ma);
145 cdata.col(my ).segment(N*my , N).real() = physical[my ]->shape;
146 cdata.col(mz ).segment(N*e , N).real() = physical[mz ]->shape
147 * inp.w() * (Ma * Ma);
148 cdata.col(mz ).segment(N*mz , N).real() = physical[mz ]->shape;
149 cdata.col(rho).segment(N*rho, N).real() = physical[rho]->shape;
150
151 // Prepare RHS for one decoupled numerical constraint per equation
152 cdata.col(rho_1+e ).segment(N*e , N).real() = numerical[e ]->shape;
153 cdata.col(rho_1+mx ).segment(N*mx , N).real() = numerical[mx ]->shape;
154 cdata.col(rho_1+my ).segment(N*my , N).real() = numerical[my ]->shape;
155 cdata.col(rho_1+mz ).segment(N*mz , N).real() = numerical[mz ]->shape;
156 cdata.col(rho_1+rho).segment(N*rho, N).real() = numerical[rho]->shape;
157
158 // Wrap data into appropriately digestible format
159 using std::size_t;
160 const array<size_t,4> sizes = {{
161 state.shape()[0], (size_t) N, (size_t) cdata.cols(), 1
162 }};
163 ic0 = new multi_array::ref<complex_t,4>(
164 cdata.data(), sizes, storage::interleaved<4>());
165 }
166
167 // Delegate to wrapped operator for the solution procedure
168 L->invert_mass_plus_scaled_operator(
169 phi, state, method, delta_t, substep_index, ic0);
170
171 // Clean up any integral constraint data wrapper we may have employed
172 delete ic0;
173
174 // Only the rank with zero-zero modes proceeds!
175 if (!rank_has_zero_zero_modes) return;
176
177 // Get an Eigen-friendly map of the zero-zero mode coefficients
178 Map<VectorXc> mean(state.origin(), state.shape()[0]*N);
179
180 // Solve the requested, possibly simultaneous constraint problem. A fancy
181 // decomposition is used for a simple 10x10 solve or to permit one or more
182 // inactive constraints via least squares. Least squares also adds
183 // robustness whenever the constraints are somehow incompatible.
184 //
185 // 1) Assemble the matrix problem for simultaneous integral constraints.
186 // Nonzero only when both the i-th and j-th constraints are enabled.
187 MatrixAr cmat(MatrixAr::Zero());
188 for (int j = 0; j < cdata.cols(); ++j) {
189 if (j<rho_1 ? physical[j]->enabled() : numerical[j-rho_1]->enabled()) {
190 for (int i = 0; i < physical.static_size; ++i) {
191 if (physical[i]->enabled()) {
192 cmat(i,j) = physical[i]->coeff.dot(
193 cdata.col(j).segment(N*i,N).real());
194 }
195 }
196 for (int i = 0; i < numerical.static_size; ++i) {
197 if (numerical[i]->enabled()) {
198 cmat(rho_1+i,j) = numerical[i]->coeff.dot(
199 cdata.col(j).segment(N*i,N).real());
200 }
201 }
202 }
203 }
204 // 2) Prepare initial values relative to desired targets when active.
205 // Otherwise, zero the associated row and column in matrix.
206 VectorAr crhs(VectorAr::Zero());
207 for (int i = 0; i < physical.static_size; ++i) {
208 if (physical[i]->enabled()) {
209 crhs[i] = physical[i]->target()
210 - physical[i]->coeff.dot(
211 mean.segment(N*i,N).real());
212 }
213 }
214 for (int i = 0; i < numerical.static_size; ++i) {
215 if (numerical[i]->enabled()) {
216 crhs[rho_1+i] = numerical[i]->target()
217 - numerical[i]->coeff.dot(
218 mean.segment(N*i,N).real());
219 }
220 }
221 // 3) Prepare matrix decomposition and solve using least squares
222 VectorAr cphi = jacobiSvd.compute(cmat).solve(crhs);
223 // 4) Add correctly scaled results to the mean state to satisfy constraints
224 mean += (cdata * cphi.asDiagonal()).rowwise().sum();
225
226 // The implicitly applied integral constraints, as point values, must be
227 // averaged across each substep to permit accounting for their impact on
228 // the Reynolds averaged equations.
229 //
230 // Per discussion with M. K. Lee, the implicit operator used to compute cphi
231 // already accounts for variable substep effects and so only a simple
232 // in-place running average is necessary:
233 //
234 // mean += (nsubsteps*(sample / delta_t) - mean) / (substep_index + 1)
235 //
236 // The delta_t accounts for step sizes already implicitly included in cphi.
237 // The correctness of this approach can be confirmed by running a 1D channel
238 // known a prior to be stationary followed by checking the total stress.
239 //
240 // Notice mx-related forcing is NOT scaled by Mach^2 when tracked
241 // because our post-processing routines will account for Mach^2 factor.
242 //
243 // Notice physical[ndx::rho] constraint lumped into Crho BY DESIGN.
244 cphi *= method.substeps() / delta_t;
245 const real_t iota = static_cast<real_t>(1) / (substep_index + 1);
246 out.fx() += iota*(
247 cphi[mx]*physical[mx]->shape
248 - out.fx()
249 );
250 out.fy() += iota*(
251 cphi[my]*physical[my]->shape
252 - out.fy()
253 );
254 out.fz() += iota*(
255 cphi[mz]*physical[mz]->shape
256 - out.fz()
257 );
258 out.f_dot_u() += iota*(
259 cphi[mx]*physical[mx]->shape*inp.u()
260 + cphi[my]*physical[my]->shape*inp.v()
261 + cphi[mz]*physical[mz]->shape*inp.w()
262 - out.f_dot_u()
263 );
264 out.qb() += iota*(
265 cphi[e]*physical[e]->shape
266 - out.qb()
267 );
268 out.CrhoE() += iota*(
269 cphi[rho_1+e ]*numerical[e ]->shape
270 - out.CrhoE()
271 );
272 out.C2rhoE() += iota*(
273 (cphi[rho_1+e ]*numerical[e ]->shape).square()
274 - out.C2rhoE()
275 );
276 out.Crhou() += iota*(
277 cphi[rho_1+mx ]*numerical[mx ]->shape
278 - out.Crhou()
279 );
280 out.C2rhou() += iota*(
281 (cphi[rho_1+mx ]*numerical[mx ]->shape).square()
282 - out.C2rhou()
283 );
284 out.Crhov() += iota*(
285 cphi[rho_1+my ]*numerical[my ]->shape
286 - out.Crhov()
287 );
288 out.C2rhov() += iota*(
289 (cphi[rho_1+my ]*numerical[my ]->shape).square()
290 - out.C2rhov()
291 );
292 out.Crhow() += iota*(
293 cphi[rho_1+mz ]*numerical[mz ]->shape
294 - out.Crhow()
295 );
296 out.C2rhow() += iota*(
297 (cphi[rho_1+mz ]*numerical[mz ]->shape).square()
298 - out.C2rhow()
299 );
300 out.Crho() += iota*(
301 cphi[rho] *physical [rho]->shape
302 + cphi[rho_1+rho]*numerical[rho]->shape
303 - out.Crho()
304 );
305 out.C2rho() += iota*( // Terms like aa + 2ab + bb
306 (cphi[rho] *physical [rho]->shape).square()
307 + 2*(cphi[rho] *physical [rho]->shape)
308 *(cphi[rho_1+rho]*numerical[rho]->shape)
309 + (cphi[rho_1+rho]*numerical[rho]->shape).square()
310 - out.C2rho()
311 );
312 out.Crhou_dot_u() += iota*(
313 cphi[rho_1+mx]*numerical[mx]->shape*inp.u()
314 + cphi[rho_1+my]*numerical[my]->shape*inp.v()
315 + cphi[rho_1+mz]*numerical[mz]->shape*inp.w()
316 - out.Crhou_dot_u()
317 );
318 out.C2rhou_dot_u() += iota*( // Terms like uu + 2uv + 2uw + vv + 2vw + ww
319 (cphi[rho_1+mx]*numerical[mx]->shape).square()
320 *inp.uu()
321 + 2*(cphi[rho_1+mx]*cphi[rho_1+my])
322 *numerical[mx]->shape
323 *numerical[my]->shape
324 *inp.uv()
325 + 2*(cphi[rho_1+mx]*cphi[rho_1+mz])
326 *numerical[mx]->shape
327 *numerical[mz]->shape
328 *inp.uw()
329 + (cphi[rho_1+my]*numerical[my]->shape).square()
330 *inp.vv()
331 + 2*(cphi[rho_1+my]*cphi[rho_1+mz])
332 *numerical[my]->shape
333 *numerical[mz]->shape
334 *inp.vw()
335 + (cphi[rho_1+mz]*numerical[mz]->shape).square()
336 *inp.ww()
337 - out.C2rhou_dot_u()
338 );
339
340 // State leaves method as coefficients in X, Y, and Z directions
341}
