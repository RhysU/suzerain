38static void parse_size_t(const std::string& s,
39 std::size_t* value,
40 const char* name)
41{
42 using std::floor;
43#pragma warning(push,disable:2259)
44 const real_t t = floor(exprparse<real_t>(s, name) + real_t(1)/2);
45#pragma warning(pop)
46 validation::ensure_nonnegative(t, name);
47 *value = t;
48}

84boost::program_options::options_description
85definition_restart::options_description()
86{
87 boost::program_options::options_description retval(
88 "Restart-writing parameters");
89
90 using boost::bind;
91 using boost::lexical_cast;
92 using boost::program_options::bool_switch;
93 using boost::program_options::value;
94 using std::string;
95 using validation::ensure_nonnegative;
96
97 retval.add_options()
98 ("metadata", value(&metadata)
99 ->default_value(metadata),
100 "Path to use when saving common metadata for output files. "
101 "Any trailing \"XXXXXX\" will be used to generate a unique name.")
102 ("uncommitted", value(&uncommitted)
103 ->default_value(uncommitted),
104 "Path to use when saving uncommitted output files. "
105 "Any trailing \"XXXXXX\" will be used to generate a unique name.")
106 ("restart_destination", value(&destination)
107 ->default_value(destination),
108 "Archiving destination to use when committing restart files. "
109 "One or more #'s must be present and will be replaced by a sequence number. "
110 "Any trailing \"XXXXXX\" will be used to generate a unique template.")
111 ("restart_retain", value<string>(NULL)
112 ->notifier(bind(&parse_size_t, _1, &retain, "restart_retain"))
113 ->default_value(lexical_cast<string>(retain)),
114 "Maximum number of committed restart files to retain")
115 ("restart_dt", value<string>(NULL)
116 ->notifier(bind(&parse_option<real_t>, _1, &dt,
117 &ensure_nonnegative<real_t>, "restart_dt"))
118 ->default_value(lexical_cast<string>(dt)),
119 "Maximum amount of simulation time between restart files")
120 ("restart_nt", value<string>(NULL)
121 ->notifier(bind(&parse_size_t, _1, &nt, "restart_nt"))
122 ->default_value(lexical_cast<string>(nt)),
123 "Maximum number of time steps between restart files")
124 ("restart_final", value<bool>(&final)
125 ->default_value(final),
126 "Should a final restart be written after advance successfully completes?")
127 ("restart_physical", bool_switch(&physical),
128 "Specify flag to save restart fields as primitive variables "
129 "stored at collocation points in physical space")
130 ;
131
132 return retval;
133}
