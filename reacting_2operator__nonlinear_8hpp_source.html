83void vfilt_form_operator_difference(const operator_base& o,
84 const contiguous_state<4, complex_t>& swave,
85 const size_t state_count,
86 const suzerain::bsplineop_luz& massluz,
87 contiguous_state<4, complex_t>& fsrcw)
88{
89 //**************************************************************
90 // Begin computing viscous filter. There are five steps. The
91 // first four must done here, while the state (swave) is still
92 // entirely coefficients. The last step (scaling by reference
93 // quantities) must be done after the reference quantities are
94 // computed below (of course).
95 //
96 // Each step is performed independently for each state variable
97 for (size_t var = ndx::e; var<state_count; ++var) {
98
99 // 1.) Accumulate D_1 swave in to fsrcw
100 o.bop_accumulate(1, 1, swave, var,
101 0, fsrcw, var );
102
103 // 2.) fsrcw <- M \ fsrcw
104 o.bop_solve(massluz, fsrcw, var);
105
106 // 3.) fsrcw <- D_1 fsrcw + 0 * fsrcw
107 o.bop_apply(1, 1, fsrcw, var);
108
109 // 4.) fsrcw <- D_2 swave - fsrcw
110 o.bop_accumulate(2, 1, swave, var,
111 -1, fsrcw, var );
112
113 }
114 // Now, fsrcw contains D_2 * swave - D_1 * M \ (D_1 * swave).
115 //
116 // To complete the filter source, we need to scale it by the
117 // appropriate reference.
118 //**************************************************************
119}

121void vfilt_apply_ref_viscous_operator(const operator_common_block &common,
122 const real_t& alpha,
123 const size_t& Ns,
124 contiguous_state<4, complex_t>& fsrcw)
125{
126 // After first traversal, have gathered reference profiles. So, we can
127 // complete the viscous filter source calculation by multiplying by the
128 // appropriate reference quantities.
129
130 // Density to be used repeatedly below before being cleared before return
131 Map<MatrixXXc> Frho(fsrcw[ndx::rho].origin(),
132 fsrcw.shape()[1],
133 fsrcw.shape()[2]*fsrcw.shape()[3]);
134
135 // Reusable accumulation buffer to avoid repeated allocation/deallocation
136 MatrixXXc tmp;
137
138 // Shorthand for common task of const referencing common.ref_something()
139 typedef ArrayXXr::ConstRowXpr const_ref_type;
140
141 // Energy
142 {
143 Map<MatrixXXc> F(fsrcw[ndx::e].origin(), Frho.rows(), Frho.cols());
144
145 // diagonal
146 const_ref_type& D(common.ref_korCv());
147 tmp.noalias() = D.matrix().asDiagonal()*F;
148
149 // x-momentum: heat flux - visc work
150 const_ref_type& nu(common.ref_nu());
151 const_ref_type& ux(common.ref_ux());
152 tmp -= ((D - nu)*ux).matrix().asDiagonal()
153 * Map<MatrixXXc>(fsrcw[ndx::mx].origin(), F.rows(), F.cols());
154
155 // y-momentum: heat flux - visc work
156 const_ref_type& uy(common.ref_uy());
157 tmp -= ((D - (alpha + 4/3.)*nu)*uy).matrix().asDiagonal()
158 * Map<MatrixXXc>(fsrcw[ndx::my].origin(), F.rows(), F.cols());
159
160 // z-momentum: heat flux - visc work
161 const_ref_type& uz(common.ref_uz());
162 tmp -= ((D - nu)*uz).matrix().asDiagonal()
163 * Map<MatrixXXc>(fsrcw[ndx::mz].origin(), F.rows(), F.cols());
164
165 // density accumulates into Drho
166 const_ref_type& e0(common.ref_es(0));
167 const ArrayXXr ke = 0.5*(ux.square() + uy.square() + uz.square());
168 ArrayXr Drho = D*(e0-ke) // heat flux
169 + nu*(2*ke + (alpha + 1/3.)*uy.square()); // visc work
170
171 // enthalpy diffusion modifies previous Drho result
172 for (std::size_t s=1; s<Ns; ++s) {
173 const_ref_type& h0(common.ref_hs(0));
174 const_ref_type& hs(common.ref_hs(s));
175 const_ref_type& cs(common.ref_cs(s));
176 const_ref_type& Ds(common.ref_Ds());
177
178 Drho -= cs*Ds*(hs - h0);
179 }
180
181 // employ Drho
182 tmp -= Drho.matrix().asDiagonal()*Frho;
183
184 // species: heat flux + enthalpy diffusion
185 for (std::size_t s=0; s<Ns-1; ++s) {
186 const_ref_type& es(common.ref_es(s+1));
187 const_ref_type& h0(common.ref_hs(0 ));
188 const_ref_type& hs(common.ref_hs(s+1));
189 const_ref_type& Ds(common.ref_Ds());
190 tmp += (D*(e0-es) + Ds*(h0-hs)).matrix().asDiagonal()
191 * Map<MatrixXXc>(fsrcw[ndx::species+s].origin(),
192 F.rows(), F.cols());
193 }
194
195 // overwrite with result
196 F = tmp;
197 }
198
199 // x-momentum
200 {
201 Map<MatrixXXc> F(fsrcw[ndx::mx].origin(), Frho.rows(), Frho.cols());
202
203 // diagonal
204 const_ref_type& D(common.ref_nu());
205 tmp.noalias() = D.matrix().asDiagonal()*F;
206
207 // density
208 const_ref_type& ux(common.ref_ux());
209 tmp -= (D*ux).matrix().asDiagonal()*Frho;
210
211 // overwrite with result
212 F = tmp;
213 }
214
215 // y-momentum
216 {
217 Map<MatrixXXc> F(fsrcw[ndx::my].origin(), Frho.rows(), Frho.cols());
218
219 // diagonal
220 const_ref_type& D(common.ref_nu());
221 tmp.noalias() = ((alpha + 4/3.)*D).matrix().asDiagonal()*F;
222
223 // density
224 const_ref_type& uy(common.ref_uy());
225 tmp -= ((alpha + 4/3.)*D*uy).matrix().asDiagonal()*Frho;
226
227 // overwrite with result
228 F = tmp;
229 }
230
231 // z-momentum
232 {
233 Map<MatrixXXc> F(fsrcw[ndx::mz].origin(), Frho.rows(), Frho.cols());
234
235 // diagonal
236 const_ref_type& D(common.ref_nu());
237 tmp.noalias() = D.matrix().asDiagonal()*F;
238
239 // density
240 const_ref_type& uz(common.ref_uz());
241 tmp -= (D*uz).matrix().asDiagonal()*Frho;
242
243 // overwrite with result
244 F = tmp;
245 }
246
247 // mass goes after species because all other equations require it
248
249 // species
250 for (unsigned int s=1; s<Ns; ++s) {
251 Map<MatrixXXc> F(fsrcw[ndx::rho+s].origin(),
252 Frho.rows(), Frho.cols());
253
254 // diagonal
255 const_ref_type& D(common.ref_Ds());
256 tmp.noalias() = D.matrix().asDiagonal()*F;
257
258 // density
259 const_ref_type& cs(common.ref_cs(s));
260 tmp -= (D*cs).matrix().asDiagonal()*Frho;
261
262 // overwrite with result
263 F = tmp;
264 }
265
266 // mass
267 {
268 Frho.setZero();
269 }
270}

277std::vector<real_t> apply_navier_stokes_spatial_operator(
278 const operator_base &o,
279 operator_common_block &common,
280 const definition_filter &fsdef,
281 const specification_largo &sgdef,
282 const std::shared_ptr<const ManufacturedSolution>& msoln,
283 const ConstitutiveModels& cmods,
284 const suzerain::bsplineop_luz& massluz,
285 const real_t time,
286 contiguous_state<4,complex_t> &swave,
287 const lowstorage::method_interface<complex_t> &method)
288{
289 SUZERAIN_TIMER_SCOPED("apply_navier_stokes_spatial_operator");
290
291 const size_t Ns = cmods.Ns();
292 assert(Ns>0);
293 assert(Ns == common.Ns);
294
295 // Shorthand
296 typedef contiguous_state<4,complex_t> state_type;
297 using std::abs;
298 using std::equal;
299 using std::max;
300 using std::min;
301 using std::size_t;
302 using std::sqrt;
303
304 // State enters method as coefficients in X, Y, and Z directions
305
306 // FIXME Ticket #2477 retrieve linearization-dependent CFL information
307 // Afterwards, change the stable time step computation accordingly
308 // This will require using common.linearization in some fashion
309 const real_t evmaxmag_real = method.evmaxmag_real();
310 const real_t evmaxmag_imag = method.evmaxmag_imag();
311
312 // Get total number of conserved state fields
313 const size_t state_count = Ns+4; // = (Ns-1) species + 5 (rho, ru, rv, rw, rE)
314
315 // Get total number of fields in aux storage
316 const size_t aux_count = (aux::species + // everything *except* species derivatives
317 dir::count*(Ns-1) ); // spatial derivatives of species
318
319 // Obtain the auxiliary storage (likely from a pool to avoid fragmenting).
320 // We assume no garbage values in the memory will impact us (for speed).
321 std::unique_ptr<state_type> _auxw_ptr(
322 support::allocate_padded_state<state_type>(
323 aux_count, o.dgrid)); // RAII
324 state_type &auxw = *_auxw_ptr; // Brevity
325
326 // Auxiliary storage for filter source:
327 // Obtain the auxiliary storage (likely from a pool to avoid fragmenting).
328 // We assume no garbage values in the memory will impact us (for speed).
329 // Number of fields for filter source is equal to number of variables
330 std::unique_ptr<state_type> _fsrcw_ptr(
331 support::allocate_padded_state<state_type>(
332 state_count, o.dgrid)); // RAII
333 state_type &fsrcw = *_fsrcw_ptr; // Brevity
334
335 // Sanity check incoming swave's and auxw's shape and contiguity
336 SUZERAIN_ENSURE(swave.shape()[0] == state_count);
337 SUZERAIN_ENSURE(swave.shape()[1] == (unsigned) o.dgrid.local_wave_extent.y());
338 SUZERAIN_ENSURE(swave.shape()[2] == (unsigned) o.dgrid.local_wave_extent.x());
339 SUZERAIN_ENSURE(swave.shape()[3] == (unsigned) o.dgrid.local_wave_extent.z());
340 SUZERAIN_ENSURE((unsigned) swave.strides()[1] == 1u);
341 SUZERAIN_ENSURE((unsigned) swave.strides()[2] == swave.shape()[1]);
342 SUZERAIN_ENSURE((unsigned) swave.strides()[3] == swave.shape()[1]*swave.shape()[2]);
343 SUZERAIN_ENSURE(equal(swave.shape() + 1, swave.shape() + 4, auxw.shape() + 1));
344 SUZERAIN_ENSURE(equal(swave.strides() + 1, swave.strides() + 4, auxw.strides() + 1));
345 SUZERAIN_ENSURE(equal(swave.shape() + 1, swave.shape() + 4, fsrcw.shape() + 1));
346 SUZERAIN_ENSURE(equal(swave.strides() + 1, swave.strides() + 4, fsrcw.strides() + 1));
347
348 // Prepare common-block-like storage used to pass details from N to L.
349 // Zeroing is done carefully as accumulated means and reference quantities
350 // must survive across nonzero substeps while instant profiles must not.
351 if (ZerothSubstep) common.set_zero(/* Ny */ swave.shape()[1]);
352 common.u().setZero();
353
354 // Maintain stable time step values to return to the caller:
355 //
356 // convtotal_* is convective stability using total velocity.
357 // It may or may not have acoustics included as necessary.
358 // convfluct_* is convective stability using only velocity fluctuations.
359 // It permits measuring any advantage in treating mean
360 // convection implicitly.
361 // diffusive_* is diffusive stability linearized however is appropriate.
362 // *_xyz_* accounts for restrictions in all three directions.
363 // It must be more retrictive than any single direction.
364 // *_x_* accounts for restrictions in streamwise direction only.
365 // *_y_* accounts for restrictions in wall-normal direction only
366 // *_z_* accounts for restrictions in spanwise direction only
367 //
368 // Results reported by the code are reported in this order.
369 array<real_t, 12> delta_t_candidates;
370 std::fill(delta_t_candidates.begin(), delta_t_candidates.end(),
371 std::numeric_limits<real_t>::max());
372 real_t &convtotal_xyz_delta_t = delta_t_candidates[ 0];
373 real_t &convfluct_xyz_delta_t = delta_t_candidates[ 1];
374 real_t &diffusive_xyz_delta_t = delta_t_candidates[ 2];
375 real_t &convtotal_x_delta_t = delta_t_candidates[ 3];
376 real_t &convfluct_x_delta_t = delta_t_candidates[ 4];
377 real_t &diffusive_x_delta_t = delta_t_candidates[ 5];
378 real_t &convtotal_y_delta_t = delta_t_candidates[ 6];
379 real_t &convfluct_y_delta_t = delta_t_candidates[ 7];
380 real_t &diffusive_y_delta_t = delta_t_candidates[ 8];
381 real_t &convtotal_z_delta_t = delta_t_candidates[ 9];
382 real_t &convfluct_z_delta_t = delta_t_candidates[10];
383 real_t &diffusive_z_delta_t = delta_t_candidates[11];
384
385 if (Filter == filter::viscous) {
386 vfilt_form_operator_difference(o, swave, state_count, massluz, fsrcw);
387 }
388
389 //********************************************************************
390 //
391 // Compute mean and dmean for slow growth
392 const std::size_t Ny = fsrcw.shape()[1];
393 MatrixXXr mean_values(Ny, 2*state_count);
394 if (sgdef.formulation.enabled()) {
395 if (o.dgrid.has_zero_zero_modes()) {
396 for (size_t var = 0; var<state_count; ++var) {
397 o.cop.accumulate(0, 1
398 , 1.0, (real_t *) &swave[var][0][0][0], 2, Ny
399 , 0.0, &mean_values(0,var), 1, Ny);
400 o.cop.accumulate(1, 1
401 , 1.0, (real_t *) &swave[var][0][0][0], 2, Ny
402 , 0.0, &mean_values(0,state_count+var), 1, Ny);
403 }
404 }
405 SUZERAIN_MPICHKR(MPI_Bcast(
406 mean_values.data(), mean_values.size(),
407 mpi::datatype<real_t>(),
408 o.dgrid.rank_zero_zero_modes, MPI_COMM_WORLD));
409 }
410 //
411 //********************************************************************
412
413 //********************************************************************
414 //
415 // From coefficients, compute values at collocation points for
416 // all state and derivatives of state *except* derivatives of
417 // total energy.
418 //
419 // This is the first wave to physical pass. There will be another
420 // to get the temperature gradient.
421 //
422
423 // NOTE: The indexing here *assumes* that the total energy is
424 // stored first.
425
426 // Energy (no derivatives)
427 o.zero_dealiasing_modes(swave, ndx::e);
428 o.bop_apply (0, 1, swave, ndx::e);
429
430
431 if (Filter == filter::cook) {
432 // FIXME: to start testing, use directly a user defined value;
433 // to do is to figure out proper coefficients for each variable;
434 // see if it works to set the coefficient for rho = 0;
435
436 const complex_t f_phi = fsdef.filter_phi;
437
438 // Filter source: compute for energy
439 fsdef.source_accumulate(o.grid, o.dgrid, f_phi, swave, ndx::e,
440 0., fsrcw, ndx::e);
441 }
442
443 // Everything else (all spatial derivatives)
444 for (size_t var = ndx::mx; var<state_count; ++var) {
445
446 // Indexing note: aux::mx is beginning of derivatives of
447 // conserved state in aux ordering. Things appearing ahead of
448 // aux::mx are not used until later.
449
450 // Compute Y derivatives of variable var at collocation points
451 // Zero wavenumbers present only for dealiasing along the way
452 o.zero_dealiasing_modes(swave, var);
453 o.bop_accumulate(1, 1, swave, var,
454 0, auxw , aux::e + dir::count*var + dir::y);
455 o.bop_apply (0, 1, swave, var);
456
457 // Compute X- and Z- derivatives of variable var at collocation points
458 // Zeros wavenumbers present only for dealiasing in the target storage
459 o.diffwave_accumulate(1, 0, 1, swave, var,
460 0, auxw , aux::e + dir::count*var + dir::x );
461 o.diffwave_accumulate(0, 1, 1, swave, var,
462 0, auxw , aux::e + dir::count*var + dir::z );
463
464 if (Filter == filter::cook) {
465 const complex_t f_phi = fsdef.filter_phi;
466 // Filter source: compute for variable var
467 fsdef.source_accumulate(o.grid, o.dgrid, f_phi, swave, var,
468 0., fsrcw, var);
469 }
470 }
471
472 physical_view<> auxp (o.dgrid, auxw );
473 physical_view<> sphys(o.dgrid, swave);
474 for (size_t i = 0; i < state_count; ++i) {
475 o.dgrid.transform_wave_to_physical(&sphys.coeffRef(i,0));
476 }
477 for (size_t i = aux::mx; i < aux_count; ++i) {
478 o.dgrid.transform_wave_to_physical(&auxp.coeffRef(i,0));
479 }
480
481 //
482 // Done with first wave to physical pass.
483 //
484 //********************************************************************
485
486 // Compute derived constants before inner loops
487 const real_t lambda1_x = o.lambda1_x;
488 const real_t lambda1_z = o.lambda1_z;
489 const real_t lambda2_x = o.lambda2_x;
490 const real_t lambda2_z = o.lambda2_z;
491
492 // Type of Boost.Accumulator to use for summation processes.
493 // Kahan summation preferred when available as incremental cost is small
494 // and we will add many small numbers to a large magnitude sum.
495 // During debugging, also make the number of samples available.
496 typedef boost::accumulators::accumulator_set<
497 real_t,
498 boost::accumulators::stats<
499 boost::accumulators::tag::sum_kahan
500#ifndef NDEBUG
501 , boost::accumulators::tag::count
502#endif
503 >
504 > summing_accumulator_type;
505
506 // Before any traversals, instantiate Eigen variable length arrays to store species
507 // specific info
508 VectorXr species(Ns); // species densities
509 VectorXr Ds (Ns); // mass diffusivities
510 VectorXr hs (Ns); // species enthalpies
511 VectorXr om (Ns); // reaction source terms
512 VectorXr cs (Ns); // species mass fractions
513 VectorXr etots (Ns); // species internal energies
514
515 Matrix3Xr grad_species (3,Ns); // spatial derivatives of species densities
516 Matrix3Xr grad_cs (3,Ns); // spatial derivatives of species mass fractions
517 Matrix3Xr sdiff (3,Ns); // diffusive fluxes for species equations
518
519
520 //********************************************************************
521 //
522 // Physical space is traversed linearly using a single offset 'offset'.
523 // The three loop structure is present to provide the global absolute
524 // positions x(i), y(j), and z(k) where necessary.
525 //
526 // Four traversals occur:
527 // (1) Computing reference quantities and mean velocity OR mean quantities
528 // (depending on linearization and which substep is being performed).
529 //
530 // (2) Computing as much of the sources and fluxes as possible
531 // given *only* conserved state and conserved state gradients.
532 // In theory this is everything. In practice, we don't do the
533 // heat flux in this pass b/c the temperature gradient isn't
534 // available.
535 //
536 // (3) (After going to wave space with T and back down to physical
537 // space with grad(T)), Compute heat flux contribution.
538 //
539 // (4) Computing any manufactured solution forcing (when enabled).
540 //
541 // The traversal pattern is best embodied by the last pass. The
542 // first passes use slightly more compact structure as only y(j)
543 // must be known within them. Study (4) and convince yourself
544 // that the rest are equivalent but lack information on x(i)
545 // and z(k).
546 //
547 // TODO: Combine some of these traversals (e.g., 1 and 2)
548
549 // Set initial guess for temperature
550 real_t Tguess = -1;
551
552 // Traversal:
553 // (1) Computing reference quantities and mean velocity OR mean
554 // velocity (depending on linearization, filtering, and which
555 // substep is being performed).
556 //
557 // Always gather all reference quantities when using implicit solves.
558 // Profiling indicates the overhead is tiny and it keeps the code readable.
559 //
560 // Also need reference quantities if we are using the viscous filter option.
561 if ( ZerothSubstep
562// && (Linearize != linearize::none || Filter == filter::viscous)
563 ) {
564
565 // Gather reference profiles and mean velocity
566
567 SUZERAIN_TIMER_SCOPED("reference quantities");
568
569 // Zero y(j) not present on this rank to avoid accumulating garbage
570 const size_t leftNotOnRank = o.dgrid.local_physical_start.y();
571 if (leftNotOnRank) common.refs.leftCols(leftNotOnRank).setZero();
572
573 // Sum reference quantities as a function of y(j) into common.ref_*
574 // See writeups/derivation.tex or rholut_imexop.h for definitions
575 for (int offset = 0, j = o.dgrid.local_physical_start.y();
576 j < o.dgrid.local_physical_end.y();
577 ++j) {
578
579 // Prepare logical indices using a struct for scoping (e.g. ref::ux).
580 struct ref { enum { ux, uy, uz,
581 uxux, uxuy, uxuz, uyuy, uyuz, uzuz,
582 p_ru, p_rw, p_rE,
583 vp_ru, vp_rw, vp_rE,
584 Cmy_rho, Ce_rho, Ce_rv,
585 nu, korCv, Ds,
586 T, gamma, a, p, p2,
587 count // Sentry
588 }; };
589
590 // An array of summing_accumulator_type holds all running sums.
591 // This gives nicer construction and allows looping over results.
592 //summing_accumulator_type acc[ref::count];
593 std::vector<summing_accumulator_type> acc(common.Nref());
594
595 const int last_zxoffset = offset
596 + o.dgrid.local_physical_extent.z()
597 * o.dgrid.local_physical_extent.x();
598 for (; offset < last_zxoffset; ++offset) {
599
600 // First, unpack conserved state...
601
602 // ... mixture density
603 const real_t rho ( sphys(ndx::rho, offset));
604 const real_t irho = 1.0/rho;
605
606 // ... momentum
607 const Vector3r m ( sphys(ndx::mx, offset),
608 sphys(ndx::my, offset),
609 sphys(ndx::mz, offset));
610
611 // ... total energy
612 const real_t e (sphys(ndx::e, offset));
613
614 // ... species densities
615 // NOTE: In species vector, idx 0 is the dilluter (the species
616 // that is not explicitly part of the state vector)
617 species(0) = rho;
618 for (unsigned int s=1; s<Ns; ++s) {
619 species(s) = sphys(ndx::species + s - 1, offset);
620
621 // dilluter density = rho_0 = rho - sum_{s=1}^{Ns-1} rho_s
622 species(0) -= species(s);
623 }
624
625 // Second, compute required derived quantities...
626
627 // ... mass fractions
628 for (unsigned int s=0; s<Ns; ++s) {
629 cs(s) = irho * species(s);
