91std::vector<std::string>
92application_base::initialize(int argc, char **argv)
93{
94#ifdef SUZERAIN_HAVE_GRVY
95 grvy_timer_init(argv[0] ? argv[0] : "NULL"); // Initialize GRVY Timers
96#endif
97 MPI_Init(&argc, &argv); // Initialize MPI...
98 wtime_mpi_init = MPI_Wtime(); // Record MPI_Init time
99 atexit((void (*) ()) MPI_Finalize); // ...finalize at exit
100 logging::initialize(MPI_COMM_WORLD, // Initialize logging
101 this->log4cxx_config().c_str());
102#ifdef SUZERAIN_HAVE_MKL // Finalize MKL at exit
103 atexit(mkl_free_buffers);
104
105#endif
106 atexit(fftw_cleanup); // Finalize FFTW at exit
107 atexit(fftw_forget_wisdom); // Finalize FFTW at exit
108#ifdef SUZERAIN_HAVE_UNDERLING
109 underling_init(&argc, &argv, 0); // Initialize underling...
110 atexit(&underling_cleanup); // ...finalize at exit
111#endif
112
113 // Hook error handling into logging infrastructure
114 gsl_set_error_handler (&mpi_abort_on_error_handler_gsl );
115 suzerain_set_error_handler (&mpi_abort_on_error_handler_suzerain );
116 esio_set_error_handler (&mpi_abort_on_error_handler_esio );
117#ifdef SUZERAIN_HAVE_UNDERLING
118 underling_set_error_handler(&mpi_abort_on_error_handler_underling);
119#endif
120
121 // Only add groups of options when non-trivial at initialization
122 if (grid) {
123 options.add_definition(*grid );
124 }
125 if (fftwdef) {
126 options.add_definition(*fftwdef);
127 options.add_options()
128#ifdef SUZERAIN_HAVE_P3DFFT
129 ("p3dfft", boost::program_options::bool_switch(&use_p3dfft),
130 "Use P3DFFT for MPI-parallel FFTs")
131#endif
132#ifdef SUZERAIN_HAVE_UNDERLING
133 ("underling", boost::program_options::bool_switch(&use_underling),
134 "Use underling for MPI-parallel FFTs")
135#endif
136 ;
137 }
138
139 // Process incoming arguments
140 std::vector<std::string> positional = options.process(argc, argv);
141
142 // Record build and invocation for posterity and to aid in debugging
143 std::ostringstream os;
144 os << argv[0];
145 for (int i = 1; i < argc; ++i) {
146 os << ' ' << argv[i];
147 }
148 INFO0(who, "Invocation: " << os.str());
149 INFO0(who, "Build: " << suzerain::version("", revstr));
150
151 // Select pencil decomposition and FFT library to use
152 options.conflicting_options("p3dfft", "underling");
153 if (!use_underling && !use_p3dfft) {
154 DEBUG0("Defaulted to using P3DFFT for parallel FFTs");
155 use_p3dfft = true;
156 }
157
158 switch (options.verbose()) {
159 case 0: break;
160 case 1: DEBUG0_ENABLE(); break;
161 default: TRACE0_ENABLE(); break;
162 }
163 switch (options.verbose_all()) {
164 case 0: break;
165 case 1: DEBUG_ENABLE(); break;
166 default: TRACE_ENABLE(); break;
167 }
168
169 return positional;
170}

191void
192application_base::load_grid_and_operators(
193 const esio_handle esioh)
194{
195 // Possibly load the grid parameters from the restart file
196 // If no restart file requested, we'd best have a sane grid instance
197 if (esioh) {
198 if (!grid) {
199 grid = std::make_shared<definition_grid>();
200 }
201 grid->load(esioh);
202 } else {
203 SUZERAIN_ENSURE(grid);
204 }
205
206 // Create the discrete B-spline operators
207 create_bsplines(grid->N.y(), grid->k, 0.0, grid->L.y(), grid->htdelta, b, cop);
208 gop.reset(new bsplineop(*b, grid->k, SUZERAIN_BSPLINEOP_GALERKIN_L2));
209}

215 // Only save grid if available

225 } else {
226 save_bsplines(esioh, *b, *cop);
227 }

231void
232application_base::establish_decomposition(
233 const bool output_size,
234 const bool output_plan,
235 const bool output_load)
236{
237 SUZERAIN_ENSURE(grid);
238
239 const std::size_t nranks = suzerain::mpi::comm_size(MPI_COMM_WORLD);
240
241 // Display information on the global degrees of freedom
242 if (output_size) {
243 INFO0(who, "Number of MPI ranks: " << nranks);
244 INFO0(who, "Grid degrees of freedom (GDOF): " << grid->N.prod());
245 INFO0(who, "GDOF by direction (XYZ): " << grid->N.x() << " "
246 << grid->N.y() << " "
247 << grid->N.z());
248 INFO0(who, "Dealiased GDOF by direction (XYZ): " << grid->dN.x() << " "
249 << grid->dN.y() << " "
250 << grid->dN.z());
251 }
252
253 // Establish the parallel decomposition and output some timing
254 if (output_plan) {
255 INFO0(who, "Preparing MPI transpose and Fourier transform plans ("
256 << c_str(fftwdef->rigor_mpi) << ", "
257 << c_str(fftwdef->rigor_fft) << ")");
258 }
259 wisdom_broadcast(fftwdef->plan_wisdom);
260 double begin = MPI_Wtime();
261 fftw_set_timelimit(fftwdef->plan_timelimit);
262#if defined(SUZERAIN_HAVE_P3DFFT) && defined(SUZERAIN_HAVE_UNDERLING)
263 if (use_p3dfft) {
264 dgrid = std::make_shared<pencil_grid_p3dfft>(
265 grid->dN, grid->P, fftwdef->rigor_fft, fftwdef->rigor_mpi);
266 } else if (use_underling) {
267 dgrid = std::make_shared<pencil_grid_underling>(
268 grid->dN, grid->P, fftwdef->rigor_fft, fftwdef->rigor_mpi);
269 } else {
270#endif
271 dgrid = std::make_shared<pencil_grid_default>(
272 grid->dN, grid->P, fftwdef->rigor_fft, fftwdef->rigor_mpi);
273#if defined(SUZERAIN_HAVE_P3DFFT) && defined(SUZERAIN_HAVE_UNDERLING)
274 }
275#endif
276 wtime_fftw_planning = MPI_Wtime() - begin;
277 if (output_plan) {
278 INFO0(who, "MPI transpose and Fourier transform planning by "
279 << dgrid->implementation() << " took "
280 << wtime_fftw_planning << " seconds");
281 if (dgrid->processor_grid.prod() != 1) {
282 INFO0(who, "Rank grid used for decomposition: "
283 << dgrid->processor_grid[0] << " "
284 << dgrid->processor_grid[1]);
285 DEBUG0(who, "Zero-zero modes located on MPI_COMM_WORLD rank "
286 << dgrid->rank_zero_zero_modes);
287 }
288 SUZERAIN_ENSURE((grid->dN == dgrid->global_physical_extent).all());
289 }
290 begin = MPI_Wtime();
291 if (wisdom_gather(fftwdef->plan_wisdom) && output_plan) {
292 INFO0(who, "FFTW wisdom gathered and saved in additional "
293 << (MPI_Wtime() - begin) << " seconds");
294 }
295
296 // Log rank-by-rank decomposition results in debug mode
297 DEBUG("Local wave start (XYZ): "
298 << dgrid->local_wave_start.transpose());
299 DEBUG("Local wave end (XYZ): "
300 << dgrid->local_wave_end.transpose());
301 DEBUG("Local wave extent (XYZ): "
302 << dgrid->local_wave_extent.transpose());
303 DEBUG("Local physical start (XYZ): "
304 << dgrid->local_physical_start.transpose());
305 DEBUG("Local physical end (XYZ): "
306 << dgrid->local_physical_end.transpose());
307 DEBUG("Local physical extent (XYZ): "
308 << dgrid->local_physical_extent.transpose());
309
310 // Display normalized workloads metrics relative to zero-zero workload
311 // (only done when the information is non-trivial in the multi-rank case)
312 if (output_load && nranks > 1) {
313 real_t sendbuf[4];
314 if (dgrid->has_zero_zero_modes()) {
315 INFO("Wave space zero-zero rank workload (XYZ): "
316 << dgrid->local_wave_extent.transpose());
317 INFO("Physical space zero-zero rank workload (XYZ): "
318 << dgrid->local_physical_extent.transpose());
319 sendbuf[0] = (real_t) dgrid->local_wave_extent.prod();
320 sendbuf[1] = (real_t) dgrid->local_physical_extent.prod();
321 }
322 SUZERAIN_MPICHKR(MPI_Bcast(sendbuf, 2,
323 suzerain::mpi::datatype<real_t>(), 0,
324 MPI_COMM_WORLD));
325 sendbuf[0] = dgrid->local_wave_extent.prod() / sendbuf[0];
326 sendbuf[1] = dgrid->local_physical_extent.prod() / sendbuf[1];
327 sendbuf[2] = -sendbuf[0];
328 sendbuf[3] = -sendbuf[1];
329
330 real_t recvbuf[4];
331 SUZERAIN_MPICHKR(MPI_Reduce(sendbuf, recvbuf, 2,
332 suzerain::mpi::datatype<real_t>(),
333 MPI_SUM, 0, MPI_COMM_WORLD));
334
335 const real_t mean_w = recvbuf[0] / nranks;
336 const real_t mean_p = recvbuf[1] / nranks;
337 SUZERAIN_MPICHKR(MPI_Reduce(sendbuf, recvbuf, 4,
338 suzerain::mpi::datatype<real_t>(),
339 MPI_MIN, 0, MPI_COMM_WORLD));
340 INFO0(who,
341 "Wave space zero-zero-normalized workloads (min/mean/max): "
342 << recvbuf[0] << ", " << mean_w << ", " << -recvbuf[2]);
343 INFO0(who,
344 "Physical space zero-zero-normalized workloads (min/mean/max): "
345 << recvbuf[1] << ", " << mean_p << ", " << -recvbuf[3]);
346 }

349void
350application_base::establish_state_storage(
351 const std::size_t linear_nfields,
352 const std::size_t nonlinear_nfields)
353{
354 SUZERAIN_ENSURE(dgrid);
355
356 // Deallocate any existing unpadded linear state not matching the request
357 if (state_linear && state_linear->shape()[0] != linear_nfields) {
358 state_linear.reset();
359 }
360
361 // Allocate the unpadded linear state to match decomposition
362 if (linear_nfields) {
363 state_linear = std::make_shared<
364 state_linear_type
365 >(to_yxz(static_cast<int>(linear_nfields), dgrid->local_wave_extent));
366 DEBUG("Linear state shape (FYXZ): "
367 << shape_array(*state_linear));
368 DEBUG("Linear state strides (FYXZ): "
369 << strides_array(*state_linear));
370 }
371
372 // Deallocate existing transformable nonlinear state not matching request
373 if (state_nonlinear && state_nonlinear->shape()[0] != nonlinear_nfields) {
374 state_nonlinear.reset();
375 }
376
377 // Allocate the transformable nonlinear state to match decomposition
378 if (nonlinear_nfields) {
379 state_nonlinear.reset(allocate_padded_state<
380 state_nonlinear_type
381 >(nonlinear_nfields, *dgrid));
382
383 DEBUG("Nonlinear state shape (FYXZ): "
384 << shape_array(*state_nonlinear));
385 DEBUG("Nonlinear state strides (FYXZ): "
386 << strides_array(*state_nonlinear));
387 }

397void
398application_base::log_discretization_quality()
399{
400 // Work in real-valued quantities as it is a bit simpler
401 bsplineop_lu boplu(*cop);
402 boplu.opform_mass(*cop);
403 double norm;
404 boplu.opnorm(norm);
405 boplu.factor();
406
407 // Compute and display discrete conservation error magnitude
408 MatrixXXr mat = MatrixXXr::Identity(b->n(), b->n());
409 boplu.solve(b->n(), mat.data(), 1, b->n()); // M^-1
410 cop->apply(1, b->n(), 1.0, mat.data(), 1, b->n()); // D*M^-1
411 boplu.solve(b->n(), mat.data(), 1, b->n()); // M^-1*D*M^-1
412 VectorXr vec(b->n());
413 b->integration_coefficients(0, vec.data());
414 vec = vec.transpose() * mat; // w^{T}*M^-1*D*M^-1
415 vec.head<1>()[0] -= -1; // Exact head
416 vec.tail<1>()[0] -= 1; // Exact tail
417 double relerr = vec.norm() / std::sqrt(real_t(2)); // Exact 2-norm
418 INFO0(who, "B-spline discrete conservation relative error near "
419 << relerr * 100 << "%");
420
421 // Compute and display condition number
422 double rcond;
423 boplu.rcond(norm, rcond);
424 INFO0(who, "B-spline mass matrix has condition number near "
425 << (1 / rcond));
426
427 // Compute collocation point spacing relative to Fourier directions...
428 real_t min_delta_y, med_delta_y, max_delta_y;
429 {
430 using namespace boost::accumulators;
431 accumulator_set< real_t, stats<tag::min, tag::median, tag::max> > acc;
432 for (int i = 0; i < b->n(); ++i) acc(b->spacing_collocation_point(i));
433 min_delta_y = extract::min (acc);
434 med_delta_y = extract::median(acc);
435 max_delta_y = extract::max (acc);
436 }
437 if (grid->N.x() > 1) { // ...but display only for non-trivial X basis...
438 const real_t inv_delta_x = grid->N.x() / grid->L.x();
439 INFO0(who, "Collocation spacing min/med/max relative to delta_x: "
440 << min_delta_y * inv_delta_x << ", "
441 << med_delta_y * inv_delta_x << ", "
442 << max_delta_y * inv_delta_x);
443 }
444 if (grid->N.z() > 1) { // ...and display only for non-trivial Z basis.
445 const real_t inv_delta_z = grid->N.z() / grid->L.z();
446 INFO0(who, "Collocation spacing min/med/max relative to delta_z: "
447 << min_delta_y * inv_delta_z << ", "
448 << med_delta_y * inv_delta_z << ", "
449 << max_delta_y * inv_delta_z);
450 }
451}
