130static inline
131suzerain_bsmbsm suzerain_bsmbsm_construct(int S, int n, int kl, int ku)
132{
133 assert(S > 0);
134 assert(n > 0);
135 assert(kl >= 0);
136 assert(ku >= 0);
137
138 suzerain_bsmbsm tmp;
139 tmp.S = S;
140 tmp.n = n;
141 tmp.kl = kl;
142 tmp.ku = ku;
143 tmp.ld = tmp.kl + 1 + tmp.ku;
144 tmp.N = tmp.S*tmp.n;
145 tmp.KL = tmp.S*(tmp.kl + 1) - 1;
146 tmp.KU = tmp.S*(tmp.ku + 1) - 1;
147 tmp.LD = tmp.KL + 1 + tmp.KU;
148 return tmp;
149}

162static inline
163int suzerain_bsmbsm_q(int S, int n, int i)
164{
165 assert(0 <= i && i < S*n);
166
167 div_t t = div(i, S);
168 return t.rem*n + t.quot;
169}
