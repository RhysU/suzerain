117boost::program_options::options_description
118definition_scenario::options_description()
119{
120 using boost::bind;
121 using boost::lexical_cast;
122 using boost::program_options::options_description;
123 using boost::program_options::typed_value;
124 using boost::program_options::value;
125 using std::unique_ptr;
126 using std::string;
127
128 options_description retval("Nondimensional scenario parameters");
129
130 // Complicated add_options() calls done to allow changing the default value
131 // displayed when the default is NaN. NaN is used as a NOP value by client
132 // code. Validation routines used below all silently allow NaNs.
133
134 unique_ptr<typed_value<string> > p;
135
136 // Re
137 p.reset(value<string>());
138 p->notifier(bind(&parse_positive, _1, &Re, name_Re));
139 if (!(boost::math::isnan)(Re)) {
140 p->default_value(lexical_cast<string>(Re));
141 }
142 retval.add_options()(name_Re, p.release(), desc_Re);
143
144 // Ma
145 p.reset(value<string>());
146 p->notifier(bind(&parse_positive, _1, &Ma, name_Ma));

251void
252definition_scenario::save(
253 const esio_handle h) const
254{
255 DEBUG0("Storing definition_scenario parameters");
256
257 // Only root writes data
258 int procid;
259 esio_handle_comm_rank(h, &procid);
260 esio_line_establish(h, 1, 0, (procid == 0 ? 1 : 0));
261
262 esio_line_write(h, name_Re, &this->Re, 0, desc_Re);
263 esio_line_write(h, name_Ma, &this->Ma, 0, desc_Ma);
264 esio_line_write(h, name_Pr, &this->Pr, 0, desc_Pr);
265 esio_line_write(h, name_bulk_rho, &this->bulk_rho, 0, desc_bulk_rho);
266 esio_line_write(h, name_bulk_rho_u, &this->bulk_rho_u, 0, desc_bulk_rho_u);
267 esio_line_write(h, name_bulk_rho_E, &this->bulk_rho_E, 0, desc_bulk_rho_E);
268 esio_line_write(h, name_alpha, &this->alpha, 0, desc_alpha);
269 esio_line_write(h, name_beta, &this->beta, 0, desc_beta);
270 esio_line_write(h, name_gamma, &this->gamma, 0, desc_gamma);
271}

273void
274definition_scenario::load(
275 const esio_handle h,
276 const bool verbose)
277{
278 DEBUG0("Loading definition_scenario parameters");
279
280 // All ranks load
281 esio_line_establish(h, 1, 0, 1);
282
283 definition_scenario t;
284 esio_line_read(h, name_Re, &t.Re, 0);
285 esio_line_read(h, name_Ma, &t.Ma, 0);
286 esio_line_read(h, name_Pr, &t.Pr, 0);
287 if (ESIO_NOTFOUND != esio_line_size(h, name_bulk_rho, NULL)) {
288 esio_line_read(h, name_bulk_rho, &t.bulk_rho, 0);
289 } else {
290 WARN0(desc_bulk_rho << " not found therefore defaulting to NaN");
291 t.bulk_rho = std::numeric_limits<real_t>::quiet_NaN();
292 }
293 if (ESIO_NOTFOUND != esio_line_size(h, name_bulk_rho_u, NULL)) {
294 esio_line_read(h, name_bulk_rho_u, &t.bulk_rho_u, 0);
295 } else {
296 WARN0(desc_bulk_rho_u << " not found therefore defaulting NaN");
297 t.bulk_rho_u = std::numeric_limits<real_t>::quiet_NaN();
298 }
299 if (ESIO_NOTFOUND != esio_line_size(h, name_bulk_rho_E, NULL)) {
300 esio_line_read(h, name_bulk_rho_E, &t.bulk_rho_E, 0);
301 } else {
302 WARN0(desc_bulk_rho_E << " not found therefore defaulting NaN");
303 t.bulk_rho_E = std::numeric_limits<real_t>::quiet_NaN();
304 }
305 esio_line_read(h, name_alpha, &t.alpha, 0);
306 esio_line_read(h, name_beta, &t.beta, 0);
307 esio_line_read(h, name_gamma, &t.gamma, 0);
308
309 this->populate(t, verbose); // Prefer this to incoming
310}
