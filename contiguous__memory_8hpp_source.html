60 template< typename I >
61 contiguous_memory(

63 typename boost::enable_if<boost::is_integral<I> >::type* dummy = 0)
64 : Allocator(),
65 pbegin_(Allocator::allocate(boost::numeric_cast<
66 typename Allocator::size_type>(count))),

72 contiguous_memory(const contiguous_memory &other)
73 : Allocator(other),
74 pbegin_(Allocator::allocate(boost::numeric_cast<

77 pend_(pbegin_ + std::distance(other.pbegin_,other.pend_))
