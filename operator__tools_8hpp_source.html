75 template<typename AlphaType, typename MultiArrayX,
76 typename BetaType, typename MultiArrayY>
77 int bop_accumulate(
78 int nderiv,
79 const AlphaType& alpha, const MultiArrayX &x, int ndx_x,
80 const BetaType& beta, MultiArrayY &y, int ndx_y) const
81 {
82 SUZERAIN_TIMER_SCOPED("operator_tools::bop_accumulate");
83
84 assert(x.shape()[1] == (unsigned) cop.n());
85 assert((unsigned) x.strides()[3] == x.shape()[2] * x.strides()[2] );
86 assert((unsigned) y.strides()[3] == y.shape()[2] * y.strides()[2] );
87 assert(std::equal(x.shape() + 1, x.shape() + 4, y.shape() + 1));
88
89 return cop.accumulate(
90 nderiv, x.shape()[2] * x.shape()[3],
91 alpha, x[ndx_x].origin(), x.strides()[1], x.strides()[2],
92 beta, y[ndx_y].origin(), y.strides()[1], y.strides()[2]);
93 }

218 void zero_dealiasing_modes(MultiArray &x,
219 int ndx_x) const
220 {
221 SUZERAIN_TIMER_SCOPED("operator_tools::zero_dealiasing_modes");
222
223 // Applying the dx^0 dz^0 operator with a scale factor of 1
