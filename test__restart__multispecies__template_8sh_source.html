1#!/bin/bash
2set -eu
3
4# Initialize test infrastructure
5source "`dirname $0`/test_setup_multispecies.sh"
6
7# Shorthand for binary under test for desired operator without statistics
8: ${OPER:=} # E.g. '--explicit' or '--implicit' or unset to use default
9reacting="prun ../reacting_advance -v $OPER --filter=viscous \
10 --statistics_dt=0 --statistics_nt=0"
11
12# These datasets are related to implicit forcing and only are meaningful when
13# using --advance_nt=N for N > 1. They must be ignored for --advance_nt=0.
14exclude_datasets="--exclude-path=/bar_f \
15 --exclude-path=/bar_qb \
16 --exclude-path=/bar_f_dot_u \
17 --exclude-path=/bar_Crho \
18 --exclude-path=/bar_Crhou \
19 --exclude-path=/bar_CrhoE \
20 --exclude-path=/bar_Crhou_dot_u \
21 --exclude-path=/bar_C2rho \
22 --exclude-path=/bar_C2rhou \
23 --exclude-path=/bar_C2rhoE \
24 --exclude-path=/bar_C2rhou_dot_u"
25exclude_datasets=$(echo $exclude_datasets | tr -d '\n' | tr -s ' ')
26
27# Run each test case in this file under the following circumstances
28# (which can be overridden by providing the environment variable METACASES).
29for METACASE in ${METACASES:= 'NP=1' 'NP=2;P=--Pa=2' 'NP=2;P=--Pb=2'}; do
30NP=
31P=
32eval "$METACASE"
33
34banner "Idempotence of restarting without time advance${OPER:+ ($OPER)}"
35(
36 cd $testdir
37 WIZ="--plan_wisdom=wisdom.init" # Prepared by test_setup.sh
38 $reacting multi0.h5 --restart_destination "a#.h5" --advance_nt=0 $WIZ $P \
39 --restart_retain=1
40 differ $exclude_datasets \
41 --exclude-path=/twopoint_kx --exclude-path=/twopoint_kz \
42 multi0.h5 a0.h5
43 differ --delta=3e-16 multi0.h5 a0.h5 /twopoint_kx
44 differ --delta=3e-16 multi0.h5 a0.h5 /twopoint_kz
45)
46
47banner "Equivalence of a field advanced both with and without a restart${OPER:+ ($OPER)}"
48(
49 cd $testdir
50 WIZ="--plan_wisdom=$(mktemp wisdom.XXXXXX)"
51 $reacting multi0.h5 --restart_destination "a#.h5" --advance_nt=2 $WIZ $P \
52 --restart_retain=1
53 $reacting a0.h5 --restart_destination "b#.h5" --advance_nt=2 $WIZ $P \
54 --restart_retain=1
55 $reacting multi0.h5 --restart_destination "c#.h5" --advance_nt=4 $WIZ $P \
56 --restart_retain=1
57
58 # Ensure simulation time "/t" matches before bothering with anything else
59 differ --use-system-epsilon --nan b0.h5 c0.h5 /t
60 differ --delta=6e-15 --nan --exclude-path=/t b0.h5 c0.h5
61)
62
63banner "Upsample/downsample both homogeneous directions${OPER:+ ($OPER)}"
64(
65 cd $testdir
66 WIZ="--plan_wisdom=$(mktemp wisdom.XXXXXX)"
67 $reacting multi0.h5 --restart_destination "a#.h5" --advance_nt=0 $WIZ $P \
68 --restart_retain=1 --Nx=$((2*$Nx)) --Nz=$((3*$Nz))
69 $reacting a0.h5 --restart_destination "b#.h5" --advance_nt=0 $WIZ $P \
70 --restart_retain=1 --Nx=$(( $Nx)) --Nz=$(( $Nz))
71 differ $exclude_datasets \
72 --exclude-path=/twopoint_kx --exclude-path=/twopoint_kz \
73 multi0.h5 b0.h5
74 differ --delta=3e-16 multi0.h5 b0.h5 /twopoint_kx
75 differ --delta=3e-16 multi0.h5 b0.h5 /twopoint_kz
76)
77
78done
