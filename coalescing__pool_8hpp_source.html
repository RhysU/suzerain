526template< typename T, class ArenaAllocator, class Policy >
527typename coalescing_pool<T,ArenaAllocator,Policy>::size_type
528coalescing_pool<T,ArenaAllocator,Policy>::blocksize(
529 size_type newsize)
530{
531 if (blocksize_ == newsize) return blocksize_; // Short circuit on NOP
532
533 if (any_blocks_used()) throw ::std::logic_error(
534 ::std::string(__PRETTY_FUNCTION__)
535 + " cannot set new block size when pool is actively in use");
536
537 newsize = ::std::max((size_type) minimum_blocksize, newsize); // Constrain
538
539 // Purge existing arenas unless newsize evenly divides blocksize_
540 if (blocksize_ % newsize) arenas.clear();
541
542 return (blocksize_ = newsize);
543}

545template< typename T, class ArenaAllocator, class Policy >
546typename coalescing_pool<T,ArenaAllocator,Policy>::size_type
547coalescing_pool<T,ArenaAllocator,Policy>::nblocks()
548{
549 if (!blocksize_) throw ::std::logic_error(::std::string(__PRETTY_FUNCTION__)
550 + " requires blocksize to be previously specified");
551
552 size_type total = 0;
553 typename arenas_type::iterator i, ie;
554 for (i = arenas.begin(), ie = arenas.end(); i != ie; ++i) {
555 total += i->nelems() / blocksize_;
556 }
557 return total;
558}

560template< typename T, class ArenaAllocator, class Policy >
561template< typename U, typename V, typename W>
562typename coalescing_pool<T,ArenaAllocator,Policy>::size_type
563coalescing_pool<T,ArenaAllocator,Policy>::nblocks(
564 U& used, V& free, W& contig)
565{
566 if (!blocksize_) throw ::std::logic_error(::std::string(__PRETTY_FUNCTION__)
567 + " requires blocksize to be previously specified");
568
569 size_type total = 0;
570 free = contig = 0;
571
572 typename arenas_type::iterator i, ie;
573 for (i = arenas.begin(), ie = arenas.end(); i != ie; ++i) {
574 total += i->nelems() / blocksize_;
575 typename fl_type::iterator j, je;
576 for (j = i->fl.begin(), je = i->fl.end(); j != je; ++j) {
577 const size_type n = j->nelems() / blocksize_;
578 free += n;
579 contig = ::std::max(contig, static_cast<W>(n));
580 }
581 }
582
583 used = total - free;
584 return total;
585}

599template< typename T, class ArenaAllocator, class Policy >
600typename coalescing_pool<T,ArenaAllocator,Policy>::size_type
601coalescing_pool<T,ArenaAllocator,Policy>::drain()
602{
603 size_type n = 0;
604
605 typename arenas_type::iterator i, ie;
606 for (i = arenas.begin(), ie = arenas.end(); i != ie;) {
607 if (i->fl.empty() || (*i->fl.begin()) != *i) {
608 ++i;
609 } else {
610 n += i->nelems() / blocksize_;
611 arenas.erase(i++);
612 }
613 }
614
615 return n;
616}

627template< typename T, class ArenaAllocator, class Policy >
628void coalescing_pool<T,ArenaAllocator,Policy>::anticipate_(
629 coalescing_pool::size_type nblocks,
630 typename coalescing_pool::arenas_type::iterator& i,
631 typename coalescing_pool::fl_type::iterator& j)
632{
633 if (!blocksize_) throw ::std::logic_error(::std::string(__PRETTY_FUNCTION__)
634 + " requires blocksize to be previously specified");
635
636 if (!nblocks) return; // Short circuit on degenerate request
637
638 const size_type nelems = nblocks * blocksize_;
639
640 // Search for first arena containing nblocks contiguous, unused blocks
641 const typename arenas_type::iterator ie = arenas.end();
642 const typename arenas_type::iterator ilb = ::std::lower_bound(
643 arenas.begin(), ie, nelems, compare_extents());
644 for (i = ilb; i != ie; ++i) {
645 const typename fl_type::iterator je = i->fl.end();
646 for (j = i->fl.begin(); j != je; ++j) {
647 if (j->e_ >= j->b_ + nelems)
648 return;
649 }
650 }
651
652 // No existing arena satisfies request; coalesce while allocating new arena
653 i = arenas.insert(ilb, new arena(::std::max(blocksize_*drain(), nelems)));
654 j = i->fl.begin();
655}

657template< typename T, class ArenaAllocator, class Policy >
658typename coalescing_pool<T,ArenaAllocator,Policy>::blocks
659coalescing_pool<T,ArenaAllocator,Policy>::acquire(
660 coalescing_pool::size_type nblocks)
661{
662 if (!blocksize_) throw ::std::logic_error(::std::string(__PRETTY_FUNCTION__)
663 + " requires blocksize to be previously specified");
664
665 if (!nblocks) return blocks(); // Short circuit on degenerate request
666
667 // Find or allocate the arena/free list from which we'll acquire nblocks
668 typename coalescing_pool::arenas_type::iterator i;
669 typename coalescing_pool::fl_type::iterator j;
670 anticipate_(nblocks, /* mutated */ i, /* mutated */ j);
671 fl_type &fl = i->fl;
672
673 // Hold out for a perfect fit in arena while tracking a best fit candidate
674 const size_type nelems = nblocks * blocksize_;
675 const typename fl_type::iterator je = fl.end();
676 typename fl_type::iterator b = je;
677 for (; j != je; ++j) {
678 const size_type jn = j->nelems();
679 if (jn == nelems) {
680 fl.erase(j);
681 return *j;
682 } else if (jn > nelems && (b == je || jn < (*b).nelems())) {
683 b = j;
684 }
685 }
686
687 if (b == je) throw ::std::runtime_error(::std::string(__PRETTY_FUNCTION__)
688 + " sanity failure: anticipate_ did not return a usable arena");
689
690 // Split head off *b, record tail in fl, and used head to satisfy request
691 blocks head((*b).b_, nelems);
692 fl.insert(b, *(new (head.e_) fl_entry(head.e_, (*b).nelems() - nelems)));
693 fl.erase(b);
694 return head;
695}

697template< typename T, class ArenaAllocator, class Policy >
698void coalescing_pool<T,ArenaAllocator,Policy>::release(
699 coalescing_pool::blocks& b)
700{
701 using ::std::invalid_argument;
702 using ::std::string;
703
704 if (!b) return; // Short circuit on release of inactive block
705
706 const typename arenas_type::iterator end = arenas.end();
707 typename arenas_type::iterator i = ::std::lower_bound(
708 arenas.begin(), end, b.nelems(), compare_extents());
709 for (; i != end; ++i) {
710 if (i->contains(b)) {
711
712 // Prepare the fl_entry header in block b and add to arena fl
713 // Double release detection necessary since copies of b may exist
714 ::std::pair<typename fl_type::iterator,bool> j
715 = i->fl.insert(*(new (b.b_) fl_entry(b.b_, b.e_)));
716 if (!j.second) throw invalid_argument(string(__PRETTY_FUNCTION__)
717 + " detected invalid argument or double release");
718
719 // Mark b as having been released
720 b.b_ = b.e_ = 0;
721
722 // Coalescing requires only one lookahead or lookbehind
723 // because we perform it on every release invocation
724
725 // Possibly coalesce the next fl entry into j
726 typename fl_type::iterator n = ::boost::next(j.first);
727 if (n != i->fl.end() && j.first->e_ == n->b_) {
728 j.first->e_ = n->e_;
729 i->fl.erase(n);
730 }
731
732 // Possibly coalesce j into the prior fl entry
733 typename fl_type::iterator p = ::boost::prior(j.first); // Invalid?
734 if (j.first != i->fl.begin() && p->e_ == j.first->b_) { // Valid!
735 p->e_ = j.first->e_;
736 i->fl.erase(j.first);
737 }
738
739 return;
740 }
741 }
742
743 throw invalid_argument(string(__PRETTY_FUNCTION__)
744 + " called on blocks from a different pool");
745}
