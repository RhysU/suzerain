53template< std::size_t Dim, typename Element >
54template< typename ExtentList >
55contiguous_state<Dim,Element>::contiguous_state(
56 const shared_range_type& storage,
57 const ExtentList& sizes)
58 : shared_range_type(storage),
59 multi_array_type(shared_range_type::begin(), sizes, storage_order_type())
60{
61 const std::ptrdiff_t minimum_size
62 = storage_order_type::compute_storage(sizes.begin());
63 if (static_cast<std::ptrdiff_t>(shared_range_type::size()) < minimum_size) {
64 std::ostringstream oss;
65 oss << __PRETTY_FUNCTION__ << " requires storage.size() >= "
66 << minimum_size << " but only " << shared_range_type::size()
67 << " was provided.";
68 throw std::invalid_argument(oss.str());
69 }

144template< typename BLASFunctor, typename Element >
145void apply(BLASFunctor functor,
146 multi_array::ref<Element,2>& x,
147 contiguous_state<2,Element>& y)
148{
149 typedef typename contiguous_state<2,Element>::index index;
150 assert(std::equal(x.shape(), x.shape() + 2, y.shape()));
151 if (SUZERAIN_UNLIKELY(0U == x.shape()[1])) return; // Sidesteps assertions
152
153 using boost::numeric_cast;
154 const index iu = numeric_cast<index>(x.index_bases()[0] + x.shape()[0]);
155
156 // Loops go from slower to faster indices for contiguous_state<2,Element>
157 for (index ix = x.index_bases()[0], iy = y.index_bases()[0];
158 ix < iu;
159 ++ix, ++iy) {
160
161 functor(x.shape()[1],
162 &x[ix][x.index_bases()[1]], x.strides()[1],
163 &y[iy][y.index_bases()[1]], y.strides()[1]);
164 }
165}

418template< std::size_t Dim, typename Element >
419void contiguous_state<Dim,Element>::add_scaled(
420 const Element& factor,
421 const contiguous_state& other)
422{
423 SUZERAIN_ENSURE_MSGEXCEPT(this != boost::addressof(other),
424 "Detected this->add_scaled(...,this)", std::invalid_argument);
425 SUZERAIN_ENSURE_EXCEPT(this->is_isomorphic(other), std::invalid_argument);
426
427 if (SUZERAIN_UNLIKELY(std::equal(other.strides(),
428 other.strides() + Dim, this->strides()))) {
429 // Identical strides between elements: use a single BLAS call
430 blas::axpy(shared_range_type::size(), factor,
431 other.shared_range_type::begin(), 1,
432 shared_range_type::begin(), 1);
433 } else {
434 // Different strides between elements: loop over BLAS calls
435 detail::apply(blas::functor::axpy<Element>(factor),
436 const_cast<contiguous_state&>(other), *this);
437 }
438}
