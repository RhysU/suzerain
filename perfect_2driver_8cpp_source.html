88std::vector<std::string>
89driver::initialize(
90 int argc,
91 char **argv)
92{
93 // Only add groups of options when non-trivial at initialization
94 if (scenario ) options.add_definition(*scenario );
95 if (isothermal) options.add_definition(*isothermal);
96 if (sg ) options.add_definition(*sg );
97 if (rad ) options.add_definition(*rad );
98 if (helm ) options.add_definition(*helm );
99 if (msoln ) options.add_definition(*msoln );
100
101 // Delegate to superclass initialization
102 std::vector<std::string> positional = super::initialize(argc, argv);
103
104 // If msoln was provided, match its contents to other members
105 if (msoln) {
106 if (scenario) msoln->match(*scenario);
107 if (grid) msoln->match(*grid);
108 }
109
110 return positional;

126void
127driver::log_manufactured_solution_absolute_error(
128 const std::string& timeprefix,
129 const real_t t,
130 const std::size_t nt)
131{
132 SUZERAIN_UNUSED(nt);
133
134 // NOP whenever a manufactured_solution is not in use
135 if (!msoln) return;
136
137 // Avoid computational cost when logging is disabled
138 support::logging::logger_type mms_abserr
139 = support::logging::get_logger("mms.abserr");
140 if (!INFO0_ENABLED(mms_abserr)) return;
141
142 // Compute $L^2_{xyz}$ of error of state against manufactured solution
143 state_nonlinear->assign_from(*state_linear);
144 accumulate_manufactured_solution(
145 1, *msoln, -1, *state_nonlinear,
146 *grid, *dgrid, *cop, *b, t);
147 const std::vector<field_L2xyz> L2
148 = compute_field_L2xyz(*state_nonlinear, *grid, *dgrid, *gop);
149
150 // Output absolute global errors for each field
151 std::ostringstream msg;
152 msg << timeprefix;
153 for (size_t k = 0; k < L2.size(); ++k) {
154 msg << ' ' << fullprec<>(L2[k].total());
155 }
156 INFO0(mms_abserr, msg.str());
157}

160void
161driver::log_linearization_error(
162 const std::string& timeprefix,
163 const real_t t,
164 const std::size_t nt)
165{
166 SUZERAIN_UNUSED(nt);
167
168 // Avoid computational cost when logging is disabled
169 support::logging::logger_type lin_abserr
170 = support::logging::get_logger("lin.abserr");
171 if (!INFO0_ENABLED(lin_abserr)) return;
172
173 SUZERAIN_TIMER_SCOPED("driver::log_linearization_error");
174
175 // Algorithm proceeds directly from necessary interface guarantees.
176 // Rewriting it symbolically from the code and grokking it is a good
177 // exercise towards understanding the low storage time interfaces.
178 //
179 // Success requires that the following preconditions all hold:
180 // i) At substep zero, this->N computes N(u) - Lu including
181 // the necessary reference quantities.
182 // ii) The reference quantities are stored in common_block and may be
183 // expressly zeroed. Zeroing them nukes linearized contributions
184 // from I{Nonlinear,Linear}Operator but not linear contributions
185 // from this->L.
186 // iii) On non-zero substeps for zero reference values, this->N computes
187 // only N(u) and this->L computes only
188 // \left(M+\varphi\mathscr{L}\right)u.
189 // iv) At the beginning of this process, state_linear contains
190 // valid information and adheres to boundary conditions.
191 // v) The explicit_operator_nonlinear application requires an auxiliary
192 // scaling factor "chi" to account for Fourier transform
193 // normalization needs.
194 // vi) Using diffwave::apply zeros wavenumbers used only for dealiasing
195 // purposes-- this data is unimportant from the perspective of
196 // measuring actual linearization error.
197 state_nonlinear->assign_from(*state_linear);
198 common_block.set_zero(grid->dN.y()); // Defensive
199 N->apply_operator(t, *state_nonlinear, *method, /*substep*/0);
200 L->accumulate_mass_plus_scaled_operator(
201 1, *state_linear, dgrid->chi(), *state_nonlinear, /*substep*/0);
202 common_block.set_zero(grid->dN.y()); // Zero reference quantities
203 L->accumulate_mass_plus_scaled_operator(
204 1, *state_linear, -1, *state_nonlinear, /*substep*/0);
205 for (size_t k = 0; k < fields.size(); ++k) {
206 diffwave::apply(0, 0, 1, (*state_nonlinear)[k].origin(),
207 grid->L.x(), grid->L.z(), dgrid->global_wave_extent.y(),
208 grid->N.x(), grid->dN.x(),
209 dgrid->local_wave_start.x(), dgrid->local_wave_end.x(),
210 grid->N.z(), grid->dN.z(),
211 dgrid->local_wave_start.z(), dgrid->local_wave_end.z());
212 }
213 state_nonlinear->exchange(*state_linear);
214 N->apply_operator(t, *state_nonlinear, *method, /*substep*/1);
215 for (size_t k = 0; k < fields.size(); ++k) {
216 diffwave::apply(0, 0, 1, (*state_nonlinear)[k].origin(),
217 grid->L.x(), grid->L.z(), dgrid->global_wave_extent.y(),
218 grid->N.x(), grid->dN.x(),
219 dgrid->local_wave_start.x(), dgrid->local_wave_end.x(),
220 grid->N.z(), grid->dN.z(),
221 dgrid->local_wave_start.z(), dgrid->local_wave_end.z());
222 }
223 state_nonlinear->add_scaled(1/dgrid->chi(), *state_linear);
224
225 // When this->L and this->N match perfectly, state_nonlinear should be
226 // identically zero at each collocation point. Seeing more than acceptable
227 // (e.g. 1e-10) floating point error indicates something is amiss.
228 std::shared_ptr<operator_tools> otool = obtain_operator_tools();
229 for (size_t f = 0; f < state_nonlinear->shape()[0]; ++f) {
230 otool->bop_solve(*otool->massluz(), *state_nonlinear, f);
231 }
232 const std::vector<field_L2xyz> L2
233 = compute_field_L2xyz(*state_nonlinear, *grid, *dgrid, *gop);
234 std::ostringstream msg;
235 msg << timeprefix;
236 for (size_t k = 0; k < L2.size(); ++k) {
237 msg << ' ' << fullprec<>(L2[k].total());
238 }
239 INFO0(lin_abserr, msg.str());
240}

242void driver::log_quantities_of_interest(
243 const std::string& timeprefix,
244 const real_t t,
245 const std::size_t nt)
246{
247 SUZERAIN_UNUSED(nt);
248
249 if (!grid) return;
250
251 SUZERAIN_TIMER_SCOPED("driver::log_quantities_of_interest");
252
253 // If possible, use existing information from mean quantities
254 // Otherwise compute from instantaneous fields stored in state_linear
255 std::shared_ptr<operator_tools> otool = obtain_operator_tools();
256 profile prof;
257 instantaneous inst;
258 if (controller && mean && mean->t == t) {
259 prof = *mean;
260 inst.copy_from(*cop, *mean);
261 } else {
262 state_nonlinear->assign_from(*state_linear);
263 prof = *take_profile(*scenario, *otool, *state_nonlinear, inst);
264 }
265
266 this->log_quantities_of_interest(timeprefix, prof, inst);
267}

311void
312driver::log_quantities_of_interest(
313 const std::string& prefix,
314 const profile& prof,
315 const instantaneous& inst,
316 const support::shared_esio_handle& esioh)
317{
318 // A horrible way to cheaply grab a shared pointer to a masslu
319 // Uses internal knowledge of obtain_operator_tools requiring a dgrid
320 std::shared_ptr<const bsplineop_lu> masslu;
321 if (dgrid) {
322 masslu = obtain_operator_tools()->masslu();
323 } else {
324 std::shared_ptr<bsplineop_lu> t(new bsplineop_lu(*cop));
325 t->factor_mass(*cop);
326 masslu = t;
327 }
328
329 // Fluctuations only make sense to log when either X or Z is nontrivial
330 const bool nontrivial_fluct_possible = grid->N.x() * grid->N.z() > 1;
331 if (nontrivial_fluct_possible) {
332
333 // Compute pointwise Reynolds-averaged and Favre-averaged fluctuation
334 // profiles as three contributions to the TKE production term.
335 ArrayX6r reynolds;
336 ArrayX6r favre;
337 ArrayX3r prodterms;
338 compute_fluctuations(*cop, *masslu, inst, reynolds, favre, prodterms);
339
340 // Find absolute of maximum of columns (with signed results)
341 // as well as the locations for those maximums.
342 Array6r amax_reynolds, yamax_reynolds;
343 for (int j = 0; j < reynolds.cols(); ++j) {
344 ArrayX6r::Index ndx;
345 reynolds.col(j).abs().maxCoeff(&ndx);
346 amax_reynolds[j] = reynolds.col(j)[ndx];
347 yamax_reynolds[j] = b->collocation_point(static_cast<int>(ndx));
348 }
349 Array6r amax_favre, yamax_favre;
350 for (int j = 0; j < favre.cols(); ++j) {
351 ArrayX6r::Index ndx;
352 favre.col(j).abs().maxCoeff(&ndx);
353 amax_favre[j] = favre.col(j)[ndx];
354 yamax_favre[j] = b->collocation_point(static_cast<int>(ndx));
355 }
356 Array3r amax_prodterms, yamax_prodterms;
357 for (int j = 0; j < prodterms.cols(); ++j) {
358 ArrayX3r::Index ndx;
359 prodterms.col(j).abs().maxCoeff(&ndx);
360 amax_prodterms[j] = prodterms.col(j)[ndx];
361 yamax_prodterms[j] = b->collocation_point(static_cast<int>(ndx));
362 }
363
364 // Compute derived information about overall production
365 ArrayX1r production = prodterms.rowwise().sum();
366 Array1r amax_production, yamax_production;
367 for (int j = 0; j < production.cols(); ++j) {
368 ArrayX1r::Index ndx;
369 production.col(j).abs().maxCoeff(&ndx);
370 amax_production[j] = production.col(j)[ndx];
371 yamax_production[j] = b->collocation_point(static_cast<int>(ndx));
372 }
373
374 // Find coefficient representations from collocation point values
375 masslu->solve(reynolds.cols(), reynolds.data(),
376 reynolds.innerStride(), reynolds.outerStride());
377 masslu->solve(favre.cols(), favre.data(),
378 favre.innerStride(), favre.outerStride());
379 masslu->solve(prodterms.cols(), prodterms.data(),
380 prodterms.innerStride(), prodterms.outerStride());
381 masslu->solve(production.cols(), production.data(),
382 production.innerStride(), production.outerStride());
383
384 // Reduce coefficients into bulk values
385 const VectorXr bulk = support::compute_bulk_weights(*b, *masslu);
386 {
387 // Broken? reynolds .matrix().applyOnTheLeft(bulk.transpose());
388 ArrayXXr tmp;
389 tmp = bulk.transpose() * reynolds .matrix(); reynolds = tmp;
390 tmp = bulk.transpose() * favre .matrix(); favre = tmp;
391 tmp = bulk.transpose() * prodterms .matrix(); prodterms = tmp;
392 tmp = bulk.transpose() * production.matrix(); production = tmp;
393 }
394
395 // Log a whole slew of information about turbulent production
396 {
397 const std::string name("prod.bulk");
398 maybe_prefix_prodterm(*this, prefix, name, header_shown[name]);
399 std::ostringstream msg;
400 msg << prefix
401 << ' ' << fullprec<>(production(0,0))
402 << ' ' << fullprec<>(prodterms(0,0))
403 << ' ' << fullprec<>(prodterms(0,1))
404 << ' ' << fullprec<>(prodterms(0,2));
405 INFO0(name, msg.str());
406 }
407 {
408 const std::string name("prod.amax");
409 maybe_prefix_prodterm(*this, prefix, name, header_shown[name]);
410 std::ostringstream msg;
411 msg << prefix
412 << ' ' << fullprec<>(amax_production[0])
413 << ' ' << fullprec<>(amax_prodterms[0])
414 << ' ' << fullprec<>(amax_prodterms[1])
415 << ' ' << fullprec<>(amax_prodterms[2]);
416 INFO0(name, msg.str());
417 }
418 {
419 const std::string name("prod.yamax");
420 maybe_prefix_prodterm(*this, prefix, name, header_shown[name]);
421 std::ostringstream msg;
422 msg << prefix
423 << ' ' << fullprec<>(yamax_production[0])
424 << ' ' << fullprec<>(yamax_prodterms[0])
425 << ' ' << fullprec<>(yamax_prodterms[1])
426 << ' ' << fullprec<>(yamax_prodterms[2]);
427 INFO0(name, msg.str());
428 }
429
430 // Log information about bulk fluctuations
431 // Reynolds and then Favre to ease impromptu comparison
432 {
433 const std::string name("reyno.bulk");
434 maybe_prefix_stresses(*this, prefix, name, header_shown[name]);
435 std::ostringstream msg;
436 msg << prefix
437 << ' ' << fullprec<>(reynolds(0,0))
438 << ' ' << fullprec<>(reynolds(0,1))
439 << ' ' << fullprec<>(reynolds(0,2))
440 << ' ' << fullprec<>(reynolds(0,3))
441 << ' ' << fullprec<>(reynolds(0,4))
442 << ' ' << fullprec<>(reynolds(0,5));
443 INFO0(name, msg.str());
444 }
445 {
446 const std::string name("favre.bulk");
447 maybe_prefix_stresses(*this, prefix, name, header_shown[name]);
448 std::ostringstream msg;
449 msg << prefix
450 << ' ' << fullprec<>(favre(0,0))
451 << ' ' << fullprec<>(favre(0,1))
452 << ' ' << fullprec<>(favre(0,2))
453 << ' ' << fullprec<>(favre(0,3))
454 << ' ' << fullprec<>(favre(0,4))
455 << ' ' << fullprec<>(favre(0,5));
456 INFO0(name, msg.str());
457 }
458
459 // Log information about maximum pointwise fluctuations
460 // Reynolds and then Favre to ease impromptu comparison
461 {
462 const std::string name("reyno.amax");
463 maybe_prefix_stresses(*this, prefix, name, header_shown[name]);
464 std::ostringstream msg;
465 msg << prefix
466 << ' ' << fullprec<>(amax_reynolds[0])
467 << ' ' << fullprec<>(amax_reynolds[1])
468 << ' ' << fullprec<>(amax_reynolds[2])
469 << ' ' << fullprec<>(amax_reynolds[3])
470 << ' ' << fullprec<>(amax_reynolds[4])
471 << ' ' << fullprec<>(amax_reynolds[5]);
472 INFO0(name, msg.str());
473 }
474 {
475 const std::string name("favre.amax");
476 maybe_prefix_stresses(*this, prefix, name, header_shown[name]);
477 std::ostringstream msg;
478 msg << prefix
479 << ' ' << fullprec<>(amax_favre[0])
480 << ' ' << fullprec<>(amax_favre[1])
481 << ' ' << fullprec<>(amax_favre[2])
482 << ' ' << fullprec<>(amax_favre[3])
483 << ' ' << fullprec<>(amax_favre[4])
484 << ' ' << fullprec<>(amax_favre[5]);
485 INFO0(name, msg.str());
486 }
487
488 // Log information about the location of maximum pointwise fluctuations
489 // Reynolds and then Favre to ease impromptu comparison
490 {
491 const std::string name("reyno.yamax");
492 maybe_prefix_stresses(*this, prefix, name, header_shown[name]);
493 std::ostringstream msg;
494 msg << prefix
495 << ' ' << fullprec<>(yamax_reynolds[0])
496 << ' ' << fullprec<>(yamax_reynolds[1])
497 << ' ' << fullprec<>(yamax_reynolds[2])
498 << ' ' << fullprec<>(yamax_reynolds[3])
499 << ' ' << fullprec<>(yamax_reynolds[4])
500 << ' ' << fullprec<>(yamax_reynolds[5]);
501 INFO0(name, msg.str());
502 }
503 {
504 const std::string name("favre.yamax");
505 maybe_prefix_stresses(*this, prefix, name, header_shown[name]);
506 std::ostringstream msg;
507 msg << prefix
508 << ' ' << fullprec<>(yamax_favre[0])
509 << ' ' << fullprec<>(yamax_favre[1])
510 << ' ' << fullprec<>(yamax_favre[2])
511 << ' ' << fullprec<>(yamax_favre[3])
512 << ' ' << fullprec<>(yamax_favre[4])
513 << ' ' << fullprec<>(yamax_favre[5]);
514 INFO0(name, msg.str());
515 }
516 }
517
518 // Prepare and log geometry-specific quantities of interest
519 if (grid->one_sided()) {
520
521 // Compute many details about the boundary layer for logging
522 suzerain_bl_local wall;
523 suzerain_bl_viscous viscous;
524 suzerain_bl_local edge;
525 suzerain_bl_local edge99;
526 suzerain_bl_thicknesses thick;
527 suzerain_bl_reynolds reynolds;
528 suzerain_bl_qoi qoi;
529 suzerain_bl_pg pg;
530 summarize_boundary_layer_nature(prof, *scenario, sg, *masslu, *b,
531 wall, viscous, thick,
532 edge, edge99,
533 reynolds, qoi, pg);
534
535 // Log messages using application-agnostic superclass functionality
536 this->log_quantities_boundary_layer(prefix,
537 &wall, &viscous, &thick,
538 &edge, &edge99,
539 &reynolds, &qoi, &pg,
540 esioh);
541
542 } else if (grid->two_sided()) {
543
544 // Compute many details about the boundary layer for logging
545 suzerain_channel_local wall;
546 suzerain_channel_viscous viscous;
547 suzerain_channel_local center;
548 suzerain_channel_qoi qoi;
549 summarize_channel_nature(prof, *scenario, *b,
550 wall, viscous, center, qoi);
551
552 // Log messages using application-agnostic superclass functionality
553 this->log_quantities_channel(prefix,
554 &wall, &viscous, &center, &qoi,
555 esioh);
556
557 } else {
558
559 SUZERAIN_ERROR_REPORT_UNIMPLEMENTED();
560
561 }

565void
566driver::compute_statistics(
567 driver::time_type t)
568{
569 SUZERAIN_TIMER_SCOPED("driver::compute_statistics");
570
571 // Obtain mean samples from instantaneous fields stored in state_linear
572 state_nonlinear->assign_from(*state_linear);
573 std::shared_ptr<operator_tools> otool = obtain_operator_tools();
574 mean = take_samples(*scenario, *sg, *otool, *b, *state_nonlinear, t);
575
576 // Obtain mean quantities computed via implicit forcing (when possible)
577 if (common_block.imp.rows() == mean->storage.rows()) {
578 mean->f().col(0) = common_block.imp.fx();
579 mean->f().col(1) = common_block.imp.fy();
580 mean->f().col(2) = common_block.imp.fz();
581 mean->f_dot_u() = common_block.imp.f_dot_u();
582 mean->qb() = common_block.imp.qb();
583 mean->CrhoE() = common_block.imp.CrhoE();
584 mean->C2rhoE() = common_block.imp.C2rhoE();
585 mean->Crhou() .col(0) = common_block.imp.Crhou();
586 mean->C2rhou().col(0) = common_block.imp.C2rhou();
587 mean->Crhou() .col(1) = common_block.imp.Crhov();
588 mean->C2rhou().col(1) = common_block.imp.C2rhov();
589 mean->Crhou() .col(2) = common_block.imp.Crhow();
590 mean->C2rhou().col(2) = common_block.imp.C2rhow();
591 mean->Crho() = common_block.imp.Crho();
592 mean->C2rho() = common_block.imp.C2rho();
593 mean->Crhou_dot_u() = common_block.imp.Crhou_dot_u();
594 mean->C2rhou_dot_u() = common_block.imp.C2rhou_dot_u();
595
596 // Convert values from collocation points to B-spline coefficients
597 bsplineop_lu mass(*cop);
598 mass.opform_mass(*cop);
599 mass.factor();
600 mass.solve(mean->implicit().cols(),
601 mean->implicit().data(),
602 mean->implicit().innerStride(),
603 mean->implicit().outerStride());
604 } else {
605 WARN0(who, "Could not obtain mean quantities set by implicit forcing");
606 }
607}

609void
610driver::save_spectra_primitive(
611 const esio_handle esioh)
612{
613 SUZERAIN_TIMER_SCOPED("driver::save_spectra_primitive");
614
615 enum { swave_count = 5 };
616 assert(static_cast<int>(ndx::e ) < swave_count);
617 assert(static_cast<int>(ndx::mx ) < swave_count);
618 assert(static_cast<int>(ndx::my ) < swave_count);
619 assert(static_cast<int>(ndx::mz ) < swave_count);
620 assert(static_cast<int>(ndx::rho) < swave_count);
621 SUZERAIN_ENSURE((int) state_nonlinear->shape()[0] == swave_count);
622
623 // Obtain information from instantaneous fields in *state_linear
624 state_nonlinear->assign_from(*state_linear);
625
626 // Convert to pointwise conserved state in physical space
627 physical_view<swave_count> sphys(*dgrid, *state_nonlinear);
628 std::shared_ptr<operator_tools> otool = obtain_operator_tools();
629 for (size_t f = 0; f < swave_count; ++f) {
630 otool->zero_dealiasing_modes( *state_nonlinear, f);
631 otool->bop_apply (0, 1, *state_nonlinear, f);
632 dgrid->transform_wave_to_physical(&sphys.coeffRef(f,0));
633 }
634
635 // Pointwise conversion from e, mx, my, mz, rho to T, u, v, w, rho
636 const real_t alpha = scenario->alpha;
637 const real_t beta = scenario->beta;
638 const real_t gamma = scenario->gamma;
639 const real_t Ma = scenario->Ma;
640 for (int offset = 0; offset < sphys.cols(); ++offset) {
641
642 // Unpack conserved state
643 const real_t e (sphys(ndx::e, offset));
644 const Vector3r m (sphys(ndx::mx, offset),
645 sphys(ndx::my, offset),
646 sphys(ndx::mz, offset));
647 const real_t rho(sphys(ndx::rho, offset));
648
649 // Compute desired primitive state
650 const Vector3r u = m / rho;
651 real_t p, T;
652 rholut::p_T(alpha, beta, gamma, Ma, rho, m, e, p, T);
653
654 // Pack desired primitive state
655 sphys(0, offset) = T;
656 sphys(1, offset) = u.x();
657 sphys(2, offset) = u.y();
658 sphys(3, offset) = u.z();
659 sphys(4, offset) = rho;
660 }
661
662 // Dealiasing modes not zeroed else can't recover m = rho * u.
663 // Convert to normalized Fourier coefficients in XZ but points in Y
664 // Leave mean else ensemble of two-point in physical space is wrong!
665 sphys *= dgrid->chi();
666 for (size_t f = 0; f < swave_count; ++f) {
667 dgrid->transform_physical_to_wave(&sphys.coeffRef(f,0));
668 }
669
670 // Only rank zero will write the results though all ranks must participate
671 const int npairs = (swave_count*(swave_count+1))/2;
672 int procid;
673 esio_handle_comm_rank(esioh, &procid);
674
675 // Compute and save the two-point correlation as (y_j, k_x, ndxpair)
676 // Ordering arises from packing of primitive state in physical space
677 {
678 std::shared_ptr<complex_t[]> twopoint_x = compute_twopoint_x(
679 *state_nonlinear, swave_count, *grid, *dgrid);
680 esio_field_establish(esioh,
681 npairs, 0, procid == 0 ? npairs : 0,
682 grid->N.x()/2+1, 0, procid == 0 ? grid->N.x()/2+1 : 0,
683 grid->N.y(), 0, procid == 0 ? grid->N.y() : 0);
684 support::complex_field_write(esioh,
685 "twopoint_kx", twopoint_x.get(), 0, 0, 0,
686 "Streamwise two-point correlations stored row-major"
687 " (/collocation_points_y, /kx, scalarpair) for scalarpair"
688 " in { T*T, T*u, T*v, T*w, T*rho, u*u, u*v, u*w, u*rho,"
689 " v*v, v*w, v*rho, w*w, w*rho, rho*rho }");
690 }
691
692 // Compute and save the two-point correlation as (y_j, k_z, ndxpair)
693 // Ordering arises from packing of primitive state in physical space
694 {
695 std::shared_ptr<complex_t[]> twopoint_z = compute_twopoint_z(
696 *state_nonlinear, swave_count, *grid, *dgrid);
697 esio_field_establish(esioh,
698 npairs, 0, procid == 0 ? npairs : 0,
699 grid->N.z(), 0, procid == 0 ? grid->N.z() : 0,
700 grid->N.y(), 0, procid == 0 ? grid->N.y() : 0);
701 support::complex_field_write(esioh,
702 "twopoint_kz", twopoint_z.get(), 0, 0, 0,
703 "Spanwise two-point correlations stored row-major"
704 " (/collocation_points_y, /kz, scalarpair) for scalarpair"
705 " in { T*T, T*u, T*v, T*w, T*rho, u*u, u*v, u*w, u*rho,"
706 " v*v, v*w, v*rho, w*w, w*rho, rho*rho }");
707 }

844// Heavy-handed, but I seemingly cannot get Intel to cooperator otherwise.
845// I want non-NaNs to be preferred to NaNs within this max operation.
846static
847real_t maxmf(const real_t a, const real_t b)
848{
849 return (boost::math::isnan)(a) ? b
850 : (boost::math::isnan)(b) ? a
851 : std::max(a, b);

854void
855driver::default_restart_interval(
856 time_type& t,
857 step_type&)
858{
859 using std::abs;
860
861 // Look for the largest magnitude, problem-dependent, macro velocity scale
862 real_t velocity = std::numeric_limits<real_t>::quiet_NaN();
863
864 // In a channel...
865 if (grid && grid->two_sided()) {
866
867 // ...any approximate bulk velocity from a driving force...
868 if (scenario) {
869
870 if ((boost::math::isnan)(scenario->bulk_rho)) {
871 TRACE0(who, "No bulk density scale available so assuming 1");
872 velocity = maxmf(abs(scenario->bulk_rho_u) / /*rho*/ 1,
873 velocity);
874 } else {
875 velocity = maxmf(abs(scenario->bulk_rho_u) / scenario->bulk_rho,
876 velocity);
877 }
878
879 }
880
881 // ...may be trumped by driving the upper and lower walls.
882 if (isothermal) {
883 velocity = maxmf(abs(isothermal->upper_u), velocity);
884 velocity = maxmf(abs(isothermal->lower_u), velocity);
885 }
886
887 }
888
889 // On a plate...
890 if (grid && grid->one_sided()) {
891
892 if (sg->formulation.enabled() && sg->baseflow) {
893 // ...prefer wall velocity from inviscid baseflow as freestream
894 // surrogate as it is independent of the wall-normal domain size...
895 largo_state freestream;
896 sg->baseflow->conserved(0.0, freestream.as_is());
897 velocity = maxmf(abs(freestream.u()), velocity);
898 } else if (isothermal) {
899 // ...taking freestream reference if-and-only-if no baseflow...
900 velocity = maxmf(abs(isothermal->upper_u), velocity);
901 }
902
903 // ...and permit a driven lower wall velocity to trump.
904 if (isothermal) {
905 velocity = maxmf(abs(isothermal->lower_u), velocity);
906 }
907
908 }
909
910 // Have a domain size and a velocity?
911 if (boost::math::isnormal(velocity)) {
912 if (grid) {
913
914 // Let's start from the associated time scale.
915 t = grid->L.x() / velocity;
916
917 // Eight restarts per flow through in a channel giving 80 restarts
918 // in 10 flow throughs, a reasonable channel simulation duration.
919 // Across O(25) flow throughs in a boundary layer campaign it
920 // produces O(200) restart files which is slightly excessive but
921 // increases data redundancy for these more expensive simulations.
922 t /= 8;
923
924 } else {
925 DEBUG0(who, "No grid details for default_restart_interval");
926 }
927 } else {
928 DEBUG0(who, "No macro scale velocity for default_restart_interval");
929 }
930}
