20/*
21 * Best-effort write(2) used only on error-reporting paths where a saved
22 * descriptor is the last resort and there is no recourse should the write
23 * itself fail. Consuming the return value here silences warn_unused_result
24 * without discarding a result we could otherwise act upon.
25 */
26static void
27best_effort_write(int fd, const void *buf, size_t count)
28{
29 const ssize_t written = write(fd, buf, count);
30 (void) written;
31}

34 * License identical to gsl_ieee_env_setup() from GSL.
35 * Copyright (c) 2012-2014, 2026 Rhys Ulerich
36 */
37void
38mpi_gsl_ieee_env_setup(const int rank)
39{
40 // Used for write(2)-based error messages
41 static const char newline[] = "\n";
42
43 // Flush stdout, stderr
44 if (fflush(stdout))
45 perror("mpi_gsl_ieee_env_setup fflush(stdout) before redirect");
46 if (fflush(stderr))
47 perror("mpi_gsl_ieee_env_setup fflush(stderr) before redirect");
48
49 // Save stdout, stderr so we may restore them later
50 int stdout_copy, stderr_copy;
51 if ((stdout_copy = dup(fileno(stdout))) < 0)
52 perror("mpi_gsl_ieee_env_setup error duplicating stdout");
53 if ((stderr_copy = dup(fileno(stderr))) < 0)
54 perror("mpi_gsl_ieee_env_setup error duplicating stderr");
55
56 // On non-root processes redirect stdout, stderr to /dev/null
57 if (rank) {
58 if (!freopen("/dev/null", "a", stdout))
59 perror("mpi_gsl_ieee_env_setup redirecting stdout");
60 if (!freopen("/dev/null", "a", stderr))
61 perror("mpi_gsl_ieee_env_setup redirecting stderr");
62 }
63
64 // Invoke gsl_ieee_env_setup on all ranks.
65 gsl_ieee_env_setup();
66
67 // Flush stdout, stderr again.
68 // Error messages sent to stderr_copy on a best-effort basis
69 if (fflush(stdout)) {
70 static const char pre[] = "mpi_gsl_ieee_env_setup fflush(stdout) after redirect: ";
71 const char *msg = strerror(errno);
72 fsync(stderr_copy);
73 best_effort_write(stderr_copy, pre, sizeof(pre));
74 best_effort_write(stderr_copy, msg, strlen(msg));
75 best_effort_write(stderr_copy, newline, strlen(newline));
76 fsync(stderr_copy);
77 }
78 if (fflush(stderr)) {
79 static const char pre[] = "mpi_gsl_ieee_env_setup fflush(stderr) after redirect: ";
80 const char *msg = strerror(errno);
81 fsync(stderr_copy);
82 best_effort_write(stderr_copy, pre, sizeof(pre));
83 best_effort_write(stderr_copy, msg, strlen(msg));
84 best_effort_write(stderr_copy, newline, strlen(newline));
85 fsync(stderr_copy);
86 }
87
88 // Restore stdout, stderr
89 // Error messages sent to stderr_copy on a best-effort basis
90 if (dup2(stdout_copy, fileno(stdout)) < 0) {
91 static const char pre[] = "mpi_gsl_ieee_env_setup reopening stdout: ";
92 const char *msg = strerror(errno);
93 fsync(stderr_copy);
94 best_effort_write(stderr_copy, pre, sizeof(pre));
95 best_effort_write(stderr_copy, msg, strlen(msg));
96 best_effort_write(stderr_copy, newline, strlen(newline));
97 fsync(stderr_copy);
98 }
99 if (dup2(stderr_copy, fileno(stderr)) < 0) {
100 static const char pre[] = "mpi_gsl_ieee_env_setup reopening stderr: ";
101 const char *msg = strerror(errno);
102 fsync(stderr_copy);
103 best_effort_write(stderr_copy, pre, sizeof(pre));
104 best_effort_write(stderr_copy, msg, strlen(msg));
105 best_effort_write(stderr_copy, newline, strlen(newline));
106 fsync(stderr_copy);
107 }
108
109 // Close saved versions of stdout, stderr
110 if (close(stdout_copy))
111 perror("mpi_gsl_ieee_env_setup closing stdout_copy");
112 if (close(stderr_copy))
113 perror("mpi_gsl_ieee_env_setup closing stderr_copy");
114
115 // Clear any errors that may have occurred on stdout, stderr
116 clearerr(stdout);
117 clearerr(stderr);
118}
