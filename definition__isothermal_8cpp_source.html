324void
325definition_isothermal::save(
326 const esio_handle h) const
327{
328 DEBUG0("Storing definition_isothermal parameters");
329
330 // Only root writes data
331 int procid;
332 esio_handle_comm_rank(h, &procid);
333
334 // scalars
335 esio_line_establish(h, 1, 0, (procid == 0 ? 1 : 0));
336 esio_line_write(h, name_lower_T, &lower_T, 0, desc_lower_T);
337 esio_line_write(h, name_lower_u, &lower_u, 0, desc_lower_u);
338 esio_line_write(h, name_lower_v, &lower_v, 0, desc_lower_v);
339 esio_line_write(h, name_lower_w, &lower_w, 0, desc_lower_w);
340 esio_line_write(h, name_lower_rho, &lower_rho, 0, desc_lower_rho);
341 esio_line_write(h, name_upper_T, &upper_T, 0, desc_upper_T);
342 esio_line_write(h, name_upper_u, &upper_u, 0, desc_upper_u);
343 esio_line_write(h, name_upper_v, &upper_v, 0, desc_upper_v);
344 esio_line_write(h, name_upper_w, &upper_w, 0, desc_upper_w);
345 esio_line_write(h, name_upper_rho, &upper_rho, 0, desc_upper_rho);
346
347 // Lower mass fractions vector (only written when non-trivial)
348 int Ns;
349 if ((Ns = lower_cs.size())) {
350 esio_line_establish(h, Ns, 0, (procid == 0 ? Ns : 0));
351 esio_line_write(h, name_lower_cs, lower_cs.data(), 1, desc_lower_cs);
352 } else {
353 WARN0("No lower mass fractions saved because lower_cs.size() == 0");
354 }
355
356 // Upper mass fractions vector (ditto, could be of different length)
357 if ((Ns = upper_cs.size())) {
358 esio_line_establish(h, Ns, 0, (procid == 0 ? Ns : 0));
359 esio_line_write(h, name_upper_cs, upper_cs.data(), 1, desc_upper_cs);
360 } else {
361 WARN0("No upper mass fractions saved because upper_cs.size() == 0");
362 }
363}

365void
366definition_isothermal::load(
367 const esio_handle h,
368 const bool verbose)
369{
370 DEBUG0("Loading definition_isothermal parameters");
371
372 definition_isothermal t;
373
374 // All ranks load
375
376 // Scalars
377 esio_line_establish(h, 1, 0, 1);
378
379 // Backwards compatibility:
380 // The old, unspecified default for apps/perfect was to use T_wall = 1.
381 // The old, apps/reacting code saved lower_T = upper_T = T_wall in "/T_wall".
382 // The loading logic honors those two older behaviors in a verbose manner.
383 real_t default_T_wall = 1;
384 if (ESIO_NOTFOUND != esio_line_size(h, "T_wall", NULL)) {
385 esio_line_read(h, "T_wall", &default_T_wall, 0);
386 INFO0("Found old-style wall temperature data = " << default_T_wall);
387 }
388
389#define LOAD_OR_DEFAULT(var, default_value) \
390 do { \
391 if (ESIO_NOTFOUND == esio_line_size(h, name_##var, NULL)) { \
392 t.var = default_value; \
393 INFO0(desc_##var << " not found therefore using " << t.var); \
394 } else { \
395 esio_line_read(h, name_##var, &t.var, 0); \
396 } \
397 } while(0)
398
399 LOAD_OR_DEFAULT(lower_T, default_T_wall);
400 LOAD_OR_DEFAULT(lower_u, 0);
401 LOAD_OR_DEFAULT(lower_v, 0);
402 LOAD_OR_DEFAULT(lower_w, 0);
403 LOAD_OR_DEFAULT(lower_rho, std::numeric_limits<real_t>::quiet_NaN());
404 LOAD_OR_DEFAULT(upper_T, default_T_wall);
405 LOAD_OR_DEFAULT(upper_u, 0);
406 LOAD_OR_DEFAULT(upper_v, 0);
407 LOAD_OR_DEFAULT(upper_w, 0);
408 LOAD_OR_DEFAULT(upper_rho, 1);
409
410#undef LOAD_OR_DEFAULT
411
412 // Lower mass fractions vector (only present when non-trivial)
413 int Ns;
414 if (ESIO_NOTFOUND == esio_line_size(h, name_lower_cs, &Ns)) {
415 INFO0(name_lower_cs << " not found therefore using 1 dilluter species");
416 t.lower_cs.assign(1U, 1.0);
417 } else {
418 t.lower_cs.resize(Ns);
419 esio_line_establish(h, Ns, 0, Ns);
420 esio_line_read(h, name_lower_cs, t.lower_cs.data(), 0);
421 }
422
423 // Upper mass fractions vector (ditto, could be be of different length)
424 if (ESIO_NOTFOUND == esio_line_size(h, name_upper_cs, &Ns)) {
425 INFO0(name_upper_cs << " not found therefore using 1 dilluter species");
426 t.upper_cs.assign(1U, 1.0);
427 } else {
428 t.upper_cs.resize(Ns);
429 esio_line_establish(h, Ns, 0, Ns);
430 esio_line_read(h, name_upper_cs, t.upper_cs.data(), 0);
431 }
432
433 this->populate(t, verbose); // Prefer this to incoming
434}
