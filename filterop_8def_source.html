94static int WORK(suzerain_filterop,_operator_assemble_cookcabot2005)(
95 const SCALAR *method_params,
96 WORK(suzerain_filterop,_workspace) *w)
97{
98 // Unpack method-specific parameters using defaults whenever NULL
99 const SCALAR alpha = method_params ? method_params[0]
100 : ((SCALAR) 66624)/100000;
101
102 static const int inc = 1; // Column vectors are contiguous in memory
103 const int n = w->n; // Filtering matrices are square
104
105 // Compute coefficients for CookCabot2005 filter
106 // NB: Convince yourself that type promotion gives full SCALAR precision
107 const SCALAR alpha_0 = 1;
108 const SCALAR alpha_1 = alpha;
109 const SCALAR alpha_2 = ( 1- alpha)/ 2;
110 const SCALAR a_0 = ( 58+105*alpha)/128;
111 const SCALAR a_1 = ( 14+ 11*alpha)/ 32;
112 const SCALAR a_2 = ( 18- 11*alpha)/ 64;
113 const SCALAR a_3 = ( 2- 3*alpha)/ 32;
114 const SCALAR a_4 = (- 2+ 3*alpha)/256;
115
116 // Place nonzero B^T values in w->B_T using w->klbt, w->kubt, and w->ldbt.
117 {
118 // Banded matrix access has form a[(ku + i)*inc + j*(lda - inc)].
119 // Incorporate the ku offset and decrement ld to speed indexing.
120 // Further, increment kl anticipating calls like imin(m, j + kl + 1).
121 int ku = w->kubt, ld = w->ldbt;
122 SCALAR * bt_j = w->B_T;
123 bt_j += ku*inc;
124 ld -= inc;
125
126 // With the row elements of B_T coded explicitly
127 // there is no need for kl+1
128 // ++kl;
129
130 assert(w->klbt == 4);
131 assert(w->kubt == 4);
132
133 for (int j = 0; j < n; bt_j += ld, ++j) {
134
135 // Start at j - ku, go to j + kl, Bandwidth is 4
136 bt_j[(j-4)*inc] = a_4;
137 bt_j[(j-3)*inc] = a_3;
138 bt_j[(j-2)*inc] = a_2;
139 bt_j[(j-1)*inc] = a_1;
140 bt_j[(j )*inc] = a_0;
141 bt_j[(j+1)*inc] = a_1;
142 bt_j[(j+2)*inc] = a_2;
143 bt_j[(j+3)*inc] = a_3;
144 bt_j[(j+4)*inc] = a_4;
145 }
146 }
147
148 // Place nonzero A^T values in w->A_T using w->klat, w->kuat, and w->ldat.
149 {
150 // Again, access has form a[(ku + i)*inc + j*(lda - inc)].
151 int kl = w->klat, ku = w->kuat, ld = w->ldat;
152 SCALAR * at_j = w->A_T + kl; // Accounts for factorization-ready data
153 at_j += ku*inc;
154 ld -= inc;
155
156 // With the row elements of A_T coded explicitly
157 // there is no need for kl+1
158 // ++kl;
159
160 assert(w->klat == 2);
161 assert(w->kuat == 2);
162
163 for (int j = 0; j < n; at_j += ld, ++j) {
164
165 // Start at j - ku, go to j + kl, Bandwidth is 2
166 at_j[(j-2)*inc] = alpha_2;
167 at_j[(j-1)*inc] = alpha_1;
168 at_j[(j )*inc] = alpha_0;
169 at_j[(j+1)*inc] = alpha_1;
170 at_j[(j+2)*inc] = alpha_2;
171 }
172 }
173
174 // Note: The matrices A_T and B_T were populated in
175 // BLAS-General-Band storage mode. The following loop
176 // would populate elements from a matrix
177 // with elements A_T[i,j] = 100*i+j:
178 // for (int j = 0; j < n; at_j += ld, ++j) {
179 // const int il = imax(0, j - ku), iu = imin(m, j + kl);
180 // for (int i = il; i < iu; ++i) {
181 // at_j[i*inc] = 100*i + j;
182 // }
183 // }
184 // where kl = w->klat + 1
185 // A sample matrix of bandwidth 2 with 7 elements
186 // generated with the loop above:
187 // const SCALAR good_A_T[] = {
188 // // ku2 ku1 diag kl1 kl2
189 // -55555, -55555, 0, 100, 200,
190 // -55555, 1, 101, 201, 301,
191 // 2, 102, 202, 302, 402,
192 // 103, 203, 303, 403, 503,
193 // 204, 304, 404, 504, 604,
194 // 305, 405, 505, 605, -55555,
195 // 406, 506, 606, -55555, -55555
196 // };
197
198 return SUZERAIN_SUCCESS;
199}

220static int WORK(suzerain_filterop,_operator_boundaries)(
221 WORK(suzerain_filterop,_workspace) *w)
222{
223 int retval = SUZERAIN_SUCCESS;
224 static const int inc = 1; // Column vectors are contiguous in memory
225 const int n = w->n; // Filtering matrices are square
226
227 // Add new boundary-type routines here, when necessary
228 switch (w->b_first) {
229 case SUZERAIN_FILTEROP_BOUNDARY_IGNORE:
230 /* Nothing to be done */
231 break;
232
233 case SUZERAIN_FILTEROP_BOUNDARY_NOFILTER:
234 // Replace near-boundary schemes on B^T
235 {
236 // Banded matrix access has form a[(ku+i)*inc + j*(lda-inc)].
237 // Incorporate the ku offset and decrement ld to speed indexing
238 int ku = w->kubt, ld = w->ldbt;
239 int nbs = imax(w->kuat, w->kubt); // nbs: near-boundary stencils
240 SCALAR * bt_j = w->B_T;
241 bt_j += ku*inc;
242 ld -= inc;
243
244 for (int j = 0; j < nbs; bt_j += ld, ++j) {
245
246 // Start at j - ku, go down to 1
247 for (int joff = w->kubt; joff > 0; --joff) {
248 bt_j[(j-joff)*inc] = 0;
249 }
250 // Diagonal element
251 bt_j[(j )*inc] = 1;
252 // Start at 1, go to j + kl
253 for (int joff = 1; joff < w->klbt+1; ++joff) {
254 bt_j[(j+joff)*inc] = 0;
255 }
256 }
257 }
258
259 // Replace near-boundary schemes on A^T
260 {
261 // Again, access has form a[(ku + i)*inc + j*(lda - inc)].
262 int kl = w->klat, ku = w->kuat, ld = w->ldat;
263 int nbs = imax(w->klat, w->klbt); // nbs: near-boundary stencils
264 SCALAR * at_j = w->A_T + kl; // For factorization-ready data
265 at_j += ku*inc;
266 ld -= inc;
267
268 // The number of near-boundary schemes that need to be replaced
269 // goes by the width of the stencil (widest side), in this case,
270 // that of B_T
271 for (int j = 0; j < nbs; at_j += ld, ++j) {
272
273 // Start at j - ku, go down to 1
274 for (int joff = w->kuat; joff > 0; --joff) {
275 at_j[(j-joff)*inc] = 0;
276 }
277 // Diagonal element
278 at_j[(j )*inc] = 1;
279 // Start at 1, go to j + kl
280 for (int joff = 1; joff < w->klat+1; ++joff) {
281 at_j[(j+joff)*inc] = 0;
282 }
283 }
284 }
285 break;
286
287 case SUZERAIN_FILTEROP_BOUNDARY_SYMMETRY:
288 // Replace near-boundary schemes on B^T
289 {
290 // Banded matrix access has form a[(ku+i)*inc + j*(lda-inc)].
291 // Incorporate the ku offset and decrement ld to speed indexing
292 int ku = w->kubt, ld = w->ldbt;
293 SCALAR * bt_j = w->B_T;
294 bt_j += ku*inc;
295 ld -= inc;
296
297 for (int j = 0; j < w->kubt; bt_j += ld, ++j) {
298
299 // Boundary element is bt_j[0];
300 for (int joff = 1; joff < w->kubt+1-j; ++joff) {
301 // Consider f(-joff) = 2*f(0) - f(joff)
302 bt_j[(0 )*inc] += 2*bt_j[(0-joff)*inc];
303 bt_j[(0+joff)*inc] -= bt_j[(0-joff)*inc];
304 // Reset value at ghost element
305 bt_j[(0-joff)*inc] = 0;
306 }
307 }
308 }
309
310 // Replace near-boundary schemes on A^T
311 {
312 // Again, access has form a[(ku + i)*inc + j*(lda - inc)].
313 int kl = w->klat, ku = w->kuat, ld = w->ldat;
314 SCALAR * at_j = w->A_T + kl; // For factorization-ready data
315 at_j += ku*inc;
316 ld -= inc;
317
318 // The number of near-boundary schemes that need to be replaced
319 // goes in general by the width of the stencil (widest side).
320 // In the symmetry case, however, the only rows that change
321 // in A_T are those with coefficients of A_T at ghost points.
322 for (int j = 0; j < w->kuat; at_j += ld, ++j) {
323
324 // Boundary element is at_j[0];
325 for (int joff = 1; joff < w->kuat+1-j; ++joff) {
326 // Consider f(-joff) = 2*f(0) - f(joff)
327 at_j[(0 )*inc] += 2*at_j[(0-joff)*inc];
328 at_j[(0+joff)*inc] -= at_j[(0-joff)*inc];
329 // Reset value at ghost element
330 at_j[(0-joff)*inc] = 0;
331 }
332 }
333 }
334 break;
335
336 case SUZERAIN_FILTEROP_BOUNDARY_PERIODIC: // TODO
337 default:
338 SUZERAIN_ERROR_NULL("Unknown first boundary treatment",
339 SUZERAIN_ESANITY);
340 }
341
342 switch (w->b_last) {
343 case SUZERAIN_FILTEROP_BOUNDARY_IGNORE:
344 /* Nothing to be done */
345 break;
346
347 case SUZERAIN_FILTEROP_BOUNDARY_NOFILTER:
348 // Replace near-boundary schemes on B^T
349 {
350 // Banded matrix access has form a[(ku+i)*inc + j*(lda-inc)].
351 // Incorporate the ku offset and decrement ld to speed indexing.
352 int ku = w->kubt, ld = w->ldbt;
353 int nbs = imax(w->klat, w->klbt); // nbs: near-boundary stencils
354 SCALAR * bt_j = w->B_T;
355
356 // Walk the upper end forwards:
357 // Point initially to element (n-nbs) in the diagonal
358 bt_j += (ku*inc + (n-nbs)*(w->ldbt)*inc);
359 ld -= inc;
360
361 // Walk the matrix forwards
362 for (int j = 0; j < nbs; bt_j += ld, ++j) {
363
364 // Start at j - ku, go down to 1
365 for (int joff = w->kubt; joff > 0; --joff) {
366 bt_j[(j-joff)*inc] = 0;
367 }
368 // Diagonal element
369 bt_j[(j )*inc] = 1;
370 // Start at 1, go to j + kl
371 for (int joff = 1; joff < w->klbt+1; ++joff) {
372 bt_j[(j+joff)*inc] = 0;
373 }
374 }
375
376 // Note: To walk the upper end backwards instead, the initial
377 // bt_j offset points initially to the last diagonal element,
378 // as
379 // bt_j += (ku*inc + (n-1)*(w->ldbt)*inc);
380 // and the outer loop changes to
381 // for (int j = 0; j > -nbs; bt_j -= ld, --j) { ... }
382
383 }
384
385 // Replace near-boundary schemes on A^T
386 {
387 // Again, access has form a[(ku + i)*inc + j*(lda - inc)].
388 int kl = w->klat, ku = w->kuat, ld = w->ldat;
389 int nbs = imax(w->klat, w->klbt); // nbs: near-boundary stencils
390 SCALAR * at_j = w->A_T + kl; // For factorization-ready data
391
392 // Walk the upper end forwards:
393 // Point initially to element (n-nbs) in the diagonal
394 at_j += (ku*inc + (n-nbs)*(w->ldat)*inc);
395 ld -= inc;
396
397 // Walk the matrix forwards
398 for (int j = 0; j < nbs; at_j += ld, ++j) {
399
400 // Start at j - ku, go down to 1
401 for (int joff = w->kuat; joff > 0; --joff) {
402 at_j[(j-joff)*inc] = 0;
403 }
404 // Diagonal element
405 at_j[(j )*inc] = 1;
406 // Start at 1, go to j + kl
407 for (int joff = 1; joff < w->klat+1; ++joff) {
408 at_j[(j+joff)*inc] = 0;
409 }
410 }
411
412 // Note: To walk the upper end backwards instead, the initial
413 // at_j offset points initially to the last diagonal element,
414 // as
415 // at_j += (ku*inc + (n-1)*(w->ldat)*inc);
416 // and the outer loop changes to
417 // for (int j = 0; j > -nbs; at_j -= ld, --j) { ... }
418 }
419 break;
420
421 case SUZERAIN_FILTEROP_BOUNDARY_SYMMETRY:
422 // Replace near-boundary schemes on B^T
423 {
424 // Banded matrix access has form a[(ku+i)*inc + j*(lda-inc)].
425 // Incorporate the ku offset and decrement ld to speed indexing
426 int ku = w->kubt, ld = w->ldbt;
427 SCALAR * bt_j = w->B_T;
428
429 // Walk the upper end backwards
430 // Point initially to element (n-1) in the diagonal
431 bt_j += (ku*inc + (n-1)*(w->ldbt)*inc);
432 ld -= inc;
433
434 for (int j = 0; j > -w->klbt; bt_j -= ld, --j) {
435
436 // Boundary element is bt_j[0]
437 for (int joff = 1; joff < (w->klbt)+1+j; ++joff) {
438 // Consider f(n-1+joff) = 2*f(n-1) - f(n-1-joff)
439 bt_j[(0 )*inc] += 2*bt_j[(0+joff)*inc];
440 bt_j[(0-joff)*inc] -= bt_j[(0+joff)*inc];
441 // Reset value at ghost element
442 bt_j[(0+joff)*inc] = 0;
443 }
444 }
445 }
446
447
448 // Replace near-boundary schemes on A^T
449 {
450 // Again, access has form a[(ku + i)*inc + j*(lda - inc)].
451 int kl = w->klat, ku = w->kuat, ld = w->ldat;
452 SCALAR * at_j = w->A_T + kl; // For factorization-ready data
453
454 // Walk the upper end backwards
455 // Point initially to element (n-1) in the diagonal
456 at_j += (ku*inc + (n-1)*(w->ldat)*inc);
457 ld -= inc;
458
459 // The number of near-boundary schemes that need to be replaced
460 // goes in general by the width of the stencil (widest side).
461 // In the symmetry case, however, the only rows that change
462 // in A_T are those with coefficients of A_T at ghost points.
463 for (int j = 0; j > -(w->klat); at_j -= ld, --j) {
464
465 // Boundary element is at_j[0];
466 for (int joff = 1; joff < (w->klat)+1+j; ++joff) {
467 // Consider f(n-1+joff) = 2*f(n-1) - f(n-1-joff)
468 at_j[(0 )*inc] += 2*at_j[(0+joff)*inc];
469 at_j[(0-joff)*inc] -= at_j[(0+joff)*inc];
470 // Reset value at ghost element
471 at_j[(0+joff)*inc] = 0;
472 }
473 }
474 }
475 break;
476
477 case SUZERAIN_FILTEROP_BOUNDARY_PERIODIC: // TODO
478 default:
479 SUZERAIN_ERROR_NULL("Unknown last boundary treatment",
480 SUZERAIN_ESANITY);
481 }
482
483 return retval;
484}

486WORK(suzerain_filterop,_workspace) *
487WORK(suzerain_filterop,_alloc)(
488 const int n,
489 const enum suzerain_filterop_method method,
490 const SCALAR *method_params,
491 const enum suzerain_filterop_boundary_treatment b_first,
492 const enum suzerain_filterop_boundary_treatment b_last)
493{
494 /* Parameter sanity checks */
495 if (n < 1) {
496 SUZERAIN_ERROR_NULL("n must be strictly positive", SUZERAIN_EINVAL);
497 }
498
499 /* Compute bandwidth-related information before allocating workspace */
500 int klat, kuat, ldat;
501 int klbt, kubt, ldbt;
502 if (SUZERAIN_SUCCESS != WORK(suzerain_filterop,_operator_bandwidths)(method,
503 method_params, &klat, &kuat, &ldat, &klbt, &kubt, &ldbt)) {
504 SUZERAIN_ERROR_NULL("Unable to compute bandwidths for method",
505 SUZERAIN_EFAILED);
506 }
507
508 /* Allocate space for the workspace struct and initialize to zero */
509 WORK(suzerain_filterop,_workspace) * const w
510 = calloc(1, sizeof(WORK(suzerain_filterop,_workspace)));
511 if (w == NULL) {
512 SUZERAIN_ERROR_NULL("failed to allocate space for workspace",
513 SUZERAIN_ENOMEM);
514 }
515
516 /* Populate initial workspace values */
517 w->method = method;
518 w->b_first = b_first;
519 w->b_last = b_last;
520 w->n = n;
521 w->klat = klat;
522 w->kuat = kuat;
523 w->ldat = ldat;
524 w->klbt = klbt;
525 w->kubt = kubt;
526 w->ldbt = ldbt;
527
528 /* Allocate one contiguous block for B and A, in that order */
529 w->B_T = suzerain_blas_calloc((w->n * w->ldbt) + (w->n * w->ldat),
530 sizeof(w->B_T[0]));
531 if (w->B_T == NULL) {
532 WORK(suzerain_filterop,_free)(w);
533 SUZERAIN_ERROR_NULL("failed to allocate space for B_T and A_T",
534 SUZERAIN_ENOMEM);
535 }
536 w->A_T = w->B_T + (w->n * w->ldbt);
537
538 /* Allocate space for the pivot vector for A_T's factorization */
539 w->ipiva = suzerain_blas_calloc(w->n, sizeof(w->ipiva[0]));
540 if (w->ipiva == NULL) {
541 WORK(suzerain_filterop,_free)(w);
542 SUZERAIN_ERROR_NULL("failed to allocate space for ipiva",
543 SUZERAIN_ENOMEM);
544 }
545
546 /* Initialize A_T and B_T depending on the method */
547 if (SUZERAIN_SUCCESS != WORK(suzerain_filterop,_operator_assemble)(
548 method_params, w)) {
549 WORK(suzerain_filterop,_free)(w);
550 SUZERAIN_ERROR_NULL("failed to assemble A_T and B_T",
551 SUZERAIN_EFAILED);
552 }
553
554 /* Apply chosen boundary treatment */
555 if (SUZERAIN_SUCCESS != WORK(suzerain_filterop,_operator_boundaries)(w)) {
556 WORK(suzerain_filterop,_free)(w);
557 SUZERAIN_ERROR_NULL("failed to modify A_T and B_T for boundaries",
558 SUZERAIN_EFAILED);
559 }
560
561 return w;
562}

564void
565WORK(suzerain_filterop,_free)(
566 WORK(suzerain_filterop,_workspace) *w)
567{
568 if (w) {
569 // B_T and A_T were allocated as a single block, so just one free
570 w->A_T = NULL;
571 suzerain_blas_free(w->B_T);
572 w->B_T = NULL;
573 suzerain_blas_free(w->ipiva);
574 w->ipiva = NULL;
575 }
576 free(w);
577}

579int
580WORK(suzerain_filterop,_factorize)(
581 WORK(suzerain_filterop,_workspace) *w)
582{
583 /* Protect the user from incorrect API usage */
584 if (w->ipiva[0] > 0) {
585 SUZERAIN_ERROR("Unable to factor an already-factored operator",
586 SUZERAIN_EINVAL);
587 }
588
589 /* Compute in-place LU factorization of the operator */
590 const int info = BLAS(suzerain_lapack_,gbtrf)(w->n, w->n, w->klat, w->kuat,
591 w->A_T, w->ldat, w->ipiva);
592 if (info) {
593 char buffer[80];
594 snprintf(buffer, sizeof(buffer),
595 STRINGIFY(BLAS(suzerain_lapack_,gbtrf))
596 " reported error %d", info);
597 SUZERAIN_ERROR(buffer, SUZERAIN_ESANITY);
598 }
599
600 /* Factorization overwrote w->ipiv[0] with something nonnegative;
601 * This is our flag to check that factorization indeed occurred */
602
603 return SUZERAIN_SUCCESS;
604}

622int
623WORK(suzerain_filterop,_solve)(
624 const int nrhs,
625 SCALAR *X,
626 const int ldx,
627 const WORK(suzerain_filterop,_workspace) * w)
628{
629 // TODO Periodic boundary treatments require additional processing
630

632 SUZERAIN_ERROR(STRINGIFY(WORK(suzerain_filterop,_factorize))
