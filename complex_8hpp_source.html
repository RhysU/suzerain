488template<class Beta, class Complex, class Alpha, class Source>
489SUZERAIN_FORCEINLINE
490typename boost::enable_if<boost::mpl::and_<
491 traits::is_complex<Beta>,
492 boost::mpl::not_<traits::is_complex<Alpha> >
493> >::type accumulate_complex(const Beta &SUZERAIN_RESTRICT beta,
494 Complex &SUZERAIN_RESTRICT dest,
495 const Alpha &SUZERAIN_RESTRICT alpha,
496 const Source &SUZERAIN_RESTRICT src)
497{
498 typedef typename traits::real<Complex>::type real_type;
499 const real_type odest_real = (real)(dest);
500 const real_type odest_imag = (imag)(dest);
501
502 (real)(dest) = (real)(beta)*odest_real - (imag)(beta)*odest_imag
503 + alpha*(real)(src);
504 (imag)(dest) = (real)(beta)*odest_imag + (imag)(beta)*odest_real
505 + alpha*(imag)(src);
506}

516template<class Beta, class Complex, class Alpha, class Source>
517SUZERAIN_FORCEINLINE
518typename boost::enable_if<boost::mpl::and_<
519 boost::mpl::not_<traits::is_complex<Beta> >,
520 traits::is_complex<Alpha>
521> >::type accumulate_complex(const Beta &SUZERAIN_RESTRICT beta,
522 Complex &SUZERAIN_RESTRICT dest,
523 const Alpha &SUZERAIN_RESTRICT alpha,
524 const Source &SUZERAIN_RESTRICT src)
525{
526 (real)(dest) = beta*(real)(dest)
527 + (real)(alpha)*(real)(src) - (imag)(alpha)*(imag)(src);
528 (imag)(dest) = beta*(imag)(dest)
529 + (real)(alpha)*(imag)(src) + (imag)(alpha)*(real)(src);
530}

540template<class Beta, class Complex, class Alpha, class Source>
541SUZERAIN_FORCEINLINE
542typename boost::enable_if<boost::mpl::and_<
543 traits::is_complex<Beta>,
544 traits::is_complex<Alpha>
545> >::type accumulate_complex(const Beta &SUZERAIN_RESTRICT beta,
546 Complex &SUZERAIN_RESTRICT dest,
547 const Alpha &SUZERAIN_RESTRICT alpha,
548 const Source &SUZERAIN_RESTRICT src)
549{
550 typedef typename traits::real<Complex>::type real_type;
551 const real_type odest_real = (real)(dest);
552 const real_type odest_imag = (imag)(dest);
553

557 + (real)(alpha)*(imag)(src) + (imag)(alpha)*(real)(src);
558}

604template<class Complex1, class Complex2, typename FPT>
605SUZERAIN_FORCEINLINE
606void assign_complex_scaled_ipower(Complex1 &SUZERAIN_RESTRICT dest,
607 const Complex2 &SUZERAIN_RESTRICT src,
608 const FPT alpha,
609 const int ipower)
610{
611 switch (ipower & 3) { // Modulo-four-like operation for 2s complement
612 case 3: // I^3 = -I = I^-1
613 (real)(dest) = alpha*real_ipower<3>(src);
614 (imag)(dest) = alpha*imag_ipower<3>(src);
615 break;
616 case 2: // I^2 = -1 = I^-2
617 (real)(dest) = alpha*real_ipower<2>(src);
618 (imag)(dest) = alpha*imag_ipower<2>(src);
619 break;
620 case 1: // I^1 = I = I^-3
621 (real)(dest) = alpha*real_ipower<1>(src);
622 (imag)(dest) = alpha*imag_ipower<1>(src);
623 break;
624 case 0: // I^0 = 1
625 (real)(dest) = alpha*real_ipower<0>(src);
626 (imag)(dest) = alpha*imag_ipower<0>(src);
627 break;
628 }
629}
