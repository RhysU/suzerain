36rigor rigor_from(const char *name)
37{
38 using std::tolower;
39 if (name) {
40 switch (tolower(name[0])) {
41 default: break;
42 case 'm': return measure;
43 case 'p': return patient;
44 case 'e':
45 switch (tolower(name[1])) {
46 default: // Assume estimate
47 case 's': return estimate;
48 case 'x': return exhaustive;
49 }
50 case 'w': return wisdom_only;
51 }
52 }
53
54 return measure;
55}
56
57rigor rigor_from(const unsigned flags)
58{
59 if (flags & FFTW_WISDOM_ONLY) {
60 return wisdom_only;
61 } else if (flags & FFTW_EXHAUSTIVE) {
62 return exhaustive;
63 } else if (flags & FFTW_PATIENT) {
64 return patient;
65 } else if (flags & FFTW_ESTIMATE) {
66 return estimate;
67 } else {
68 return measure; // Default
69 }
70}
