41banner "Checking zero-zero and Nyquist wavenumbers are strictly real-valued"

45 # Build a single h5dump invocation dumping the relevant wavenumbers

48 [ "$((Nx%2))" -eq 0 ] && realmodes+=("0,$((Nx/2)),0")
49 [ "$((Nz%2))" -eq 0 ] && realmodes+=("$((Nz/2)),0,0")
50 [ "$((Nx%2))" -eq 0 -a "$((Nz%2))" ] && realmodes+=("$((Nz/2)),$((Nx/2)),0")
51 for field in rho rho_u rho_v rho_w rho_E

53 for realmode in "${realmodes[@]}"

56 done
57 done
58 cmd+=(-w 8 -m %22.14g mms0.h5)
59 echo ${cmd[*]}

63 # awk program keeping only the h5dump lines between 'DATA {' and '}'
64 declare -r awk_h5dump_dataonly=' \
65 $0 ~ /^[[:space:]]*DATA[[:space:]]*{[[:space:]]*$/ {keep = 1; next} \
66 $0 ~ /^[[:space:]]*}[[:space:]]*$/ {keep = 0} \
67 keep == 1 { print $0 }'

69 # Ensure that the entire fourth column (imag component) is identically zero
70 awk $awk_h5dump_dataonly realmodes | awk '$4 != 0.0 { exit 1 }' \
