49std::vector<field_extrema_xz>
50compute_field_extrema_xz(
51 contiguous_state<4,complex_t>& swave,
52 const specification_grid& grid,
53 const pencil_grid& dgrid,
54 const bsplineop& cop)
55{
56 // State enters as coefficients in X, Y, and Z stored swave[F][Y][X][Z]
57
58 // TODO Consider using std::vector for auto-zeroed buffer storage
59 // Contiguous temporary storage for accumulating/broadcasting results
60 // Each fast columns stores one wall-normal swaths of data for a scalar
61 const int ny = boost::numeric_cast<int>(swave.shape()[1]);
62 const int nf = boost::numeric_cast<int>(swave.shape()[0]);
63 const int buf2size = ny * 2*nf;
64 std::unique_ptr<val_index[]> buf2(new val_index[buf2size]);
65 // Storage for negatives fraction
66 const int bufnsize = ny * nf;
67 std::unique_ptr<real_t[]> bufn(new real_t [bufnsize]);
68
69 // Initialize storage with "uninteresting" min and max values
70 // so that the final Allreduce ignores contributions from
71 // ranks that have no business contributing to certain y_j.
72 // extrema
73 for (int i = 0; i < buf2size/2; ++i) {
74 buf2[i].val = +std::numeric_limits<real_t>::infinity();
75 buf2[i].index = 0;
76 }
77 for (int i = buf2size/2; i < buf2size; ++i) {
78 buf2[i].val = -std::numeric_limits<real_t>::infinity();
79 buf2[i].index = 0;
80 }
81 // negatives
82 for (int i = 0; i < bufnsize; ++i) {
83 bufn[i] = 0.;
84 }
85
86 // For each field...
87 operator_tools o(grid, dgrid, cop);
88 physical_view<> sphys(dgrid, swave);
89 const int dNx = dgrid.global_physical_extent.x();
90 const int dNz = dgrid.global_physical_extent.z();
91 for (int f = 0; f < boost::numeric_cast<int>(swave.shape()[0]); ++f) {
92
93 // Transform to physical space (dealiasing along the way)...
94 o.zero_dealiasing_modes(swave, f);
95 o.bop_apply(0, 1.0, swave, f);
96 dgrid.transform_wave_to_physical(&sphys.coeffRef(f,0));
97
98 // Loop over Y...
99 for (int offset = 0, j = dgrid.local_physical_start.y();
100 j < dgrid.local_physical_end.y();
101 ++j) {
102
103 // Loop over X and Z...
104 for (int k = o.dgrid.local_physical_start.z();
105 k < o.dgrid.local_physical_end.z();
106 ++k) {
107
108 for (int i = o.dgrid.local_physical_start.x();
109 i < o.dgrid.local_physical_end.x();
110 ++i, /* NB */ ++offset) {
111
112 int xz_global_offset = i + k * dNx;
113
114 // Retrieve scalar-valued state at point...
115 const real_t val = sphys(f, offset);
116
117 // ...and tracking minimum value at this wall-normal spot
118 // being careful to preserve NaNs during local reduction.
119 // On ties, prefer the smallest global index for
120 // deterministic results across MPI decompositions.
121
122 // minimum value
123 if (val < buf2[j+f*ny].val
124 || (val == buf2[j+f*ny].val
125 && xz_global_offset < buf2[j+f*ny].index)) {
126 buf2[j+f*ny].val = val;
127 buf2[j+f*ny].index = xz_global_offset;
128 }
129
130 // maximum value
131 if (val > buf2[j+(f+nf)*ny].val
132 || (val == buf2[j+(f+nf)*ny].val
133 && xz_global_offset < buf2[j+(f+nf)*ny].index)) {
134 buf2[j+(f+nf)*ny].val = val;
135 buf2[j+(f+nf)*ny].index = xz_global_offset;
136 }
137
138 // count of negative values
139 if (val < 0.) {
140 bufn[j+f*ny] += 1.;
141 }
142
143 }
144
145 }
146
147 }
148
149 }
150
151 // Allreduce the information in a single operation playing a (-max) trick
152 for (int i = buf2size/2; i < buf2size; ++i) {
153 buf2[i].val *= -1;
154 }
155 SUZERAIN_MPICHKR(MPI_Allreduce(MPI_IN_PLACE, &buf2[0], buf2size,
156 MPI_DOUBLE_INT,
157 MPI_MINLOC, MPI_COMM_WORLD));
158 for (int i = buf2size/2; i < buf2size; ++i) {
159 buf2[i].val *= -1;
160 }
161
162 // Allreduce the negatives and compute fraction of points
163 SUZERAIN_MPICHKR(MPI_Allreduce(MPI_IN_PLACE, &bufn[0], bufnsize,
164 mpi::datatype_of(bufn[0]),
165 MPI_SUM, MPI_COMM_WORLD));
166 for (int i = 0; i < bufnsize; ++i) {
167 bufn[i] = bufn[i] / (dNx * dNz * ny);
168 }
169
170 // Copy the wall-normal profiles into the returned vector...
171 std::vector<field_extrema_xz> retval(swave.shape()[0]);
172 for (int f = 0; f < nf; ++f) {
173 retval[f].min.resize(ny);
174 retval[f].max.resize(ny);
175 retval[f].imin.resize(ny);
176 retval[f].imax.resize(ny);
177 retval[f].kmin.resize(ny);
178 retval[f].kmax.resize(ny);
179 retval[f].fneg.resize(ny);
180
181 for (int j = 0; j < ny; ++j) {
182 // value
183 retval[f].min(j) = buf2[ f *ny + j].val;
184 retval[f].max(j) = buf2[(nf+f)*ny + j].val;
185
186 // x-index = xz_global % dNx
187 retval[f].imin(j) = buf2[ f *ny + j].index % dNx;
188 retval[f].imax(j) = buf2[(nf+f)*ny + j].index % dNx;
189
190 // z-index = (xz_global - i-index) / dNx
191 retval[f].kmin(j) = (buf2[ f *ny + j].index
192 - retval[f].imin(j)) / dNx;
193 retval[f].kmax(j) = (buf2[(nf+f)*ny + j].index
194 - retval[f].imax(j)) / dNx;
195
196 // negatives
197 retval[f].fneg(j) = bufn[ f *ny + j];
198 }
199 }
200
201 // ...and hope the compiler is smart enough to elide the copy.
202 return retval;
203}

205std::vector<field_extrema_xyz>
206compute_field_extrema_xyz(
207 contiguous_state<4,complex_t>& swave,
208 const specification_grid& grid,
209 const pencil_grid& dgrid,
210 const bsplineop& cop)
211{
212 // State enters as coefficients in X, Y, and Z stored swave[F][Y][X][Z]
213
214 // TODO Rather than delegating, it may be faster to inline and modify
215 std::vector<field_extrema_xz> wallnormal
216 = compute_field_extrema_xz(swave, grid, dgrid, cop);
217
218 // Reduce the wall-normal profiles down to a single value
219 std::vector<field_extrema_xyz> retval(wallnormal.size());
220 int ny = boost::numeric_cast<int>(swave.shape()[1]);
221 for (size_t f = 0; f < swave.shape()[0]; ++f) {
222 retval[f].min = +std::numeric_limits<real_t>::infinity();
223 retval[f].max = -std::numeric_limits<real_t>::infinity();
224 retval[f].fneg = 0.;
225
226 for (int j = 0; j < ny; ++j) {
227
228 // minimum value (on ties, prefer smallest j)
229 if (wallnormal[f].min(j) < retval[f].min) {
230 retval[f].min = wallnormal[f].min(j);
231 retval[f].imin = wallnormal[f].imin(j);
232 retval[f].jmin = j;
233 retval[f].kmin = wallnormal[f].kmin(j);
234 }
235
236 // maximum value (on ties, prefer smallest j)
237 if (wallnormal[f].max(j) > retval[f].max) {
238 retval[f].max = wallnormal[f].max(j);
239 retval[f].imax = wallnormal[f].imax(j);
240 retval[f].jmax = j;
241 retval[f].kmax = wallnormal[f].kmax(j);
242 }
243
244 // negatives
245 retval[f].fneg += wallnormal[f].fneg(j);
246
247 }
248 }
249
250 return retval;
251}
