36bsmbsm_solver*
37bsmbsm_solver::build(
38 const suzerain_bsmbsm& bsmbsm,
39 const specification_zgbsv& spec,
40 const int nrhs)
41{
42 switch (spec.method()) {
43
44 case specification_zgbsv::zgbsv:
45 return new bsmbsm_solver_zgbsv(bsmbsm, spec, nrhs);
46
47 case specification_zgbsv::zgbsvx:
48 return new bsmbsm_solver_zgbsvx(bsmbsm, spec, nrhs);
49
50 case specification_zgbsv::zcgbsvx:
51 return new bsmbsm_solver_zcgbsvx(bsmbsm, spec, nrhs);
52
53 }
54
55 throw std::invalid_argument("bsmbsm_solver::build: unknown spec.method()");
56}

60 const specification_zgbsv& spec,
61 const int nrhs)

66 , PAPT(LU.data() + KL, LU.rows() - KL, LU.cols(), LU.colStride()) // Alias

76 ipiv.setConstant(-12345);
77#endif
78}

92bsmbsm_solver::apprx(const bool acceptable)
93{
94 const bool old = apprx_;
95 if (spec.reuse()) {
96 apprx_ = apprx_ && acceptable;
97 if (!apprx_) {
98 fact_ = default_fact();

114int
115bsmbsm_solver::solve_internal(const char trans,
116 const int nrhs)
117{
118 if (SUZERAIN_UNLIKELY(nrhs < 1 || nrhs > PB.cols()))
119 throw std::invalid_argument("Invalid nrhs supplied to solve()");
120 const int info = solve_hook(trans, nrhs); // Invoke subclass-specific hook
121 if (info == 0) return info; // Eagerly return on success
122
123 // Otherwise, loudly report any errors that occurred during the solve
124 char buffer[128];
125 if (info < 0) {
126 snprintf(buffer, sizeof(buffer),
127 "%s reported error in argument %d",
128 spec.mname(), -info);
129 SUZERAIN_ERROR(buffer, SUZERAIN_ESANITY);
130 } else if (info <= N) {
131 snprintf(buffer, sizeof(buffer),
132 "%s reported singularity in PAP^T row %d"
133 " corresponding to A row %d for state scalar %d",
134 spec.mname(), info - 1, suzerain_bsmbsm_q(S, n, info - 1),
135 suzerain_bsmbsm_q(S, n, info - 1) / n);
136 SUZERAIN_ERROR(buffer, SUZERAIN_ESANITY);
137 } else {
138 snprintf(buffer, sizeof(buffer),
139 "%s reported unknown error %d", spec.mname(), info);
140 SUZERAIN_ERROR(buffer, SUZERAIN_ESANITY);
141 }
142}

155int
156bsmbsm_solver_zgbsv::solve_hook(
157 const char trans,
158 const int nrhs)
159{
160 assert(apprx_ == false);
161 // return suzerain_lapackext_zgbsv(&fact_, trans, N, KL, KU, nrhs,
162 // LU.data(), LU.colStride(), ipiv.data(),
163 // PB.data(), PB.colStride());
164
165 // Logic below should be exactly equivalent to above commented out
166 // call to suzerain_lapackext_zgbsv. Inlined here to enable
167 // separate timing of factorize and backsub steps.
168 int info = 0;
169 if (toupper(fact_) == 'N') {
170 SUZERAIN_TIMER_SCOPED("bsmbsm_solver_zgbsv::suzerain_lapack_zgbtrf");
171 info = suzerain_lapack_zgbtrf(N, N, KL, KU,
172 LU.data(), LU.colStride(), ipiv.data());
173 fact_ = 'F';
174 }
175 if (!info) {
176 SUZERAIN_TIMER_SCOPED("bsmbsm_solver_zgbsv::suzerain_lapack_zgbtrs");
177 info = suzerain_lapack_zgbtrs(trans, N, KL, KU, nrhs,
178 LU.data(), LU.colStride(), ipiv.data(),
179 PB.data(), PB.colStride());
180 }
181 return info;
182}

186 const specification_zgbsv& spec,
187 const int nrhs)
188 : bsmbsm_solver(bsmbsm, spec, nrhs)
189 , equed_('N') // Default for non-factorized per zgbsvx
190 , rcwork_(N, 3) // Stores (r, c, rwork) x nrhs
191 , rcond_(std::numeric_limits<double>::quiet_NaN())
192 , err_(2, nrhs) // Stores (ferr, berr) x nrhs
193 , work_(N, 2) // Per zgbsvx requirements
194 , PAPT_(LD, N) // Operator storage for out-of-place factorization
195 , PX_(N, nrhs) // Solution storage for out-of-place solution
196{
197 if (spec.method() != specification_zgbsv::zgbsvx)
198 throw std::invalid_argument("Invalid method in bsmbsm_solver_zgbsvx");
199
200 // See Eigen "Changing the mapped array" documentation for details
201 assert(spec.in_place() == false);
202 new (&PAPT) PAPT_type(PAPT_.data(), PAPT_.rows(),
203 PAPT_.cols(), PAPT_.colStride());
204 new (&PX) PX_type(PX_.data(), PX_.rows(), PX_.cols());
205
206 // Defensively set NaNs or NaN-like values on debug builds
207#ifndef NDEBUG
208 rcwork_.setConstant(std::numeric_limits<double>::quiet_NaN());
209 err_ .setConstant(std::numeric_limits<double>::quiet_NaN());
210 work_ .setConstant(suzerain::complex::NaN<complex_double>());
211 PAPT_ .setConstant(suzerain::complex::NaN<complex_double>());
212 PX_ .setConstant(suzerain::complex::NaN<complex_double>());
213#endif
214}

252int
253bsmbsm_solver_zgbsvx::solve_hook(
254 const char trans,
255 const int nrhs)
256{
257#ifndef NDEBUG
258 if (fact_ != 'F') { // NaN r, c, and rwork on 'N' or 'E'...
259 rcwork_ .setConstant(std::numeric_limits<double>::quiet_NaN());
260 } else { // ...but maintain r and c on 'F' per ZGBSVX
261 rwork_().setConstant(std::numeric_limits<double>::quiet_NaN());
262 }
263 err_ .setConstant(std::numeric_limits<double>::quiet_NaN());
264 work_.setConstant(suzerain::complex::NaN<complex_double>());
265#endif
266
267 // Perform the requested solve
268 assert(apprx_ == false);
269 const int info = suzerain_lapack_zgbsvx(fact_, trans, N, KL, KU, nrhs,
270 PAPT.data(), PAPT.colStride(), LU.data(), LU.colStride(),
271 ipiv.data(), &equed_, r_().data(), c_().data(), PB.data(),
272 PB.colStride(), PX.data(), PX.colStride(), &rcond_,
273 ferr_().data(), berr_().data(), work_.data(), rwork_().data());
274 fact_ = 'F';
275
276 // Track statistics for each right hand side per stats_type/stats_names
277 double samples[stats_type::static_size] = {
278 (equed_ == 'R' || equed_ == 'B') ? 1.0 : 0.0,
279 (equed_ == 'C' || equed_ == 'B') ? 1.0 : 0.0,
280 (1 / rcond_),
281 /* ferr below */ std::numeric_limits<double>::quiet_NaN(),
282 /* berr below */ std::numeric_limits<double>::quiet_NaN()
283 };
284 for (int j = 0; j < nrhs; ++j) {
285 samples[3] = ferr_()[j];
286 samples[4] = berr_()[j];
287 stats(samples);
288 }
289
290 return info;
291}

343std::vector<std::string>
344bsmbsm_solver_zcgbsvx::summarize_statistics() const
345{
346 std::vector<std::string> retval = bsmbsm_solver::summarize_statistics();
347
348 if (stats.count() > 0) { // Add operational details when solves occurred
349 std::ostringstream msg;
350 msg.precision(std::numeric_limits<double>::digits10/2 + 1);
351 retval.reserve(retval.size() + stats_type::static_size);
352 for (std::size_t i = 0; i < stats_type::static_size; ++i) {
353 // Suppress by-definition uninteresting statistics
354 if (i == 0 && (spec.siter() < 0 || spec.diter() < 0)) continue;
355 if (i == 1 && (spec.siter() < 0 || spec.diter() < 0)) continue;
356 if (i == 2 && !spec.reuse() ) continue;
357#pragma warning(push,disable:1572)
358 if (i == 3 && spec.tolsc() == 0) continue; // Implies afrob == -1
359#pragma warning(pop)
360 if (i == 4 && spec.siter() < 0) continue;
361 if (i == 5 && spec.diter() < 0) continue;
362 msg.str("");
363 msg << "Min/avg/max/std of "
364 << stats_names[i]
365 << ": "
366 << stats.min(i) << ", "
367 << stats.avg(i) << ", "
368 << stats.max(i) << ", "
369 << stats.std(i);
370 retval.push_back(msg.str());
371 }
372 }
373
374 return retval;
375}

377int
378bsmbsm_solver_zcgbsvx::solve_hook(
379 const char trans,
380 const int nrhs)
381{
382#ifndef NDEBUG
383 iter_ .setConstant(-12345);
384 tolscres_.setConstant(std::numeric_limits<double>::quiet_NaN());
385 r_ .setConstant(suzerain::complex::NaN<complex_double>());
386#endif
387
388 // Perform the requested solve processing each right hand side in turn
389 int info = 0, j = -1;
390 while (!info && ++j < nrhs) {
391
392 // Reset specification-related constants for each right hand side
393 siter_()[j] = spec.siter();
394 diter_()[j] = spec.diter();
395 tolsc_()[j] = spec.tolsc();
396 info = suzerain_lapackext_zcgbsvx(&fact_, &apprx_, spec.aiter(), trans,
397 N, KL, KU, PAPT.data(), &afrob_,
398 LU.data(), ipiv.data(),
399 PB.data(), PX.data(),
400 &siter_()[j], &diter_()[j],
401 &tolsc_()[j], r_.data(), &res_()[j]);
402
403 // Track statistics for each right hand side per stats_type/stats_names
404 double samples[stats_type::static_size] = {
405 (fact_ == 'S') ? 1.0 : 0.0, (fact_ == 'D') ? 1.0 : 0.0,
406 static_cast<double>(apprx_), afrob_,
407 static_cast<double>(siter_()[j]), static_cast<double>(diter_()[j]),
408 tolsc_()[j], res_()[j]
409 };
410 stats(samples);
411 }
412
413 return info;
414}
