135instantaneous&
136instantaneous::copy_from(
137 const bsplineop& cop,
138 const samples& that)
139{
140 SUZERAIN_ENSURE(cop.get()->method == SUZERAIN_BSPLINEOP_COLLOCATION_GREVILLE);
141
142 // Ensure adequate storage for the assignment
143 // FIXME Usage of q::count brittle wrt subclasses-- NoChange preferable
144#ifndef NDEBUG
145 setConstant(that.storage.rows(), q::count,
146 std::numeric_limits<Scalar>::quiet_NaN());
147#else
148 resize (that.storage.rows(), q::count);
149#endif
150
151 // Convert data, quantity-by-quantity, from coefficients to point values
152 // Could perform multiple matvecs here in one shot , but why be error prone?
153 // Additionally, in-place matvec operations require an auxiliary buffer,
154 // so we wouldn't be saving much data movement in an amortizing operation.
155 cop.accumulate(0, 1.0, that.rho() .col(0).data(), 1, 0.0, this->rho() .data(), 1);
156 cop.accumulate(0, 1.0, that.p() .col(0).data(), 1, 0.0, this->p() .data(), 1);
157 cop.accumulate(0, 1.0, that.p2() .col(0).data(), 1, 0.0, this->p2() .data(), 1);
158 cop.accumulate(0, 1.0, that.u() .col(0).data(), 1, 0.0, this->u() .data(), 1);
159 cop.accumulate(0, 1.0, that.u() .col(1).data(), 1, 0.0, this->v() .data(), 1);
160 cop.accumulate(0, 1.0, that.u() .col(2).data(), 1, 0.0, this->w() .data(), 1);
161 cop.accumulate(0, 1.0, that.u_u() .col(0).data(), 1, 0.0, this->uu() .data(), 1);
162 cop.accumulate(0, 1.0, that.u_u() .col(1).data(), 1, 0.0, this->uv() .data(), 1);
163 cop.accumulate(0, 1.0, that.u_u() .col(2).data(), 1, 0.0, this->uw() .data(), 1);
164 cop.accumulate(0, 1.0, that.u_u() .col(3).data(), 1, 0.0, this->vv() .data(), 1);
165 cop.accumulate(0, 1.0, that.u_u() .col(4).data(), 1, 0.0, this->vw() .data(), 1);
166 cop.accumulate(0, 1.0, that.u_u() .col(5).data(), 1, 0.0, this->ww() .data(), 1);
167 cop.accumulate(0, 1.0, that.rho_u() .col(0).data(), 1, 0.0, this->rhou() .data(), 1);
168 cop.accumulate(0, 1.0, that.rho_u() .col(1).data(), 1, 0.0, this->rhov() .data(), 1);
169 cop.accumulate(0, 1.0, that.rho_u() .col(2).data(), 1, 0.0, this->rhow() .data(), 1);
170 cop.accumulate(0, 1.0, that.rho_E() .col(0).data(), 1, 0.0, this->rhoE() .data(), 1);
171 cop.accumulate(0, 1.0, that.rho_u_u().col(0).data(), 1, 0.0, this->rhouu().data(), 1);
172 cop.accumulate(0, 1.0, that.rho_u_u().col(1).data(), 1, 0.0, this->rhouv().data(), 1);
173 cop.accumulate(0, 1.0, that.rho_u_u().col(2).data(), 1, 0.0, this->rhouw().data(), 1);
174 cop.accumulate(0, 1.0, that.rho_u_u().col(3).data(), 1, 0.0, this->rhovv().data(), 1);
175 cop.accumulate(0, 1.0, that.rho_u_u().col(4).data(), 1, 0.0, this->rhovw().data(), 1);
176 cop.accumulate(0, 1.0, that.rho_u_u().col(5).data(), 1, 0.0, this->rhoww().data(), 1);
177 cop.accumulate(0, 1.0, that.rho_E_E().col(0).data(), 1, 0.0, this->rhoEE().data(), 1);
178
179 return *this;
180}
