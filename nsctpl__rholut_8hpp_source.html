38template <typename Scalar>
39template <typename T1, typename T2, typename T3, typename T4>
40Scalar primitive<Scalar>::operator()(T1 x, T2 y, T3 z, T4 t) const
41{
42 using ::std::cos;
43 using ::std::sin;
44 const Scalar twopi_invLx = twopi / *Lx;
45 const Scalar twopi_invLy = twopi / *Ly;
46 const Scalar twopi_invLz = twopi / *Lz;
47
48 return a_0*cos(g_0 + f_0*t) + a_x*cos(c_x + b_x*twopi_invLx*x)*cos(g_x + f_x*t) + a_y*cos(g_y + f_y*t)*cos(c_y + b_y*twopi_invLy*y) + a_z*cos(g_z + f_z*t)*cos(c_z + b_z*twopi_invLz*z) + a_xy*cos(g_xy + f_xy*t)*cos(c_xy + b_xy*twopi_invLx*x)*cos(e_xy + d_xy*twopi_invLy*y) + a_xz*cos(g_xz + f_xz*t)*cos(c_xz + b_xz*twopi_invLx*x)*cos(e_xz + d_xz*twopi_invLz*z) + a_yz*cos(c_yz + b_yz*twopi_invLy*y)*cos(e_yz + d_yz*twopi_invLz*z)*cos(g_yz + f_yz*t);
49}

51template <typename Scalar>
52template <typename T1, typename T2, typename T3, typename T4>
53Scalar primitive<Scalar>::_t(T1 x, T2 y, T3 z, T4 t) const
54{
55 using ::std::cos;
56 using ::std::sin;
57 const Scalar twopi_invLx = twopi / *Lx;
58 const Scalar twopi_invLy = twopi / *Ly;
59 const Scalar twopi_invLz = twopi / *Lz;
60
61 return -a_0*f_0*sin(g_0 + f_0*t) - a_x*f_x*cos(c_x + b_x*twopi_invLx*x)*sin(g_x + f_x*t) - a_y*f_y*cos(c_y + b_y*twopi_invLy*y)*sin(g_y + f_y*t) - a_z*f_z*cos(c_z + b_z*twopi_invLz*z)*sin(g_z + f_z*t) - a_xy*f_xy*cos(c_xy + b_xy*twopi_invLx*x)*cos(e_xy + d_xy*twopi_invLy*y)*sin(g_xy + f_xy*t) - a_xz*f_xz*cos(c_xz + b_xz*twopi_invLx*x)*cos(e_xz + d_xz*twopi_invLz*z)*sin(g_xz + f_xz*t) - a_yz*f_yz*cos(c_yz + b_yz*twopi_invLy*y)*cos(e_yz + d_yz*twopi_invLz*z)*sin(g_yz + f_yz*t);
62}

272NSCTPL_RHOLUT_MANUFACTURED_SOLUTION_TPDECLARATION
273template <typename T1, typename T2, typename T3, typename T4>
274Scalar manufactured_solution<NSCTPL_RHOLUT_MANUFACTURED_SOLUTION_TPNAMES>::e(
275 T1 x, T2 y, T3 z, T4 t) const
276{
277 const Scalar T = this->T(x, y, z, t);
278 const Scalar u = this->u(x, y, z, t);
279 const Scalar v = this->v(x, y, z, t);
280 const Scalar w = this->w(x, y, z, t);
281 const Scalar e = T / (gamma * (gamma - 1))
282 + (Ma * Ma) * (u*u + v*v + w*w ) / 2;
283 return e;
284}

339NSCTPL_RHOLUT_MANUFACTURED_SOLUTION_TPDECLARATION
340template <typename T1, typename T2, typename T3, typename T4>
341Scalar manufactured_solution<NSCTPL_RHOLUT_MANUFACTURED_SOLUTION_TPNAMES>::grad_e(
342 T1 x, T2 y, T3 z, T4 t, int index) const
343{
344 const Scalar u = this->u(x, y, z, t);
345 const Scalar v = this->v(x, y, z, t);
346 const Scalar w = this->w(x, y, z, t);
347 Scalar dT, du, dv, dw;
348 switch (index) {
349 case (IndexBase + 0):
350 dT = this->T._x(x, y, z, t);
351 du = this->u._x(x, y, z, t);
352 dv = this->v._x(x, y, z, t);
353 dw = this->w._x(x, y, z, t);
354 break;
355 case (IndexBase + 1):
356 dT = this->T._y(x, y, z, t);
357 du = this->u._y(x, y, z, t);
358 dv = this->v._y(x, y, z, t);
359 dw = this->w._y(x, y, z, t);
360 break;
361 case (IndexBase + 2):
362 dT = this->T._z(x, y, z, t);
363 du = this->u._z(x, y, z, t);
364 dv = this->v._z(x, y, z, t);
365 dw = this->w._z(x, y, z, t);
366 break;
367 default: return ::std::numeric_limits<Scalar>::signaling_NaN();
368 }
369 const Scalar de = dT / (gamma * (gamma - 1))
370 + (Ma * Ma) * (u*du + v*dv + w*dw);
371 return de;
372}

374NSCTPL_RHOLUT_MANUFACTURED_SOLUTION_TPDECLARATION
375template <typename T1, typename T2, typename T3, typename T4>
376Scalar manufactured_solution<NSCTPL_RHOLUT_MANUFACTURED_SOLUTION_TPNAMES>::grad_p(
377 T1 x, T2 y, T3 z, T4 t, int index) const
378{
379 const Scalar rho = this->rho(x, y, z, t); // shadow
380 const Scalar T = this->T(x, y, z, t);
381 Scalar drho, dT;
382 switch (index) {
383 case (IndexBase + 0):
384 dT = this->T._x(x, y, z, t);
385 drho = this->rho._x(x, y, z, t);
386 break;
387 case (IndexBase + 1):
388 dT = this->T._y(x, y, z, t);
389 drho = this->rho._y(x, y, z, t);
390 break;
391 case (IndexBase + 2):
392 dT = this->T._z(x, y, z, t);
393 drho = this->rho._z(x, y, z, t);
394 break;
395 default: return ::std::numeric_limits<Scalar>::signaling_NaN();
396 }
397 const Scalar dp = (drho * T + rho * dT) / gamma;
398 return dp;
399}
