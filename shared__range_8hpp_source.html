24// IMPLEMENTATION COMMENTS:
25//
26// Olaf van der Spek suggested this concept to boost-users
27// (http://boost.2283326.n4.nabble.com/RFC-A-better-shared-array-td3895154.html)
28// and provided an initial implementation at http://pastebin.com/wkdLVqM1.
29//
30// Per discussion with Peter Dimov in that thread,
31// (std::shared_ptr<T[]>/iterator_range) state was used for more generality instead
32// of the more space efficient (std::shared_ptr<T[]>/size_t) state choice. The
33// assumption is that someone doing lots of lightweight "range work" is
34// explicitly handling resource management while someone doing someone doing
35// lots of lightweight "array work" is similarly explicitly handling lengths.
36//
37// More of the std::shared_ptr<T[]> stored as shared_range.p_ (e.g. use_count, get)
38// was deliberately NOT exposed as I could not reconcile those semantics with a
39// desired to have shared_range truly be an iterator_range (and therefore
40// support construction and assignment from a generic ForwardRange).
41//
42// Full std::shared_ptr-like Allocator support (in particular, simultaneous header
43// and storage allocation) seems to require
44// https://svn.boost.org/trac/boost/ticket/5924 which (small world) Olaf also
45// requested.
46//
47// See http://www.boost.org/doc/libs/release/libs/range/doc/html/range/upgrade.html
48// for version-to-version changes in Boost.Range that impact the
49// implementation. Several places in the implementation rely on <algorithm>
50// rather than Range algorithms to improve version-independence.
51//
52// No enclosing namespace is assumed in the implementation. Hence ::std:: and
53// ::boost:: appears where one might expect std:: and boost::. The
54// implementation therefore can be #included into any namespace.
55
65template< class T >
66class shared_range
67 : public ::boost::iterator_range<T*>
68{
69public: // typedefs
70
72 typedef typename ::boost::iterator_range<T*>::iterator_range iterator_range;
73
75 typedef T element_type;
76
77public: // construction
78
80 shared_range()
81 : iterator_range(),
82 p_()
83 {}
84
86 // See std::shared_ptr(T*) for resource management details.
87 shared_range(T *b, ::std::size_t sz)
88 : iterator_range(b, b + sz),
89 p_(b)
90 {}

93 // See std::shared_ptr(T*,D) for resource management details.
94 template< class D >
95 shared_range(T *b, ::std::size_t sz, D d)

97 p_(b, d)
98 {}
99
101 // See std::shared_ptr(T*) for resource management details of \c b.
102 shared_range(T *b, T *e)
103 : iterator_range(b, e),
104 p_(b)
105 {}
106
108 // See std::shared_ptr(T*, D) for resource management details of \c b.
109 template< class D >
110 shared_range(T *b, T *e, D d)
111 : iterator_range(b, e),
112 p_(b, d)
113 {}
114
116 // See std::shared_ptr(std::shared_ptr const &) for resource management details.
117 shared_range(const ::std::shared_ptr<T[]>& p, ::std::size_t sz)
118 : iterator_range(p.get(), p.get() + sz),
119 p_(p)

153
155 shared_range& operator=(shared_range o) // pass-by-value for elision
156 {

159 }

163 // Provided for compatibility with \c iterator_range.
164 template< class ForwardRange >

170 }

174 // Provided for compatibility with \c iterator_range.
175 template< class ForwardRange >
176 shared_range& operator=( const ForwardRange& r )
177 {
178 this->iterator_range::operator=(r);

183public: // swap, reset
184
186 void swap(shared_range& o)
187 {
188 // It would be nice if iterator_range provided a member swap...
189 ::boost::core::invoke_swap(static_cast<iterator_range&>(*this),
190 static_cast<iterator_range&>(o));
191 p_.swap(o.p_);
192 }

266// To check if \c l and \c r are identical, use <tt>l.equal(r)</tt>.
267template< class T >
268inline bool operator==(const shared_range<T>& l, const shared_range<T>& r)
269{
270 if (l.equal(r)) return true; // Short circuit full range comparison
271
272 typedef typename shared_range<T>::iterator_range iterator_range;
273 return static_cast<const iterator_range&>(l)
274 == static_cast<const iterator_range&>(r);
275}

278// To check if \c l and \c r are not identical, use <tt>!l.equal(r)</tt>.
279template< class T >
280inline bool operator!=(const shared_range<T>& l, const shared_range<T>& r)
281{

304// Preserves allocation size information for use in Allocator::deallocate.
305// Necessary to use a custom allocator in conjunction with std::shared_ptr.
306template<class Allocator>
307struct allocator_deleter : public Allocator
308{
309 typename Allocator::size_type sz;
310
311 allocator_deleter(typename Allocator::size_type sz) : sz(sz) {}
312
313 void operator()(typename Allocator::pointer p)
314 {
315 if (p) {
316 for (typename Allocator::size_type i = 0; i < sz; ++i)
317 Allocator::destroy(p + i);
318
319 Allocator::deallocate(p, sz);
320 }
321 }
322};
323
324}

327template< class A >
328shared_range<typename A::value_type> allocate_shared_range(
329 A a,
330 ::std::size_t sz)
331{
332 typedef typename A::value_type value_type;
333 shared_range<value_type> t(a.allocate(sz), sz, allocator_deleter<A>(sz));
334 ::std::uninitialized_fill(t.begin(), t.end(), value_type());
335 return t;
336}

339// Argument \c val is the copy construction source. Allocator \c a
340// is \em not used to allocate the shared_range details, only the storage
341// used to contain the range.
342template< class A, class V >
343shared_range<typename A::value_type> allocate_shared_range(
344 A a,
345 ::std::size_t sz,
346 const V& val,
347 typename A::const_pointer hint = 0)
348{
349 shared_range<typename A::value_type> t(
350 a.allocate(sz, hint), sz, allocator_deleter<A>(sz));
351 ::std::uninitialized_fill(t.begin(), t.end(), val);
352 return t;
353}

356template< class ForwardRange >
357shared_range<typename ::boost::range_value<ForwardRange>::type>
358clone_shared_range(const ForwardRange &r)
359{
360 // TODO Employ uninitialized_copy without making allocator assumptions?
361 // See http://stackoverflow.com/questions/7822925 for details/concerns.
362 // Would become a single pass operation at the cost of a custom deleter.
363 const ::std::size_t sz = ::boost::distance(r);
364 typedef typename ::boost::range_value<ForwardRange>::type value_type;
365 shared_range<value_type> t = make_shared_range<value_type>(sz);
366 ::std::copy(::boost::begin(r), ::boost::end(r), t.begin());
367 return t;
368}

371// Allocator \c a is \em not used to allocate the shared_range details, only
372// the storage used to contain the range.
373template< class A, class ForwardRange >
374shared_range<typename A::value_type> clone_shared_range(
375 A a, const ForwardRange &r, typename A::const_pointer hint = 0)
376{
377 const typename A::size_type sz = ::boost::distance(r);
378 shared_range<typename A::value_type> t(
379 a.allocate(sz, hint), sz, allocator_deleter<A>(sz));
380 ::std::uninitialized_copy(::boost::begin(r), ::boost::end(r), t.begin());
381 return t;
382}
