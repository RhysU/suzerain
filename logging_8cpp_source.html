103void SubversiveASHEL::addAppenderEvent(const Logger *logger,
104 const Appender *appender)
105{
106 if (!this->target.length())
107 throw RuntimeException("SubversiveASHEL::setTarget() not called");
108
109 // Do nothing if logger is at or below this->target.
110 LoggerPtr ancestor = Logger::getLogger(logger->getName());
111 do {
112 if (0 == this->target.compare(ancestor->getName())) return;
113 } while ((ancestor = ancestor->getParent()));
114
115 // Find the AppenderPtr matching the raw pointer
116 LoggerPtr loggerPtr = Logger::getLogger(logger->getName());
117 AppenderPtr appenderPtr;
118 for (auto& a : loggerPtr->getAllAppenders()) {
119 if (a.get() == appender) {
120 appenderPtr = a;
121 break;
122 }
123 }
124 if (!appenderPtr) return;
125
126 // Suppress adding of appenders outside of this->target's subloggers
127 loggerPtr->removeAppender(appenderPtr);
128
129 // Add appenders intended for RootLogger to this->target
130 if (!loggerPtr->getParent()) {
131 Logger::getLogger(target)->addAppender(appenderPtr);
132 }
133}

169void initialize(MPI_Comm, const char * const default_conf)
170{
171 static const char log4j_config_envvar[] = "log4j.configuration";
172 static const char log4cxx_config_envvar[] = "LOG4CXX_CONFIGURATION";
173
174 // Programmatically enforce execution before log4cxx auto-configuration.
175 if (LogManager::getLoggerRepository()->isConfigured()) {
176 throw std::logic_error(
177 "Apache log4cxx subsystem initialized before logger::initialize()."
178 "\nThis is a CODING ERROR and will drastically hurt scalability.");
179 }
180
181 // Ensure MPI has been initialized
182 {
183 int flag;
184 MPI_Initialized(&flag);
185 assert(flag);
186 if (!flag) throw std::logic_error("MPI_Init has not been invoked");
187 }
188
189 // Obtain our rank within MPI_COMM_WORLD
190 int worldrank;
191 MPI_Comm_rank(MPI_COMM_WORLD, &worldrank);
192
193 // Build a rank-specific name for the "allranks" LoggerPtr instance
194 LogString worldrankname;
195 {
196 std::ostringstream oss;
197 oss << 'r' << worldrank;
198 worldrankname = oss.str();
199 }
200
201 // For ranks > 0 create a HierarchyEventListener managed by the repository
202 if (worldrank > 0) {
203 auto p = std::make_shared<SubversiveASHEL>();
204 p->setTarget(worldrankname);
205 LogManager::getLoggerRepository()->addHierarchyEventListener(p);
206 }
207
208 // Check that we have not accidentally triggered configuration.
209 // Paranoia due to "When the LogManager class is loaded into memory the
210 // default initialization procedure is initiated" in LogManager Doxygen.
211 if (LogManager::getLoggerRepository()->isConfigured()) {
212 throw std::logic_error("log4cxx heisen-configuration detected");
213 }
214
215 // Separate logic for rank zero and higher ranks. Broadcasting of
216 // buflen serves to enforce rank zero logic running first. Only rank
217 // zero hits the filesystem searching for configuration files.
218 if (worldrank == 0) {
219
220 std::string configpath;
221 Pool pool;
222 File file;
223
224 // Lookup logic for configpath based on log4cxx source:
225 // 1) Check environment variable LOG4CXX_CONFIGURATION
226 // 2) Check environment variable log4j.configuration
227 // 3) Check existence of log4cxx.properties
228 // 4) Check existence of log4j.properties
229 // See log4cxx/trunk/src/main/cpp/defaultconfigurator.cpp. Notice that
230 // XML-based files are skipped as log4cxx seems to have no in-memory
231 // DOMConfigurator::doConfigure(...) functionality and hitting disk
232 // in any fashion slows down our startup for large rank counts.
233 if (getenv(log4cxx_config_envvar)) {
234 configpath = getenv(log4cxx_config_envvar);
235 }
236 if (configpath.empty() && getenv(log4j_config_envvar)) {
237 configpath = getenv(log4j_config_envvar);
238 }
239 if (configpath.empty()) {
240 if (file.setPath("log4cxx.properties").exists(pool)) {
241 configpath = "log4cxx.properties";
242 } else if (file.setPath("log4j.properties").exists(pool)) {
243 configpath = "log4j.properties";
244 }
245 }
246
247 // Overwrite default configuration using file contents, if possible.
248 std::vector<unsigned char> buf;
249 if (default_conf) {
250 buf.assign(default_conf, default_conf + strlen(default_conf));
251 }
252 if (!configpath.empty() && file.setPath(configpath).exists(pool)) {
253 // Hideous ifdef to workaround log4cxx's inconsistent char usage.
254 buf.resize(file.length(pool));
255 FileInputStream fis(file);
256#if CHAR_MIN == 0 /* "char" denotes "unsigned char" */
257 ByteBuffer bb(reinterpret_cast<char *>(&buf.front()), buf.size());
258 fis.read(bb);
259#else /* "char" denotes "signed char" */
260 std::vector<char> tmp(buf.size());
261 ByteBuffer bb(&tmp.front(), tmp.size());
262 fis.read(bb);
263 // Not the least bit ASCII 128+ safe but throws if issue arises
264 // (Boost-less std::transform + boost::numeric::converter).
265 std::vector<char>::const_iterator first = tmp.begin();
266 std::vector<unsigned char>::iterator result = buf.begin();
267 while (first != tmp.end()) {
268 if (*first < 0) throw std::runtime_error(
269 "Logging configuration file must be ASCII 128");
270 *result++ = static_cast<unsigned char>(*first++);
271 }
272#endif
273 }
274
275 // Broadcast configuration buffer length
276 int buflen = buf.size();
277 MPI_Bcast(&buflen, 1, MPI_INT, 0, MPI_COMM_WORLD);
278 if (buflen) {
279
280 // Configure logging subsystem using the nontrivial buffer contents
281 InputStreamPtr bais(new ByteArrayInputStream(buf));
282 Properties props;
283 PropertyConfigurator pc;
284 try {
285 props.load(bais);
286 pc.doConfigure(props, LogManager::getLoggerRepository());
287 } catch (Exception &e) {
288 BasicConfigurator::configure(); // Fallback
289 buf.push_back('\0');
290 LOG4CXX_WARN(Logger::getRootLogger(),
291 "Logging configuration from "
292 << (configpath.size() ? configpath : "default_conf")
293 << "invalid: " << e.what() << "\n" << &buf.front());
294 buf.pop_back();
295 }
296
297 // Broadcast configuration buffer now that rank zero has finished
298 MPI_Bcast(&buf.front(), buflen, MPI_UNSIGNED_CHAR, 0, MPI_COMM_WORLD);
299
300 } else {
301 // No buffer contents so use a vanilla log4cxx configuration
302 BasicConfigurator::configure();
303 }
304
305 } else {
306
307 // Higher ranks receive buffer length
308 int buflen;
309 MPI_Bcast(&buflen, 1, MPI_INT, 0, MPI_COMM_WORLD);
310 if (buflen) {
311
312 // Configure logging subsystem using incoming buffer contents
313 std::vector<unsigned char> buf(buflen);
314 MPI_Bcast(&buf.front(), buflen, MPI_UNSIGNED_CHAR, 0, MPI_COMM_WORLD);
315 InputStreamPtr bais(new ByteArrayInputStream(buf));
316 Properties props;
317 PropertyConfigurator pc;
318 try {
319 props.load(bais);
320 pc.doConfigure(props, LogManager::getLoggerRepository());
321 } catch (Exception & /* ignored */) {
322 BasicConfigurator::configure(); // Fallback
323 }
324
325 } else {
326 // No buffer contents so use a vanilla log4cxx configuration
327 BasicConfigurator::configure();
328 }
329
330 }
331
332 // Assert that all ranks indeed have initialized logging
333 assert(LogManager::getLoggerRepository()->isConfigured());
334
335 // "rankzero" is a disguise for the log4cxx::RootLogger where
336 // SubversiveASHEL::addAppenderEvent() ensures nothing is emitted
337 // on non-root ranks while still leaving isLevelEnabledFor() intact.
338 rankzero = Logger::getRootLogger();
339
340 // "allranks" emits events on all ranks by having SubversiveASHEL co-opt
341 // appenders from RootLogger on non-root ranks.
342 allranks = Logger::getLogger(worldrankname);
343
344 // On non-zero ranks the RootLogger may have no appenders (which is fine).
345 // However, it causes a worrisome one-time message like the following
346 // log4cxx: No appender could be found for logger (root).
347 // log4cxx: Please initialize the log4cxx system properly.
348 // to appear once from each rank. Temporarily disable log4cxx's LogLog,
349 // emit a message into the ether, and turn LogLog back on again.
350 if (worldrank > 0 && rankzero->getAllAppenders().empty()) {
351 LogLog::setQuietMode(true);
352 rankzero->fatal("Sorry, baby, but I had to crash that Honda.");
353 LogLog::setQuietMode(false);
354 }
355}
