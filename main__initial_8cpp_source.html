308 // Primitive state to be initialized depending on scenario
309 // Pressure and temperature convenient for later adding pulses
310 ArrayXr u = ArrayXr::Constant(Ny, numeric_limits<real_t>::quiet_NaN());
311 ArrayXr v = ArrayXr::Constant(Ny, numeric_limits<real_t>::quiet_NaN());
312 ArrayXr w = ArrayXr::Constant(Ny, numeric_limits<real_t>::quiet_NaN());
313 ArrayXr T = ArrayXr::Constant(Ny, numeric_limits<real_t>::quiet_NaN());
314 ArrayXr p = ArrayXr::Constant(Ny, numeric_limits<real_t>::quiet_NaN());
315
316 if ( options.variables().count("Re_x")
317 && !sg->formulation.enabled()) {
318 INFO(who, "Initializing a classical boundary layer profile");
319
320 INFO(who, "Blasius base profile for u, v, and T uses Re_x = "
321 << Re_x << ", Pr = " << scenario->Pr);
322 INFO(who, "Linear ramp uses for w from "
323 << isothermal->lower_w << " to " << isothermal->upper_w);
324 std::shared_ptr<gsl_spline> fit_u(suzerain_blasius_u(Re_x),
325 gsl_spline_free);
326 std::shared_ptr<gsl_spline> fit_v(suzerain_blasius_v(Re_x),
327 gsl_spline_free);
328 std::shared_ptr<gsl_spline> fit_T(suzerain_blasius_T(Re_x, scenario->Pr),
329 gsl_spline_free);
330 std::shared_ptr<gsl_interp_accel> accel(gsl_interp_accel_alloc(),
331 gsl_interp_accel_free);
332
333 INFO0(who, "Computing mean freestream behavior per plate scenario");
334 // See treatment_plate.tex for origin of T-dependent scaling
335 // Blasius profile fixes upper_{u,v} for the freestream...
336 isothermal->upper_u = sqrt(isothermal->upper_T); // Eqn (6)
337 isothermal->upper_v = gsl_spline_eval(fit_v.get(),
338 grid->L.y(), accel.get())
339 * isothermal->upper_u;
340 isothermal->upper_rho = pow(isothermal->upper_T,
341 scenario->beta - 0.5); // Eqn (8)
342 // ...where we've blissfully ignored lower_{u,v} (tra la la)
343
344 // Denormalize calls honor lower, upper values for u, w, T.
345 // Wall-normal v does not use one as fit_v does not achieve 1
346 for (int j = 0; j < Ny; ++j) {
347 const double y_j = b->collocation_point(j);
348 u[j] = isothermal->denormalize_u(
349 gsl_spline_eval(fit_u.get(), y_j, accel.get()));
350 v[j] = gsl_spline_eval(fit_v.get(), y_j, accel.get())
351 * isothermal->upper_u;
352 w[j] = isothermal->denormalize_w(y_j / grid->L.y());
353 T[j] = isothermal->denormalize_T(
354 gsl_spline_eval(fit_T.get(), y_j, accel.get()));
355 }
356
357 // Find pressure using constant density taken from freestream
358 p = isothermal->upper_rho * T / scenario->gamma;
359 INFO(who, "Base field uses p from " << p(0) << " to " << p(Ny - 1));
360
361 } else if ( options.variables().count("Re_x")
362 && sg->formulation.enabled()) {
363 INFO(who, "Initializing approximate homogenized boundary layer");
364
365 INFO(who, "Blasius base profile for u and T uses Re_x = "
366 << Re_x << ", Pr = " << scenario->Pr);
367 INFO(who, "Linear ramp uses for v from "
368 << isothermal->lower_v << " to " << isothermal->upper_v);
369 INFO(who, "Linear ramp uses for w from "
370 << isothermal->lower_w << " to " << isothermal->upper_w);
371 std::shared_ptr<gsl_spline> fit_u(suzerain_blasius_u(Re_x),
372 gsl_spline_free);
373 std::shared_ptr<gsl_spline> fit_T(suzerain_blasius_T(Re_x, scenario->Pr),
374 gsl_spline_free);
375 std::shared_ptr<gsl_interp_accel> accel(gsl_interp_accel_alloc(),
376 gsl_interp_accel_free);
377
378 INFO(who, "Computing mean freestream behavior per plate scenario");
379 // See treatment_plate.tex for origin of T-dependent scaling
380 // Blasius profile fixes upper_u for the freestream...
381 isothermal->upper_u = sqrt(isothermal->upper_T); // Eqn (6)
382 isothermal->upper_rho = pow(isothermal->upper_T,
383 scenario->beta - 0.5); // Eqn (8)
384 // ...where we've blissfully ignored lower_u (tra la la)
385
386 // Denormalize calls honor lower, upper values for u, v, w, T.
387 for (int j = 0; j < Ny; ++j) {
388 const double y_j = b->collocation_point(j);
389 u[j] = isothermal->denormalize_u(
390 gsl_spline_eval(fit_u.get(), y_j, accel.get()));
391 v[j] = isothermal->denormalize_v(y_j / grid->L.y());
392 w[j] = isothermal->denormalize_w(y_j / grid->L.y());
393 T[j] = isothermal->denormalize_T(
394 gsl_spline_eval(fit_T.get(), y_j, accel.get()));
395 }
396
397 // Find pressure using constant density taken from freestream
398 p = isothermal->upper_rho * T / scenario->gamma;
399 INFO(who, "Base field uses p from " << p(0) << " to " << p(Ny - 1));
400
401 } else {
402 INFO(who, "Initializing channel flow profile");
403
404 INFO(who, "Base field starts from linear ramps for u, v, w, and T");
405 for (int j = 0; j < Ny; ++j) {
406 const double norm_y_j = b->collocation_point(j) / grid->L.y();
407 u[j] = isothermal->denormalize_u(norm_y_j);
408 v[j] = isothermal->denormalize_v(norm_y_j);
409 w[j] = isothermal->denormalize_w(norm_y_j);
410 T[j] = isothermal->denormalize_T(norm_y_j);
411 }
412 INFO(who, "Base field uses u from " << u(0) << " to " << u(Ny - 1));
413 INFO(who, "Base field uses v from " << v(0) << " to " << v(Ny - 1));
414 INFO(who, "Base field uses w from " << w(0) << " to " << w(Ny - 1));
415 INFO(who, "Base field uses T from " << T(0) << " to " << T(Ny - 1));
416
417 INFO(who, "Parabolic profile will be added with npower = " << npower);
418 INFO(who, "Finding normalization so u = (y*(L-y))^npower integral is 1");
419 real_t normalization = numeric_limits<real_t>::quiet_NaN();
420 if (npower == 1) {
421 // Mathematica: (Integrate[(x (L-x)),{x,0,L}]/L)^(-1)
422 normalization = 6 / pow(grid->L.y(), 2);
423 } else if (npower > 0) {
424 // Mathematica: (Integrate[(x (L - x))^n, {x, 0, L}]/L)^(-1)
425 // - (Gamma[-n] Gamma[3/2+n])
426 // / (2^(-1-2 n) L^(1+2 n) \[Pi]^(3/2) Csc[n \[Pi]])
427 const real_t num1 = sin(npower * pi<real_t>());
428 const real_t num2 = tgamma(-npower);
429 const real_t num3 = tgamma(real_t(3)/2 + npower);
430 const real_t denom1 = pow(2, -1-2*npower);
431 const real_t denom2 = pow(Ly, 1+2*npower);
432 const real_t denom3 = pow(pi<real_t>(), real_t(3)/2);
433 normalization = - (num1 * num2 * num3 * Ly)
434 / (denom1 * denom2 * denom3);
435 } else {
436 // Degenerate npower == 0 case avoids tgamma(...) domain issues
437 normalization = 0;
438 }
439
440 INFO(who, "Adding the requested parabolic streamwise velocity profile");
441 for (int j = 0; j < Ny; ++j) {
442 const real_t y_j = b->collocation_point(j);
443 u(j) += ( scenario->bulk_rho_u
444 - (isothermal->lower_u + isothermal->upper_u)/2)
445 * normalization
446 * pow(y_j * (Ly - y_j), npower);
447 }
448
449 // Find pressure from temperature and constant density assumption
450 p = scenario->bulk_rho * T / scenario->gamma;
451 INFO(who, "Base field uses p from " << p(0) << " to " << p(Ny - 1));
452
453 }
454
455 // Shout, as appropriate, if non-trivial radial flow without slow growth
456 if (rad->trivial()) {
457 DEBUG(who, "Trivial radial flow problem parameters ignored");
458 } else if (sg->formulation.enabled()) {
459 INFO(who, "Non-trivial radial flow active");
460 } else {
461 WARN(who, "Non-trivial radial flow active but without Largo formulation");
462 }
463
464 if (options.variables().count("acoustic_strength")) {
465 const real_t left = (Ly / 2) * (1 - acoustic_support);
466 const real_t right = (Ly / 2) * (1 + acoustic_support);
467 INFO(who, "Adding wall-normal acoustic pulse with strength "
468 << acoustic_strength
469 << " on (" << left << ", " << right << ")");
470 for (int j = 0; j < Ny; ++j) {// Baum et al JCP 1994 pp 254-5
471 const real_t v0 = v(j);
472 const real_t rho0 = scenario->gamma * p(j) / T(j);
473 v(j) += math::bump::shifted(b->collocation_point(j),
474 left, right, acoustic_strength,
475 real_t(0), acoustic_stiffness);
476 p(j) += rho0 * std::sqrt(T(j)) * scenario->Ma * (v(j) - v0);
477 T(j) = (scenario->gamma * p(j))
478 / (rho0 + scenario->Ma * (v(j) - v0));
479 }
480 }
481
482 if (options.variables().count("entropy_strength")) {
483 const real_t left = (Ly / 2) * (1 - entropy_support);
484 const real_t right = (Ly / 2) * (1 + entropy_support);
485 INFO(who, "Adding wall-normal entropy pulse with strength "
486 << entropy_strength
487 << " on (" << left << ", " << right << ")");
488 for (int j = 0; j < Ny; ++j) { // Baum et al JCP 1994 pp 254-5
489 T(j) += math::bump::shifted(b->collocation_point(j),
490 left, right, entropy_strength,
491 real_t(0), entropy_stiffness);
492 }
493 }
494
495 INFO(who, "Computing density from pressure and temperature profiles");
496 ArrayXr rho = scenario->gamma * p / T;
497 T.resize(0); // Mark irrelevant for further use
498
499 INFO(who, "Computing total energy and momentum from primitive state");
500 ArrayXr rho_E(Ny);
501 for (int j = 0; j < Ny; ++j) {
502 const Vector3r m(rho(j) * u(j),
503 rho(j) * v(j),
504 rho(j) * w(j));
505 rho_E(j) = rholut::energy_internal(scenario->gamma, p(j))
506 + rholut::energy_kinetic (scenario->Ma, rho(j), m);
507 }
508 ArrayXr rho_u = rho * u;
509 ArrayXr rho_v = rho * v;
510 ArrayXr rho_w = rho * w;
511 u.resize(0); // Mark irrelevant for further use
512 v.resize(0); // Mark irrelevant for further use
513 w.resize(0); // Mark irrelevant for further use
514 p.resize(0); // Mark irrelevant for further use
515
516 INFO(who, "Converting conserved state to B-spline coefficients");
517 suzerain::bsplineop_lu masslu(*cop);
518 masslu.factor_mass(*cop);
519 masslu.solve(1, rho_E.data(), 1, rho_E.size());
520 masslu.solve(1, rho_u.data(), 1, rho_u.size());
521 masslu.solve(1, rho_v.data(), 1, rho_v.size());
522 masslu.solve(1, rho_w.data(), 1, rho_w.size());
523 masslu.solve(1, rho .data(), 1, rho .size());
524
525 INFO(who, "Copying the coefficients directly into the zero-zero modes");
526 Map<VectorXc>((*state_linear)[ndx::e ].origin(), Ny)
527 = rho_E.cast<complex_t>();
528 Map<VectorXc>((*state_linear)[ndx::mx ].origin(), Ny)
529 = rho_u.cast<complex_t>();
530 Map<VectorXc>((*state_linear)[ndx::my ].origin(), Ny)
531 = rho_v.cast<complex_t>();
532 Map<VectorXc>((*state_linear)[ndx::mz ].origin(), Ny)
533 = rho_w.cast<complex_t>();
534 Map<VectorXc>((*state_linear)[ndx::rho].origin(), Ny)
535 = rho .cast<complex_t>();
536
537 }
538
539 INFO0(who, "Saving the newly initialized state to disk");
540 if (msoln) {
541 save_restart(mms, restart_file, clobber);
542 } else {
543 save_restart(0, restart_file, clobber);
544 }
545
546 return EXIT_SUCCESS;
