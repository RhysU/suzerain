36std::string error_string(const int errorcode)
37{
38 std::ostringstream retval;
39
40 char *errorstring = NULL;
41 int resultlen;
42 const int status = MPI_Error_string(errorcode, errorstring, &resultlen);
43 switch (status) {
44 case MPI_SUCCESS:
45 retval << errorstring;
46 break;
47 case MPI_ERR_ARG:
48 retval << "No error string known for code " << errorcode;
49 break;
50 default:
51 retval << "Unknown error " << status
52 << " when retrieving string for code " << errorcode;
53 }
54
55 return retval.str();
56}

82std::string comm_rank_identifier(MPI_Comm comm)
83{
84 // All MPI calls below fail for MPI_COMM_NULL,
85 // so handle it as a special case.
86 if (comm == MPI_COMM_NULL) {
87 return "NULL";
88 }
89
90 // We want the name for comm as a C-style string at p_buffer
91 // with resultlen = strlen(p_buffer)
92 char buffer[MPI_MAX_OBJECT_NAME];
93 char * p_buffer = buffer;
94 int resultlen;
95 const int error = MPI_Comm_get_name(comm, buffer, &resultlen);
96 if (error) throw std::runtime_error(error_string(error));
97 if (resultlen == 0) {
98 // Providing a default value if MPI had none.
99 // comm is an opaque handle (a pointer under some MPI implementations),
100 // so format it through uintptr_t rather than %X on a bare argument.
101 resultlen = snprintf(buffer, sizeof(buffer),
102 "c0x%" PRIxPTR, (uintptr_t) comm);
103 assert(resultlen >= 0);
104 } else {
105 // Brevity, trim any leading MPI_COMM_ business by modifying p_buffer
106 static const char prefix[] = "MPI_COMM_";
107 static const int prefix_len = sizeof(prefix) - 1;
108 if (0 == strncasecmp(buffer, prefix, prefix_len)) {
109 p_buffer += prefix_len;
110 resultlen -= prefix_len;
111 }
112 }
113
114 // Look up comm's size and this process' rank
115 int size, rank;
116 char sep;
117 if (comm == MPI_COMM_SELF) {
118 // size and rank are useless for MPI_COMM_SELF,
119 // so provide details from MPI_COMM_WORLD instead
120 sep = 'w';
121 size = comm_size(MPI_COMM_WORLD);
122 rank = comm_rank(MPI_COMM_WORLD);
123 } else {
124 sep = 'r';
125 size = comm_size(comm);
126 rank = comm_rank(comm);
127 }
128
129 // Pack up and return a std::string containing name and rank
130 // Take care so that all results from the same comm are the same length
131 std::ostringstream oss;
132 oss.write(p_buffer, resultlen);
133 oss << sep << std::setfill('0')
134#pragma warning(push,disable:810 2259)
135 << std::setw(ceil(log10(size)))
136#pragma warning(pop)
137 << rank;
138 return oss.str();
139}
