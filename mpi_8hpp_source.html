111template<class Integer, class ForwardIterator>
112void dims_create(const Integer nnodes,
113 const ForwardIterator dimsBegin,
114 const ForwardIterator dimsEnd) {
115
116 typedef typename
117 std::iterator_traits<ForwardIterator>::value_type value_type;
118
119 // numeric_cast and/or copy input types to type int to match MPI API
120 using boost::numeric_cast;
121 const int int_nnodes = numeric_cast<int>(nnodes);
122 const int int_ndims = numeric_cast<int>(std::distance(dimsBegin,dimsEnd));
123 std::unique_ptr<int[]> int_dims(new int[int_ndims]);
124 std::transform(dimsBegin, dimsEnd, int_dims.get(),
125 boost::numeric::converter<int,value_type>());
126
127 // Invoke the MPI API
128 const int status = MPI_Dims_create(int_nnodes, int_ndims, int_dims.get());
129 if (status != MPI_SUCCESS) throw std::runtime_error(error_string(status));
130
131 // Copy the output back to the caller
132 std::transform(int_dims.get(), int_dims.get() + int_ndims, dimsBegin,
133 boost::numeric::converter<value_type,int>());
134}

149template< typename charT, typename traits >
150std::basic_ostream<charT,traits>& sleep_barrier(
151 MPI_Comm comm = MPI_COMM_SELF,
152 const int sleep_rank = 0,
153 std::basic_ostream<charT,traits> &os = std::cerr)
154{
155 if (sleep_rank < 0) {
156 throw (std::runtime_error("sleep_rank < 0"));
157 }
158 if (sleep_rank >= comm_size(comm)) {
159 throw (std::runtime_error("sleep_rank > comm_size(comm)"));
160 }
161
162 // Retrieve the current hostname in a beyond-paranoid manner, continuing
163 // the sleep_barrier iff the hostname retrieval succeeds.
164 char hostname[256];
165 {
166 const std::string id = comm_rank_identifier(comm);
167 // Ensure null termination in event of truncation
168 hostname[sizeof(hostname)-1] = 0;
169 errno = 0;
170 int local_gethostname_err = gethostname(hostname, sizeof(hostname)-1);
171 if (local_gethostname_err) {
172 os << "Error from gethostname on " << id
173 << ":" << strerror(errno)
174 << "; skipping sleep_barrier." << std::endl;
175 }
176 int global_gethostname_err = 0;
177 const int err = MPI_Allreduce(&local_gethostname_err,
178 &global_gethostname_err,
179 1, MPI_INT, MPI_LOR, comm);
180 if (err) throw std::runtime_error(error_string(err));
181 if (global_gethostname_err) return os;
182 }
183
184 // Force all processes to sync up prior to sleeping
185 {
186 const int err = MPI_Barrier(comm);
187 if (err) throw std::runtime_error(error_string(err));
188 }
189
190 // Sleep indefinitely if we're the lucky process.
191 // Set i nonzero using a debugger to exit infinite loop below.
192 if (comm_rank(comm) == sleep_rank) {
193 os << "sleep_barrier: pid " << getpid() << " on host "
194 << hostname << " ready for attach." << std::endl;
195 int i = 0;
196 while (0 == i) {
197 sleep(5);
198 }
199 }
200
201 // Force all processes to sync up after sleeping
202 {
203 const int err = MPI_Barrier(comm);
204 if (err) throw std::runtime_error(error_string(err));
205 }
206
207 return os;
208}
