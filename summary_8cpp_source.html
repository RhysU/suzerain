66std::ostream&
67summary::write(std::ostream &out, bool header, bool csv) const
68{
69 typedef fullprec<storage_type::Scalar> fp;
70 using std::setw;
71 using std::right;
72 using std::max;
73
74 const char * const sep = csv ? ", " : " ";
75
76 // Compute maximum field width to anticipate
77 // (Could be done at compile-time, but, meh).
78 size_t width = fp::width;
79 for (size_t i = 0; i < nscalars::total; ++i) {
80 width = max(width, strlen(name[i]));
81 }
82
83 if (header) {
84 for (size_t j = 0; j < nscalars::total - 1; ++j) {
85 out << setw(width) << right << name[j] << sep;
86 }
87 out << setw(width) << right << name[nscalars::total-1] << '\n';
88 }
89
90 const std::string pad(width - fp::width,' ');
91 for (int i = 0; i < storage.rows(); ++i) {
92 for (size_t j = 0; j < nscalars::total-1; ++j) {
93 out << pad << fp(storage(i,j)) << sep;
94 }
95 out << pad << fp(storage(i, nscalars::total-1)) << '\n';
96 }
97
98 return out;
99}
