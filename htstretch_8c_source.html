39double
40suzerain_htstretch1(const double delta,
41 const double L,
42 const double x)
43{
44 assert(L > 0);
45 assert(0 <= x && x <= L);
46#pragma warning(push,disable:1572)
47 if (SUZERAIN_UNLIKELY(delta == 0.0)) {
48#pragma warning(pop)
49 return x / L;
50 } else {
51 return 1 + tanh(delta*(x/L - 1))/tanh(delta);
52 }
53}

55double
56suzerain_htstretch1_ddelta(const double delta,
57 const double L,
58 const double x)
59{
60 assert(L > 0);
61 assert(0 <= x && x <= L);
62#pragma warning(push,disable:1572)
63 if (SUZERAIN_UNLIKELY(delta == 0.0)) {
64#pragma warning(pop)
65 return 0.0;
66 } else {
67 const double xoverLlessone = x/L-1;
68 const double cosh_expr = cosh(delta*xoverLlessone);
69 const double tanh_expr = tanh(delta*xoverLlessone);
70 const double sinh_expr = sinh(delta);
71 return xoverLlessone/(tanh(delta)*cosh_expr*cosh_expr)
72 - tanh_expr/(sinh_expr*sinh_expr);
73 }
74}

76double
77suzerain_htstretch1_dL(const double delta,
78 const double L,
79 const double x)
80{
81 assert(L > 0);
82 assert(0 <= x && x <= L);
83#pragma warning(push,disable:1572)
84 if (SUZERAIN_UNLIKELY(delta == 0.0)) {
85#pragma warning(pop)
86 return -x / (L * L);
87 } else {
88 const double xoverLlessone = x/L-1;
89 const double cosh_expr = cosh(delta*xoverLlessone);
90 return -x*delta/(L*L*tanh(delta)*cosh_expr*cosh_expr);
91 }
92}

111double
112suzerain_htstretch2(const double delta,
113 const double L,
114 const double x)
115{
116 assert(L > 0);
117 assert(0 <= x && x <= L);
118#pragma warning(push,disable:1572)
119 if (SUZERAIN_UNLIKELY(delta == 0.0)) {
120#pragma warning(pop)
121 return x / L;
122 } else {
123 return 0.5*(1 + tanh(delta*(x/L - 0.5))/tanh(delta/2.));
124 }
125}

127double
128suzerain_htstretch2_ddelta(const double delta,
129 const double L,
130 const double x)
131{
132 assert(L > 0);
133 assert(0 <= x && x <= L);
134 const double xoverLlesshalf = x/L - 0.5;
135 const double sinh_halfdelta = sinh(delta/2);
136 const double cosh_term = cosh(delta*xoverLlesshalf);
137#pragma warning(push,disable:1572)
138 if (SUZERAIN_UNLIKELY(delta == 0.0)) {
139#pragma warning(pop)
140 return 0.0;
141 } else {
142 return - ((L-2*x)*sinh(delta)+L*sinh((2*x/L-1)*delta))
143 / (8*L*cosh_term*cosh_term*sinh_halfdelta*sinh_halfdelta);
144 }
145}

147double
148suzerain_htstretch2_dL(const double delta,
149 const double L,
150 const double x)
151{
152 assert(L > 0);
153 assert(0 <= x && x <= L);
154 const double xoverLlesshalf = x/L - 0.5;
155 const double cosh_term = cosh(delta*xoverLlesshalf);
156#pragma warning(push,disable:1572)
157 if (SUZERAIN_UNLIKELY(delta == 0.0)) {
158#pragma warning(pop)
159 return -x / (L * L);
160 } else {
161 return -x*delta/(tanh(delta/2)*2*L*L*cosh_term*cosh_term);
162 }
163}

241static
242int find_delta(gsl_function_fdf *problem_fdf,
243 const double delta_lower,
244 const double delta_upper,
245 const double epsabs,
246 const int maxiter,
247 double *delta)
248{
249 // Initial bounds sanity check.
250 // Hopefully ensures root existence for hyperbolic tangent functions,
251 // At least for the class of things we care about here.
252 const double crit_val
253 = ((delta_problem_params*) problem_fdf->params)->crit_val;
254 const double crit_val_lower = GSL_FN_FDF_EVAL_F(problem_fdf, delta_lower);
255 const double crit_val_upper = GSL_FN_FDF_EVAL_F(problem_fdf, delta_upper);
256 if (crit_val >= crit_val_lower) {
257 SUZERAIN_ERROR("(crit_x, crit_val) incompatible with delta_lower",
258 SUZERAIN_FAILURE);
259 }
260 if (crit_val <= crit_val_upper) {
261 SUZERAIN_ERROR("(crit_x, crit_val) incompatible with delta_upper",
262 SUZERAIN_FAILURE);
263 }
264
265 // Initialize a bracketing solver
266 gsl_function problem_f = { problem_fdf->f, problem_fdf->params };
267 gsl_root_fsolver *fsolver = gsl_root_fsolver_alloc(gsl_root_fsolver_brent);
268 if (!fsolver) {
269 SUZERAIN_ERROR("Unable to allocate solver", SUZERAIN_ENOMEM);
270 }
271 if (gsl_root_fsolver_set(fsolver, &problem_f, delta_lower, delta_upper)) {
272 gsl_root_fsolver_free(fsolver);
273 SUZERAIN_ERROR("Unable to initialize solver", SUZERAIN_ESANITY);
274 }
275
276 // Iterate until either iteration or tolerance criteria is met
277 int iter = 0;
278 do {
279 *delta = gsl_root_fsolver_root(fsolver);
280 const double residual = GSL_FN_EVAL(&problem_f, *delta);
281 if (GSL_SUCCESS == gsl_root_test_residual(residual, epsabs))
282 break;
283
284 const int err = gsl_root_fsolver_iterate(fsolver);
285 if (err) {
286 gsl_root_fsolver_free(fsolver);
287 SUZERAIN_ERROR("Error during solver iteration", err);
288 }
289 } while (iter++ < maxiter);
290
291 // Tear down solver and finish up
292 gsl_root_fsolver_free(fsolver);
293 return (iter < maxiter) ? SUZERAIN_SUCCESS : SUZERAIN_EMAXITER;
294}

296int
297suzerain_htstretch1_find_delta(const double L,
298 const double x_crit,
299 const double s_crit,
300 const double epsabs,
301 const int maxiter,
302 double *delta)
303{
304 assert(L > 0);
305 assert(0.0 <= x_crit && x_crit <= L);
306 assert(0.0 <= s_crit && s_crit <= 1.0);
307 assert(epsabs > 0);
308 assert(maxiter > 0);
309 assert(delta != NULL);
310
311 delta_problem_params dpp = { L, x_crit, s_crit };
312 gsl_function_fdf fdf = { &htstretch1_delta_problem_f,
313 &htstretch1_delta_problem_df,
314 &htstretch1_delta_problem_fdf,
315 &dpp };
316
317 return find_delta(&fdf, 1e-8, 1e3, epsabs, maxiter, delta);
318}
