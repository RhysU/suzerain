38// Find pointwise solution details given state, Ma2=Ma**2, gamm1=gam-1
39// Compare nozzle_details function source within notebooks/radialflow.m
40static inline
41void nozzle_details(const double R,
42 const double u,
43 const double rho,
44 const double p,
45 const double Ma02,
46 const double gam1,
47 double *up,
48 double *rhop,
49 double *pp,
50 double *a2)
51{
52 const double u2 = u*u;
53 const double C = (2/Ma02 + gam1*(1 - u2));
54 *up = (u*C) / (R*(2*u2 - C));
55 *pp = -Ma02*rho*u*(*up);
56 *a2 = 1 + Ma02/2*gam1*(1 - u2);
57 *rhop = *pp / *a2;
58 SUZERAIN_UNUSED(p);
59}

62typedef struct params_type {

64 double gam1; // gamma - 1

67// Find [u; rho; p]' given r, x=[u; rho; p], Ma02=Ma0**2, gam1=gamma-1
68// Compare nozzle_rhs function source within notebooks/radialflow.m
69static int nozzle_rhs(double R,
70 const double y[],
71 double dydt[],
72 void *params)
73{
74 double a2;
75 nozzle_details(R, y[0], y[1], y[2],
76 ((params_type *)params)->Ma02,
77 ((params_type *)params)->gam1,
78 &dydt[0], &dydt[1], &dydt[2], &a2);
79 return GSL_SUCCESS;
80}

82// Compare nozzle function source within notebooks/radialflow.m
83suzerain_radialflow_solution *
84suzerain_radialflow_solver(
85 const double Ma0,
86 const double gamma,
87 const double u1,
88 const double rho1,
89 const double p1,
90 const double * const R,
91 const size_t size)
92{
93 // Sanity check incoming arguments and realizability
94 if (Ma0 <= 0) SUZERAIN_ERROR_NULL("Ma0 <= 0", SUZERAIN_EDOM);
95 if (gamma <= 1) SUZERAIN_ERROR_NULL("gamma <= 1", SUZERAIN_EDOM);
96 if (rho1 <= 0) SUZERAIN_ERROR_NULL("rho1 <= 0", SUZERAIN_EDOM);
97 if (p1 <= 0) SUZERAIN_ERROR_NULL("p1 <= 0", SUZERAIN_EDOM);
98 if (size == 0) SUZERAIN_ERROR_NULL("size == 0", SUZERAIN_EINVAL);
99 if (!(u1*u1 < 2 / (Ma0*Ma0 * (gamma-1)) + 1)) {
100 char msg[128];
101 snprintf(msg, sizeof(msg),
102 "Ma0=%g, gamma=%g, u1=%g imply a*a <= 0", Ma0, gamma, u1);
103 SUZERAIN_ERROR_NULL(msg, SUZERAIN_EDOM);
104 }
105
106 // Allocate solution storage into which integrator results will be saved
107 suzerain_radialflow_solution * const s = malloc(
108 sizeof(suzerain_radialflow_solution)
109 + size * sizeof(suzerain_radialflow_state));
110 if (!s) SUZERAIN_ERROR_NULL("Solution allocation failed", SUZERAIN_ENOMEM);
111 s->state[0].R = s->state[0].u = s->state[0].rho = s->state[0].p = GSL_NAN;
112
113 // Use GNU Scientific Library ODE integrator on [u; rho; p]' system
114 // being careful to account for increasing vs decreasing R[0], R[1], ...
115 double current_R = R[0];
116 double y[3] = { u1, rho1, p1 };
117 params_type params = { Ma0*Ma0, gamma - 1 };
118 gsl_odeiv2_system sys = { &nozzle_rhs, 0, sizeof(y)/sizeof(y[0]), &params};
119 const double abstol = 0;
120 const double reltol = GSL_SQRT_DBL_EPSILON;
121 const double hstart = (size == 1) ? +GSL_SQRT_DBL_EPSILON
122 : (R[0] <= R[1]) ? +GSL_SQRT_DBL_EPSILON
123 : -GSL_SQRT_DBL_EPSILON;
124 gsl_odeiv2_driver * driver = gsl_odeiv2_driver_alloc_y_new(
125 &sys, gsl_odeiv2_step_rkf45, hstart, abstol, reltol);
126 if (!driver) {
127 free(s);
128 SUZERAIN_ERROR_NULL("Driver allocation failed", GSL_EFAILED);
129 }
130 int error = 0;
131 for (size_t i = 1; i < size && !error; ++i) { // Advance state to R[i]
132 error = gsl_odeiv2_driver_apply(driver, &current_R, R[i], y);
133 s->state[i].R = current_R;
134 s->state[i].u = y[0];
135 s->state[i].rho = y[1];
136 s->state[i].p = y[2];
137 }
138 gsl_odeiv2_driver_free(driver);
139 if (error) {
140 char msg[128];
141 snprintf(msg, sizeof(msg), "gsl_odeiv2_driver_apply reports %d: %s",
142 error, gsl_strerror(error));
143 free(s);
144 SUZERAIN_ERROR_NULL(msg, GSL_EFAILED);
145 }
146
147 // Save scenario parameters and initial conditions (marking solution valid)
148 s->Ma0 = Ma0;
149 s->gamma = gamma;
150 s->size = size;
151 s->state[0].R = R[0];
152 s->state[0].u = u1;
153 s->state[0].rho = rho1;
154 s->state[0].p = p1;
155
156 // Compute derivatives and sound speed squared directly from state
157 for (size_t i = 0; i < size; ++i) {
158 nozzle_details(s->state[i].R,
159 s->state[i].u,
160 s->state[i].rho,
161 s->state[i].p,
162 s->Ma0 * s->Ma0,
163 s->gamma - 1,
164 &s->state[i].up,
165 &s->state[i].rhop,
166 &s->state[i].pp,
167 &s->state[i].a2);
168 }
169
170 return s;
171}

173inline // Permits inlining by functions appearing later in this file
174double
175suzerain_radialflow_delta(
176 const suzerain_radialflow_solution * s,
177 const size_t i)
178{
179 assert(i < s->size);
180 const double R0 = GSL_MIN(s->state[0].R, s->state[s->size-1].R);
181 return sqrt(gsl_pow_2(s->state[i].R) - gsl_pow_2(R0));
182}

184double
185suzerain_radialflow_qoi_Mae(
186 const suzerain_radialflow_solution * s,
187 const size_t i)
188{
189 assert(i < s->size);
190 const double R0 = GSL_MIN(s->state[0].R, s->state[s->size-1].R);
191 return (s->Ma0 * R0 * fabs(s->state[i].u) )
192 / ( s->state[i].R * sqrt(s->state[i].a2));
193}

195double
196suzerain_radialflow_qoi_pexi(
197 const suzerain_radialflow_solution * s,
198 const size_t i)
199{
200 assert(i < s->size);
201 const double delta = suzerain_radialflow_delta(s, i);
202 const double R0 = GSL_MIN(s->state[0].R, s->state[s->size-1].R);
203 return - (s->state[i].R * delta * s->state[i].up)
204 / (R0 * fabs(s->state[i].u));
205}

207double
208suzerain_radialflow_qoi_Te(
209 const suzerain_radialflow_solution * s,
210 const size_t i,
211 const double Ma)
212{
213 assert(i < s->size);
214 return s->gamma * gsl_pow_2(Ma / s->Ma0)
215 * (s->state[i].p / s->state[i].rho);
216}

218int
219suzerain_radialflow_qoi_match(
220 const double delta,
221 const double gamma,
222 const double Ma_e,
223 const double p_exi,
224 double *Ma0,
225 double *R0,
226 double *R,
227 double *uR,
228 double *rhoR,
229 double *pR)
230{
231
232 // Solve quadratic for R0
233 double x[2];
234 const double Ma_e2 = gsl_pow_2(Ma_e );
235 const double delta2 = gsl_pow_2(delta);
236 const int nreal = gsl_poly_solve_quadratic(
237 - fabs(Ma_e2 - 1) * fabs(p_exi),
238 delta,
239 - Ma_e2 * delta2 * fabs(p_exi) * GSL_SIGN(Ma_e2 - 1),
240 x+0, x+1);
241
242 // If real roots exist...
243 if (nreal > 0) {
244 // ...take the largest one possible...
245 *R0 = x[nreal - 1];
246 *Ma0 = 1 / sqrt(1/Ma_e2 + (gamma - 1)*delta2/gsl_pow_2(*R0)/2);
247 *R = sqrt(gsl_pow_2(*R0) + delta2);
248 *uR = - (*R / *R0) * GSL_SIGN(p_exi*(Ma_e2 - 1));
249 *rhoR = 1;
250 *pR = *rhoR * gsl_pow_2(*Ma0) / gamma / Ma_e2;
251 } else {
252 // ...otherwise defensive failure...
253 *R0 = GSL_NAN;
254 *Ma0 = GSL_NAN;
255 *R = GSL_NAN;
256 *uR = GSL_NAN;
257 *rhoR = GSL_NAN;
258 *pR = GSL_NAN;
259 }
260 // ...notice success requires a positive R0!
261 return *R0 >= 0 ? SUZERAIN_SUCCESS : SUZERAIN_EFAILED;
262}

264inline // Suggests inlining within suzerain_radialflow_cartesian_conserved
265void
266suzerain_radialflow_cartesian_primitive(
267 const suzerain_radialflow_solution * s,
268 const size_t i,
269 const double Ma,
270 double * const rho,
271 double * const u,
272 double * const v,
273 double * const p,
274 double * const a,
275 double * const rho_xi,
276 double * const u_xi,
277 double * const v_xi,
278 double * const p_xi,
279 double * const a_xi,
280 double * const rho_y,
281 double * const u_y,
282 double * const v_y,
283 double * const p_y,
284 double * const a_y)
285{
286 assert(i < s->size);
287 // If u(R0) >= 0, compute at coords (x,delta). Otherwise (-x,delta).
288 // Possibly flipping from the first to the second quadrant permits
289 // flow to always be in positive x direction regardless of sonic-ness.
290 const double x = (s->state[0].u >= 0 ? 1 : -1) * s->state[0].R;
291 const double x_inv_R = x / s->state[i].R;
292 const double x2 = gsl_pow_2(x);
293 const double y = suzerain_radialflow_delta(s, i);
294 const double y2 = gsl_pow_2(y);
295 const double y_inv_R = y / s->state[i].R;
296 const double inv_R = 1 / s->state[i].R;
297 const double inv_R2 = gsl_pow_2(inv_R);
298 const double Ma2Ma02 = gsl_pow_2(Ma / s->Ma0);
299
300 const suzerain_radialflow_state * const t = &s->state[i];
301 *rho = t->rho;
302 *u = t->u * x_inv_R;
303 *v = t->u * y_inv_R;
304 *p = t->p * Ma2Ma02;
305 *rho_xi = t->rhop * x_inv_R;
306 *u_xi = inv_R2 * (x2 * t->up + y2 * inv_R * t->u);
307 *v_xi = x_inv_R*y_inv_R * ( t->up - inv_R * t->u);
308 *p_xi = t->pp * Ma2Ma02 * x_inv_R;
309 *rho_y = t->rhop * y_inv_R;
310 *u_y = x_inv_R*y_inv_R * ( t->up - inv_R * t->u);
311 *v_y = inv_R2 * (y2 * t->up + x2 * inv_R * t->u);
312 *p_y = t->pp * Ma2Ma02 * y_inv_R;
313
314 // Beware (*a) contains Ma/Ma0*a when computing ap, *a_xi, and *a_y
315 *a = sqrt(t->a2 * Ma2Ma02);
316 const double ap = (1 - s->gamma) / 2 * Ma * Ma * t->u * t->up / (*a);
317 *a_xi = x_inv_R * ap;
318 *a_y = y_inv_R * ap;
319}

321void
322suzerain_radialflow_cartesian_conserved(
323 const suzerain_radialflow_solution * s,
324 const size_t i,
325 const double Ma,
326 double * const r,
327 double * const ru,
328 double * const rv,
329 double * const rE,
330 double * const p,
331 double * const r_xi,
332 double * const ru_xi,
333 double * const rv_xi,
334 double * const rE_xi,
335 double * const p_xi,
336 double * const r_y,
337 double * const ru_y,
338 double * const rv_y,
339 double * const rE_y,
340 double * const p_y)
341{
342 // Delegate to compute local primitive state and {p,a}_{,_xi,_y}
343 double rho, u, v, a, rho_xi, u_xi, v_xi, a_xi, rho_y, u_y, v_y, a_y;
344 suzerain_radialflow_cartesian_primitive(
345 s, i, Ma, &rho, &u, &v, p, &a,
346 &rho_xi, &u_xi, &v_xi, p_xi, &a_xi,
347 &rho_y, &u_y, &v_y, p_y, &a_y);
348
349 // Though expected from formulation, method does not assume H = constant
350 // This permits both detecting formulation and coding errors as well as
351 // correctly propagating any roundoff-ish drift in H from solution.
352 const double invgam1 = 1/(s->gamma - 1);
353 const double Ma22 = Ma*Ma/2;
354 const double H = a*a*invgam1 + Ma22*(u*u + v*v);
355 const double H_xi = 2*(a*a_xi*invgam1 + Ma22*(u*u_xi + v*v_xi));
356 const double H_y = 2*(a*a_y *invgam1 + Ma22*(u*u_y + v*v_y ));
357
358 // Convert to conserved state using \rho H = \rho E + p
359 *r = rho;
360 *ru = rho*u;
361 *rv = rho*v;
362 *rE = rho*H - *p;
363
364 // Convert streamwise primitive derivatives to conserved derivatives
365 *r_xi = rho_xi;
366 *ru_xi = rho*u_xi + rho_xi*u;
367 *rv_xi = rho*v_xi + rho_xi*v;
368 *rE_xi = rho*H_xi + rho_xi*H - *p_xi;
369
370 // Convert wall-normal primitive derivatives to conserved derivatives
371 *r_y = rho_y ;
372 *ru_y = rho*u_y + rho_y*u;
373 *rv_y = rho*v_y + rho_y*v;
374 *rE_y = rho*H_y + rho_y*H - *p_y;
375}
