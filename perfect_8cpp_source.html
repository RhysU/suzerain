93void
94adjust_scenario(contiguous_state<4,complex_t> &swave,
95 const definition_scenario& scenario,
96 const specification_grid& grid,
97 const pencil_grid& dgrid,
98 const bsplineop& cop,
99 const real_t old_Ma,
100 const real_t old_gamma)
101{
102 // We are only prepared to handle a fixed number of fields in this routine
103 enum { state_count = 5 };
104
105 // Ensure state storage meets this routine's assumptions
106 SUZERAIN_ENSURE( swave.shape()[0] == state_count);
107 SUZERAIN_ENSURE(numeric_cast<int>(swave.shape()[1]) == dgrid.local_wave_extent.y());
108 SUZERAIN_ENSURE(numeric_cast<int>(swave.shape()[2]) == dgrid.local_wave_extent.x());
109 SUZERAIN_ENSURE(numeric_cast<int>(swave.shape()[3]) == dgrid.local_wave_extent.z());
110
111 bool quickreturn = true;
112#pragma warning(push,disable:1572)
113 if (old_Ma != scenario.Ma) {
114#pragma warning(pop)
115 INFO0("Changing state Mach number from "
116 << old_Ma << " to " << scenario.Ma);
117 quickreturn = false;
118 }
119#pragma warning(push,disable:1572)
120 if (old_gamma != scenario.gamma) {
121#pragma warning(pop)
122 INFO0("Changing state ratio of specific heats from "
123 << old_gamma << " to " << scenario.gamma);
124 quickreturn = false;
125 }
126 if (quickreturn) {
127 DEBUG0("Not rescaling energy as scenario parameters have not changed");
128 return;
129 }
130
131 // Convert state to physical space collocation points
132 operator_tools otool(grid, dgrid, cop);
133 physical_view<state_count> sphys(dgrid, swave);
134 for (size_t k = 0; k < state_count; ++k) {
135 otool.bop_apply(0, 1.0, swave, k);
136 dgrid.transform_wave_to_physical(&sphys.coeffRef(k,0));
137 }
138
139 // Adjust total energy by the necessary amount at every collocation point
140 // This procedure is not the cheapest or least numerically noisy,
141 // but it does re-use existing compute kernels in a readable way.
142 INFO0("Holding density and temperature constant during changes");
143 for (int offset = 0, j = dgrid.local_physical_start.y();
144 j < dgrid.local_physical_end.y();
145 ++j) {
146 const int last_zxoffset = offset
147 + dgrid.local_physical_extent.z()
148 * dgrid.local_physical_extent.x();
149 for (; offset < last_zxoffset; ++offset) {
150
151 const real_t e (sphys(ndx::e, offset));
152 const Vector3r m (sphys(ndx::mx, offset),
153 sphys(ndx::my, offset),
154 sphys(ndx::mz, offset));
155 const real_t rho(sphys(ndx::rho, offset));
156
157 // Compute temperature using old_gamma, old_Ma
158 real_t p, T;
159 rholut::p_T(scenario.alpha, scenario.beta, old_gamma, old_Ma,
160 rho, m, e, /*out*/ p, /*out*/ T);
161 // Compute total energy from new gamma, Ma, rho, T
162 rholut::p(scenario.gamma, rho, T, /*out*/ p);
163 sphys(ndx::e, offset) = rholut::energy_internal(scenario.gamma, p)
164 + rholut::energy_kinetic(scenario.Ma, rho, m);
165 }
166 }
167
168 // Convert state back to wave space coefficients in X, Y, and Z
169 // building FFT normalization constant into the mass matrix
170 bsplineop_luz massluz(cop);
171 const complex_t scale_factor = 1 / dgrid.chi();
172 massluz.opform(1, &scale_factor, cop);
173 massluz.factor();
174 for (size_t i = 0; i < state_count; ++i) {
175 dgrid.transform_physical_to_wave(&sphys.coeffRef(i, 0));
176 otool.bop_solve(massluz, swave, i);
177 }
178}

180void
181add_noise(contiguous_state<4,complex_t> &state,
182 const specification_noise& noise,
183 const definition_scenario& scenario,
184 const specification_grid& grid,
185 const pencil_grid& dgrid,
186 const bsplineop& cop,
187 bspline &b)
188{
189 // FIXME Needs to be made aware of channel vs flat plate
190
191 // We are only prepared to handle a fixed number of fields in this routine
192 enum { state_count = 5 };
193
194 // Ensure state storage meets this routine's assumptions
195 SUZERAIN_ENSURE( state.shape()[0] == state_count);
196 SUZERAIN_ENSURE(numeric_cast<int>(state.shape()[1]) == dgrid.local_wave_extent.y());
197 SUZERAIN_ENSURE(numeric_cast<int>(state.shape()[2]) == dgrid.local_wave_extent.x());
198 SUZERAIN_ENSURE(numeric_cast<int>(state.shape()[3]) == dgrid.local_wave_extent.z());
199
200 // Ensure we were handed collocation-based operator matrices
201 SUZERAIN_ENSURE(cop.get()->method == SUZERAIN_BSPLINEOP_COLLOCATION_GREVILLE);
202
203 const int Ny = grid.N.y();
204
205#pragma warning(push,disable:1572)
206 if (noise.percent == 0) {
207#pragma warning(pop)
208 DEBUG0("Zero noise added to velocity fields");
209 return;
210 }
211
212 using inorder::wavenumber_abs;
213 using inorder::wavenumber_max;
214 using inorder::wavenumber_translatable;
215 const real_t twopi = 2 * boost::math::constants::pi<real_t>();
216
217 // Evaluate maximum fluctuation magnitude based on percentage of
218 // centerline mean streamwise velocity and broadcast result.
219 // Alright, alright... actually an approximate mean velocity.
220 real_t maxfluct;
221 if (dgrid.has_zero_zero_modes()) {
222 complex_t momentum, density;
223 const real_t centerline = grid.L.y() / 2;
224 b.linear_combination(
225 0, &state[ndx::mx][0][0][0], centerline, &momentum);
226 INFO0("Centerline mean streamwise momentum at y = "
227 << centerline << " is " << momentum);
228 b.linear_combination(
229 0, &state[ndx::rho][0][0][0], centerline, &density);
230 INFO0("Centerline mean density at y = "
231 << centerline << " is " << density);
232 maxfluct = noise.percent / 100 * (abs(momentum) / abs(density));
233 }
234 SUZERAIN_MPICHKR(MPI_Bcast(&maxfluct, 1,
235 mpi::datatype_of(maxfluct),
236 dgrid.rank_zero_zero_modes, MPI_COMM_WORLD));
237 INFO0("Adding velocity perturbations with maximum magnitude " << maxfluct);
238
239 // Compute and display kxfrac_min, kxfrac_max constraints
240#pragma warning(push,disable:2259)
241 const int dkx_max = noise.kxfrac_max * wavenumber_max(grid.N.x());
242 const int dkx_min = noise.kxfrac_min * wavenumber_max(grid.N.x());
243#pragma warning(pop)
244#pragma warning(push,disable:1572)
245 if (noise.kxfrac_max != 1 || noise.kxfrac_min != 0) {
246#pragma warning(pop)
247 INFO0("Perturbations added only to absolute X wavenumbers in range ["
248 << dkx_min << ":" << dkx_max << "]");
249 }
250
251 // Compute and display kzfrac_min, kzfrac_max constraints
252#pragma warning(push,disable:2259)
253 const int dkz_max = noise.kzfrac_max * wavenumber_max(grid.N.z());
254 const int dkz_min = noise.kzfrac_min * wavenumber_max(grid.N.z());
255#pragma warning(pop)
256#pragma warning(push,disable:1572)
257 if (noise.kzfrac_max != 1 || noise.kzfrac_min != 0) {
258#pragma warning(pop)
259 INFO0("Perturbations added only to absolute Z wavenumbers in range ["
260 << dkz_min << ":" << dkz_max << "]");
261 }
262
263 // Form mass matrix to convert (wave, collocation point values, wave)
264 // perturbations to (wave, coefficients, wave)
265 bsplineop_luz massluz(cop);
266 massluz.factor_mass(cop);
267
268 // Set L'Ecuyer et al.'s rngstream seed. Use a distinct Substream for each
269 // wall-normal pencil to ensure process is a) repeatable despite changes in
270 // processor count, b) easy to code, and c) embarrassingly parallel.
271 rngstream rng;
272 {
273 array<unsigned long,6> seed;
274 std::fill(seed.begin(), seed.end(), noise.seed);
275 rng.SetSeed(seed.data());
276 }
277 rng.IncreasedPrecis(true); // Use more bits of resolution
278
279 // Approach is the following:
280 // 0) Allocate storage for state and three additional scalar fields.
281 // 1) Generate a random vector-valued field \tilde{A}.
282 // \tilde{A}'s x- and z-derivatives have zero mean by periodicity;
283 // 2) Zero first two B-spline coefficients near walls so partial_y
284 // \tilde{A} vanishes at the wall
285 // 3) This step intentionally left blank.
286 // 4) Compute curl A in physical space and rescale so maximum
287 // pointwise norm of A is maxfluct. curl A is solenoidal
288 // and now has the velocity perturbation properties we desire.
289 // 5) Store curl A in physical space in the three scalar fields.
290 // 6) Copy state into auxiliary state storage and bring to
291 // physical space.
292 // 7) At each point compute velocity and internal energy. Perturb
293 // velocity and compute new total energy using perturbed
294 // velocities.
295 // 8) Bring perturbed state information back to wavespace.
296 // 9) Overwrite state storage with the new perturbed state.
297
298 // 0) Allocate storage for state and three additional scalar fields.
299 std::unique_ptr<contiguous_state<4,complex_t> > _s_ptr( // RAII
300 support::allocate_padded_state<contiguous_state<4,complex_t> >(
301 state_count + 3, dgrid));
302 contiguous_state<4,complex_t> &s = *_s_ptr; // Shorthand
303 std::fill(s.range().begin(), s.range().end(), 0); // Zero memory
304
305 // 1) Generate a random vector-valued field \tilde{A}.
306 // For each scalar component of \tilde{A}...
307 for (size_t l = 0; l < 3; ++l) {
308
309 for (int k = 0; k < grid.dN.z(); ++k) {
310 if (!wavenumber_translatable(grid.N.z(), grid.dN.z(), k)) continue;
311
312
313 for (int i = 0; i < grid.dN.x(); ++i) {
314 if (!wavenumber_translatable(grid.N.x(), grid.dN.x(), i)) continue;
315
316 // ...and advance rngstream to the (i, ., k) substream...
317 // ...(necessary for processor-topology independence)...
318 rng.ResetNextSubstream();
319
320 // ...but only the rank holding the (i, ., k) pencil continues.
321 if ( k < dgrid.local_wave_start.z()
322 || k >= dgrid.local_wave_end.z()
323 || i < dgrid.local_wave_start.x()
324 || i >= dgrid.local_wave_end.x()) continue;
325
326 // Satisfy fluct_kzfrac_min, fluct_kzfrac_max constraints
327 if (wavenumber_abs(grid.dN.z(), k) < dkz_min) continue;
328 if (wavenumber_abs(grid.dN.z(), k) > dkz_max) continue;
329
330 // Satisfy fluct_kxfrac_min, fluct_kxfrac_max constraints
331 if (wavenumber_abs(grid.dN.x(), i) < dkx_min) continue;
332 if (wavenumber_abs(grid.dN.x(), i) > dkx_max) continue;
333
334 // Compute local indices for global (i, ., k).
335 const int local_i = i - dgrid.local_wave_start.x();
336 const int local_k = k - dgrid.local_wave_start.z();
337
338 // ...generate coeffs with well-defined pseudorandom order.
339 // 2) Zero first two B-spline coefficients near walls
340 // so partial_y \tilde{A} vanishes at the wall
341 s[2*l][0][local_i][local_k] = 0;
342 s[2*l][1][local_i][local_k] = 0;
343 for (int j = 2; j < Ny - 2; ++j) {
344 const real_t magnitude = rng.RandU01();
345 const real_t phase = rng.RandU01() * twopi;
346 s[2*l][j][local_i][local_k] = std::polar(magnitude, phase);
347 }
348 s[2*l][Ny - 2][local_i][local_k] = 0;
349 s[2*l][Ny - 1][local_i][local_k] = 0;
350
351 } // end X
352
353 } // end Z
354
355 } // end scalar components of A
356
357 // 4) Compute curl A in physical space and rescale so maximum
358 // pointwise norm of A is maxfluct. curl A is solenoidal
359 // and now has the velocity perturbation properties we desire.
360
361 // Copy s[2l] into s[2l+1] as we need two copies to compute curl A
362 for (size_t l = 0; l < 3; ++l) s[2*l+1] = s[2*l];
363
364 // Prepare physical-space view of the wave-space storage
365 physical_view<state_count+3> p(dgrid, s);
366
367 // Initializing operator_base to access decomposition-ready utilities
368 operator_tools otool(grid, dgrid, cop);
369
370 // From Ax in s[0] compute \partial_y Ax
371 otool.bop_apply(1, 1.0, s, 0);
372 dgrid.transform_wave_to_physical(&p.coeffRef(0,0));
373
374 // From Ax in s[1] compute \partial_z Ax
375 otool.bop_apply(0, 1.0, s, 1);
376 otool.diffwave_apply(0, 1, 1.0, s, 1);
377 dgrid.transform_wave_to_physical(&p.coeffRef(1,0));
378
379 // From Ay in s[2] compute \partial_x Ay
380 otool.bop_apply(0, 1.0, s, 2);
381 otool.diffwave_apply(1, 0, 1.0, s, 2);
382 dgrid.transform_wave_to_physical(&p.coeffRef(2,0));
383
384 // From Ay in s[3] compute \partial_z Ay
385 otool.bop_apply(0, 1.0, s, 3);
386 otool.diffwave_apply(0, 1, 1.0, s, 3);
387 dgrid.transform_wave_to_physical(&p.coeffRef(3,0));
388
389 // From Az in s[4] compute \partial_x Az
390 otool.bop_apply(0, 1.0, s, 4);
391 otool.diffwave_apply(1, 0, 1.0, s, 4);
392 dgrid.transform_wave_to_physical(&p.coeffRef(4,0));
393
394 // From Az in s[5] compute \partial_y Az
395 otool.bop_apply(1, 1.0, s, 5);
396 dgrid.transform_wave_to_physical(&p.coeffRef(5,0));
397
398 // Store curl A in s[{5,6,7}] and find global maximum magnitude of curl A
399 real_t maxmagsquared = 0;
400 for (int offset = 0, j = dgrid.local_physical_start.y();
401 j < dgrid.local_physical_end.y();
402 ++j) {
403
404 for (int k = dgrid.local_physical_start.z();
405 k < dgrid.local_physical_end.z();
406 ++k) {
407
408 for (int i = dgrid.local_physical_start.x();
409 i < dgrid.local_physical_end.x();
410 ++i, /* NB */ ++offset) {
411
412 // Assert curl A is identically zero at the walls
413 if (j == 0 || j == Ny - 1) {
414#pragma warning(push,disable:1572)
415 assert(p(0, offset) == 0.0);
416 assert(p(1, offset) == 0.0);
417 assert(p(2, offset) == 0.0);
418 assert(p(3, offset) == 0.0);
419 assert(p(4, offset) == 0.0);
420 assert(p(5, offset) == 0.0);
421#pragma warning(pop)
422 }
423
424 // The definition of curl gives the following components
425 // 1) \partial_y A_z - \partial_z A_y
426 // 2) \partial_z A_x - \partial_x A_z
427 // 3) \partial_x A_y - \partial_y A_x
428 // where the mean of \partial_x and \partial_z terms must be
429 // zero by periodicity. Components 1 and 3 may have nonzero
430 // mean because they wall-normal derivatives contributions.
431 const Vector3r curlA(p(5, offset) - p(3, offset),
432 p(1, offset) - p(4, offset),
433 p(2, offset) - p(0, offset));
434
435 // 5) Store curl A in physical space in the 3 scalar fields.
436 //
437 // A nonzero mean in the x, y, and z directions is,
438 // respectively, "corrected" by bulk forcing, the wall, and
439 // viscous effects. The spanwise viscous effects presumably
440 // have the slowest timescale so rotate the components from 123
441 // to 312 to reduce the simulation time before stationarity.
442 // This rotation may introduce acoustic noise.
443 p(state_count + 0, offset) = curlA.z();
444 p(state_count + 1, offset) = curlA.x();
445 p(state_count + 2, offset) = curlA.y();
446
447 maxmagsquared = math::maxnan(
448 maxmagsquared, curlA.squaredNorm());
449
450 } // end X
451
452 } // end Z
453
454 } // end Y
455 SUZERAIN_MPICHKR(MPI_Allreduce(MPI_IN_PLACE, &maxmagsquared, 1,
456 mpi::datatype<real_t>::value,
457 MPI_MAX, MPI_COMM_WORLD));
458
459 // Rescale curl A components so max ||curl A|| == maxfluct
460 p.row(state_count + 0) *= (maxfluct / std::sqrt(maxmagsquared));
461 p.row(state_count + 1) *= (maxfluct / std::sqrt(maxmagsquared));
462 p.row(state_count + 2) *= (maxfluct / std::sqrt(maxmagsquared));
463
464 // 6) Copy state into auxiliary state storage and bring to
465 // physical space.
466 for (size_t i = 0; i < state_count; ++i) {
467 s[i] = state[i];
468 otool.bop_apply(0, 1.0, s, i);
469 dgrid.transform_wave_to_physical(&p.coeffRef(i,0));
470 }
471
472 // 7) At each point compute velocity and internal energy. Perturb
473 // velocity and compute new total energy using perturbed
474 // velocities.
475 const real_t Ma = scenario.Ma;
476 for (int offset = 0, j = dgrid.local_physical_start.y();
477 j < dgrid.local_physical_end.y();
478 ++j) {
479
480 for (int k = dgrid.local_physical_start.z();
481 k < dgrid.local_physical_end.z();
482 ++k) {
483
484 for (int i = dgrid.local_physical_start.x();
485 i < dgrid.local_physical_end.x();
486 ++i, /* NB */ ++offset) {
487
488 // Retrieve internal energy
489 real_t e(p(ndx::e, offset));
490 Vector3r m(p(ndx::mx, offset),
491 p(ndx::my, offset),
492 p(ndx::mz, offset));
493 const real_t rho(p(ndx::rho, offset));
494 const real_t e_int = rholut::energy_internal(Ma, rho, m, e);
495
496 // Perturb momentum and compute updated total energy
497 m.x() += rho * p(state_count + 0, offset);
498 m.y() += rho * p(state_count + 1, offset);
499 m.z() += rho * p(state_count + 2, offset);
500 const real_t e_kin = rholut::energy_kinetic(Ma, rho, m);
501 e = e_int + e_kin;
502
503 // Store results back to state fields
504 p(ndx::e, offset) = e;
505 p(ndx::mx, offset) = m.x();
506 p(ndx::my, offset) = m.y();
507 p(ndx::mz, offset) = m.z();
508
509 } // end X
510
511 } // end Z
512
513 } // end Y
514
515 // 8) Bring perturbed state information back to wavespace (no rho!)
516 // Build FFT normalization constant into Y direction's mass matrix.
517 const complex_t scale_factor = 1 / dgrid.chi();
518 massluz.opform(1, &scale_factor, cop);
519 massluz.factor();
520 dgrid.transform_physical_to_wave(&p.coeffRef(ndx::e , 0)); // X, Z
521 dgrid.transform_physical_to_wave(&p.coeffRef(ndx::mx, 0)); // X, Z
522 dgrid.transform_physical_to_wave(&p.coeffRef(ndx::my, 0)); // X, Z
523 dgrid.transform_physical_to_wave(&p.coeffRef(ndx::mz, 0)); // X, Z
524 otool.bop_solve(massluz, s, ndx::e ); // Y
525 otool.bop_solve(massluz, s, ndx::mx); // Y
526 otool.bop_solve(massluz, s, ndx::my); // Y
527 otool.bop_solve(massluz, s, ndx::mz); // Y
528
529 // 9) Overwrite state storage with the new perturbed state (not rho!)
530 state[ndx::e ] = s[ndx::e ];
531 state[ndx::mx] = s[ndx::mx];
532 state[ndx::my] = s[ndx::my];
533 state[ndx::mz] = s[ndx::mz];
534}

536// This looks like logic from navier_stokes.hpp but does not belong there.
537// Reading through that file, especially the apply_operator implementation, is
538// recommended before reviewing this logic. This routine is definitely
539// suboptimal but is expected to be invoked relatively infrequently.
540std::unique_ptr<samples>
541take_samples(const definition_scenario &scenario,
542 const specification_largo &sg,
543 const operator_tools &otool,
544 const bspline &b,
545 contiguous_state<4,complex_t> &swave,
546 const real_t t)
547{
548 // State enters method as coefficients in X, Y, and Z directions
549 SUZERAIN_TIMER_SCOPED("perfect::take_samples");
550
551 // Shorthand for the operator_tools member(s) commonly used
552 const pencil_grid &dgrid = otool.dgrid;
553
554 // We are only prepared to handle a fixed number of fields in this routine
555 enum { state_count = 5 };
556
557 // Shorthand
558 const std::size_t Ny = swave.shape()[1];
559 namespace acc = boost::accumulators;
560 typedef contiguous_state<4,complex_t> state_type;
561
562 // We need auxiliary scalar-field storage. Prepare logical indices using a
563 // struct for scoping (e.g. aux::rho_y). Ordering will match usage below.
564 struct aux { enum {
565 e_y, e_x, e_z,
566 mx_y, mx_x, mx_z,
567 my_y, my_x, my_z,
568 mz_y, mz_x, mz_z,
569 rho_y, rho_x, rho_z,
570 count // Sentry

1024 acc[ref::rho2_dzw](rho2 * grad_u(2,2));

1027 acc[ref::rho_dxu_dyv](rho * grad_u(0,0) * grad_u(1,1));

1030 acc[ref::rho_dyv_dzw](rho * grad_u(1,1) * grad_u(2,2));
1031 acc[ref::rho_dzw_dzw](rho * grad_u(2,2) * grad_u(2,2));

1034 acc[ref::rho2_dxu_dyv](rho2 * grad_u(0,0) * grad_u(1,1));
1035 acc[ref::rho2_dxu_dzw](rho2 * grad_u(0,0) * grad_u(2,2));

1037 acc[ref::rho2_dyv_dzw](rho2 * grad_u(1,1) * grad_u(2,2));
1038 acc[ref::rho2_dzw_dzw](rho2 * grad_u(2,2) * grad_u(2,2));

1041 largo_state state(e, m.x(), m.y(), m.z(), rho, p);
1042 largo_state slowgrowth_f;

1044

1046 acc[ref::SrhoE ](slowgrowth_f.e );
1047 acc[ref::Srhou ](slowgrowth_f.mx );
1048 acc[ref::Srhov ](slowgrowth_f.my );
1049 acc[ref::Srhow ](slowgrowth_f.mz );
1050 acc[ref::Srho ](slowgrowth_f.rho);
1051 acc[ref::Srhou_dot_u]( slowgrowth_f.mx*u.x()
1052 + slowgrowth_f.my*u.y()
1053 + slowgrowth_f.mz*u.z());

1055 // Gather squared slow growth forcing to permit computing variances
1056 acc[ref::S2rhoE ](std::pow(slowgrowth_f.e , 2));
1057 acc[ref::S2rhou ](std::pow(slowgrowth_f.mx , 2));
1058 acc[ref::S2rhov ](std::pow(slowgrowth_f.my , 2));
1059 acc[ref::S2rhow ](std::pow(slowgrowth_f.mz , 2));
1060 acc[ref::S2rho ](std::pow(slowgrowth_f.rho, 2));
1061 acc[ref::S2rhou_dot_u](std::pow( slowgrowth_f.mx*u.x()
1062 + slowgrowth_f.my*u.y()
1063 + slowgrowth_f.mz*u.z(), 2));
1064 } // end X // end Z
1065
1066 // All accumulators should have seen a consistent number of samples
1067#ifndef NDEBUG
1068 const std::size_t miscreant = inconsistent_accumulation_count(acc);
1069 assert(!miscreant);
1070#endif
1071
1072 // Extract y-specific sums into MPI-reduction-ready storage for y(j)
1073 using boost::accumulators::sum;
1074#define EXTRACT(quantity, component, offset, description) \
1075 ret->quantity()(j, offset) = sum(acc[ref::component]);
1076 SUZERAIN_SAMPLES_COMPONENTS_FOR_EACH(
1077 EXTRACT, SUZERAIN_SAMPLES_PHYSICAL)
1078#undef EXTRACT
1079
1080 } // end Y
1081
1082 // Notice dgrid.rank_zero_zero_modes already contains "wave-sampled"
1083 // quantities while other ranks have zeros in those locations.
1084
1085 // Allreduce to obtain global sums on every rank
1086 SUZERAIN_MPICHKR(MPI_Allreduce(
1087 MPI_IN_PLACE, ret->storage.data(), ret->storage.size(),
1088 mpi::datatype<samples::storage_type::Scalar>::value,
1089 MPI_SUM, MPI_COMM_WORLD));
1090
1091 // Physical space sums, which are at collocation points, need to be
1092 // scaled by the dealiased extents and converted to coefficients.
1093 ret->physical() *= dgrid.chi();
1094 otool.masslu()->solve(ret->physical().outerSize(),
1095 ret->physical().data(),
1096 ret->physical().innerStride(),
1097 ret->physical().outerStride());
1098
1099 // Fill with NaNs those samples that were not computed by this method
1100 ret->implicit().setConstant(std::numeric_limits<real_t>::quiet_NaN());
1101
1102 return ret;
1103}

1109 const operator_tools &otool,
1110 contiguous_state<4,complex_t> &swave)
1111{
1112 // State enters method as coefficients in X, Y, and Z directions
1113 SUZERAIN_TIMER_SCOPED("perfect::take_profile");
1114
1115 // Shorthand for the operator_tools member(s) commonly used
1116 const pencil_grid &dgrid = otool.dgrid;

1119 enum { state_count = 5 };
1120 assert(static_cast<int>(ndx::e ) < state_count);
1121 assert(static_cast<int>(ndx::mx ) < state_count);
1122 assert(static_cast<int>(ndx::my ) < state_count);

1129 SUZERAIN_ENSURE(numeric_cast<int>(swave.shape()[1]) == dgrid.local_wave_extent.y());
1130 SUZERAIN_ENSURE(numeric_cast<int>(swave.shape()[2]) == dgrid.local_wave_extent.x());

1137 std::unique_ptr<profile> ret = std::make_unique<profile>(Ny);

1139 // Obtain samples available in wave-space from mean conserved state.
1140 // These coefficients are inherently averaged across the X-Z plane.
1141 if (dgrid.has_zero_zero_modes()) {
1142 ret->rho_u().col(0) = Map<VectorXc>(swave[ndx::mx ].origin(), Ny).real();
1143 ret->rho_u().col(1) = Map<VectorXc>(swave[ndx::my ].origin(), Ny).real();

1150 otool.zero_dealiasing_modes(swave, i);

1152 dgrid.transform_wave_to_physical(&sphys.coeffRef(i,0));
1153 }
1154
1155 // Physical space is traversed linearly using a single offset 'offset'.
1156 // The two loop structure is used as positions x(i), and z(k) unnecessary.
1157 for (int offset = 0, j = dgrid.local_physical_start.y();
1158 j < dgrid.local_physical_end.y();
1159 ++j) {
1160
1161 // Type used to accumulate means versus wall-normal position
1162 typedef boost::accumulators::accumulator_set<
1163 real_t, boost::accumulators::stats<
1164 boost::accumulators::tag::sum_kahan
1165 >
1166 > accumulator_type;
1167
1168 // Accumulators for each mean quantity computed in physical space.
1169 // For example, quantity "foo" has accumulator "sum_foo".
1170 // Declared within Y loop so they are reset on each Y iteration.
1171#define DECLARE(r, data, tuple) \
1172 accumulator_type BOOST_PP_CAT(sum_,BOOST_PP_TUPLE_ELEM(2, 0, tuple)) \
1173 [BOOST_PP_TUPLE_ELEM(2, 1, tuple)];
1174 BOOST_PP_SEQ_FOR_EACH(DECLARE,, SUZERAIN_PROFILE_PHYSICAL)
1175#undef DECLARE
1176
1177 // Iterate across the j-th ZX plane
1178 const int last_zxoffset = offset
1179 + dgrid.local_physical_extent.z()
1180 * dgrid.local_physical_extent.x();

1182
1183 // Unpack conserved state
1184 const real_t e(sphys(ndx::e, offset));
1185 const Vector3r m(sphys(ndx::mx, offset),
1186 sphys(ndx::my, offset),
1187 sphys(ndx::mz, offset));
1188 const real_t rho(sphys(ndx::rho, offset));

1190 // Compute quantities of interest
1191 const Vector3r u = rholut::u(rho, m);
1192 real_t p, T, mu, lambda;
1193 rholut::p_T_mu_lambda(scenario.alpha, scenario.beta,
1194 scenario.gamma, scenario.Ma,
1195 rho, m, e, p, T, mu, lambda);
1196
1197 // Accumulate into sum_XXX using function syntax.
1198 sum_a [0](std::sqrt(T));
1199 sum_H0[0]((e + p) / rho);
1200 sum_ke[0](u.squaredNorm() / 2);
1201 sum_mu[0](mu);
1202 sum_T [0](T);
1203 sum_u [0](u.x());
1204 sum_u [1](u.y());
1205
1206 } // end X // end Z
1207
1208 // Move y-specific sums into MPI-reduction-ready storage for y(j) using
1209 // Eigen comma initialization syntax. Yes, this is not stride 1.
1210#define EXTRACT_SUM(z, n, q) boost::accumulators::sum(sum_##q[n])
1211#define MOVE_SUM_INTO_TMP(r, data, tuple) \
1212 ret->BOOST_PP_TUPLE_ELEM(2, 0, tuple)().row(j) << \
1213 BOOST_PP_ENUM(BOOST_PP_TUPLE_ELEM(2, 1, tuple), \
1214 EXTRACT_SUM, BOOST_PP_TUPLE_ELEM(2, 0, tuple));
1215 BOOST_PP_SEQ_FOR_EACH(MOVE_SUM_INTO_TMP,, SUZERAIN_PROFILE_PHYSICAL)
1216#undef EXTRACT_SUM
1217#undef MOVE_SUM_INTO_TMP
1218
1219 } // end Y
1220
1221 // Notice dgrid.rank_zero_zero_modes already contains "wave-sampled"
1222 // profile while other ranks have zeros in those locations.
1223
1224 // Allreduce to obtain global sums on every rank
1225 SUZERAIN_MPICHKR(MPI_Allreduce(
1226 MPI_IN_PLACE, ret->storage.data(), ret->storage.size(),
1227 mpi::datatype<profile::storage_type::Scalar>::value,
1228 MPI_SUM, MPI_COMM_WORLD));
1229
1230 // Physical space sums, which are at collocation points, need to be
1231 // scaled by the dealiased extents and converted to coefficients.
1232 ret->physical() *= dgrid.chi();
1233 otool.masslu()->solve(ret->physical().outerSize(),
1234 ret->physical().data(),
1235 ret->physical().innerStride(),
1236 ret->physical().outerStride());
1237
1238 // BEWARE: The 4-argument take_profile(...) below relies on
1239 // the fact that swave has been converted to physical space
1240 // as of the end of this method.
1241
1242 return ret;
1243}

1370 acc[references::q::rhow ](m.z() );
1371 acc[references::q::rhoE ](e );

1373 acc[references::q::rhouv](m.x()* m.y() / rho);

1391 refs.row(i)[j] = boost::accumulators::sum(acc[i]);
1392 }
1393
1394 } // end Y
1395
1396 // Allreduce and scale refs sums to obtain means on all ranks
1397 SUZERAIN_MPICHKR(MPI_Allreduce(MPI_IN_PLACE, refs.data(), refs.size(),
1398 mpi::datatype<references::Scalar>::value, MPI_SUM, MPI_COMM_WORLD));

1504compute_fluctuations(const bsplineop& cop,

1508 ArrayX6r &favre,
1509 ArrayX3r &prodterms)

1512 reynolds.resize(inst.rows(), NoChange);
1513 reynolds.col(0) = inst.uu() - inst.u().square();

1519
1520 // Prepare pointwise \bar{\rho} \widetilde{u_i'' u_j''}
1521 // which is NOT the final result for the method
1522 favre.resize(inst.rows(), NoChange);
1523 favre.col(0)=(inst.rhouu() - inst.rhou().square() /inst.rho());
1524 favre.col(1)=(inst.rhouv() - inst.rhou()*inst.rhov()/inst.rho());
1525 favre.col(2)=(inst.rhouw() - inst.rhou()*inst.rhow()/inst.rho());
1526 favre.col(3)=(inst.rhovv() - inst.rhov().square() /inst.rho());
1527 favre.col(4)=(inst.rhovw() - inst.rhov()*inst.rhow()/inst.rho());
1528 favre.col(5)=(inst.rhovw() - inst.rhow().square() /inst.rho());
1529
1530 // Compute the three separate contributions to production:
1531 // - \bar{\rho} \widetilde{u''\otimes{}u''} : \nabla\tilde{u} =
1532 // - bar_rho(y)*( tilde_upp_vpp*tilde_u__y
1533 // + tilde_vpp_vpp*tilde_v__y
1534 // + tilde_vpp_wpp*tilde_w__y )
1535 //
1536 // Begin by forming pointwise \partial_y \tilde{u}_i
1537 prodterms.resize(inst.rows(), NoChange);
1538 prodterms.col(0) = inst.rhou();
1539 prodterms.col(1) = inst.rhov();
1540 prodterms.col(2) = inst.rhow();
1541 prodterms.colwise() /= inst.rho();
1542 masslu.solve(prodterms.cols(), prodterms.data(),
1543 prodterms.innerStride(), prodterms.outerStride());
1544 cop.apply(1, prodterms.cols(), 1.0, prodterms.data(),
1545 prodterms.innerStride(), prodterms.outerStride());
1546 // End by scaling with the relevant fluctuation data.
1547 // Notice \bar{\rho} already present in contents of tmp.
1548 prodterms.col(0) *= - favre.col(1);
1549 prodterms.col(1) *= - favre.col(3);
1550 prodterms.col(2) *= - favre.col(4);
1551
1552 // Divide \bar{\rho} \widetilde{u_i'' u_j''} by \bar{\rho}
1553 // to get the desired Favre-averaged fluctuations.
1554 favre.colwise() /= inst.rho();
1555}

1559 const definition_scenario &scenario,
1560 const std::shared_ptr<specification_largo> &sg,

1592 wall.mu = prof.mu()[0];
1593 wall.Pr = scenario.Pr;
1594 wall.rho = prof.rho()[0];
1595 assert(&(wall.T) + 1 == &(wall.T__y)); // Next, compute both T and T__y
1596 b.linear_combination(1, prof.T().col(0).data(), 0.0, &(wall.T), 1);
1597 assert(&(wall.u) + 1 == &(wall.u__y)); // Next, compute both u and u__y
1598 b.linear_combination(1, prof.u().col(0).data(), 0.0, &(wall.u), 1);
1599 wall.v = prof.u().col(1)[0];
1600 wall.y = 0.0;
1601
1602 // Compute viscous quantities based only on wall information
1603 if (const int err = suzerain_bl_compute_viscous(
1604 scenario.Re,
1605 &wall,
1606 &viscous)) {
1607 WARNONCE0("profile",
1608 "suzerain_bl_compute_viscous(...) returned "
1609 << err << " (" << suzerain_strerror(err) << ")");
1610 }
1611
1612 // Prepare inviscid baseflow coefficients if necessary
1613 ArrayX4r coeffs_inviscid; // Columns are H0, rhou, u, v
1614 if (sg && sg->formulation.enabled() && sg->baseflow) {
1615 coeffs_inviscid.resize(b.n(), NoChange);
1616
1617 // Retrieve values on collocation points
1618 largo_state state;
1619 baseflow_interface &baseflow = *sg->baseflow;
1620 for (int j = 0; j < b.n(); ++j) {
1621 const real_t y_j = b.collocation_point(j);
1622 baseflow.conserved(y_j, state.as_is());
1623 baseflow.pressure (y_j, state.p);
1624 coeffs_inviscid(j, 0) = state.H0();
1625 coeffs_inviscid(j, 1) = state.mx;
1626 coeffs_inviscid(j, 2) = state.u();
1627 coeffs_inviscid(j, 3) = state.v();
1628 }

1631 masslu.solve(coeffs_inviscid.outerSize(),
1632 coeffs_inviscid.data(),
1633 coeffs_inviscid.innerStride(),
1634 coeffs_inviscid.outerStride());

1641 prof.ke().data(),
1642 prof.rho_u().col(0).data(),
1643 prof.u().col(0).data(),
1644 &thick,
1645 b.bw)) {

1652 prof.H0().data(),
1653 prof.ke().data(),
1654 prof.rho_u().col(0).data(),
1655 prof.u().col(0).data(),
1656 coeffs_inviscid.innerStride(),
1657 coeffs_inviscid.col(0).data(),
1658 coeffs_inviscid.col(1).data(),
1659 coeffs_inviscid.col(2).data(),
1660 coeffs_inviscid.col(3).data(),
1661 &thick,
1662 b.bw)) {
1663 WARNONCE0("profile",

1666 }

1670 // Evaluate state at the edge99 (y=thick.delta99) from B-spline coefficients
1671 // As this is futzy, use a loop to repeat the computations for each target
1672 {
1673 const double y[2] = { thick.delta, thick.delta99 };
1674 suzerain_bl_local *e[2] = { &edge, &edge99 };
1675 for (size_t i = 0; i < sizeof(y)/sizeof(y[0]); ++i) {
1676 std::fill_n(reinterpret_cast<double *>(e[i]),
1677 sizeof(*e[i])/sizeof(double),
1678 std::numeric_limits<double>::quiet_NaN());
1679 e[i]->gamma = scenario.gamma;
1680 e[i]->Pr = scenario.Pr;
1681 e[i]->y = y[i];

1683 // NOP as fill_n above ensured NaN propagated correctly
1684 } else {
1685 // Later, could more quickly evaluate basis once and then re-use that
1686 // result to repeatedly form the necessary linear combinations.
1687 b.linear_combination(0, prof.a().data(), y[i], &e[i]->a);
1688 b.linear_combination(0, prof.mu().data(), y[i], &e[i]->mu);
1689 b.linear_combination(0, prof.rho().data(), y[i], &e[i]->rho);
1690 assert(&e[i]->T + 1 == &e[i]->T__y); // Next, both T and T__y
1691 b.linear_combination(1, prof.T().col(0).data(), y[i], &e[i]->T, 1);
1692 assert(&e[i]->u + 1 == &e[i]->u__y); // Next, both u and u__y
1693 b.linear_combination(1, prof.u().col(0).data(), y[i], &e[i]->u, 1);
1694 b.linear_combination(0, prof.u().col(1).data(), y[i], &e[i]->v);
1695 }
1696 }
1697 }
1698
1699 // Compute Reynolds numbers
1700 if (0 == coeffs_inviscid.size()) {
1701 if (const int err = suzerain_bl_compute_reynolds(
1702 scenario.Re,
1703 &edge,
1704 &edge99,
1705 &thick,
1706 &reynolds)) {
1707 WARNONCE0("profile",
1708 "suzerain_bl_compute_reynolds(...) returned "
1709 << err << " (" << suzerain_strerror(err) << ")");
1710 }
1711 } else {
1712 if (const int err = suzerain_bl_compute_reynolds_baseflow(
1713 scenario.Re,
1714 prof.H0().data(),
1715 prof.ke().data(),
1716 prof.rho_u().col(0).data(),
1717 prof.u().col(0).data(),
1718 1,
1719 coeffs_inviscid.col(0).data(),
1720 coeffs_inviscid.col(1).data(),
1721 coeffs_inviscid.col(2).data(),
1722 coeffs_inviscid.col(3).data(),
1723 &edge,
1724 &edge99,
1725 &reynolds,
1726 b.bw)) {
1727 WARNONCE0("profile",
1728 "suzerain_bl_compute_reynolds_baseflow(...) returned "
1729 << err << " (" << suzerain_strerror(err) << ")");
1730 }
1731 }
1732
1733 // Compute general quantities of interest
1734 suzerain_bl_compute_qoi(scenario.Ma, scenario.Re,
1735 &wall, &viscous, &edge99, &thick, &qoi);
1736
1737 // Mean pressure and streamwise velocity gradients come from slow growth
1738 double edge99_p__x = 0, edge99_u__x = 0;
1739 if (sg && sg->formulation.enabled() && sg->baseflow) {
1740 const double delta99 = thick.delta99;
1741 if (SUZERAIN_UNLIKELY((isnan)(delta99))) {
1742 edge99_p__x = edge99_u__x = std::numeric_limits<real_t>::quiet_NaN();
1743 } else {
1744 largo_state base, dy, dx; // as_is()
1745 sg->baseflow->conserved(delta99,
1746 base.as_is(), dy.as_is(), dx.as_is());
1747 sg->baseflow->pressure (delta99,
1748 base.p, dy.p, dx.p);
1749 edge99_p__x = dx.p; // Direct
1750 edge99_u__x = (dx.mx - dx.rho*base.mx/base.rho)/base.rho; // Chained
1751 }
1752 }
1753 if (const int err = suzerain_bl_compute_pg(
1754 scenario.Ma,
1755 scenario.Re,
1756 &wall,
1757 &viscous,
1758 &edge99,
1759 edge99_p__x,
1760 edge99_u__x,
1761 &thick,
1762 &pg)) {
1763 WARNONCE0("profile",
1764 "suzerain_bl_compute_pg(...) returned "
1765 << err << " (" << suzerain_strerror(err) << ")");
1766 }
1767
1768# undef WARNONCE0
1769}
