49struct lazy_pow_
50{
51 template <typename X, typename Y>
52 struct result { typedef X type; };
53
54 template <typename X, typename Y>
55 X operator()(X x, Y y) const

172 boost::spirit::qi::rule<
173 Iterator, FPT(), boost::spirit::ascii::space_type
174 > expression, term, factor, primary;
175
176 grammar() : grammar::base_type(expression)
177 {
178 using boost::spirit::qi::real_parser;
179 using boost::spirit::qi::real_policies;
180 real_parser<FPT,real_policies<FPT> > real;
181
182 using boost::spirit::qi::_1;
183 using boost::spirit::qi::_2;
184 using boost::spirit::qi::_3;
185 using boost::spirit::qi::no_case;
186 using boost::spirit::qi::_val;
187
188 boost::phoenix::function<lazy_pow_> lazy_pow;
189 boost::phoenix::function<lazy_ufunc_> lazy_ufunc;
190 boost::phoenix::function<lazy_bfunc_> lazy_bfunc;
191
192 expression =
193 term [_val = _1]
194 >> *( ('+' >> term [_val += _1])
195 | ('-' >> term [_val -= _1])
196 )
197 ;
198
199 term =
200 factor [_val = _1]
201 >> *( ('*' >> factor [_val *= _1])
202 | ('/' >> factor [_val /= _1])
203 )
204 ;
205
206 factor =
207 primary [_val = _1]
208 >> *( ("**" >> factor [_val = lazy_pow(_val, _1)])
209 )
210 ;
211
212 primary =
213 real [_val = _1]
214 | '(' >> expression [_val = _1] >> ')'
215 | ('-' >> primary [_val = -_1])
216 | ('+' >> primary [_val = _1])
217 | (no_case[ufunc] >> '(' >> expression >> ')')
218 [_val = lazy_ufunc(_1, _2)]
219 | (no_case[bfunc] >> '(' >> expression >> ','
220 >> expression >> ')')
221 [_val = lazy_bfunc(_1, _2, _3)]
222 | no_case[constant] [_val = _1]
223 ;
224
225 }
226};
