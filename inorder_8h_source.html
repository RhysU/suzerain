91inline
92int suzerain_inorder_wavenumber(const int N, const int i)

179inline
180int suzerain_inorder_wavenumber_nyquist(const int N, const int w)
181{
182 // Deliberately no assertions on out-of-range wavenumbers!
183 return (w == N/2)*(1 - (N % 2));
184}

257inline
258int suzerain_inorder_wavenumber_valid(const int N, const int w)
259{
260 return suzerain_inorder_wavenumber_min(N) <= w

281inline
282int suzerain_inorder_wavenumber_diff(const int N, const int dN, const int i)
283{
284 assert(N <= dN);
285 assert(0 <= i && i < dN);
286 if (i < (N+1)/2) {
287 return i;
288 } else if (i >= dN - (N-1)/2) {
289 return -dN+i;
290 } else {
291 return 0;
292 }
293}
