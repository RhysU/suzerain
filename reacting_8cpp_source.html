72void add_species_fields( const std::vector<std::string>& species_names,
73 std::vector<support::field>& fields )
74{
75 // Nothing to do if we only have 1 specie
76 if (species_names.size()==1) return;
77
78 // number of species
79 const size_t Ns = species_names.size();
80
81 // number of other variables
82 const size_t Nv = fields.size();
83
84 // total number of field variables we need
85 const size_t Nf = Nv + Ns - 1;
86
87 // get space for species
88 fields.resize(Nf);
89
90 // initialize species
91 for (size_t i = Nv; i < Nf; ++i) {
92 // yes: +1 b/c first in species list is the diluter
93 fields[i].identifier = "rho_" + species_names[i-Nv+1];
94 fields[i].description += "Dimensional ";
95 fields[i].description += species_names[i-Nv+1];
96 fields[i].description += " density";
97 fields[i].location = "rho_" + species_names[i-Nv+1];
98 }

196void
197add_noise(contiguous_state<4,complex_t> &state,
198 const specification_noise& noise,
199 const specification_grid& grid,
200 const pencil_grid& dgrid,
201 const bsplineop& cop,
202 bspline &b)
203{
204 // FIXME Needs to be made aware of channel vs flat plate
205
206 // We are only prepared to handle a fixed number of fields in this routine
207 enum { state_count = 5 };
208
209 // Ensure state storage meets this routine's assumptions
210 SUZERAIN_ENSURE( state.shape()[0] == state_count);
211 SUZERAIN_ENSURE(numeric_cast<int>(state.shape()[1]) == dgrid.local_wave_extent.y());
212 SUZERAIN_ENSURE(numeric_cast<int>(state.shape()[2]) == dgrid.local_wave_extent.x());
213 SUZERAIN_ENSURE(numeric_cast<int>(state.shape()[3]) == dgrid.local_wave_extent.z());
214
215 // Ensure we were handed collocation-based operator matrices
216 SUZERAIN_ENSURE(cop.get()->method == SUZERAIN_BSPLINEOP_COLLOCATION_GREVILLE);
217
218 const int Ny = grid.N.y();
219
220#pragma warning(push,disable:1572)
221 if (noise.percent == 0) {
222#pragma warning(pop)
223 DEBUG0("Zero noise added to velocity fields");
224 return;
225 }
226
227 using inorder::wavenumber_abs;
228 using inorder::wavenumber_max;
229 using inorder::wavenumber_translatable;
230 const real_t twopi = 2 * boost::math::constants::pi<real_t>();
231
232 // Evaluate maximum fluctuation magnitude based on percentage of
233 // centerline mean streamwise velocity and broadcast result.
234 // Alright, alright... actually an approximate mean velocity.
235 real_t maxfluct;
236 if (dgrid.has_zero_zero_modes()) {
237 complex_t momentum, density;
238 const real_t centerline = grid.L.y() / 2;
239 b.linear_combination(
240 0, &state[ndx::mx][0][0][0], centerline, &momentum);
241 INFO0("Centerline mean streamwise momentum at y = "
242 << centerline << " is " << momentum);
243 b.linear_combination(
244 0, &state[ndx::rho][0][0][0], centerline, &density);
245 INFO0("Centerline mean density at y = "
246 << centerline << " is " << density);
247 maxfluct = noise.percent / 100 * (abs(momentum) / abs(density));
248 }
249 SUZERAIN_MPICHKR(MPI_Bcast(&maxfluct, 1,
250 mpi::datatype_of(maxfluct),
251 dgrid.rank_zero_zero_modes, MPI_COMM_WORLD));
252 INFO0("Adding velocity perturbations with maximum magnitude " << maxfluct);
253
254 // Compute and display kxfrac_min, kxfrac_max constraints
255#pragma warning(push,disable:2259)
256 const int dkx_max = noise.kxfrac_max * wavenumber_max(grid.N.x());
257 const int dkx_min = noise.kxfrac_min * wavenumber_max(grid.N.x());
258#pragma warning(pop)
259#pragma warning(push,disable:1572)
260 if (noise.kxfrac_max != 1 || noise.kxfrac_min != 0) {
261#pragma warning(pop)
262 INFO0("Perturbations added only to absolute X wavenumbers in range ["
263 << dkx_min << ":" << dkx_max << "]");
264 }
265
266 // Compute and display kzfrac_min, kzfrac_max constraints
267#pragma warning(push,disable:2259)
268 const int dkz_max = noise.kzfrac_max * wavenumber_max(grid.N.z());
269 const int dkz_min = noise.kzfrac_min * wavenumber_max(grid.N.z());
270#pragma warning(pop)
271#pragma warning(push,disable:1572)
272 if (noise.kzfrac_max != 1 || noise.kzfrac_min != 0) {
273#pragma warning(pop)
274 INFO0("Perturbations added only to absolute Z wavenumbers in range ["
275 << dkz_min << ":" << dkz_max << "]");
276 }
277
278 // Form mass matrix to convert (wave, collocation point values, wave)
279 // perturbations to (wave, coefficients, wave)
280 bsplineop_luz massluz(cop);
281 massluz.factor_mass(cop);
282
283 // Set L'Ecuyer et al.'s rngstream seed. Use a distinct Substream for each
284 // wall-normal pencil to ensure process is a) repeatable despite changes in
285 // processor count, b) easy to code, and c) embarrassingly parallel.
286 rngstream rng;
287 {
288 array<unsigned long,6> seed;
289 std::fill(seed.begin(), seed.end(), noise.seed);
290 rng.SetSeed(seed.data());
291 }
292 rng.IncreasedPrecis(true); // Use more bits of resolution
293
294 // Approach is the following:
295 // 0) Allocate storage for state and three additional scalar fields.
296 // 1) Generate a random vector-valued field \tilde{A}.
297 // \tilde{A}'s x- and z-derivatives have zero mean by periodicity;
298 // 2) Zero first two B-spline coefficients near walls so partial_y
299 // \tilde{A} vanishes at the wall
300 // 3) This step intentionally left blank.
301 // 4) Compute curl A in physical space and rescale so maximum
302 // pointwise norm of A is maxfluct. curl A is solenoidal
303 // and now has the velocity perturbation properties we desire.
304 // 5) Store curl A in physical space in the three scalar fields.
305 // 6) Copy state into auxiliary state storage and bring to
306 // physical space.
307 // 7) At each point compute velocity and internal energy. Perturb
308 // velocity and compute new total energy using perturbed
309 // velocities.
310 // 8) Bring perturbed state information back to wavespace.
311 // 9) Overwrite state storage with the new perturbed state.
312
313 // 0) Allocate storage for state and three additional scalar fields.
314 std::unique_ptr<contiguous_state<4,complex_t> > _s_ptr( // RAII
315 support::allocate_padded_state<contiguous_state<4,complex_t> >(
316 state_count + 3, dgrid));
317 contiguous_state<4,complex_t> &s = *_s_ptr; // Shorthand
318 std::fill(s.range().begin(), s.range().end(), 0); // Zero memory
319
320 // 1) Generate a random vector-valued field \tilde{A}.
321 // For each scalar component of \tilde{A}...
322 for (size_t l = 0; l < 3; ++l) {
323
324 for (int k = 0; k < grid.dN.z(); ++k) {
325 if (!wavenumber_translatable(grid.N.z(), grid.dN.z(), k)) continue;
326
327
328 for (int i = 0; i < grid.dN.x(); ++i) {
329 if (!wavenumber_translatable(grid.N.x(), grid.dN.x(), i)) continue;
330
331 // ...and advance rngstream to the (i, ., k) substream...
332 // ...(necessary for processor-topology independence)...
333 rng.ResetNextSubstream();
334
335 // ...but only the rank holding the (i, ., k) pencil continues.
336 if ( k < dgrid.local_wave_start.z()
337 || k >= dgrid.local_wave_end.z()
338 || i < dgrid.local_wave_start.x()
339 || i >= dgrid.local_wave_end.x()) continue;
340
341 // Satisfy fluct_kzfrac_min, fluct_kzfrac_max constraints
342 if (wavenumber_abs(grid.dN.z(), k) < dkz_min) continue;
343 if (wavenumber_abs(grid.dN.z(), k) > dkz_max) continue;
344
345 // Satisfy fluct_kxfrac_min, fluct_kxfrac_max constraints
346 if (wavenumber_abs(grid.dN.x(), i) < dkx_min) continue;
347 if (wavenumber_abs(grid.dN.x(), i) > dkx_max) continue;
348
349 // Compute local indices for global (i, ., k).
350 const int local_i = i - dgrid.local_wave_start.x();
351 const int local_k = k - dgrid.local_wave_start.z();
352
353 // ...generate coeffs with well-defined pseudorandom order.
354 // 2) Zero first two B-spline coefficients near walls
355 // so partial_y \tilde{A} vanishes at the wall
356 s[2*l][0][local_i][local_k] = 0;
357 s[2*l][1][local_i][local_k] = 0;
358 for (int j = 2; j < Ny - 2; ++j) {
359 const real_t magnitude = rng.RandU01();
360 const real_t phase = rng.RandU01() * twopi;
361 s[2*l][j][local_i][local_k] = std::polar(magnitude, phase);
362 }
363 s[2*l][Ny - 2][local_i][local_k] = 0;
364 s[2*l][Ny - 1][local_i][local_k] = 0;
365
366 } // end X
367
368 } // end Z
369
370 } // end scalar components of A
371
372 // 4) Compute curl A in physical space and rescale so maximum
373 // pointwise norm of A is maxfluct. curl A is solenoidal
374 // and now has the velocity perturbation properties we desire.
375
376 // Copy s[2l] into s[2l+1] as we need two copies to compute curl A
377 for (size_t l = 0; l < 3; ++l) s[2*l+1] = s[2*l];
378
379 // Prepare physical-space view of the wave-space storage
380 physical_view<state_count+3> p(dgrid, s);
381
382 // Initializing operator_base to access decomposition-ready utilities
383 operator_tools otool(grid, dgrid, cop);
384
385 // From Ax in s[0] compute \partial_y Ax
386 otool.bop_apply(1, 1.0, s, 0);
387 dgrid.transform_wave_to_physical(&p.coeffRef(0,0));
388
389 // From Ax in s[1] compute \partial_z Ax
390 otool.bop_apply(0, 1.0, s, 1);
391 otool.diffwave_apply(0, 1, 1.0, s, 1);
392 dgrid.transform_wave_to_physical(&p.coeffRef(1,0));
393
394 // From Ay in s[2] compute \partial_x Ay
395 otool.bop_apply(0, 1.0, s, 2);
396 otool.diffwave_apply(1, 0, 1.0, s, 2);
397 dgrid.transform_wave_to_physical(&p.coeffRef(2,0));
398
399 // From Ay in s[3] compute \partial_z Ay
400 otool.bop_apply(0, 1.0, s, 3);
401 otool.diffwave_apply(0, 1, 1.0, s, 3);
402 dgrid.transform_wave_to_physical(&p.coeffRef(3,0));
403
404 // From Az in s[4] compute \partial_x Az
405 otool.bop_apply(0, 1.0, s, 4);
406 otool.diffwave_apply(1, 0, 1.0, s, 4);
407 dgrid.transform_wave_to_physical(&p.coeffRef(4,0));
408
409 // From Az in s[5] compute \partial_y Az
410 otool.bop_apply(1, 1.0, s, 5);
411 dgrid.transform_wave_to_physical(&p.coeffRef(5,0));
412
413 // Store curl A in s[{5,6,7}] and find global maximum magnitude of curl A
414 real_t maxmagsquared = 0;
415 for (int offset = 0, j = dgrid.local_physical_start.y();
416 j < dgrid.local_physical_end.y();
417 ++j) {
418
419 for (int k = dgrid.local_physical_start.z();
420 k < dgrid.local_physical_end.z();
421 ++k) {
422
423 for (int i = dgrid.local_physical_start.x();
424 i < dgrid.local_physical_end.x();
425 ++i, /* NB */ ++offset) {
426
427 // Assert curl A is identically zero at the walls
428 if (j == 0 || j == Ny - 1) {
429#pragma warning(push,disable:1572)
430 assert(p(0, offset) == 0.0);
431 assert(p(1, offset) == 0.0);
432 assert(p(2, offset) == 0.0);
433 assert(p(3, offset) == 0.0);
434 assert(p(4, offset) == 0.0);
435 assert(p(5, offset) == 0.0);
436#pragma warning(pop)
437 }
438
439 // The definition of curl gives the following components
440 // 1) \partial_y A_z - \partial_z A_y
441 // 2) \partial_z A_x - \partial_x A_z
442 // 3) \partial_x A_y - \partial_y A_x
443 // where the mean of \partial_x and \partial_z terms must be
444 // zero by periodicity. Components 1 and 3 may have nonzero
445 // mean because they wall-normal derivatives contributions.
446 const Vector3r curlA(p(5, offset) - p(3, offset),
447 p(1, offset) - p(4, offset),
448 p(2, offset) - p(0, offset));
449
450 // 5) Store curl A in physical space in the 3 scalar fields.
451 //
452 // A nonzero mean in the x, y, and z directions is,
453 // respectively, "corrected" by bulk forcing, the wall, and
454 // viscous effects. The spanwise viscous effects presumably
455 // have the slowest timescale so rotate the components from 123
456 // to 312 to reduce the simulation time before stationarity.
457 // This rotation may introduce acoustic noise.
458 p(state_count + 0, offset) = curlA.z();
459 p(state_count + 1, offset) = curlA.x();
460 p(state_count + 2, offset) = curlA.y();
461
462 maxmagsquared = math::maxnan(
463 maxmagsquared, curlA.squaredNorm());
464
465 } // end X
466
467 } // end Z
468
469 } // end Y
470 SUZERAIN_MPICHKR(MPI_Allreduce(MPI_IN_PLACE, &maxmagsquared, 1,
471 mpi::datatype<real_t>::value,
472 MPI_MAX, MPI_COMM_WORLD));
473
474 // Rescale curl A components so max ||curl A|| == maxfluct
475 p.row(state_count + 0) *= (maxfluct / std::sqrt(maxmagsquared));
476 p.row(state_count + 1) *= (maxfluct / std::sqrt(maxmagsquared));
477 p.row(state_count + 2) *= (maxfluct / std::sqrt(maxmagsquared));
478
479 // 6) Copy state into auxiliary state storage and bring to
480 // physical space.
481 for (size_t i = 0; i < state_count; ++i) {
482 s[i] = state[i];
483 otool.bop_apply(0, 1.0, s, i);
484 dgrid.transform_wave_to_physical(&p.coeffRef(i,0));
485 }
486
487 // 7) At each point compute velocity and internal energy. Perturb
488 // velocity and compute new total energy using perturbed
489 // velocities.
490
491 for (int offset = 0, j = dgrid.local_physical_start.y();
492 j < dgrid.local_physical_end.y();
493 ++j) {
494
495 for (int k = dgrid.local_physical_start.z();
496 k < dgrid.local_physical_end.z();
497 ++k) {
498
499 for (int i = dgrid.local_physical_start.x();
500 i < dgrid.local_physical_end.x();
501 ++i, /* NB */ ++offset) {
502
503 // Retrieve internal energy
504 real_t e(p(ndx::e, offset));
505 Vector3r m(p(ndx::mx, offset),
506 p(ndx::my, offset),
507 p(ndx::mz, offset));
508 const real_t rho(p(ndx::rho, offset));
509 // Setting Ma=1 makes this correct for dimensional case
510 const real_t e_int = rholut::energy_internal(1.0, rho, m, e);
511
512 // Perturb momentum and compute updated total energy
513 m.x() += rho * p(state_count + 0, offset);
514 m.y() += rho * p(state_count + 1, offset);
515 m.z() += rho * p(state_count + 2, offset);
516 // Setting Ma=1 makes this correct for dimensional case
517 const real_t e_kin = rholut::energy_kinetic(1.0, rho, m);
518 e = e_int + e_kin;
519
520 // Store results back to state fields
521 p(ndx::e, offset) = e;
522 p(ndx::mx, offset) = m.x();
523 p(ndx::my, offset) = m.y();
524 p(ndx::mz, offset) = m.z();
525
526 } // end X
527
528 } // end Z
529
530 } // end Y
531
532 // 8) Bring perturbed state information back to wavespace (no rho!)
533 // Build FFT normalization constant into Y direction's mass matrix.
534 const complex_t scale_factor = 1 / dgrid.chi();
535 massluz.opform(1, &scale_factor, cop);
536 massluz.factor();
537 dgrid.transform_physical_to_wave(&p.coeffRef(ndx::e , 0)); // X, Z
538 dgrid.transform_physical_to_wave(&p.coeffRef(ndx::mx, 0)); // X, Z
539 dgrid.transform_physical_to_wave(&p.coeffRef(ndx::my, 0)); // X, Z
540 dgrid.transform_physical_to_wave(&p.coeffRef(ndx::mz, 0)); // X, Z
541 otool.bop_solve(massluz, s, ndx::e ); // Y
542 otool.bop_solve(massluz, s, ndx::mx); // Y
543 otool.bop_solve(massluz, s, ndx::my); // Y
544 otool.bop_solve(massluz, s, ndx::mz); // Y
545
546 // 9) Overwrite state storage with the new perturbed state (not rho!)
547 state[ndx::e ] = s[ndx::e ];
548 state[ndx::mx] = s[ndx::mx];
549 state[ndx::my] = s[ndx::my];
550 state[ndx::mz] = s[ndx::mz];
551}
