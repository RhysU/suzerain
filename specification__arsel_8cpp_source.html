58const std::string&
59specification_arsel::criterion(const std::string& abbrev)
60{
61 const best_model_function::type best_model
62 = best_model_function::lookup(abbrev, submean);
63 if (!best_model) {
64 throw std::invalid_argument(
65 "Unknown model selection criterion abbreviation");
66 }
67 bmf = reinterpret_cast<void *>(best_model);
68 return this->abbrev = abbrev;
69}
