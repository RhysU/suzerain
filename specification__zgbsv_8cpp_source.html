56specification_zgbsv::specification_zgbsv(const std::string& spec)
57{
58 using namespace std;
59
60 // Default construct instance
61 specification_zgbsv s;
62
63 // Advance past any leading whitespace
64 std::string::const_iterator iter = spec.begin(), end = spec.end();
65 while (iter != end && isspace(*iter)) ++iter;
66
67 // If nontrivial content, parse with a Spirit grammar modifying defaults
68 if (iter != end) {
69 using boost::phoenix::ref;
70 using boost::spirit::qi::_1;
71 using boost::spirit::qi::bool_;
72 using boost::spirit::qi::char_;
73 using boost::spirit::qi::double_;
74 using boost::spirit::qi::int_;
75 using boost::spirit::qi::no_case;
76
77 const bool r = boost::spirit::qi::phrase_parse(iter, end, (
78 // Grammar Begin
79 ( no_case["zgbsvx" ] [ ref(s.method_) = specification_zgbsv::zgbsvx ]
80 >> -( ( char_(',') >> no_case["equil"] >> '='
81 >> no_case[bool_] [ref(s.equil_) = _1])
82 )
83 )
84 | ( no_case["zgbsv" ] [ ref(s.method_) = specification_zgbsv::zgbsv ])
85 | ( no_case["zcgbsvx"] [ ref(s.method_) = specification_zgbsv::zcgbsvx ]
86 >> -( ( char_(',') >> no_case["reuse"] >> '='
87 >> no_case[bool_] [ref(s.reuse_) = _1])
88 ^ ( char_(',') >> no_case["aiter"] >> '='
89 >> int_ [ref(s.aiter_) = _1])
90 ^ ( char_(',') >> no_case["siter"] >> '='
91 >> int_ [ref(s.siter_) = _1])
92 ^ ( char_(',') >> no_case["diter"] >> '='
93 >> int_ [ref(s.diter_) = _1])
94 ^ ( char_(',') >> no_case["tolsc"] >> '='
95 >> double_[ref(s.tolsc_) = _1])
96 )
97 )
98 // Grammar End
99 ), boost::spirit::ascii::space);
100
101 // Ensure the grammar matched and the entire specification was consumed
102 if (!r || iter != end) {
103 ostringstream os;
104 os << "spec_zgbsv specification '" << spec
105 << "' invalid beginning with '";
106 copy(iter, end, ostream_iterator<string::value_type>(os));
107 os << '\'';
108 throw invalid_argument(os.str());
109 }
110 }
111
112 // Assign the mutated defaults to *this
113 *this = s;
114}
