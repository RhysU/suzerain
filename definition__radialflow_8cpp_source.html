96boost::program_options::options_description
97definition_radialflow::options_description()
98{
99 using boost::bind;
100 using boost::lexical_cast;
101 using boost::program_options::options_description;
102 using boost::program_options::typed_value;
103 using boost::program_options::value;
104 using std::unique_ptr;
105 using std::string;
106 using validation::ensure_nonnegative;
107 using validation::ensure_positive;
108
109 options_description retval(
110 "Radial flow problem reference parameters"
111 " for homogenized boundary layers with baseflow");
112
113 unique_ptr<typed_value<string> > p;
114
115 // deltae
116 p.reset(value<string>());
117 p->notifier(bind(&validate_option<double>, _1, &deltae,
118 &ensure_positive<double>, name_deltae));
119 if (!(boost::math::isnan)(deltae)) {
120 p->default_value(lexical_cast<string>(deltae));
121 }
122 retval.add_options()(name_deltae, p.release(), desc_deltae);
123
124 // gamma
125 p.reset(value<string>());
126 p->notifier(bind(&validate_option<double>, _1, &gamma,
127 &ensure_positive<double>, name_gamma));
128 if (!(boost::math::isnan)(gamma)) {
129 p->default_value(lexical_cast<string>(gamma));
130 }
131 retval.add_options()(name_gamma, p.release(), desc_gamma);
132
133 // Mae
134 p.reset(value<string>());
135 p->notifier(bind(&validate_option<double>, _1, &Mae,
136 &ensure_positive<double>, name_Mae));
137 if (!(boost::math::isnan)(Mae)) {
138 p->default_value(lexical_cast<string>(Mae));
139 }
140 retval.add_options()(name_Mae, p.release(), desc_Mae);
141
142 // pexi
143 p.reset(value<string>());
144 p->notifier(bind(&parse_option<double>, _1, &pexi, name_pexi));
145 if (!(boost::math::isnan)(pexi)) {
146 p->default_value(lexical_cast<string>(pexi));
147 }
148 retval.add_options()(name_pexi, p.release(), desc_pexi);
149
150 return retval;
151}

175void
176definition_radialflow::save(
177 const esio_handle h) const
178{
179 // Save nothing if there's nothing interesting to save
180 if (this->trivial()) {
181 return;
182 }
183
184 DEBUG0("Storing definition_radialflow parameters");
185
186 // Write out the "container" holding all other settings
187 const int one = 1;
188 int procid;
189 esio_handle_comm_rank(h, &procid);
190 esio_line_establish(h, 1, 0, (procid == 0 ? 1 : 0));
191 esio_line_write(h, location, &one, 0,
192 "Is a radial flow definition in use?");
193
194 // Write out information as attributes within the container
195 esio_attribute_write(h, location, attr_deltae, &deltae);
196 esio_attribute_write(h, location, attr_gamma, &gamma);
197 esio_attribute_write(h, location, attr_Mae, &Mae);
198 esio_attribute_write(h, location, attr_pexi, &pexi);
199}

201void
202definition_radialflow::load(
203 const esio_handle h,
204 const bool verbose)
205{
206 definition_radialflow t;
207
208 // Only proceed if a definition is active in the restart
209 int in_use = 0;
210 esio_line_establish(h, 1, 0, 1); // All ranks load any data
211 if (ESIO_NOTFOUND != esio_line_size(h, location, NULL)) {
212 esio_line_read(h, location, &in_use, 0);
213 }
214 if (!in_use) {
215 return;
216 }
217
218 DEBUG0("Loading definition_radialflow parameters");
219
220 // Read in information as attributes within the container
221 esio_attribute_read(h, location, attr_deltae, &t.deltae);
222 esio_attribute_read(h, location, attr_gamma, &t.gamma);
223 esio_attribute_read(h, location, attr_Mae, &t.Mae);
224 esio_attribute_read(h, location, attr_pexi, &t.pexi);
225
226 return this->populate(t, verbose);
227}
