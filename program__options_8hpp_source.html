220 // Routine is Copyright Vladimir Prus 2002-2004.
221 // Distributed under the Boost Software License, Version 1.0.

239 template<typename String>

242 {
243 // Routine is Copyright Vladimir Prus 2002-2004.
244 // Distributed under the Boost Software License, Version 1.0.
245 // (See http://www.boost.org/LICENSE_1_0.txt)
246 if (variables_.count(for_what) && !variables_[for_what].defaulted())
247 if ( variables_.count(required_option) == 0
248 || variables_[required_option].defaulted())
249 throw std::invalid_argument(std::string("Option '") + for_what
250 + "' requires option '" + required_option + "'.");
