96 bsmbsm_solver::build(suzerain_bsmbsm_construct(

98 cop.max_kl(), cop.max_ku()), spec, 1));
99
100 species_solver.push_back(tmp);

126void isothermal_hybrid_linear_operator::apply_mass_plus_scaled_operator(
127 const complex_t &phi,
128 multi_array::ref<complex_t,4> &state,
129 const std::size_t substep_index) const
130{
131 SUZERAIN_TIMER_SCOPED("apply_mass_plus_scaled_operator");
132 SUZERAIN_ENSURE(common.linearization != linearize::none);
133
134 // Shorthand
135 using inorder::wavenumber;
136 using inorder::wavenumber_absmin;
137 SUZERAIN_UNUSED(substep_index);
138
139 // We are only prepared to handle rho_E, rho_u, rho_v, rho_w, rho!
140 assert(static_cast<int>(ndx::e ) < flow_solver->S);
141 assert(static_cast<int>(ndx::mx ) < flow_solver->S);
142 assert(static_cast<int>(ndx::my ) < flow_solver->S);
143 assert(static_cast<int>(ndx::mz ) < flow_solver->S);
144 assert(static_cast<int>(ndx::rho) < flow_solver->S);
145
146 // TODO: Assert species solver sized correctly
147
148 // Wavenumber traversal modeled after those found in suzerain/diffwave.c
149 const int Ny = dgrid.global_wave_extent.y();
150 const int Nx = grid.N.x();
151 const int dNx = grid.dN.x();
152 const int dkbx = dgrid.local_wave_start.x();
153 const int dkex = dgrid.local_wave_end.x();
154 const int Nz = grid.N.z();
155 const int dNz = grid.dN.z();
156 const int dkbz = dgrid.local_wave_start.z();
157 const int dkez = dgrid.local_wave_end.z();
158
159 // Sidesteps assertions when local rank contains no wavespace information
160 if (SUZERAIN_UNLIKELY(0U == state.shape()[1])) return;
161
162 // Incoming state has wall-normal pencils of interleaved state scalars?
163 SUZERAIN_ENSURE(state.shape() [0] == (unsigned) (flow_solver->S+
164 species_solver.size()));
165 SUZERAIN_ENSURE(state.strides()[0] == (unsigned) Ny);
166 SUZERAIN_ENSURE(state.shape() [1] == (unsigned) Ny);
167 SUZERAIN_ENSURE(state.strides()[1] == 1);
168
169 // Compute total state size for pencil
170 int Ntot = flow_solver->N;
171 if (species_solver.size()>0) {
172 // All species have same size
173 Ntot += species_solver.size()*species_solver[0]->N;
174 }
175
176 // Scratch for "in-place" suzerain_reacting_imexop_accumulate usage
177 VectorXc tmp(Ntot);
178 suzerain_reacting_imexop_scenario s(this->imexop_s());
179 suzerain_reacting_imexop_ref ref;
180 suzerain_reacting_imexop_refld ld;
181 common.imexop_ref(ref, ld);
182
183 // Vector to save top boundary state in case we need it to
184 // overwrite action of linear operator (for used in boundary layer
185 // case)
186 VectorXc top_state(state.shape()[0]);
187
188 // Iterate across local wavenumbers and apply operator "in-place"
189 // Short circuit "continues" occur for Nyquist and non-dealiased modes...
190 // ...where the former will be zeroed during the later invert call.
191 switch (common.linearization) {
192
193 case linearize::rhome_y:
194 {
195 for (int n = dkbz; n < dkez; ++n) {
196 const int wn = wavenumber(dNz, n);
197 if (std::abs(wn) > wavenumber_absmin(Nz)) continue;
198
199 for (int m = dkbx; m < dkex; ++m) {
200 const int wm = wavenumber(dNx, m);
201 if (std::abs(wm) > wavenumber_absmin(Nx)) continue;
202
203 // Get pointer to (.,m,n)-th state pencil
204 complex_t * const p = &state[0][0][m - dkbx][n - dkbz];
205
206 // if BL, save top state
207 if (grid.one_sided())
208 for (std::size_t i=0; i<state.shape()[0]; ++i)
209 top_state[i] = state[i][Ny-1][m - dkbx][n - dkbz];
210
211 // Copy pencil into temporary storage
212 blas::copy(Ntot, p, 1, tmp.data(), 1);
213
214 // Accumulate result back into state storage
215 SUZERAIN_TIMER_SCOPED("suzerain_reacting_imexop_accumulate");
216
217 // Flow equations
218 suzerain_reacting_flow_imexop_accumulate(
219 phi, &s, &ref, &ld, cop.get(),
220 0, // imagzero=false FIXME: update after removing from reacting_imexop
221 tmp.data() + ndx::e * Ny,
222 tmp.data() + ndx::mx * Ny,
223 tmp.data() + ndx::my * Ny,
224 tmp.data() + ndx::mz * Ny,
225 tmp.data() + ndx::rho * Ny,
226 0.0,
227 p + ndx::e * Ny,
228 p + ndx::mx * Ny,
229 p + ndx::my * Ny,
230 p + ndx::mz * Ny,
231 p + ndx::rho * Ny);
232
233 // Species equations
234 for (std::size_t alfa=0; alfa<species_solver.size(); ++alfa) {
235 suzerain_reacting_species_imexop_accumulate(
236 phi, &s, &ref, &ld, cop.get(),
237 0, // imagzero=false FIXME: update after removing from reacting_imexop
238 tmp.data() + (ndx::species+alfa) * Ny,
239 0.0,
240 p + (ndx::species+alfa) * Ny);
241 }
242
243 // if BL, replace top state
244 if (grid.one_sided())
245 for (std::size_t i=0; i<state.shape()[0]; ++i)
246 state[i][Ny-1][m - dkbx][n - dkbz] = top_state[i];
247
248 }
249 }
250 break;
251 }
252
253 default:
254 SUZERAIN_ERROR_VOID_UNIMPLEMENTED();
255
256 }

259void isothermal_hybrid_linear_operator::accumulate_mass_plus_scaled_operator(
260 const complex_t &phi,
261 const multi_array::ref<complex_t,4> &input,
262 const complex_t &beta,
263 contiguous_state<4,complex_t> &output,
264 const std::size_t substep_index) const
265{
266 SUZERAIN_TIMER_SCOPED("accumulate_mass_plus_scaled_operator");
267 SUZERAIN_ENSURE(common.linearization != linearize::none);
268
269 // Shorthand
270 using inorder::wavenumber;
271 using inorder::wavenumber_absmin;
272 SUZERAIN_UNUSED(substep_index);
273
274 // We are only prepared to handle rho_E, rho_u, rho_v, rho_w, rho!
275 assert(static_cast<int>(ndx::e ) < flow_solver->S);
276 assert(static_cast<int>(ndx::mx ) < flow_solver->S);
277 assert(static_cast<int>(ndx::my ) < flow_solver->S);
278 assert(static_cast<int>(ndx::mz ) < flow_solver->S);
279 assert(static_cast<int>(ndx::rho) < flow_solver->S);
280
281 // TODO: Assert species solver sized correctly
282
283 // Wavenumber traversal modeled after those found in suzerain/diffwave.c
284 const int Ny = dgrid.global_wave_extent.y();
285 const int Nx = grid.N.x();
286 const int dNx = grid.dN.x();
287 const int dkbx = dgrid.local_wave_start.x();
288 const int dkex = dgrid.local_wave_end.x();
289 const int Nz = grid.N.z();
290 const int dNz = grid.dN.z();
291 const int dkbz = dgrid.local_wave_start.z();
292 const int dkez = dgrid.local_wave_end.z();
293
294 // Sidesteps assertions when local rank contains no wavespace information
295 if (SUZERAIN_UNLIKELY(0U == input.shape()[1])) return;
296
297 // Input and output state storage has contiguous wall-normal scalars?
298 // Furthermore, input has contiguous wall-normal pencils of all state?
299 SUZERAIN_ENSURE(output.is_isomorphic(input));
300 SUZERAIN_ENSURE(input.shape() [0] == (unsigned) (flow_solver->S +
301 species_solver.size()));
302 SUZERAIN_ENSURE(input.strides() [0] == (unsigned) Ny);
303 SUZERAIN_ENSURE(input.shape() [1] == (unsigned) Ny);
304 SUZERAIN_ENSURE(input.strides() [1] == 1);
305 SUZERAIN_ENSURE(output.strides()[1] == 1);
306
307 // Scratch for suzerain_reacting_imexop_accumulate usage
308 suzerain_reacting_imexop_scenario s(this->imexop_s());
309 suzerain_reacting_imexop_ref ref;
310 suzerain_reacting_imexop_refld ld;
311 common.imexop_ref(ref, ld);
312
313 // Vector to save top boundary state in case we need it to
314 // overwrite action of linear operator (for used in boundary layer
315 // case)
316 VectorXc top_input (input.shape() [0]);
317 VectorXc top_output(output.shape()[0]);
318
319 // Iterate across local wavenumbers and apply operator "in-place"
320 // Short circuit "continues" occur for Nyquist and non-dealiased modes...
321 // ...where the former will be zeroed during the later invert call.
322 switch (common.linearization) {
323
324 case linearize::rhome_y:
325 {
326 for (int n = dkbz; n < dkez; ++n) {
327 const int wn = wavenumber(dNz, n);
328 if (std::abs(wn) > wavenumber_absmin(Nz)) continue;
329
330 for (int m = dkbx; m < dkex; ++m) {
331 const int wm = wavenumber(dNx, m);
332 if (std::abs(wm) > wavenumber_absmin(Nx)) continue;
333
334
335 // if BL, save top state
336 if (grid.one_sided()) {
337 for (std::size_t i=0; i<input.shape()[0]; ++i) {
338 top_input[i] = input [i][Ny-1][m - dkbx][n - dkbz];
339 top_output[i] = output[i][Ny-1][m - dkbx][n - dkbz];
340 }
341 }
342
343 // Accumulate result
344 SUZERAIN_TIMER_SCOPED("suzerain_reacting_imexop_accumulate");
345
346 // Flow equations
347 suzerain_reacting_flow_imexop_accumulate(
348 phi, &s, &ref, &ld, cop.get(),
349 0, // imagzero=false FIXME: update after removing from reacting_imexop
350 &input [ndx::e ][0][m - dkbx][n - dkbz],
351 &input [ndx::mx ][0][m - dkbx][n - dkbz],
352 &input [ndx::my ][0][m - dkbx][n - dkbz],
353 &input [ndx::mz ][0][m - dkbx][n - dkbz],
354 &input [ndx::rho][0][m - dkbx][n - dkbz],
355 beta,
356 &output[ndx::e ][0][m - dkbx][n - dkbz],
357 &output[ndx::mx ][0][m - dkbx][n - dkbz],
358 &output[ndx::my ][0][m - dkbx][n - dkbz],
359 &output[ndx::mz ][0][m - dkbx][n - dkbz],
360 &output[ndx::rho ][0][m - dkbx][n - dkbz]);
361
362 // Species equations
363 for (std::size_t alfa=0; alfa<species_solver.size(); ++alfa) {
364 suzerain_reacting_species_imexop_accumulate(
365 phi, &s, &ref, &ld, cop.get(),
366 0, // imagzero=false FIXME: update after removing from reacting_imexop
367 &input [ndx::species+alfa][0][m - dkbx][n - dkbz],
368 beta,
369 &output[ndx::species+alfa][0][m - dkbx][n - dkbz]);
370 }
371
372 // if BL, replace top
373 if (grid.one_sided()) {
374 // protect against 0*NaN = Nan
375 if (std::norm(beta)==0) {
376 for (std::size_t i=0; i<input.shape()[0]; ++i) {
377 output[i][Ny-1][m - dkbx][n - dkbz] = top_input[i];
378 }
379 }
380 else {
381 for (std::size_t i=0; i<input.shape()[0]; ++i) {
382 output[i][Ny-1][m - dkbx][n - dkbz] =
383 top_input[i] + beta*top_output[i];
384 }
385 }
386 }
387
388
389 }
390 }
391 break;
392 }
393
394 default:
395 SUZERAIN_ERROR_VOID_UNIMPLEMENTED();
396 }

416class IsothermalNoSlipPATPTEnforcer
417{
418 enum { nsides = 2, nmomentum = 3 };
419
420 // Indices within PA^TP^T at which to apply boundary conditions
421 // Computed once within constructor and then repeatedly used
422 int rho[nsides], noslip[nsides][nmomentum], e[nsides];
423
424 // Precomputed coefficient based on the isothermal equation of state
425 real_t e_tot;
426
427 // Number of wall BCs (must be 1 or 2 depending on channel or
428 // boundary layer)
429 int nwalls;
430
431 // Store velocity values in a vector to loop through components
432 std::vector<real_t> velw;
433
434 // To flip sign of v at upper wall for channel case
435 std::vector<real_t> vsgn;
436
437public:
438
439 IsothermalNoSlipPATPTEnforcer(const suzerain_bsmbsm &A_T,
440 const real_t &u_wall,
441 const real_t &v_wall,
442 const real_t &w_wall,
443 const real_t &e_tot,
444 const int nwalls)
445 : // Pass internal energy and compute total energy to use in the object
446 e_tot(e_tot+0.5*(u_wall*u_wall + v_wall*v_wall + w_wall*w_wall))
447 , nwalls(nwalls)
448
449 {
450 // Starting offset to named scalars in interleaved_state pencil
451 const int e0 = static_cast<int>(ndx::e ) * A_T.n;
452 const int mx0 = static_cast<int>(ndx::mx ) * A_T.n;
453 const int my0 = static_cast<int>(ndx::my ) * A_T.n;
454 const int mz0 = static_cast<int>(ndx::mz ) * A_T.n;
455 const int rho0 = static_cast<int>(ndx::rho) * A_T.n;
456
457 // Relative to foo0 what is the offset to lower, upper walls
458 const int wall[nsides] = { 0, A_T.n - 1};
459
460 // Prepare indices within PA^TP^T corresponding to the walls.
461 // Uses that {A^T}_{i,j} maps to {PA^TP^T}_{{q^-1}(i),{q^(-1)}(j)}.
462 for (int i = 0; i < nsides; ++i) {
463 e [i] = suzerain_bsmbsm_qinv(A_T.S, A_T.n, e0 + wall[i]);
464 noslip[i][0] = suzerain_bsmbsm_qinv(A_T.S, A_T.n, mx0 + wall[i]);
465 noslip[i][1] = suzerain_bsmbsm_qinv(A_T.S, A_T.n, my0 + wall[i]);
466 noslip[i][2] = suzerain_bsmbsm_qinv(A_T.S, A_T.n, mz0 + wall[i]);
467 rho [i] = suzerain_bsmbsm_qinv(A_T.S, A_T.n, rho0 + wall[i]);
468 }
469
470 // Initialize wall velocity vector
471 velw.resize(3);
472 velw[0] = u_wall;
473 velw[1] = v_wall;
474 velw[2] = w_wall;
475
476 // Initialize sign constant for velocity components
477 // NOTE: we need to declare the sign for the three components
478 // Indices go by eqn+nmomentum*wall, with
479 // eqn {u,v,w} = {1,2,3} and
480 // wall{bottom, top} = {0,1}
481 vsgn.resize(nsides*nmomentum);
482 // u-velocity
483 vsgn[0] = 1;
484 vsgn[3] = 1;
485 // v-velocity
486 vsgn[1] = 1;
487 vsgn[4] = -1;
488 // w-velocity
489 vsgn[2] = 1;
490 vsgn[5] = 1;
491 }

497 template<typename T>
498 void rhs(T * const b)
499 {
500 for (int wall = 0; wall < nwalls; ++wall) {
501 for (int eqn = 0; eqn < nmomentum; ++eqn) {
502 b[noslip[wall][eqn]] = 0;
503 }
504 b[e[wall]] = 0;
505 }
506 }
507
514 template<typename T>
515 void op(const suzerain_bsmbsm& A_T, T * const patpt, int patpt_ld)
516 {
517 // Attempt made to not unnecessarily disturb matrix conditioning.
518
519 for (int wall = 0; wall < nwalls; ++wall) {
520 int begin, end;
521
522 // Zero off-diagonals so momentum equations are const*rho{u,v,w}=0.
523 for (size_t eqn = 0; eqn < nmomentum; ++eqn) {
524 T * const col = (T *) suzerain_gbmatrix_col(
525 A_T.N, A_T.N, A_T.KL, A_T.KU, (void *) patpt, patpt_ld,
526 sizeof(T), noslip[wall][eqn], &begin, &end);
527 // Necessary to ensure constraint possible in degenerate case
528 complex_t &rhocoeff = col[rho[wall]];
529 if (rhocoeff == complex_t(0)) rhocoeff = 1;
530 for (int i = begin; i < end; ++i) {
531 if (i == noslip[wall][eqn]) {
532 col[i] = rhocoeff;
533 } else if (i == rho[wall]) {
534 col[i] = - rhocoeff * vsgn[eqn+nmomentum*wall] * velw[eqn];
535 } else {
536 col[i] = 0;
537 }
538
539 }
540 }
541
542 // Set constraint const*rho - const*gamma*(gamma-1)*rhoE = 0
543 T * const col = (T *) suzerain_gbmatrix_col(
544 A_T.N, A_T.N, A_T.KL, A_T.KU, (void *) patpt, patpt_ld,
545 sizeof(T), e[wall], &begin, &end);
546 // Necessary to ensure constraint possible in degenerate case
547 complex_t &rhocoeff = col[rho[wall]];
548 if (rhocoeff == complex_t(0)) rhocoeff = 1;
549 // Scan row and adjust coefficients for constraint
550 for (int i = begin; i < end; ++i) {
551 if (i == e[wall]) {
552 col[i] = -rhocoeff/e_tot; // TODO: Check me!
553 } else if (i == rho[wall]) {
554 // NOP
555 } else {
556 col[i] = 0;
557 }
558 }
559 }
560 }

570class MassFractionPATPTEnforcer
571{
572 enum { nsides = 2, nmomentum = 3 };
573
574 // Indices within PA^TP^T at which to apply boundary conditions
575 // Computed once within constructor and then repeatedly used
576 int rho_s[nsides];
577 int Ny;
578
579 // Precomputed coefficient based on the isothermal equation of state
580 const std::vector<real_t>& wall_mass_fractions;
581
582 // Number of wall BCs (must be 1 or 2 depending on channel or
583 // boundary layer)
584 int nwalls;
585
586public:
587
588 MassFractionPATPTEnforcer(const suzerain_bsmbsm &A_T,
589 const std::vector<real_t>& wall_mass_fractions,
590 const int nwalls)
591 : Ny(A_T.n)
592 , wall_mass_fractions(wall_mass_fractions)
593 , nwalls(nwalls)
594 {
595 // Starting offset to named scalars in interleaved_state pencil
596 const int rho_s0 = 0;
597
598 // Relative to foo0 what is the offset to lower, upper walls
599 const int wall[nsides] = { 0, A_T.n - 1};
600
601 // Prepare indices within PA^TP^T corresponding to the walls.
602 // Uses that {A^T}_{i,j} maps to {PA^TP^T}_{{q^-1}(i),{q^(-1)}(j)}.
603 for (int i = 0; i < nsides; ++i) {
604 rho_s[i] = suzerain_bsmbsm_qinv(A_T.S, A_T.n, rho_s0 + wall[i]);
605 }
606 }
607
611 template<typename T>
612 void rhs(T * const bspec, T * const xflow, const int alfa)
613 {
614 // Note two assumptions...
615 // 1.) alfa+1 implies wall_mass_fractions[0] is diluter
616 // 2.) incoming xflow is non-permuted flow solution
617
618 for (int wall = 0; wall < nwalls; ++wall) {
619 bspec[rho_s[wall]] =
620 xflow[ndx::rho*Ny+wall*(Ny-1)]*wall_mass_fractions[alfa+1];
621 }
622 }
623
627 template<typename T>
628 void rhs_non_permuted(T * const bspec, T * const xflow, const int alfa)
629 {
630 // Note two assumptions...
631 // 1.) alfa+1 implies wall_mass_fractions[0] is diluter
632 // 2.) incoming xflow is non-permuted flow solution
633
634 for (int wall = 0; wall < nwalls; ++wall) {
635 bspec[wall*(Ny-1)] =
636 xflow[ndx::rho*Ny+wall*(Ny-1)]*wall_mass_fractions[alfa+1];
637 }
638 }

649 // No attempt made to not unnecessarily disturb matrix conditioning.
650
651 for (int wall = 0; wall < nwalls; ++wall) {
652 int begin, end;
653
654 T * const col = (T *) suzerain_gbmatrix_col(
655 A_T.N, A_T.N, A_T.KL, A_T.KU, (void *) patpt, patpt_ld,

657 // Scan row and adjust coefficients for constraint
658 for (int i = begin; i < end; ++i) {
659 if (i == rho_s[wall]) {
660 col[i] = 1.0;
661 } else {
662 col[i] = 0;
663 }
664 }
665 }

669void isothermal_hybrid_linear_operator::invert_mass_plus_scaled_operator(
670 const complex_t &phi,
671 multi_array::ref<complex_t,4> &state,
672 const lowstorage::method_interface<complex_t> &method,
673 const component delta_t,
674 const std::size_t substep_index,
675 multi_array::ref<complex_t,4> *ic0) const
676{
677 // State enters method as coefficients in X and Z directions
678 // State enters method as collocation point values in Y direction
679
680 SUZERAIN_TIMER_SCOPED("invert_mass_plus_scaled_operator");
681 SUZERAIN_ENSURE(common.linearization != linearize::none);
682
683 // Shorthand
684 using inorder::wavenumber;
685 using inorder::wavenumber_absmin;
686 SUZERAIN_UNUSED(method);
687 SUZERAIN_UNUSED(delta_t);
688 SUZERAIN_UNUSED(substep_index);
689
690 // number of flow equations
691 enum {Nf=5};
692
693 // We are only prepared to handle rho_E, rho_u, rho_v, rho_w, rho!
694 assert(static_cast<int>(ndx::e ) < flow_solver->S);
695 assert(static_cast<int>(ndx::mx ) < flow_solver->S);
696 assert(static_cast<int>(ndx::my ) < flow_solver->S);
697 assert(static_cast<int>(ndx::mz ) < flow_solver->S);
698 assert(static_cast<int>(ndx::rho) < flow_solver->S);
699
700 // TODO: Assert species solver sized correctly
701
702 // Wavenumber traversal modeled after those found in suzerain/diffwave.c
703 const int Ny = dgrid.global_wave_extent.y();
704 const int Nx = grid.N.x();
705 const int dNx = grid.dN.x();
706 const int dkbx = dgrid.local_wave_start.x();
707 const int dkex = dgrid.local_wave_end.x();
708 const int Nz = grid.N.z();
709 const int dNz = grid.dN.z();
710 const int dkbz = dgrid.local_wave_start.z();
711 const int dkez = dgrid.local_wave_end.z();
712
713 // Sidesteps assertions when local rank contains no wavespace information
714 if (SUZERAIN_UNLIKELY(0U == state.shape()[1])) return;
715
716 // Incoming state has wall-normal pencils of interleaved state scalars?
717 SUZERAIN_ENSURE(state.shape() [1] == (unsigned) Ny);
718 SUZERAIN_ENSURE(state.strides()[1] == 1);
719 SUZERAIN_ENSURE(state.strides()[0] == (unsigned) Ny);
720 SUZERAIN_ENSURE(state.shape() [0] == (unsigned) (flow_solver->S +
721 species_solver.size()));
722
723 // Compute how many additional mean constraints we must solve
724 // Ensure conformant, mean constraints are arriving on the correct rank
725 const std::size_t nconstraints = ic0 ? ic0->shape()[2]*ic0->shape()[3] : 0;
726 if (nconstraints) {
727 SUZERAIN_ENSURE(dgrid.has_zero_zero_modes());
728 SUZERAIN_ENSURE(ic0->shape() [1] == (unsigned) Ny);
729 SUZERAIN_ENSURE(ic0->strides()[1] == 1);
730 SUZERAIN_ENSURE(ic0->strides()[0] == (unsigned) Ny);
731 SUZERAIN_ENSURE(ic0->shape() [0] == (unsigned) (flow_solver->S +
732 species_solver.size()));
733 }
734
735 // treatment_channel step (3) performs the operator solve which for the
736 // implicit treatment must be combined with boundary conditions
737
738 // Pack reference details for suzerain_reacting_imexop routines
739 suzerain_reacting_imexop_scenario s(this->imexop_s());
740 suzerain_reacting_imexop_ref ref;
741 suzerain_reacting_imexop_refld ld;
742 common.imexop_ref(ref, ld);
743
744 // Prepare an almost functor mutating RHS and PA^TP^T to enforce BCs.
745 // FIXME: permit upper_T, upper_cs, and upper_{u,v,w}
746 int nwalls=2;
747 if (grid.one_sided()) nwalls=1;
748
749 IsothermalNoSlipPATPTEnforcer flow_bc_enforcer(
750 *flow_solver,
751 isospec.lower_u, isospec.lower_v, isospec.lower_w,
752 cmods.e_from_T(isospec.lower_T, isospec.lower_cs), nwalls);
753
754 std::shared_ptr<MassFractionPATPTEnforcer> species_bc_enforcer;
755 if (species_solver.size()>0) {
756 // FIXME: permit upper_T and upper_cs
757 species_bc_enforcer =
758 std::make_shared<MassFractionPATPTEnforcer>(
759 *(species_solver[0]), isospec.lower_cs, nwalls);
760 }
761
762 // Prepare a scratch buffer for packc/packf usage
763 //
764 // NOTE: buf sized for usage with flow operator is always big
765 // enough for usage with decoupled species operators... beware if
766 // you do anything else
767 ArrayXXc buf(flow_solver->ld, flow_solver->n);
768
769 // Iterate across local wavenumbers and "invert" operator "in-place"
770 //
771 // The loop structure changes markedly on rhome_xyz vs rhome_y,
772 // but the basic sequence of steps remains identical in each case.
773 switch (common.linearization) {
774
775 case linearize::rhome_y:
776 {
777 // Form complex-valued, wavenumber-independent PA^TP^T
778 static const char trans = 'T';
779 if (flow_solver->spec.in_place()) { // Pack for in-place LU
780 SUZERAIN_TIMER_SCOPED("implicit operator assembly (packf)");
781 suzerain_reacting_flow_imexop_packf(
782 phi, &s, &ref, &ld, cop.get(),
783 ndx::e, ndx::mx, ndx::my, ndx::mz, ndx::rho,
784 buf.data(), flow_solver.get(), flow_solver->LU.data());
785 } else { // Pack for out-of-place LU
786 SUZERAIN_TIMER_SCOPED("implicit operator assembly (packc)");
787 suzerain_reacting_flow_imexop_packc(
788 phi, &s, &ref, &ld, cop.get(),
789 ndx::e, ndx::mx, ndx::my, ndx::mz, ndx::rho,
790 buf.data(), flow_solver.get(), flow_solver->PAPT.data());
791 }
792
793 for (std::size_t alfa=0; alfa<species_solver.size(); ++alfa) {
794 if (species_solver[alfa]->spec.in_place()) { // Pack for in-place LU
795 SUZERAIN_TIMER_SCOPED("implicit operator assembly (packf)");
796 suzerain_reacting_species_imexop_packf(
797 phi, &s, &ref, &ld, cop.get(),
798 buf.data(), species_solver[alfa].get(),
799 species_solver[alfa]->LU.data());
800 } else { // Pack for out-of-place LU
801 SUZERAIN_TIMER_SCOPED("implicit operator assembly (packc)");
802 suzerain_reacting_species_imexop_packc(
803 phi, &s, &ref, &ld, cop.get(),
804 buf.data(), species_solver[alfa].get(),
805 species_solver[alfa]->PAPT.data());
806 }
807 }
808
809 // Apply boundary conditions to PA^TP^T
810 {
811 SUZERAIN_TIMER_SCOPED("implicit operator BCs");
812 flow_bc_enforcer.op(*flow_solver, flow_solver->PAPT.data(),
813 flow_solver->PAPT.colStride());
814 }
815
816 // If one-sided, set 5x5 PAPT block corresponding to
817 // freestream boundary (i.e., the top) to identity matrix
818 if (grid.one_sided()) {
819 const int nr = flow_solver->KL+flow_solver->KU+1;
820 const int ncb = flow_solver->S*(flow_solver->n-1);
821
822 // zero the whole block
823 flow_solver->PAPT.block(0 , ncb, nr, Nf).setZero();
824
825 // set diagonal to 1
826 flow_solver->PAPT.block(flow_solver->KL, ncb, 1 , Nf).setOnes();
827 }
828
829 // Inform the flow_solver about the new, unfactorized operator
830 flow_solver->supplied_PAPT();
831
832
833 for (std::size_t alfa=0; alfa<species_solver.size(); ++alfa) {
834
835 {
836 // Apply species BCs to operator
837 SUZERAIN_TIMER_SCOPED("implicit operator BCs");
838 species_bc_enforcer->op(*(species_solver[alfa]),
839 species_solver[alfa]->PAPT.data(),
840 species_solver[alfa]->PAPT.colStride());
841 }
842
843 // If one-sided, set 1x1 PAPT block corresponding to
844 // freestream boundary (i.e., the top) to identity matrix
845 if (grid.one_sided()) {
846 const int KL = species_solver[alfa]->KL;
847 const int KU = species_solver[alfa]->KU;
848
849 const int nr = KL+KU+1;
850 const int ncb =
851 species_solver[alfa]->S*(species_solver[alfa]->n-1);
852
853 // zero the whole block
854 species_solver[alfa]->PAPT.block(0 , ncb, nr, 1).setZero();
855 // set diagonal to 1
856 species_solver[alfa]->PAPT.block(KL, ncb, 1 , 1).setOnes();
857 }
858
859 // Inform species_solver about new, unfactorized operator
860 species_solver[alfa]->supplied_PAPT();
861 }
862
863 // Species solver supplied call
864
865 // Solve using a new right hand side for each wavenumber pair km, kn
866 for (int n = dkbz; n < dkez; ++n) {
867 const int wn = wavenumber(dNz, n);
868
869 for (int m = dkbx; m < dkex; ++m) {
870 const int wm = wavenumber(dNx, m);
871
872 // Get pointer to (.,m,n)-th state pencil
873 complex_t * const p = &state[0][0][m - dkbx][n - dkbz];
874
875 // Continue didn't yet occur for Nyquist/dealiasing modes...
876 if ( std::abs(wn) > wavenumber_absmin(Nz)
877 || std::abs(wm) > wavenumber_absmin(Nx)) {
878 memset(p, 0, flow_solver->N*sizeof(p[0])); // ...so we can zero,
879 continue; // and short circuit.
880 }
881
882 // Form right hand side, apply BCs, factorize, and solve.
883 // Beware that much ugly, ugly magic is hidden just below.
884
885 // Flow RHS, BCs, solve
886 flow_solver->supply_B(p);
887
888 // Boundary conditions
889 {
890 SUZERAIN_TIMER_SCOPED("implicit right hand side BCs");
891 flow_bc_enforcer.rhs(flow_solver->PB.data());
892 }
893
894 flow_solver->solve(trans);
895 flow_solver->demand_X(p);
896
897
898 // Species RHSs, BCs, solves
899 for (std::size_t alfa=0; alfa<species_solver.size(); ++alfa) {
900 species_solver[alfa]->supply_B(p + (ndx::species+alfa)*Ny);
901
902 {
903 SUZERAIN_TIMER_SCOPED("implicit right hand side BCs");
904 species_bc_enforcer->rhs(species_solver[alfa]->PB.data(),
905 p, alfa);
906 }
907
908 species_solver[alfa]->solve(trans);
909 species_solver[alfa]->demand_X(p + (ndx::species+alfa)*Ny);
910 }
911
912 }
913 }
914
915 // Compute total state size for pencil
916 int Ntot = flow_solver->N;
917 if (species_solver.size()>0) {
918 // All species have same size
919 Ntot += species_solver.size()*species_solver[0]->N;
920 }
921
922 // If necessary, solve any required integral constraints
923 for (std::size_t i = 0; i < nconstraints; ++i) {
924 SUZERAIN_TIMER_SCOPED("implicit constraint solution");
925 flow_solver->supply_B(ic0->data() + i * Ntot);
926 flow_bc_enforcer.rhs(flow_solver->PB.data());
927 flow_solver->solve(trans);
928 flow_solver->demand_X(ic0->data() + i * Ntot);
929
930
931 // Handle effects of constraints on species wall BCs
932 for (std::size_t alfa=0; alfa<species_solver.size(); ++alfa) {
933 species_bc_enforcer->rhs_non_permuted(
934 ic0->data() + i * Ntot + (ndx::species+alfa)*Ny,
935 ic0->data() + i * Ntot,
936 alfa);
937 }
938 }
939
940 break;
941 }
942
943
944 default:
945 SUZERAIN_ERROR_VOID_UNIMPLEMENTED();
946 }
947
948 // State leaves method as coefficients in X, Y, and Z directions
949}
