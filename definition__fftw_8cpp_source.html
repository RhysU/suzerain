52boost::program_options::options_description
53definition_fftw::options_description()
54{
55 namespace po = boost::program_options;
56 using boost::bind;
57 using std::string;
58 using std::ostringstream;
59
60 ostringstream oss;
61 oss << " {" << fftw::c_str(fftw::estimate)
62 << ", " << fftw::c_str(fftw::measure)
63 << ", " << fftw::c_str(fftw::patient)
64 << ", " << fftw::c_str(fftw::exhaustive)
65 << ", or " << fftw::c_str(fftw::wisdom_only)
66 << "}";
67 const string rigor_options = oss.str();
68
69 string rigor_fft_description = "FFTW FFT planning rigor ";
70 rigor_fft_description += rigor_options;
71
72 string rigor_mpi_description = "FFTW MPI planning rigor ";
73 rigor_mpi_description += rigor_options;
74
75 po::options_description retval("FFTW planning options:");
76
77 retval.add_options()
78 ("rigor_fft",
79 po::value<std::string>(NULL)
80 ->notifier(bind(&definition_fftw::normalize_rigor_fft, this, _1))
81 ->default_value(fftw::c_str(rigor_fft)),
82 rigor_fft_description.c_str())
83 ("rigor_mpi",
84 po::value<std::string>(NULL)
85 ->notifier(bind(&definition_fftw::normalize_rigor_mpi, this, _1))
86 ->default_value(fftw::c_str(rigor_mpi)),
87 rigor_mpi_description.c_str())
88 ("plan_wisdom",
89 po::value(&plan_wisdom),
90 "File for accumulating FFTW planning wisdom")
91 ("plan_timelimit",
92 po::value(&plan_timelimit)
93 ->default_value(FFTW_NO_TIMELIMIT, "unlimited"),
94 "Maximum seconds for creating any FFTW plan")
95 ;
96
97 return retval;
98}
