83boost::program_options::options_description
84definition_channel::options_description()
85{
86 using boost::bind;
87 using boost::lexical_cast;
88 using boost::program_options::options_description;
89 using boost::program_options::typed_value;
90 using boost::program_options::value;
91 using std::unique_ptr;
92 using std::string;
93
94 options_description retval("Channel flow scenario parameters");
95
96 // Complicated add_options() calls done to allow changing the default value
97 // displayed when the default is NaN. NaN is used as a NOP value by client
98 // code. Validation routines used below all silently allow NaNs.
99
100 unique_ptr<typed_value<string> > p;
101
102 // bulk_rho
103 p.reset(value<string>());
104 p->notifier(bind(&parse_nonnegative, _1, &bulk_rho, name_bulk_rho));
105 if (!(boost::math::isnan)(bulk_rho)) {
106 p->default_value(lexical_cast<string>(bulk_rho));
107 }
108 retval.add_options()(name_bulk_rho, p.release(), desc_bulk_rho);
109
110 // bulk_rho_u
111 p.reset(value<string>());
112 p->notifier(bind(&parse_nonnegative, _1, &bulk_rho_u, name_bulk_rho_u));
113 if (!(boost::math::isnan)(bulk_rho_u)) {
114 p->default_value(lexical_cast<string>(bulk_rho_u));
115 }
116 retval.add_options()(name_bulk_rho_u, p.release(), desc_bulk_rho_u);
117
118 // bulk_rho_E
119 p.reset(value<string>());
120 p->notifier(bind(&parse_nonnegative, _1, &bulk_rho_E, name_bulk_rho_E));
121 if (!(boost::math::isnan)(bulk_rho_E)) {
122 p->default_value(lexical_cast<string>(bulk_rho_E));
123 }
124 retval.add_options()(name_bulk_rho_E, p.release(), desc_bulk_rho_E);
125
126 return retval;
127}

157void
158definition_channel::save(
159 const esio_handle h) const
160{
161 DEBUG0("Storing definition_channel parameters");
162
163 // Only root writes data
164 int procid;
165 esio_handle_comm_rank(h, &procid);
166
167 // scalars
168 esio_line_establish(h, 1, 0, (procid == 0 ? 1 : 0));
169 esio_line_write(h, name_bulk_rho, &this->bulk_rho, 0, desc_bulk_rho);
170 esio_line_write(h, name_bulk_rho_u, &this->bulk_rho_u, 0, desc_bulk_rho_u);
171 esio_line_write(h, name_bulk_rho_E, &this->bulk_rho_E, 0, desc_bulk_rho_E);
172}

174void
175definition_channel::load(
176 const esio_handle h,
177 const bool verbose)
178{
179 DEBUG0("Loading definition_channel parameters");
180
181 definition_channel t;
182
183 // All ranks load
184
185 // Scalars
186 esio_line_establish(h, 1, 0, 1);
187 esio_line_read(h, name_bulk_rho, &t.bulk_rho, 0);
188 esio_line_read(h, name_bulk_rho_u, &t.bulk_rho_u, 0);
189 if (ESIO_NOTFOUND != esio_line_size(h, name_bulk_rho_E, NULL)) {
190 esio_line_read(h, name_bulk_rho_E, &t.bulk_rho_E, 0);
191 } else {
192 INFO0(desc_bulk_rho_E << " not found; defaulting to disabled");
193 t.bulk_rho_E = std::numeric_limits<real_t>::quiet_NaN();
194 }
195
196 this->populate(t, verbose); // Prefer this to incoming
197}
