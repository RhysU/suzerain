43template<typename T>
44static void parse_option(const std::string& s,
45 T* value,
46 void (*validator)(T, const char*),
47 const char* name)
48{
49#pragma warning(push,disable:2259)
50 const T t = exprparse<real_t>(s, name);
51#pragma warning(pop)
52 validator(t, name);
53 *value = t;
54}

113{
114 using boost::bind;
115 using boost::lexical_cast;
116 using boost::program_options::bool_switch;
117 using boost::program_options::value;
118 using std::string;
119 using validation::ensure_nonnegative;

122 boost::program_options::options_description retval(desc_location);

201void
202definition_helm::save(
203 const esio_handle h) const
204{
205 DEBUG0("Storing definition_helm parameters");
206
207 // Only root writes the containing location with
208 // a 0 or 1 indicating whether or not helm was enabled.
209 int procid;
210 esio_handle_comm_rank(h, &procid);
211 esio_line_establish(h, 1, 0, (procid == 0 ? 1 : 0));
212 const int enabled_int = enabled;
213 esio_line_write(h, location, &enabled_int, 0, desc_location);
214
215 // Everyone writes the metadata
216 esio_attribute_write(h, location, attr_kp, &this->h.kp);
217 esio_attribute_write(h, location, attr_r, &this->r );
218 esio_attribute_write(h, location, attr_Td, &this->h.Td);
219 esio_attribute_write(h, location, attr_Tf, &this->h.Tf);
220 esio_attribute_write(h, location, attr_Ti, &this->h.Ti);
221 esio_attribute_write(h, location, attr_Tt, &this->h.Tt);
222}

224void
225definition_helm::load(
226 const esio_handle h,
227 const bool verbose)
228{
229 definition_helm t;
230
231 // When possible, load settings from restart.
232
233 // All ranks load any data with short circuit when not present
234 esio_line_establish(h, 1, 0, 1);
235 if (ESIO_NOTFOUND == esio_line_size(h, location, NULL)) {
236 return;
237 }
238
239 DEBUG0("Loading definition_helm parameters");
240 esio_attribute_read(h, location, attr_kp, &t.h.kp);
241 esio_attribute_read(h, location, attr_r, &t.r );
242 esio_attribute_read(h, location, attr_Td, &t.h.Td);
243 esio_attribute_read(h, location, attr_Tf, &t.h.Tf);
244 esio_attribute_read(h, location, attr_Ti, &t.h.Ti);
245 esio_attribute_read(h, location, attr_Tt, &t.h.Tt);
246
247 // Prefer incoming to temporary
248 this->populate(t, verbose);
249}
