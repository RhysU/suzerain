37// Helper method to turn, e.g., "USR1, sigusr2, 3 , Term "
38// into signal numbers, e.g. {SIGUSR1, SIGUSR2, 3, SIGTERM}.
39static std::vector<int> parse_spec(const std::string& name,
40 const std::string& spec)
41{
42 using boost::algorithm::is_any_of;
43 using boost::algorithm::split;
44 using boost::algorithm::trim;
45 using boost::bind;
46 using std::for_each;
47 using std::locale;
48 using std::remove;
49 using std::string;
50 using std::vector;
51
52 // Split comma-separated spec into whitespace-trimmed, non-empty tokens
53 vector<string> tokens;
54 split(tokens, spec, is_any_of(","));
55 for_each(tokens.begin(), tokens.end(), bind(trim<string>, _1, locale()));
56 tokens.erase(remove(tokens.begin(), tokens.end(), ""), tokens.end());
57
58 vector<int> retval;
59
60 // Convert symbolic names or integer strings into signal numbers
61 for (vector<string>::iterator iter = tokens.begin();
62 iter != tokens.end(); ++iter)
63 {
64
65 const string& token = *iter;
66
67 // First, try to lookup a named signal number
68 int signum = suzerain_signal_number(token.c_str());
69 if (signum == SUZERAIN_FAILURE)
70 {
71 // That failing, next, try to parse token as a positive integer
72 errno = 0;
73 char* end;
74 const long l = strtol(token.c_str(), &end, 0);
75 if (errno != 0 || *end != '\0' || l <= 0 || l > INT_MAX)
76 {
77 std::ostringstream oss;
78 oss << "Signal specification '"
79 << token
80 << "' provided for "
81 << name
82 << " is neither a known signal"
83 " nor a strictly positive integer";
84 throw std::invalid_argument(oss.str());
85 }
86 signum = l;
87 }
88 retval.push_back(signum);
89 }
90
91 return retval;
92}

113definition_signal::options_description()
114{
115 using boost::algorithm::join;
116 using boost::program_options::options_description;
117 using boost::program_options::value;
118 using std::back_insert_iterator;
119 using std::string;

125 // Build human-readable default strings based on the current values

140
141 vector<string> names_teardown;
142 transform(teardown.begin(), teardown.end(),
143 back_insert_iterator<vector<string> >(names_teardown),
144 &suzerain_signal_name);
145
146 vector<string> names_halt;
147 transform(halt.begin(), halt.end(),
148 back_insert_iterator<vector<string> >(names_halt),
149 &suzerain_signal_name);
150
151 // Construct the collection of options to be returned
152 retval.add_options()
153 ("signal_status", value<std::string>()

158 ("signal_restart", value<std::string>()

160 ->notifier(boost::bind(&definition_signal::parse_restart, this, _1)),
161 "Write restart file on any signal in this comma-separated list")
162
163 ("signal_statistics", value<std::string>()
164 ->default_value(join(names_statistics, ","))
165 ->notifier(boost::bind(&definition_signal::parse_statistics, this, _1)),
166 "Write statistics file on any signal in this comma-separated list")
167
168 ("signal_teardown", value<std::string>()
169 ->default_value(join(names_teardown, ","))
170 ->notifier(boost::bind(&definition_signal::parse_teardown, this, _1)),
171 "Tear down simulation on any signal in this comma-separated list")
172
173 ("signal_halt", value<std::string>()
174 ->default_value(join(names_halt, ","))
175 ->notifier(boost::bind(&definition_signal::parse_halt, this, _1)),
176 "Halt simulation on any signal in this comma-separated list")
177 ;
178
179 return retval;
180}
