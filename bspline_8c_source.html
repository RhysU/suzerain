46int
47suzerain_bspline_linear_combination(
48 const size_t nderiv,
49 const double * coeffs,
50 const size_t npoints,
51 const double * points,
52 double * values,
53 const size_t ldvalues,
54 gsl_matrix *dB,
55 gsl_bspline_workspace *w)
56{
57 /* Parameter sanity checks */
58 if (SUZERAIN_UNLIKELY(ldvalues && ldvalues < npoints)) {
59 SUZERAIN_ERROR("ldvalues too small for npoints", SUZERAIN_EINVAL);
60 }
61
62 /* ldvalues == 0 signals that we only want derivative nderiv */
63 const size_t jstart = (ldvalues == 0) ? nderiv : 0;
64 const size_t korder = gsl_bspline_order(w);
65
66 size_t istart;
67 for (size_t i = 0; i < npoints; ++i) {
68
69 gsl_bspline_basis_deriv(points[i], nderiv, dB, &istart, w);
70
71 const double * coeff_start = coeffs + istart;
72
73 for (size_t j = jstart; j <= nderiv; ++j) {
74 double dot = 0.0;
75 for (size_t k = 0; k < korder; ++k) {
76 dot += coeff_start[k] * (dB->data + j)[k * dB->tda];
77 }
78 values[i + j * ldvalues] = dot;
79 }
80 }
81
82 return SUZERAIN_SUCCESS;
83}

86suzerain_bspline_linear_combination_complex(
87 const size_t nderiv,
88 const complex_double *coeffs,

97 if (SUZERAIN_UNLIKELY(ldvalues && ldvalues < npoints)) {
98 SUZERAIN_ERROR("ldvalues too small for npoints", SUZERAIN_EINVAL);
99 }
100
101 /* ldvalues == 0 signals that we only want derivative nderiv */
102 const size_t jstart = (ldvalues == 0) ? nderiv : 0;
103 const size_t korder = gsl_bspline_order(w);
104
105 size_t istart;
106 for (size_t i = 0; i < npoints; ++i) {
107
108 gsl_bspline_basis_deriv(points[i], nderiv, dB, &istart, w);
109
110 const complex_double * coeff_start = coeffs + istart;
111
112 for (size_t j = jstart; j <= nderiv; ++j) {
113 double dotr = 0.0, doti = 0.0;
114 for (size_t k = 0; k < korder; ++k) {
115 const double basis_value = (dB->data + j)[k * dB->tda];
116 dotr += creal(coeff_start[k]) * basis_value;
117 doti += cimag(coeff_start[k]) * basis_value;
118 }
119 values[i + j*ldvalues] = complex_double(dotr, doti);
120 }
121 }
122
123 return SUZERAIN_SUCCESS;
124}

135// A GSL-ready way to evaluate a B-spline function plus offset at some point x
136static
137double linear_combination_function(

139{
140 double retval = GSL_NAN;

147// A workhorse method for suzerain_bspline_crossing avoiding allocation.
148// Exists to permit suzerain_bspline_crossing_first to be less expensive.
149// Argument s must be a valid gsl_root_fsolver instance on entry.
150static int
151suzerain_bspline_crossing_internal(
152 const size_t nderiv,
153 const double * coeffs,
154 const double value,
155 double * lower,
156 double * upper,
157 const size_t maxiter,
158 const double epsabs,
159 const double epsrel,
160 double * location,
161 gsl_matrix *dB,
162 gsl_bspline_workspace *w,
163 gsl_root_fsolver *s)
164{
165 // Wrap the incoming parameters into an gsl_function for evaluation
166 linear_combination_params params = { nderiv, coeffs, dB, w, -value };
167 gsl_function f = { linear_combination_function, &params};
168
169 // Initialize fsolver on [lower, upper]
170 // Bracketing, rather than fdfsolver, avoids exiting user-specified region
171 gsl_error_handler_t * h = gsl_set_error_handler_off(); // Push handler
172 int status = gsl_root_fsolver_set(s, &f, *lower, *upper);
173 status = (GSL_SUCCESS == status) ? GSL_CONTINUE : status;
174
175 // Proceed until success, failure, or maximum iterations reached
176 // On success, overwrite *location as described in API documentation.
177 *location = GSL_NAN;
178 for (size_t iter = 1; iter < maxiter && status == GSL_CONTINUE; ++iter) {
179 if (GSL_SUCCESS != (status = gsl_root_fsolver_iterate(s))) break;
180 *lower = gsl_root_fsolver_x_lower(s);
181 *upper = gsl_root_fsolver_x_upper(s);
182 status = gsl_root_test_interval(*lower, *upper, epsabs, epsrel);
183 }
184 if (status == GSL_SUCCESS) {
185 *location = gsl_root_fsolver_root(s);
186 }
187 gsl_set_error_handler(h); // Pop handler
188 return status;
189}

191int
192suzerain_bspline_crossing(
193 const size_t nderiv,
194 const double * coeffs,
195 const double value,
196 double * lower,
197 double * upper,
198 const size_t maxiter,
199 const double epsabs,
200 const double epsrel,
201 double * location,
202 gsl_matrix *dB,
203 gsl_bspline_workspace *w)
204{
205 // Initialize fsolver to use Brent-Dekker
206 // Bracketing, rather than fdfsolver, avoids exiting user-specified region
207 gsl_root_fsolver * const s = gsl_root_fsolver_alloc(gsl_root_fsolver_brent);
208 if (SUZERAIN_UNLIKELY(s == NULL)) {
209 SUZERAIN_ERROR("Could not obtain gsl_root_fsolver", SUZERAIN_ENOMEM);
210 }
211
212 // Invoke internal method relying on pre-existing workspace
213 const int status = suzerain_bspline_crossing_internal(
214 nderiv, coeffs, value, lower, upper, maxiter,
215 epsabs, epsrel, location, dB, w, s);
216
217 // Clean up and return
218 gsl_root_fsolver_free(s);
219 return status;
220}

222int
223suzerain_bspline_integration_coefficients(
224 const size_t nderiv,
225 double * coeffs,
226 size_t inc,
227 double lo,
228 double hi,
229 gsl_matrix *dB,
230 gsl_bspline_workspace *w)
231{
232 /* Obtain an appropriate order Gauss-Legendre integration rule */
233 gsl_integration_glfixed_table * const tbl
234 = gsl_integration_glfixed_table_alloc((gsl_bspline_order(w) - nderiv + 1)/2);
235 if (SUZERAIN_UNLIKELY(tbl == NULL)) {
236 SUZERAIN_ERROR("failed to obtain Gauss-Legendre rule from GSL",
237 SUZERAIN_ESANITY);
238 }
239
240 /* Zero integration coefficient values */
241 for (size_t i = 0; i < gsl_bspline_ncontrol(w); ++i) {
242 coeffs[i * inc] = 0.0;
243 }
244
245 /* Accumulate the breakpoint-by-breakpoint contributions to coeffs */
246 double xj = 0, wj = 0;
247 for (size_t i = 0; i < (w->nbreak - 1); ++i) {
248
249 /* Determine "order-matching intersection" of i-th */
250 /* breakpoint interval with integration bounds (lo, hi). */
251 /* Function omsect() is just the magic we need here to */
252 /* make partial integration intervals and hi < lo work. */
253 double a, b;
254 if (omsect(gsl_bspline_breakpoint(i, w),
255 gsl_bspline_breakpoint(i+1, w),
256 lo, hi, &a, &b)) {
257
258 for (size_t j = 0; j < tbl->n; ++j) {
259
260 /* Get j-th Gauss point xj and weight wj */
261 gsl_integration_glfixed_point(a, b, j, &xj, &wj, tbl);
262
263 /* Evaluate basis functions at point xj */
264 size_t kstart;
265 gsl_bspline_basis_deriv(xj, nderiv, dB, &kstart, w);
266 const size_t kend = kstart + gsl_bspline_order(w) - 1;
267
268 /* Accumulate weighted basis evaluations into coeffs */
269 for (size_t k = kstart; k <= kend; ++k) {
270 coeffs[k * inc] += wj * gsl_matrix_get(dB,
271 k - kstart, nderiv);
272 }
273 }
274
275 }
276 }
277
278 /* Free integration rule resources */
279 gsl_integration_glfixed_table_free(tbl);
280
281 return SUZERAIN_SUCCESS;
282}

284double
285suzerain_bspline_distance(
286 const gsl_bspline_workspace *a,
287 const gsl_bspline_workspace *b)
288{
289 double retval = GSL_DBL_MAX;
290 const size_t a_nknot = a->knots->size;
291 const size_t b_nknot = b->knots->size;
292 if ( gsl_bspline_order(a) == gsl_bspline_order(b)
293 && gsl_bspline_ncontrol(a) == gsl_bspline_ncontrol(b)
294 && a_nknot == b_nknot) {
295 retval = 0;
296 for (size_t i = 0; i < a_nknot; ++i) {
297 const double dist_i = fabs( gsl_vector_get(a->knots, i)
298 - gsl_vector_get(b->knots, i));
299 retval = GSL_MAX(retval, dist_i);
300 }
301 }
302 return retval;
303}

307double
308suzerain_bspline_spacing_greville_abscissae(
309 size_t i,
310 gsl_bspline_workspace *w)
311{
312 /* Find nearest abscissae indices silently folding back into range */
313 const size_t n = gsl_bspline_ncontrol(w);
314 size_t im = (i == 0 ) ? 1 : i - 1;
315 size_t ip = (i == n - 1) ? n - 2 : i + 1;
316
317 /* Compute and return the minimum distance */
318 double x = gsl_bspline_greville_abscissa(i, w);
319 double xm = gsl_bspline_greville_abscissa(im, w);
320 double xp = gsl_bspline_greville_abscissa(ip, w);
321
322 double dxm = fabs(x - xm);
323 double dxp = fabs(xp - x);
324 return dxm < dxp ? dxm : dxp;
325}

327double
328suzerain_bspline_spacing_breakpoints(
329 size_t i,
330 gsl_bspline_workspace *w)
331{
332 // Make indices {first,...,last} span the required portion of the knots
333 size_t stride = w->knots->stride;
334 size_t k = gsl_bspline_order(w);
335 double * first = w->knots->data + (i )*stride;
336 double * last = w->knots->data + (i+k)*stride;
337
338 // Determine the minimum non-zero spacing between adjacent knots
339 double retval = GSL_DBL_MAX;
340 while (first != last) {
341
342 double left = *first;
343 first += stride;
344 double right = *first;
345
346 double dist = right - left;
347 if (dist > 0 && dist < retval) {
348 retval = dist;
349 }
350 }
351 return retval;
352}
