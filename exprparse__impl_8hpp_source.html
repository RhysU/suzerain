38template<typename FPT>
39FPT exprparse_impl(const char *s, const char *name)
40{
41 using namespace std;
42 const char * iter = s;
43 const char * const end = iter + strlen(s);
44 FPT result;
45
46 if (!exprgrammar::parse(iter, end, result)) {
47 ostringstream m;
48 if (name) {
49 m << "exprparse error in " << name << " at '";
50 } else {
51 m << "exprparse error at '";
52 }
53 copy(iter, end, ostream_iterator<const char>(m));
54 m << '\'';
55 throw invalid_argument(m.str());
56 }
57
58 if (iter != end) {
59 ostringstream m;
60 m << "exprparse halted at position " << distance(s, iter) << " in ";
61 if (name) {
62 m << name << ' ';
63 }
64 m << '\'' << s << '\'';
65 throw invalid_argument(m.str());
66 }
67
68 return result;
69}

118template<typename T>
119void exprparse_range_impl(const std::string& s,
120 T *valmin, T *valmax,
121 const T defaultmin, const T defaultmax,
122 const T absmin, const T absmax,
123 const char *name)
124{
125 // Sanity check incoming arguments and provide default name behavior
126 using namespace std;
127 if (!(absmin <= defaultmin))
128 throw invalid_argument("!(absmin <= defaultmin)");
129 if (!(defaultmin <= defaultmax))
130 throw invalid_argument("!(defaultmin <= defaultmax)");
131 if (!(defaultmax <= absmax ))
132 throw invalid_argument("!(defaultmax <= absmax)");
133 if (!name) name = "exprparse_range input";
134
135 // Split s on a mandatory colon into whitespace-trimmed s_{min,max}
136 const size_t colonpos = s.find_first_of(':');
137 if (colonpos == string::npos) {
138 throw invalid_argument(string(name)
139 + " not in format \"low:high\", \"[low]:high\", or low:[high].");
140 }
141 string s_min(s, 0, colonpos);
142 string s_max(s, colonpos + 1);
143 boost::algorithm::trim(s_min);
144 boost::algorithm::trim(s_max);
145
146 // Parse recognized formats into valmin and valmax
147 if (s_min.length() == 0 && s_max.length() == 0) {
148 throw invalid_argument(string(name)
149 + " not in format \"low:high\", \"[low]:high\", or low:[high].");
150 } else if (s_min.length() == 0) {
151 *valmin = defaultmin;
152 *valmax = exprparse_impl<T>(s_max, name);
153 } else if (s_max.length() == 0) {
154 *valmin = exprparse_impl<T>(s_min, name);
155 *valmax = defaultmax;
156 } else {
157 *valmin = exprparse_impl<T>(s_min, name);
158 *valmax = exprparse_impl<T>(s_max, name);
159 }
160
161 // Ensure valmin <= valmax
162 if (*valmin > *valmax) swap(*valmin, *valmax);
163
164 // Validate range is within [absmin, absmax]
165 if (*valmin < absmin || absmax < *valmax) {
166 ostringstream m;
167 m << name << " value [" << *valmin << ":" << *valmax
168 << "] is outside valid range [" << absmin << ":" << absmax << "]";
169 throw invalid_argument(m.str());
170 }
171}
