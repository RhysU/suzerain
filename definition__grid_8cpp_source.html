44 const std::string& s,

158boost::program_options::options_description

160{
161 using boost::bind;
162 using boost::lexical_cast;
163 using boost::program_options::options_description;
164 using boost::program_options::typed_value;

171 options_description retval(
172 "Mixed Fourier/B-spline computational grid definition");
173
174 // Used to help resolve pointers-to-members taking strings
175 specification_grid& (specification_grid::*f)(const string&) = NULL;
176
177 unique_ptr<typed_value<string> > p;
178
179 // Lx
180 p.reset(value<string>());
181 p->notifier(bind(&validate_option<real_t>, _1, &L.x(),
182 &ensure_positive<real_t>, name_Lx));
183 if (!(boost::math::isnan)(L.x())) {
184 p->default_value(lexical_cast<string>(L.x()));
185 }
186 retval.add_options()(name_Lx, p.release(), desc_Lx);
187
188 // Nx
189 p.reset(value<string>());
190 f = &specification_grid::Nx;
191 p->notifier(bind(f, this, _1));
192 if (N.x()) {
193 p->default_value(lexical_cast<string>(N.x()));
194 }
195 retval.add_options()(name_Nx, p.release(), desc_Nx);

202 p->default_value(lexical_cast<string>(DAF.x()));

204 retval.add_options()(name_DAFx, p.release(), desc_DAFx);

370void
371definition_grid::save(
372 const esio_handle h) const
373{
374 // Only root writes data
375 int procid;
376 esio_handle_comm_rank(h, &procid);
377
378 DEBUG0("Saving definition_grid parameters");
379
380 esio_line_establish(h, 1, 0, (procid == 0 ? 1 : 0));
381
382 esio_line_write(h, name_Lx, &this->L.x(), 0, desc_Lx );
383 esio_line_write(h, name_Nx, &this->N.x(), 0, desc_Nx );
384 esio_line_write(h, name_DAFx, &this->DAF.x(), 0, desc_DAFx );
385 esio_line_write(h, name_Ly, &this->L.y(), 0, desc_Ly );
386 esio_line_write(h, name_Ny, &this->N.y(), 0, desc_Ny );
387 esio_line_write(h, name_k, &this->k, 0, desc_k );
388 esio_line_write(h, name_htdelta, &this->htdelta, 0, desc_htdelta);
389 esio_line_write(h, name_Lz, &this->L.z(), 0, desc_Lz );
390 esio_line_write(h, name_Nz, &this->N.z(), 0, desc_Nz );
391 esio_line_write(h, name_DAFz, &this->DAF.z(), 0, desc_DAFz );
392
393 DEBUG0("Storing wavenumber vectors for Fourier bases");
394 ArrayXc cbuf(std::max(N.x()/2+1, N.z()));
395
396 // Obtain Hermitian symmetric X wavenumbers via computing Re(1*(i*kx)/i)
397 // Notice N.x() and dN.x() represent logical, not stored, extents.
398 cbuf.fill(complex_t(1,0));
399 diffwave::apply(1, 0, complex_t(0,-1), cbuf.data(),
400 L.x(), L.z(),
401 1, N.x(), dN.x(), 0, N.x()/2+1, 1, 1, 0, 1);
402 esio_line_establish(h, N.x()/2+1, 0, (procid == 0 ? N.x()/2+1 : 0));
403 esio_line_write(h, "kx", reinterpret_cast<real_t *>(cbuf.data()), 2,
404 "Wavenumbers in streamwise Hermitian-symmetric X direction");
405
406 // Obtain Z wavenumbers via computing Re(1*(i*kz)/i)
407 cbuf.fill(complex_t(1,0));
408 diffwave::apply(0, 1, complex_t(0,-1), cbuf.data(),
409 L.x(), L.z(),
410 1, 1, 1, 0, 1, N.z(), N.z(), 0, N.z());
411 esio_line_establish(h, N.z(), 0, (procid == 0 ? N.z() : 0));
412 esio_line_write(h, "kz", reinterpret_cast<real_t *>(cbuf.data()), 2,
413 "Wavenumbers in spanwise Z direction");
414
415 DEBUG0("Storing collocation point vectors for Fourier bases");
416 ArrayXr rbuf(std::max(dN.x(), dN.z()));
417
418 // Save collocation points in x
419 for (int i = 0; i < dN.x(); ++i) {
420 rbuf[i] = this->x(i);
421 }
422 esio_line_establish(h, dN.x(), 0, (procid == 0 ? dN.x() : 0));
423 esio_line_write(h, "collocation_points_x", rbuf.data(), 0,
424 "Collocation points for the dealiased, streamwise X direction");
425
426 // Save collocation points in z
427 for (int k = 0; k < dN.z(); ++k) {
428 rbuf[k] = this->z(k);
429 }
430 esio_line_establish(h, dN.z(), 0, (procid == 0 ? dN.z() : 0));
431 esio_line_write(h, "collocation_points_z", rbuf.data(), 0,
432 "Collocation points for the dealiased, spanwise Z direction");
433}

435void
436definition_grid::load(
437 const esio_handle h,
438 const bool verbose)
439{
440 DEBUG0("Loading definition_grid parameters");
441
442 esio_line_establish(h, 1, 0, 1); // All ranks load
443
444 definition_grid t;
445 esio_line_read(h, name_Lx, &t.L.x(), 0);
446 {
447 int value;
448 esio_line_read(h, name_Nx, &value, 0);
449 t.Nx(value);
450 }
451 {
452 double factor;
453 esio_line_read(h, name_DAFx, &factor, 0);
454 t.DAFx(factor);
455 }
456 esio_line_read(h, name_Ly, &t.L.y(), 0);
457 {
458 int value;
459 esio_line_read(h, name_Ny, &value, 0);
460 t.Ny(value);
461 }
462 esio_line_read(h, name_k, &t.k, 0);
463 esio_line_read(h, name_htdelta, &t.htdelta, 0);
464 esio_line_read(h, name_Lz, &t.L.z(), 0);
465 {
466 int value;
467 esio_line_read(h, name_Nz, &value, 0);
468 t.Nz(value);
469 }
470 {
471 double factor;
472 esio_line_read(h, name_DAFz, &factor, 0);
473 t.DAFz(factor);
474 }
475 this->populate(t, verbose); // Prefer incoming to temporary
476}
