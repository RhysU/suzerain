42std::vector<field_L2xyz>
43compute_field_L2xyz(
44 const contiguous_state<4,complex_t> &state,
45 const specification_grid& grid,
46 const pencil_grid& dgrid,
47 const bsplineop& gop)
48{
49 // Ensure state storage meets this routine's assumptions
50 // Notice state.shape()[0] may be any value
51 using boost::numeric_cast;
52 assert(numeric_cast<int>(state.shape()[1]) == dgrid.local_wave_extent.y());
53 assert(numeric_cast<int>(state.shape()[2]) == dgrid.local_wave_extent.x());
54 assert(numeric_cast<int>(state.shape()[3]) == dgrid.local_wave_extent.z());
55
56 // Ensure we were handed Galerkin field_L2xyz operator matrices
57 SUZERAIN_ENSURE(gop.get()->method==SUZERAIN_BSPLINEOP_GALERKIN_L2);
58
59 // Only want non-dealiased X-direction modes to contribute to L2
60 // Compute wavenumber translation logistics for X direction
61 int fxb[2], fxe[2], mxb[2], mxe[2];
62 inorder::wavenumber_translate(grid.N.x(),
63 grid.dN.x(),
64 dgrid.local_wave_start.x(),
65 dgrid.local_wave_end.x(),
66 fxb[0], fxe[0], fxb[1], fxe[1],
67 mxb[0], mxe[0], mxb[1], mxe[1]);
68 // X contains only positive wavenumbers => second range must be empty
69 assert(fxb[1] == fxe[1]);
70 assert(mxb[1] == mxe[1]);
71
72 // Only want non-dealiased Z-direction modes to contribute to L2
73 // Compute wavenumber translation logistics for Z direction
74 // One or both ranges may be empty
75 int fzb[2], fze[2], mzb[2], mze[2];
76 inorder::wavenumber_translate(grid.N.z(),
77 grid.dN.z(),
78 dgrid.local_wave_start.z(),
79 dgrid.local_wave_end.z(),
80 fzb[0], fze[0], fzb[1], fze[1],
81 mzb[0], mze[0], mzb[1], mze[1]);
82
83 // Temporary storage for inner product computations
84 //
85 // Though u^H M u is always real for SPD M, floating point errors may
86 // produce a slightly non-SPD M so we permit complex intermediate results
87 // up until the return structure is packed.
88 VectorXc tmp;
89 tmp.setZero(grid.N.y());
90
91 // Contiguous temporary storage for accumulating and broadcasting results
92 VectorXc buf(2*state.shape()[0]);
93 Map<VectorXc> total2(buf.data() + 0, state.shape()[0]);
94 Map<VectorXc> mean2 (buf.data() + state.shape()[0], state.shape()[0]);
95
96 // Compute the local L2 contribution towards each L^2_xyz norm squared
97 // Computation uses partial sums at each loop to reduce swamping which is
98 // more-or-less recursive summation using large partitioning factors.
99 total2.setZero();
100 for (size_t k = 0; k < state.shape()[0]; ++k) {
101 complex_t jsum = 0;
102 for (int j = 0; j < 2; ++j) {
103 complex_t nsum = 0;
104 for (int n = mzb[j]; n < mze[j]; ++n) {
105 complex_t msum = 0;
106 for (int m = mxb[0]; m < mxe[0]; ++m) {
107 const complex_t * u_mn
108 = &state[k][0][m - dgrid.local_wave_start.x()]
109 [n - dgrid.local_wave_start.z()];
110 gop.accumulate(0, 1.0, u_mn, 1, 0.0, tmp.data(), 1);
111 complex_t dot = blas::dot(grid.N.y(), u_mn, 1,
112 tmp.data(), 1);
113 if (m > 0 && m < grid.dN.x()/2) {
114 dot *= 2;
115 }
116 msum += dot;
117 }
118 nsum += msum;
119 }
120 jsum += nsum;
121 }
122 total2[k] += jsum;
123 }
124 total2 *= grid.L.x() * grid.L.z();
125
126 // Compute the mean contributions to each L^2_xyz norm squared.
127 if (dgrid.has_zero_zero_modes()) {
128 for (size_t k = 0; k < state.shape()[0]; ++k) {
129 const complex_t * u_mn = &state[k][0][0][0];
130 gop.accumulate(0, 1.0, u_mn, 1, 0.0, tmp.data(), 1);
131 mean2[k] = blas::dot(grid.N.y(), u_mn, 1, tmp.data(), 1);
132 }
133 mean2 *= grid.L.x() * grid.L.z();
134 } else {
135 mean2.setZero();
136 }
137
138 // Allreduce total2 and mean2 simultaneously.
139 // (Though it moves more overall data, Allreduce chosen as Redmine #3079
140 // suspected a race-related hang from earlier Reduce/Bcast implementation).
141 //
142 // Reduction operation is complex-valued, but OpenMPI pre-1.7.3 can bomb
143 // unless we keep it real: https://svn.open-mpi.org/trac/ompi/ticket/3127.
144 SUZERAIN_MPICHKR(MPI_Allreduce(MPI_IN_PLACE, buf.data(),
145 (sizeof(complex_t)/sizeof(real_t))*buf.size(),
146 mpi::datatype<real_t>(), MPI_SUM, MPI_COMM_WORLD));
147
148 // Obtain fluctuating2 = total2 - mean2 and pack the return structure
149 std::vector<field_L2xyz> retval(state.shape()[0]);
150 for (size_t k = 0; k < retval.size(); ++k) {
151 retval[k].mean = std::sqrt(std::abs(mean2[k] ));
152 retval[k].fluctuating = std::sqrt(std::abs(total2[k] - mean2[k]));
153 }
154
155 return retval;
156}

162std::vector<field_L2xz>
163compute_field_L2xz(
164 const contiguous_state<4,complex_t> &state,
165 const specification_grid& grid,
166 const pencil_grid& dgrid,
167 const bsplineop& cop)
168{
169 // Ensure state storage meets this routine's assumptions
170 // Notice state.shape()[0] may be any value
171 using boost::numeric_cast;
172 assert(numeric_cast<int>(state.shape()[1]) == dgrid.local_wave_extent.y());
173 assert(numeric_cast<int>(state.shape()[2]) == dgrid.local_wave_extent.x());
174 assert(numeric_cast<int>(state.shape()[3]) == dgrid.local_wave_extent.z());
175
176 // Ensure we were handed collocation-based operator matrices
177 SUZERAIN_ENSURE(cop.get()->method==SUZERAIN_BSPLINEOP_COLLOCATION_GREVILLE);
178
179 // Only want non-dealiased X-direction modes to contribute to L2
180 // Compute wavenumber translation logistics for X direction
181 int fxb[2], fxe[2], mxb[2], mxe[2];
182 inorder::wavenumber_translate(grid.N.x(),
183 grid.dN.x(),
184 dgrid.local_wave_start.x(),
185 dgrid.local_wave_end.x(),
186 fxb[0], fxe[0], fxb[1], fxe[1],
187 mxb[0], mxe[0], mxb[1], mxe[1]);
188 // X contains only positive wavenumbers => second range must be empty
189 assert(fxb[1] == fxe[1]);
190 assert(mxb[1] == mxe[1]);
191
192 // Only want non-dealiased Z-direction modes to contribute to L2
193 // Compute wavenumber translation logistics for Z direction
194 // One or both ranges may be empty
195 int fzb[2], fze[2], mzb[2], mze[2];
196 inorder::wavenumber_translate(grid.N.z(),
197 grid.dN.z(),
198 dgrid.local_wave_start.z(),
199 dgrid.local_wave_end.z(),
200 fzb[0], fze[0], fzb[1], fze[1],
201 mzb[0], mze[0], mzb[1], mze[1]);
202
203 // Temporary complex-valued storage for matrix-vector products
204 ArrayXc tmp;
205 tmp.setZero(grid.N.y());
206
207 // Contiguous temporary storage for accumulating and broadcasting results
208 // Fast columns store wall-normal swaths of data.
209 // Real-valued since cwiseAbs2 always produces real-valued results.
210 ArrayXXr buf(grid.N.y(), 2*state.shape()[0]);
211 ArrayXXr::ColsBlockXpr total2 = buf.leftCols (state.shape()[0]);
212 ArrayXXr::ColsBlockXpr mean2 = buf.rightCols(state.shape()[0]);
213
214 // Compute the local L2 contribution towards each L^2_xz norm squared
215 // Computation uses partial sums at each loop to reduce swamping which is
216 // more-or-less recursive summation using large partitioning factors.
217 ArrayXr jsum, nsum, msum;
218 total2.setZero();
219 for (size_t k = 0; k < state.shape()[0]; ++k) {
220 jsum.setZero(grid.N.y());
221 for (int j = 0; j < 2; ++j) {
222 nsum.setZero(grid.N.y());
223 for (int n = mzb[j]; n < mze[j]; ++n) {
224 msum.setZero(grid.N.y());
225 for (int m = mxb[0]; m < mxe[0]; ++m) {
226 const complex_t * u_mn
227 = &state[k][0][m - dgrid.local_wave_start.x()]
228 [n - dgrid.local_wave_start.z()];
229 if (m > 0 && m < grid.dN.x()/2) {
230 // Conjugate symmetry uses root-2 so Abs2 below gives 2
231 cop.accumulate(0, M_SQRT2, u_mn, 1, 0., tmp.data(), 1);
232 } else {
233 cop.accumulate(0, 1.0, u_mn, 1, 0., tmp.data(), 1);
234 }
235 msum += tmp.abs2();
236 }
237 nsum += msum;
238 }
239 jsum += nsum;
240 }
241 total2.col(k) += jsum;
242 }
243 total2 *= grid.L.x() * grid.L.z();
244
245 // Compute the mean contributions to each L^2_xyz norm squared.
246 if (dgrid.has_zero_zero_modes()) {
247 for (size_t k = 0; k < state.shape()[0]; ++k) {
248 const complex_t * u_mn = &state[k][0][0][0];
249 cop.accumulate(0, 1.0, u_mn, 1, 0.0, tmp.data(), 1);
250 mean2.col(k) = tmp.cwiseAbs2();
251 }
252 mean2 *= grid.L.x() * grid.L.z();
253 } else {
254 mean2.setZero();
255 }
256
257 // Allreduce total2 and mean2 simultaneously.
258 // (Though it moves more overall data, Allreduce chosen as Redmine #3079
259 // suspected a race-related hang from earlier Reduce/Bcast implementation).
260 SUZERAIN_MPICHKR(MPI_Allreduce(MPI_IN_PLACE, buf.data(),
261 buf.size(), mpi::datatype<real_t>(), MPI_SUM, MPI_COMM_WORLD));
262
263 // Obtain fluctuating2 = total2 - mean2 and pack the return structure
264 std::vector<field_L2xz> retval(state.shape()[0]);
265 for (size_t k = 0; k < retval.size(); ++k) {
266 retval[k].mean = mean2.col(k).sqrt();
267 retval[k].fluctuating = (total2.col(k) - mean2.col(k)).sqrt();
268 }
269
270 return retval;
271}

287 assert(numeric_cast<int>(state.shape()[2]) == dgrid.local_wave_extent.x());

296 dgrid.local_wave_end.x(),

300 assert(fxb[1] == fxe[1]);

308 grid.dN.z(),
309 dgrid.local_wave_start.z(),
310 dgrid.local_wave_end.z(),

315 // Fast columns store wall-normal swaths of data.
316 // Real-valued since cwiseAbs2 always produces real-valued results.
317 ArrayXXr buf(grid.N.y(), 2*state.shape()[0]);
318 ArrayXXr::ColsBlockXpr total2 = buf.leftCols (state.shape()[0]);
319 ArrayXXr::ColsBlockXpr mean2 = buf.rightCols(state.shape()[0]);
320

322 // Computation uses partial sums at each loop to reduce swamping which is

330 for (int n = mzb[j]; n < mze[j]; ++n) {
331 msum.setZero(grid.N.y());
332 for (int m = mxb[0]; m < mxe[0]; ++m) {
333 Map<const ArrayXc> u_mn(
334 &state[k][0][m - dgrid.local_wave_start.x()]
335 [n - dgrid.local_wave_start.z()],
336 grid.N.y());
337 if (m > 0 && m < grid.dN.x()/2) { // Conjugate symmetry
338 msum += 2*u_mn.abs2();
339 } else {
340 msum += u_mn.abs2();
341 }
342 }
343 nsum += msum;
344 }
345 jsum += nsum;
346 }
347 total2.col(k) += jsum;
348 }

357 mean2 *= grid.L.x() * grid.L.z();
358 } else {
359 mean2.setZero();
360 }
361

363 // (Though it moves more overall data, Allreduce chosen as Redmine #3079
364 // suspected a race-related hang from earlier Reduce/Bcast implementation).

372 retval[k].fluctuating = (total2.col(k) - mean2.col(k)).sqrt();
373 }
374
375 return retval;
376}

378void
379compute_twopoint_xlocal(
380 const contiguous_state<4,complex_t> &state,
381 const contiguous_state<4,complex_t>::index si,
382 const contiguous_state<4,complex_t>::index sj,
383 const specification_grid& grid,
384 const pencil_grid& dgrid,
385 complex_t * const out)
386{
387 // Ensure state matches assumptions; state.shape()[0] may be any value
388 using boost::numeric_cast;
389 SUZERAIN_ENSURE((int) state.shape()[1] == dgrid.local_wave_extent.y());
390 SUZERAIN_ENSURE((int) state.shape()[2] == dgrid.local_wave_extent.x());
391 SUZERAIN_ENSURE((int) state.shape()[3] == dgrid.local_wave_extent.z());
392 SUZERAIN_ENSURE(state.strides()[1] == (int) 1);
393 SUZERAIN_ENSURE(state.strides()[2] == (int) state.shape()[1]);
394 SUZERAIN_ENSURE(state.strides()[3] == (int) ( state.shape()[1]
395 *state.shape()[2]));
396
397 // Grab relevant parallel decomposition information
398 const int Ny = dgrid.global_wave_extent.y();
399 const int Nx = grid.N.x();
400 const int dNx = grid.dN.x();
401 const int dkbx = dgrid.local_wave_start.x();
402 const int dkex = dgrid.local_wave_end.x();
403 const int dkbz = dgrid.local_wave_start.z();
404 const int dkez = dgrid.local_wave_end.z();
405
406 // Prepare a view of the output storage and zero it prior to accumulation
407 Map<MatrixXXc> o(out, Ny, Nx/2+1);
408 o.setZero();
409
410 // Sum rank-local contribution to Fourier-transformed two-point correlation
411 // We /do/ sum dealiasing/Nyquist modes so clear a priori if that bugs you
412 // (though notice only Nx/2+1 and not dNx/2+1 modes are reported in result)
413 for (int n = dkbz; n < dkez; ++n) {
414 for (int m = dkbx; m < dkex; ++m) {
415 const int wm = inorder::wavenumber(dNx, m);
416 if (wm >= o.cols()) continue; // Ignored
417
418 Map<const VectorXc> u_mn(&state[si][0][m - dkbx][n - dkbz], Ny);
419 Map<const VectorXc> v_mn(&state[sj][0][m - dkbx][n - dkbz], Ny);
420
421 o.col(wm) += u_mn.conjugate().cwiseProduct(v_mn);
422 }
423 }
424
425 // Ensure real-valued R_uv(x) arises from Fourier-transforming result
426 // (occurs trivially for R_uu but dealiasing can break property for R_uv)
427 o.col(0).imag().setZero();
428 if (Nx%2 == 0) {
429 o.col(Nx/2).imag().setZero();
430 }
431}

433void
434compute_twopoint_zlocal(
435 const contiguous_state<4,complex_t> &state,
436 const contiguous_state<4,complex_t>::index si,
437 const contiguous_state<4,complex_t>::index sj,
438 const specification_grid& grid,
439 const pencil_grid& dgrid,
440 complex_t * const out)
441{
442 // Ensure state matches assumptions; state.shape()[0] may be any value
443 using boost::numeric_cast;
444 SUZERAIN_ENSURE((int) state.shape()[1] == dgrid.local_wave_extent.y());
445 SUZERAIN_ENSURE((int) state.shape()[2] == dgrid.local_wave_extent.x());
446 SUZERAIN_ENSURE((int) state.shape()[3] == dgrid.local_wave_extent.z());
447 SUZERAIN_ENSURE(state.strides()[1] == (int) 1);
448 SUZERAIN_ENSURE(state.strides()[2] == (int) state.shape()[1]);
449 SUZERAIN_ENSURE(state.strides()[3] == (int) ( state.shape()[1]
450 *state.shape()[2]));
451
452 // Grab relevant parallel decomposition information
453 const int Ny = dgrid.global_wave_extent.y();
454 const int dNx = grid.dN.x();
455 const int dkbx = dgrid.local_wave_start.x();
456 const int dkex = dgrid.local_wave_end.x();
457 const int Nz = grid.N.z();
458 const int dNz = grid.dN.z();
459 const int dkbz = dgrid.local_wave_start.z();
460 const int dkez = dgrid.local_wave_end.z();
461
462 // Prepare a view of the output storage and zero it prior to accumulation
463 Map<MatrixXXc> o(out, Ny, Nz);
464 o.setZero();
465
466 // Sum rank-local contribution to Fourier-transformed two-point correlation
467 // We /do/ sum dealiasing/Nyquist modes so clear a priori if that bugs you
468 for (int n = dkbz; n < dkez; ++n) {
469 if (!inorder::wavenumber_translatable(Nz, dNz, n)) continue; // Ignored
470 const int wn_idx = inorder::index(Nz, inorder::wavenumber(dNz, n));
471
472 for (int m = dkbx; m < dkex; ++m) {
473
474 Map<const VectorXc> u_mn(&state[si][0][m - dkbx][n - dkbz], Ny);
475 Map<const VectorXc> v_mn(&state[sj][0][m - dkbx][n - dkbz], Ny);
476
477 if (m == 0) {
478 o.col(wn_idx) += u_mn.conjugate().cwiseProduct(v_mn);
479 } else if (inorder::wavenumber_nyquist_index(dNx, m)) {
480 o.col(wn_idx) += u_mn.cwiseProduct(v_mn.conjugate());
481 } else {
482 // Admittedly, there are faster ways to compute this product
483 o.col(wn_idx) += u_mn.conjugate().cwiseProduct(v_mn)
484 + u_mn.cwiseProduct(v_mn.conjugate());
485 }
486 }
487 }
488
489 // Ensure real-valued R_uv(z) arises from Fourier-transforming result
490 // (occurs trivially for R_uu but dealiasing can break property for R_uv)
491 o.col(0).imag().setZero();
492 if (Nz%2 == 0) {
493 o.col(Nz/2).imag().setZero();
494 }
495}

497std::shared_ptr<complex_t[]>
498compute_twopoint_x(
499 const contiguous_state<4,complex_t> &state,
500 const contiguous_state<4,complex_t>::index nf,
501 const specification_grid& grid,
502 const pencil_grid& dgrid)
503{
504 SUZERAIN_ENSURE((int) state.shape()[0] <= (int) nf);
505
506 // Allocate contiguous storage for all the pairwise results
507 // As two-point is real-valued, only positive wavenumbers computed
508 const int npairs = (nf*(nf+1))/2;
509 const int bufpair = grid.N.y() * (grid.N.x()/2+1);
510 std::shared_ptr<complex_t[]> retval(
511 (complex_t*)suzerain_blas_malloc(sizeof(complex_t)*bufpair*npairs),
512 suzerain_blas_free);
513
514 // Compute result for each unique pair of scalar fields
515 for (int si = 0; si < nf; ++si) {
516 for (int sj = si; sj < nf; ++sj) {
517 const int ndxpair = nf*si + sj - (si*(si+1))/2;
518 compute_twopoint_xlocal(state, si, sj, grid, dgrid,
519 &retval[bufpair*ndxpair]);
520 }
521 }
522
523 // Perform the global summation across all pairs at once
524 //
525 // Reduction operation is complex-valued, but OpenMPI pre-1.7.3 can bomb
526 // unless we keep it real: https://svn.open-mpi.org/trac/ompi/ticket/3127.
527 SUZERAIN_MPICHKR(MPI_Allreduce(MPI_IN_PLACE, retval.get(),
528 (sizeof(complex_t)/sizeof(real_t))*bufpair*npairs,
529 mpi::datatype<real_t>(), MPI_SUM, MPI_COMM_WORLD));
530
531 return retval;
532}

534std::shared_ptr<complex_t[]>
535compute_twopoint_z(
536 const contiguous_state<4,complex_t> &state,
537 const contiguous_state<4,complex_t>::index nf,
538 const specification_grid& grid,
539 const pencil_grid& dgrid)
540{
541 SUZERAIN_ENSURE((int) state.shape()[0] <= (int) nf);
542
543 // Allocate contiguous storage for all the pairwise results
544 // As two-point is real-valued, only positive wavenumbers computed
545 const int npairs = (nf*(nf+1))/2;
546 const int bufpair = grid.N.y() * grid.N.z();
547 std::shared_ptr<complex_t[]> retval(
548 (complex_t*)suzerain_blas_malloc(sizeof(complex_t)*bufpair*npairs),
549 suzerain_blas_free);
550
551 // Compute result for each unique pair of scalar fields
552 for (int si = 0; si < nf; ++si) {
553 for (int sj = si; sj < nf; ++sj) {
554 const int ndxpair = nf*si + sj - (si*(si+1))/2;
555 compute_twopoint_zlocal(state, si, sj, grid, dgrid,
556 &retval[bufpair*ndxpair]);
557 }
558 }
559
560 // Perform the global summation across all pairs at once
561 //
562 // Reduction operation is complex-valued, but OpenMPI pre-1.7.3 can bomb
563 // unless we keep it real: https://svn.open-mpi.org/trac/ompi/ticket/3127.
564 SUZERAIN_MPICHKR(MPI_Allreduce(MPI_IN_PLACE, retval.get(),
565 (sizeof(complex_t)/sizeof(real_t))*bufpair*npairs,
566 mpi::datatype<real_t>(), MPI_SUM, MPI_COMM_WORLD));
567
568 return retval;
569}
