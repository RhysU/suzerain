90 template< typename EigenArray >
91 bool operator()(const std::string& name, const EigenArray& dat) const
92 {
93 int procid;
94 esio_handle_comm_rank(esioh, &procid);
95 esio_field_establish(esioh,
96 1, 0, (procid == 0 ? 1 : 0),
97 dat.cols(), 0, (procid == 0 ? dat.cols() : 0),
98 dat.rows(), 0, (procid == 0 ? dat.rows() : 0));
99
100 std::string key(prefix);
101 key.append(name);
102 return ESIO_SUCCESS == esio_field_write(
103 esioh, key.c_str(), dat.data(), 0, 0, 0,
104 "Mean quantity sample stored using row-major indices (B-spline"
105 " coefficient, tensor component, sample number) where the"
106 " B-spline basis is defined by /Ny, /breakpoints_y, and /knots");
107 }
108
109private:
110
111 std::string who;
112 esio_handle esioh;
113 std::string prefix;
114

130class quantities_loader
131{
132
133public:
134
135 quantities_loader(const std::string& who,
136 const esio_handle esioh,
137 const std::string& prefix)
138 : who(who), esioh(esioh), prefix(prefix) {}
139
140 template< typename EigenArray >
141 bool operator()(const std::string& name, const EigenArray& dat_) const {
142
143 // http://eigen.tuxfamily.org/dox/TopicFunctionTakingEigenTypes.html
144 EigenArray& dat = const_cast<EigenArray&>(dat_);
145
146 int procid;
147 esio_handle_comm_rank(esioh, &procid);
148
149 std::string key(prefix);
150 key.append(name);
151
152 int cglobal, bglobal, aglobal;
153 if ( ESIO_SUCCESS == esio_field_size(esioh, key.c_str(),
154 &cglobal, &bglobal, &aglobal)
155 && ( EigenArray::ColsAtCompileTime == Dynamic
156 || dat.cols() == bglobal)) {
157 dat.resize(aglobal, bglobal);
158 esio_field_establish(esioh,
159 1, 0, (procid == 0 ? 1 : 0),
160 dat.cols(), 0, (procid == 0 ? dat.cols() : 0),
161 dat.rows(), 0, (procid == 0 ? dat.rows() : 0));
162 return ESIO_SUCCESS == esio_field_read(
163 esioh, key.c_str(), dat.data(), 0, 0, 0);
164 } else {
165 WARN0(who, "Unable to load " << key
166 << " for nscalar = " << dat.cols());
167 dat.fill(std::numeric_limits<
168 typename EigenArray::Scalar>::quiet_NaN());
169 return false;
170 }
171 }
172
173private:
174
175 std::string who;
176 esio_handle esioh;
177 std::string prefix;
178
179};

181bool quantities_base::load(const esio_handle h)
182{
183 // Were any quantities loaded from file?
184 bool success = false;
185
186 // Defensively NaN out all storage in the instance prior to load.
187 this->storage.fill(std::numeric_limits<real_t>::quiet_NaN());
188
189 int cglobal, bglobal, aglobal;
190 if (ESIO_SUCCESS == esio_field_size(h, "bar_rho",
191 &cglobal, &bglobal, &aglobal)) {
192 this->storage.resize(aglobal, NoChange);
193 quantities_loader f(this->who, h, "bar_");
194 success = this->foreach(f);
195 } else {
196 WARN0(who, "No mean quantity samples loaded--"
197 " unable to anticipate storage needs");
198 }
199
200 return success;
201}

247bool quantities::load(const esio_handle h)
248{
249 bool retval = super::load(h);
250
251 // Defensively NaN out all storage in the instance prior to load.
252 this->species_storage.fill(std::numeric_limits<real_t>::quiet_NaN());
253
254 DEBUG0("Loading species statistics.");
255
256 int cglobal, bglobal, aglobal;
257 if (ESIO_SUCCESS == esio_field_size(h, "bar_rho_s",
258 &cglobal, &bglobal, &aglobal)) {
259 this->species_storage.resize(aglobal, 12*bglobal);
260
261 if (this->Ns==0)
262 { this->Ns = bglobal; }
263 else
264 { assert(static_cast<int>(this->Ns)==bglobal); }
265
266 quantities_loader f("quantities", h, "bar_");
267 retval &= f("rho_s", this->rho_s(0, Ns));
268 retval &= f("cs_s" , this->cs_s (0, Ns));
269 retval &= f("om_s" , this->om_s (0, Ns));
270 retval &= f("rho_s_u" , this->rho_s_u ());
271 retval &= f("rho_Ds_grad_cs" , this->rho_Ds_grad_cs ());
272 retval &= f("rho_Ds_grad_cs_hs", this->rho_Ds_grad_cs_hs());
273 } else {
274 WARN0("quantities", "No mean quantity samples loaded--"
275 " unable to anticipate storage needs");
276 }
277
278 return retval;
279}

282// This looks like logic from operator_nonlinear.hpp but does not belong there.
283// Reading through that file, especially the apply_operator implementation, is
284// recommended before reviewing this logic. This routine is definitely
285// suboptimal but is expected to be invoked relatively infrequently.
286quantities sample_quantities(
287 const antioch_constitutive& cmods,
288 const specification_grid &grid,
289 const pencil_grid &dgrid,
290 const bsplineop &cop,
291 contiguous_state<4,complex_t> &swave,
292 const real_t t)
293{
294 // FIXME: This function supports reacting in that all calculations
295 // here use appropriate constitutive laws, etc. However, not all
296 // stats we want, particularly species-related quantities, are
297 // currently computed.
298 if (cmods.Ns()>1) {
299 WARN0("sample_quantitites",
300 "sample_quantities does not fully support reacting flow yet!");
301 }
302
303 const size_t Ns = cmods.Ns();
304 const size_t state_count = Ns+4;
305
306 // Shorthand
307 const std::size_t Ny = swave.shape()[1];
308 namespace acc = boost::accumulators;
309 typedef contiguous_state<4,complex_t> state_type;
310
311 // State enters method as coefficients in X, Y, and Z directions
312
313 // We need auxiliary scalar-field storage. Prepare logical indices using a
314 // struct for scoping (e.g. aux::rho_y). Ordering will match usage below.
315
316 // Indices for directions
317 struct dir { enum {
318 y,
319 x,
320 z,
321 count
322 }; };
323
324 // Indices for auxiliary storage for state derivatives
325 // and heat flux related quantities
326 struct aux { enum {
327 T = 0,
328 gT = 1,
329 e = 1 + 1*dir::count,
330 mx = 1 + 2*dir::count,
331 my = 1 + 3*dir::count,
332 mz = 1 + 4*dir::count,
333 rho = 1 + 5*dir::count,
334 species = 1 + 6*dir::count
335 }; };
336
337 // Get total number of fields in aux storage
338 const size_t aux_count = (aux::species +
339 dir::count*(Ns-1) );
340
341 // Obtain the auxiliary storage (likely from a pool to avoid fragmenting).
342 // We assume no garbage values in the memory will impact us (for speed).
343 std::unique_ptr<state_type> _auxw_ptr(
344 support::allocate_padded_state<state_type>(
345 aux_count, dgrid)); // RAII
346 state_type &auxw = *_auxw_ptr; // Shorthand
347
348 // Ensure state storage meets this routine's assumptions
349
350 // Sanity check incoming swave's and auxw's shape and contiguity
351 SUZERAIN_ENSURE( swave.shape()[0] == state_count);
352 SUZERAIN_ENSURE(numeric_cast<int>(swave.shape()[1]) == dgrid.local_wave_extent.y());
353 SUZERAIN_ENSURE(numeric_cast<int>(swave.shape()[2]) == dgrid.local_wave_extent.x());
354 SUZERAIN_ENSURE(numeric_cast<int>(swave.shape()[3]) == dgrid.local_wave_extent.z());
355 SUZERAIN_ENSURE((unsigned) swave.strides()[1] == 1u);
356 SUZERAIN_ENSURE((unsigned) swave.strides()[2] == swave.shape()[1]);
357 SUZERAIN_ENSURE((unsigned) swave.strides()[3] == swave.shape()[1]*swave.shape()[2]);
358 SUZERAIN_ENSURE(std::equal(swave.shape() + 1, swave.shape() + 4,
359 auxw.shape() + 1));
360 SUZERAIN_ENSURE(std::equal(swave.strides() + 1, swave.strides() + 4,
361 auxw.strides() + 1));
362
363 // Rank-specific details accumulated in ret to be MPI_Allreduce later
364 //quantities ret(t, Ny);
365 quantities ret(t, Ny, Ns);
366
367 // Obtain samples available in wave-space from mean conserved state.
368 // These coefficients are inherently averaged across the X-Z plane.
369 if (dgrid.has_zero_zero_modes()) {
370 ret.rho_E() = Map<VectorXc>(swave[ndx::e ].origin(), Ny).real();
371 ret.rho_u().col(0) = Map<VectorXc>(swave[ndx::mx ].origin(), Ny).real();
372 ret.rho_u().col(1) = Map<VectorXc>(swave[ndx::my ].origin(), Ny).real();
373 ret.rho_u().col(2) = Map<VectorXc>(swave[ndx::mz ].origin(), Ny).real();
374 ret.rho() = Map<VectorXc>(swave[ndx::rho].origin(), Ny).real();
375 }
376
377 // Obtain access to helper routines for differentiation
378 operator_tools otool(grid, dgrid, cop);
379
380 for (size_t var = 0; var<state_count; ++var) {
381
382 // Compute Y derivatives of variable var at collocation points
383 // Zero wavenumbers present only for dealiasing along the way
384 otool.zero_dealiasing_modes(swave, var);
385 otool.bop_accumulate(1, 1, swave, var,
386 0, auxw , aux::e + dir::count*var + dir::y);
387 otool.bop_apply (0, 1, swave, var);
388
389 // Compute X- and Z- derivatives of variable var at collocation points
390 // Zeros wavenumbers present only for dealiasing in the target storage
391 otool.diffwave_accumulate(1, 0, 1, swave, var,
392 0, auxw , aux::e + dir::count*var + dir::x );
393 otool.diffwave_accumulate(0, 1, 1, swave, var,
394 0, auxw , aux::e + dir::count*var + dir::z );
395
396 }
397
398 // Collectively convert swave and auxw to physical space using parallel
399 // FFTs. In physical space, we'll employ views to reshape the 4D row-major
400 // (F, Y, Z, X) with contiguous (Y, Z, X) into a 2D (F, Y*Z*X) layout where
401 // we know F a priori. Reducing the dimensionality encourages linear
402 // access and eases indexing overhead.
403 physical_view<> auxp(dgrid, auxw);
404 physical_view<>sphys(dgrid, swave);
405 for (std::size_t i = 0; i < state_count; ++i) {
406 dgrid.transform_wave_to_physical(&sphys.coeffRef(i,0));
407 }
408 for (std::size_t i = 0; i < aux_count; ++i) {
409 dgrid.transform_wave_to_physical(&auxp.coeffRef(i,0));
410 }
411
412
413
414 // Before traversal, instantiate Eigen variable length arrays
415 // to store species specific info
416 VectorXr species(Ns); // species densities
417 VectorXr Ds (Ns); // mass diffusivities
418 VectorXr hs (Ns); // species enthalpies
419 VectorXr oms (Ns); // reaction source terms
420 VectorXr cs (Ns); // species mass fractions
421 VectorXr etots (Ns); // species internal energies
422
423 Matrix3Xr grad_species (3,Ns); // spatial derivatives of species densities
424 Matrix3Xr grad_cs (3,Ns); // spatial derivatives of species mass fracs
425 Matrix3Xr sdiff (3,Ns); // diffusive fluxes for species equations
426
427
428 //------------------------------------------------------------------
429 // Adapted from nonlinear, to get temperature gradient and heat flux
430 //------------------------------------------------------------------
431 //
432 // (1) Compute temperature and store the field
433 // (2) Take temperature from physical to wave space.
434 // (3) Differentiate temperature in wave space and bring grad(T)
435 // back to physical space.
436 // (4) (After going back to wave space with T and back down the
437 // grad(T)), Compute heat flux along wit all the other
438 // variables for sampling.
439
440 // (1) Traversal to compute temperature
441 real_t Tguess = -1;
442 for (int offset = 0, j = dgrid.local_physical_start.y();
443 j < dgrid.local_physical_end.y();
444 ++j) {
445
446 real_t T = -1;
447 for (int k = dgrid.local_physical_start.z();
448 k < dgrid.local_physical_end.z();
449 ++k) {
450
451 for (int i = dgrid.local_physical_start.x();
452 i < dgrid.local_physical_end.x();
453 ++i, /* NB */ ++offset) {
454
455 // Unpack total energy, momentum, and density
456 const real_t e (sphys(ndx::e, offset));
457 const Vector3r m (sphys(ndx::mx, offset),
458 sphys(ndx::my, offset),
459 sphys(ndx::mz, offset));
460 const real_t rho(sphys(ndx::rho, offset));
461
462 const real_t irho = 1.0/rho;
463
464 // Unpack species variables
465 // NOTE: In species vector, idx 0 is the dilluter (the species
466 // that is not explicitly part of the state vector)
467 // Compute mass fractions
468 species(0) = rho;
469 for (unsigned int s=1; s<Ns; ++s) {
470 species(s) = sphys(ndx::species + s - 1, offset);
471 cs(s) = irho * species(s);
472
473 // dilluter density = rho_0 = rho - sum_{s=1}^{Ns-1} rho_s
474 species(0) -= species(s);
475 }
476 cs(0) = irho * species(0);
477
478 // Compute temperature from internal energy
479 // (assuming thermal equilibrium)
480 const real_t re_internal = e -
481 0.5*irho*(m[0]*m[0] + m[1]*m[1] + m[2]*m[2]);
482 T = cmods.sm_thermo->T_from_e_tot(irho*re_internal, cs, Tguess);
483 Tguess = T;
484
485 auxp(aux::T , offset) = T;
486 } // end x
487 } // end z
488 } // end y
489
490 // (2a) Temperature from physical to wave space (still collocation in y)
491 otool.dgrid.transform_physical_to_wave(&auxp.coeffRef(aux::T,0));
492
493 // (2b) Apply inverse mass matrix to get to pure coefficient space
494 otool.bop_solve(*otool.massluz(), auxw, aux::T);
495
496 // ... and zero wavenumbers present only for dealiasing
497 // while simultaneously normalizing FFT
498 otool.diffwave_apply(0, 0, otool.dgrid.chi(), auxw, aux::T);
499
500 // (3) Get derivatives.
501 //
502 // Compute Y derivatives of variable var at collocation points
503 otool.bop_accumulate(1, 1, auxw, aux::T, 0, auxw, aux::gT + dir::y);
504
505 // Get to collocation points in preparation for computing x and z
506 // derivatives
507 otool.bop_apply (0, 1, auxw, aux::T);
508
509 // Compute X- and Z- derivatives of variable var at collocation points
510 // Zeros wavenumbers present only for dealiasing in the target storage
511 otool.diffwave_accumulate(1, 0, 1, auxw, aux::T, 0, auxw, aux::gT+dir::x);
512 otool.diffwave_accumulate(0, 1, 1, auxw, aux::T, 0, auxw, aux::gT+dir::z);
513
514 // FFTs to get to physical space
515 otool.dgrid.transform_wave_to_physical(&auxp.coeffRef(aux::gT+dir::y,0));
516 otool.dgrid.transform_wave_to_physical(&auxp.coeffRef(aux::gT+dir::x,0));
517 otool.dgrid.transform_wave_to_physical(&auxp.coeffRef(aux::gT+dir::z,0));
518
519 // Bring T to physical space as well to use the exact value as guess
520 // I'm not sure if doing this is less expensive though ...
521 otool.dgrid.transform_wave_to_physical(&auxp.coeffRef(aux::T,0));
522
523 // Now have temperature gradient at collocation points.
524 //-----------------------------------------------------------------
525
526
527 // (4) Physical space is traversed linearly using a single offset 'offset'.
528 // The three loop structure is present to provide the global absolute
529 // positions x(i), y(j), and z(k) where necessary.
530 for (int offset = 0, j = dgrid.local_physical_start.y();
531 j < dgrid.local_physical_end.y();
532 ++j) {
533
534 // Type used to accumulate mean quantities versus wall-normal position
535 typedef acc::accumulator_set<
536 real_t, acc::stats<acc::tag::sum_kahan>
537 > accumulator_type;
538
539 // Accumulators for each mean quantity computed in physical space.
540 // For example, quantity "foo" has accumulator "sum_foo".
541 // Declared within Y loop so they are reset on each Y iteration.
542#define DECLARE(r, data, tuple) \
543 accumulator_type BOOST_PP_CAT(sum_,BOOST_PP_TUPLE_ELEM(2, 0, tuple)) \
544 [BOOST_PP_TUPLE_ELEM(2, 1, tuple)];
545 BOOST_PP_SEQ_FOR_EACH(DECLARE,,
546 SUZERAIN_REACTING_FLOW_QUANTITIES_PHYSICAL)
547#undef DECLARE
548
549 // Vectors of species quantity accumulators
550 std::vector<accumulator_type> sum_rho_s (Ns);
551 std::vector<accumulator_type> sum_cs_s (Ns);
552 std::vector<accumulator_type> sum_om_s (Ns);
553 std::vector<accumulator_type> sum_rho_s_u (Ns*dir::count);
554 std::vector<accumulator_type> sum_rho_Ds_grad_cs (Ns*dir::count);
555 std::vector<accumulator_type> sum_rho_Ds_grad_cs_hs(Ns*dir::count);
556
557 for (int k = dgrid.local_physical_start.z();
558 k < dgrid.local_physical_end.z();
559 ++k) {
560
561 for (int i = dgrid.local_physical_start.x();
562 i < dgrid.local_physical_end.x();
563 ++i, /* NB */ ++offset) {
564
565 // Unpack total energy-related quantities
566 const real_t e(sphys(ndx::e, offset));
567 const Vector3r grad_e(auxp(aux::e+dir::x, offset),
568 auxp(aux::e+dir::y, offset),
569 auxp(aux::e+dir::z, offset));
570
571 // Unpack momentum-related quantities
572 const Vector3r m(sphys(ndx::mx, offset),
573 sphys(ndx::my, offset),
574 sphys(ndx::mz, offset));
575 const real_t div_m = ( auxp(aux::mx+dir::x, offset)
576 + auxp(aux::my+dir::y, offset)
577 + auxp(aux::mz+dir::z, offset));
578
579 const Matrix3r grad_m;
580 const_cast<Matrix3r&>(grad_m) <<
581 auxp(aux::mx+dir::x, offset),
582 auxp(aux::mx+dir::y, offset),
583 auxp(aux::mx+dir::z, offset),
584 auxp(aux::my+dir::x, offset),
585 auxp(aux::my+dir::y, offset),
586 auxp(aux::my+dir::z, offset),
587 auxp(aux::mz+dir::x, offset),
588 auxp(aux::mz+dir::y, offset),
589 auxp(aux::mz+dir::z, offset);
590
591 // Unpack density-related quantities
592 const real_t rho(sphys(ndx::rho, offset));
593 const real_t irho = 1.0/rho;
594
595 const Vector3r grad_rho(auxp(aux::rho+dir::x, offset),
596 auxp(aux::rho+dir::y, offset),
597 auxp(aux::rho+dir::z, offset));
598
599
600 // Unpack species variables
601 // NOTE: In species vector, idx 0 is the dilluter (the species
602 // that is not explicitly part of the state vector)
603 species(0) = rho;
604
605 grad_species(0,0) = grad_rho(0);
606 grad_species(1,0) = grad_rho(1);
607 grad_species(2,0) = grad_rho(2);
608
609 for (unsigned int s=1; s<Ns; ++s) {
610 species(s) = sphys(ndx::species + s - 1, offset);
611
612 unsigned int si = aux::species + (s-1)*dir::count;
613 grad_species(0,s) = auxp(si + dir::x, offset);
614 grad_species(1,s) = auxp(si + dir::y, offset);
615 grad_species(2,s) = auxp(si + dir::z, offset);
616
617 // dilluter density = rho_0 = rho - sum_{s=1}^{Ns-1} rho_s
618 species(0) -= species(s);
619
620 grad_species(0,0) -= grad_species(0,s);
621 grad_species(1,0) -= grad_species(1,s);
622 grad_species(2,0) -= grad_species(2,s);
623 }
624
625 // Compute mass fractions and mass fraction gradients
626 for (unsigned int s=0; s<Ns; ++s) {
627
628 cs(s) = irho * species(s);
629
630 grad_cs(0,s) = irho*(grad_species(0,s) - cs(s)*grad_rho(0));
631 grad_cs(1,s) = irho*(grad_species(1,s) - cs(s)*grad_rho(1));
