152 /* Compile-time totals of the number of scalars sampled at each point */
153 struct nscalars { enum {
154#define EXTRACT(r, data, tuple) BOOST_PP_TUPLE_ELEM(2, 1, tuple)
155#define SUM(s, state, x) BOOST_PP_ADD(state, x)
156
157 wave = BOOST_PP_SEQ_FOLD_LEFT(SUM, 0, BOOST_PP_SEQ_TRANSFORM(
158 EXTRACT,,SUZERAIN_REACTING_FLOW_QUANTITIES_WAVE)),
159
160 physical = BOOST_PP_SEQ_FOLD_LEFT(SUM, 0, BOOST_PP_SEQ_TRANSFORM(
161 EXTRACT,,SUZERAIN_REACTING_FLOW_QUANTITIES_PHYSICAL)),
162
163 implicit = BOOST_PP_SEQ_FOLD_LEFT(SUM, 0, BOOST_PP_SEQ_TRANSFORM(
164 EXTRACT,,SUZERAIN_REACTING_FLOW_QUANTITIES_IMPLICIT)),
165
166 total = BOOST_PP_SEQ_FOLD_LEFT(SUM, 0, BOOST_PP_SEQ_TRANSFORM(
167 EXTRACT,,SUZERAIN_REACTING_FLOW_QUANTITIES))
