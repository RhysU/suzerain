404 // Mark Entry as noncopyable to avoid accidental performance hits
405 struct Entry : public boost::noncopyable {
406 bool periodic;
407 TimeType every_dt, next_t;
408 StepType every_nt, next_nt;
409 boost::function<StopType (TimeType t, StepType nt)> callback;
410 };
411
412 // ptr_container to explicitly manage noncopyable instances with minimal
413 // overhead and to automatically free contained resources on destruction.
414 // ptr_vector to keep data as contiguous as possible in memory.
415 typedef boost::ptr_vector<Entry> EntryList;

417 typename boost::function<TimeType (TimeType)> stepper_;
418 TimeType min_dt_, max_dt_, current_t_, current_dt_;
419 EntryList entries_;

421 // Maintain running statistics on the actual time step sizes taken
422 // Also stores the current time step counter as a convenient side effect
423 boost::accumulators::accumulator_set<
424 TimeType,
425 boost::accumulators::features<
426 boost::accumulators::tag::count,
427 boost::accumulators::tag::min,
428 boost::accumulators::tag::mean,
429 boost::accumulators::tag::max,
430 boost::accumulators::tag::lazy_variance

434 template< typename T >
435 static T add_and_coerce_overflow_to_max(T a, T b)
436 {
437 // Overflow occurs when max < a + b or equivalently max - b < a.
438 // This overflow detection works if-and-only-if b is nonnegative!
439 if (SUZERAIN_UNLIKELY(std::numeric_limits<T>::max() - b < a)) {
440 return std::numeric_limits<T>::max();
441 } else {
442 return a + b;
443 }
444 }

462template< typename TimeType, typename StepType, typename StopType >
463template< typename CallbackType >
464void timecontroller<TimeType,StepType,StopType>::add_callback(
465 TimeType what_t,
466 StepType what_nt,
467 CallbackType callback)
468{
469 // Callbacks established in the past are likely usage errors (<). The
470 // advance() method does not perform callbacks prior to advancing
471 // simulation time, so we cannot handle "immediate" callback requests (==).
472 // Combined, these provide the (<=) constraints just below.
473 if (what_t <= current_t_) {
474 throw std::invalid_argument("what_t <= current_t()");
475 }
476 if (what_nt <= current_nt()) {
477 throw std::invalid_argument("what_nt <= current_nt()");
478 }
479
480 // Avoid runtime costs for callbacks that should never occur.
481 if ( what_t == std::numeric_limits<TimeType>::max()
482 && what_nt == std::numeric_limits<StepType>::max()) {
483 return;
484 }
485
486 Entry *e = new Entry; // Allocate Entry on heap
487 e->periodic = false;
488 e->every_dt = 0;
489 e->every_nt = 0;
490 e->next_t = what_t;
491 e->next_nt = what_nt;
492 e->callback = callback;
493 entries_.push_back(e); // Transfer Entry memory ownership
494}

507template< typename TimeType, typename StepType, typename StopType >
508template< typename CallbackType >
509void timecontroller<TimeType,StepType,StopType>::add_periodic_callback(
510 TimeType first_dt,
511 TimeType every_dt,
512 StepType every_nt,
513 CallbackType callback)
514{
515 // first_dt after every_dt to improve messages when delegation target
516 if (every_dt <= 0) throw std::invalid_argument("every_dt <= 0");
517 if (first_dt <= 0) throw std::invalid_argument("first_dt <= 0");
518 if (every_nt <= 0) throw std::invalid_argument("every_nt <= 0");
519
520 // Avoid runtime costs for callbacks that should never occur.
521 // Notice check occurs before add_and_coerce_overflow_to_max.
522 if ( every_dt == std::numeric_limits<TimeType>::max()
523 && every_nt == std::numeric_limits<StepType>::max()) {
524 return;
525 }
526
527 Entry *e = new Entry; // Allocate Entry on heap
528 e->periodic = true;
529 e->every_dt = every_dt;
530 e->every_nt = every_nt;
531 e->next_t = add_and_coerce_overflow_to_max(current_t_, first_dt);
532#pragma warning(push,disable:1572)
533 if (e->next_t == current_t_) {
534 // Pathological, first_dt is O(eps) and current_t_+first_dt was NOP.
535 // We re-register the event to occur at the 2nd and subsequent freq.
536 // Presumably, any status/restart/whatever we should save in O(eps) can
537 // be recovered from what the restart file triggering this case.
538 e->next_t = add_and_coerce_overflow_to_max(current_t_, every_dt);
539 if (e->next_t == current_t_) {
540 throw std::invalid_argument(
541 "Insufficient precision in current_t_"
542 " to track events of frequency every_dt");
543 }
544 }
545#pragma warning(pop)
546 e->next_nt = add_and_coerce_overflow_to_max(current_nt(), every_nt);
547 e->callback = callback;
548 entries_.push_back(e); // Transfer Entry memory ownership
549}

551template< typename TimeType, typename StepType, typename StopType >
552StopType timecontroller<TimeType,StepType,StopType>::advance(
553 const TimeType final_t,
554 const StepType final_nt)
555{
556 SUZERAIN_TIMER_SCOPED("timecontroller::advance");
557
558 SUZERAIN_ENSURE(min_dt_ <= max_dt_);
559 using std::min;
560
561 // Maintain the next simulation time something interesting must happen
562 TimeType next_event_t = std::numeric_limits<TimeType>::max();
563
564 // Find the simulation time of the first callback
565 for (typename EntryList::iterator iter = entries_.begin();
566 iter != entries_.end();
567 ++iter) {
568 next_event_t = min(next_event_t, (*iter).next_t);
569 }
570
571 // Advance time until done or we abort for some reason
572 while (current_t_ < final_t && current_nt() < final_nt) {
573
574 // Determine maximum possible step size allowed by all event criteria
575 const TimeType possible_dt
576 = min(max_dt_, min(final_t, next_event_t) - current_t_);
577 // Check possible_dt is strictly positive in two steps as
578 // a failure in one versus the other is informative for debugging.
579#pragma warning(push,disable:1572)
580 SUZERAIN_ENSURE(possible_dt != 0);
581 SUZERAIN_ENSURE(possible_dt > 0);
582#pragma warning(pop)
583
584 // Take a single step, record new step and time, check the step was
585 // finite, and only then update statistics. (see Redmine #3034).
586 // By design, !isfinite(current_dt_) propagates into current_t_.
587 current_dt_ = stepper_(possible_dt);
588 current_t_ += current_dt_;
589 if (SUZERAIN_UNLIKELY(!(boost::math::isfinite)(current_t_))) {
590 return false;
591 }
592 dt_stats(current_dt_);
593 SUZERAIN_ENSURE(current_dt_ <= possible_dt);
594
595 // Check callbacks and determine next callback simulation time
596 bool called_periodic = false;
597 next_event_t = std::numeric_limits<TimeType>::max();
598 typename EntryList::iterator iter = entries_.begin();
599 while (iter != entries_.end()) {
600
601 // Callback required?
602 bool call = false;
603 if (current_nt() == (*iter).next_nt) {
604 call = true;
605 } else if (called_periodic && (*iter).periodic) {
606 // Redmine #3116: Special handling for multiple periodic events
607 // possibly occurring to combat next_t drift due to their
608 // every_t being multiples of each other (but to only machine
609 // precision). Selects min_dt as the "merging" tolerance
610 // as the user has already said timesteps less than min_dt
611 // are to be avoided per some physically-relevant criterion.
612 if ((*iter).next_t - current_t_ < min_dt_) {
613 call = true;
614 }
615 } else if (current_t_ == (*iter).next_t) {
616 call = true;
617 }
618
619 // Perform callback and associated bookkeeping
620 if (call) {
621
622 // Tracks if any periodic callback has occurred at this time
623 if ((*iter).periodic) called_periodic = true;
624
625 // Must perform state updates prior to any possible abort
626 const StopType keep_advancing
627 = (*iter).callback(current_t_, current_nt());
628
629 // Remove single-shot Entry from further consideration
630 if (SUZERAIN_UNLIKELY(!((*iter).periodic))) {
631 iter = entries_.erase(iter);
632 if (SUZERAIN_UNLIKELY(!keep_advancing)) {
633 return false;
634 }
635 continue;
636 }
637
638 // Update periodic Entry with time of next required callback
639 (*iter).next_t = add_and_coerce_overflow_to_max(
640 current_t_, (*iter).every_dt);
641 (*iter).next_nt = add_and_coerce_overflow_to_max(
642 current_nt(), (*iter).every_nt);
643
644 if (SUZERAIN_UNLIKELY(!keep_advancing)) {
645 return false;
646 }
647 }
648
649 // Update next_event_t based on this Entry's needs
650 next_event_t = min(next_event_t, (*iter).next_t);
651
652 // Next!
653 ++iter;
654 }
655
656 // Abort if the step size was too small, but only if driven by physics
657 if (SUZERAIN_UNLIKELY( current_dt_ < min_dt_
658#pragma warning(push,disable:1572)
659 && current_dt_ != possible_dt)) {
660#pragma warning(pop)
661 return false; // On return current_dt() < min_dt()
662 }
663 }
664
665 // Successfully advanced up to provided criteria
666 return StopType(1);
667}
