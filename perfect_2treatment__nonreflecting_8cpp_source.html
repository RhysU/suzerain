79std::vector<real_t>
80treatment_nonreflecting::apply_operator(
81 const real_t time,
82 contiguous_state<4,complex_t> &swave,
83 const lowstorage::method_interface<complex_t> &method,
84 const std::size_t substep_index) const
85{
86 // Simply delegate all processing unless the grid is one-sided.
87 if (!grid.one_sided()) {
88 return N->apply_operator(time, swave, method, substep_index);
89 }
90 // On a one-sided grid, the work is much more interesting...
91 SUZERAIN_TIMER_SCOPED("treatment_nonreflecting::apply_operator");
92
93 // Implementation approach:
94 // 1) Preserve the wavenumber-dependent state at the upper boundary
95 // 2) Invoke the wrapped nonlinear operator in the usual fashion
96 // 3) Build the various matrices we need from the reference state.
97 // 4) Prepare pre-computable products of the various matrices
98 // 5) Modify the right hand side in a wave-number dependent fashion.
99 //
100 // Implementation currently works for only the 5-equation case
101 // (though, given correct Giles matrices, nothing prevents extension).
102 SUZERAIN_ENSURE(swave.shape()[0] == 5U);
103
104 // State enters method as coefficients in X, Y, and Z directions
105
106 // If the k_x and k_z first-order NRBC terms are to be applied explicitly,
107 // make a local, stride-1 copy of -I times upper boundary state values.
108 // Notice that boundary coefficients are 1-1 with boundary point values.
109 // That is, applying the mass matrix is an ignorable NOP at the boundary.
110 Matrix5Xc negI_stash;
111 if (linearization == linearize::none) {
112 negI_stash.resize(NoChange, swave.shape()[2] * swave.shape()[3]);
113 const complex_t negI_unit(0, -1);
114 const int ku = boost::numeric_cast<int>(swave.shape()[0]);
115 const int l = boost::numeric_cast<int>(swave.shape()[1]) - 1; // Upper
116 const int mu = boost::numeric_cast<int>(swave.shape()[2]);
117 const int nu = boost::numeric_cast<int>(swave.shape()[3]);
118 for (int k = 0; k < ku; ++k) {
119 for (int n = 0; n < nu; ++n) {
120 for (int m = 0; m < mu; ++m) {
121 negI_stash(k, m + mu*n) = negI_unit * swave[k][l][m][n];
122 }
123 }
124 }
125 }
126
127 // Invoke the wrapped nonlinear operator (which may compute references!)
128 const std::vector<real_t> retval = N->apply_operator(
129 time, swave, method, substep_index);
130
131 // State is now coefficients in X and Z directions
132 // State is now collocation point values in Y direction
133
134 // Prepare the matrices required to implement the boundary condition.
135 // The hideous const_cast is required due to timestepping API--
136 // The alternative would be making the members mutable, but catching
137 // any accidental modification elsewhere is worth this const_cast.
138 if (substep_index == 0) {
139 const_cast<treatment_nonreflecting*>(this)
140 ->compute_giles_matrices_upper();
141 }
142
143 // Wavenumber traversal modeled after those found in suzerain/diffwave.c
144 const int Ny = dgrid.global_wave_extent.y();
145 const int dNx = grid.dN.x();
146 const int dkbx = dgrid.local_wave_start.x();
147 const int dkex = dgrid.local_wave_end.x();
148 const int dNz = grid.dN.z();
149 const int dkbz = dgrid.local_wave_start.z();
150 const int dkez = dgrid.local_wave_end.z();
151
152 // Traverse wavenumbers updating the RHS with the Giles boundary condition
153 // TODO Could short circuit some of this traversal using dealiasing
154 const Matrix5r ImPG_VL_S_RY = VL_S_RY - PG_VL_S_RY;
155 if (negI_stash.size()) {
156
157 // "Implementation primarily within the nonlinear explicit operator"
158 // which also looks appropriate for wall-normal implicit treatment.
159 // However, Redmine #3114 shows the wall-normal implicit case to be
160 // ill-posed and hence the alternate, lower-order else block is applied.
161 assert(linearization == linearize::none);
162
163 const real_t twopioverLx_by_chi = twopiover(grid.L.x()) / dgrid.chi();
164 const real_t twopioverLz_by_chi = twopiover(grid.L.z()) / dgrid.chi();
165 const int mu = boost::numeric_cast<int>(swave.shape()[2]);
166
167 for (int n = dkbz; n < dkez; ++n) {
168 const int wn = inorder::wavenumber(dNz, n);
169 const real_t kn_by_chi = twopioverLz_by_chi*wn;
170
171 for (int m = dkbx; m < dkex; ++m) {
172 const int wm = inorder::wavenumber(dNx, m);
173 const real_t km_by_chi = twopioverLx_by_chi*wm;
174
175 // Accumulates kn/km NRBC terms before possible ImPG swamping.
176 // The -I scaling has been accommodated within negI_stash.
177 Vector5c tmp = kn_by_chi
178 * PG_BG_VL_S_RY.cast<complex_t>()
179 * negI_stash.col((m - dkbx) + mu*(n - dkbz));
180 tmp.noalias() += km_by_chi
181 * PG_CG_VL_S_RY.cast<complex_t>()
182 * negI_stash.col((m - dkbx) + mu*(n - dkbz));
183
184 // Pack upper boundary nonlinear RHS into contiguous buffer
185 Vector5c N;
186 N[0] = swave[ndx::e ][Ny - 1][m - dkbx][n - dkbz];
187 N[1] = swave[ndx::mx ][Ny - 1][m - dkbx][n - dkbz];
188 N[2] = swave[ndx::my ][Ny - 1][m - dkbx][n - dkbz];
189 N[3] = swave[ndx::mz ][Ny - 1][m - dkbx][n - dkbz];
190 N[4] = swave[ndx::rho][Ny - 1][m - dkbx][n - dkbz];
191
192 // Project out unwanted characteristics from RHS and accumulate
193 // followed by projecting result back to conserved state
194 tmp.noalias() += ImPG_VL_S_RY.cast<complex_t>() * N;
195 N.noalias() = inv_VL_S_RY.cast<complex_t>() * tmp;
196
197 // Unpack new upper boundary RHS from the contiguous buffer
198 swave[ndx::e ][Ny - 1][m - dkbx][n - dkbz] = N[0];
199 swave[ndx::mx ][Ny - 1][m - dkbx][n - dkbz] = N[1];
200 swave[ndx::my ][Ny - 1][m - dkbx][n - dkbz] = N[2];
201 swave[ndx::mz ][Ny - 1][m - dkbx][n - dkbz] = N[3];
202 swave[ndx::rho][Ny - 1][m - dkbx][n - dkbz] = N[4];
203
204 }
205 }
206
207 } else {
208
209 // "Implementation primarily within the linear implicit operator"
210 // which is appropriate for linear implicit work in three directions
211 // as well as wall-normal-only implicitness per Redmine #3114.
212 assert( linearization == linearize::rhome_xyz
213 || linearization == linearize::rhome_y);
214
215 const Matrix5r upper_nrbc_n = inv_VL_S_RY * ImPG_VL_S_RY;
216 for (int n = dkbz; n < dkez; ++n) {
217 for (int m = dkbx; m < dkex; ++m) {
218
219 // Pack upper boundary RHS into a contiguous buffer
220 Vector5c N;
221 N[0] = swave[ndx::e ][Ny - 1][m - dkbx][n - dkbz];
222 N[1] = swave[ndx::mx ][Ny - 1][m - dkbx][n - dkbz];
223 N[2] = swave[ndx::my ][Ny - 1][m - dkbx][n - dkbz];
224 N[3] = swave[ndx::mz ][Ny - 1][m - dkbx][n - dkbz];
225 N[4] = swave[ndx::rho][Ny - 1][m - dkbx][n - dkbz];
226
227 // Modify the packed RHS to remove unwanted characteristics
228 N.applyOnTheLeft(upper_nrbc_n.cast<complex_t>());
229
230 // Unpack new upper boundary RHS from the contiguous buffer
231 swave[ndx::e ][Ny - 1][m - dkbx][n - dkbz] = N[0];
232 swave[ndx::mx ][Ny - 1][m - dkbx][n - dkbz] = N[1];
233 swave[ndx::my ][Ny - 1][m - dkbx][n - dkbz] = N[2];
234 swave[ndx::mz ][Ny - 1][m - dkbx][n - dkbz] = N[3];
235 swave[ndx::rho][Ny - 1][m - dkbx][n - dkbz] = N[4];
236
237 }
238 }
239
240 }
241
242 return retval;
243}
