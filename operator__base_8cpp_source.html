32operator_base::operator_base(
33 const specification_grid &grid,
34 const pencil_grid &dgrid,
35 const bsplineop &cop,
36 bspline &b)
37 : operator_tools(grid, dgrid, cop)
38 , one_over_delta_x(grid.N.x() /* !dN.x() */ / grid.L.x())
39 , lambda1_x(boost::math::constants::pi<real_t>() * one_over_delta_x)
40 , lambda2_x(lambda1_x * lambda1_x)
41 , one_over_delta_z(grid.N.z() /* !dN.z() */ / grid.L.z())
42 , lambda1_z(boost::math::constants::pi<real_t>() * one_over_delta_z)
43 , lambda2_z(lambda1_z * lambda1_z)
44 , y_(boost::extents[boost::multi_array_types::extent_range(
45 dgrid.local_physical_start.y(),
46 dgrid.local_physical_end.y())])
47 , one_over_delta_y_(
48 boost::extents[boost::multi_array_types::extent_range(
49 dgrid.local_physical_start.y(),
50 dgrid.local_physical_end.y())])
51 , lambda1_y_(
52 boost::extents[boost::multi_array_types::extent_range(
53 dgrid.local_physical_start.y(),
54 dgrid.local_physical_end.y())])
55 , lambda2_y_(
56 boost::extents[boost::multi_array_types::extent_range(
57 dgrid.local_physical_start.y(),
58 dgrid.local_physical_end.y())])
59{
60 const real_t pi = boost::math::constants::pi<real_t>();
61
62 // Compute the B-spline-dependent correction factor to obtain
63 // good maximum eigenvalue estimates given wall-normal inhomogeneity.
64 // See model document for definitions of C^{(1)} and C^{(2)}.
65 real_t C1, Clow1;
66 real_t C2, Clow2;
67 if (grid.htdelta >= 0) { // See grid.two_sided(), grid.one_sided()
68 suzerain_bspline_htstretch2_evdeltascale_greville_abscissae(
69 1, b.k(), +grid.htdelta, b.n(), &C1, &Clow1, NULL);
70 suzerain_bspline_htstretch2_evdeltascale_greville_abscissae(
71 2, b.k(), +grid.htdelta, b.n(), &C2, &Clow2, NULL);
72 } else {
73 suzerain_bspline_htstretch1_evdeltascale_greville_abscissae(
74 1, b.k(), -grid.htdelta, b.n(), &C1, &Clow1, NULL);
75 suzerain_bspline_htstretch1_evdeltascale_greville_abscissae(
76 2, b.k(), -grid.htdelta, b.n(), &C2, &Clow2, NULL);
77 }
78
79 // In practice, directly using C^{(1)} and C^{(2)} is too aggressive
80 // as it requires using inconsistent safety factors when computing
81 // convectively- or diffusively-limited test problems. Softening
82 // in the following manner seems to permit a single safety factor
83 // to address both convective and diffusive stability restrictions.
84 C1 = std::pow(C1, 33 / 64.);
85 Clow1 = std::pow(Clow1, 33 / 64.);
86 C2 = std::pow(C2, 27 / 64.);
87 Clow2 = std::pow(Clow2, 27 / 64.);
88
89 // Compute collocation point-based information local to this rank
90 for (int j = dgrid.local_physical_start.y();
91 j < dgrid.local_physical_end.y();
92 ++j) {
93
94 // Collocation point locations
95 y_[j] = b.collocation_point(j);
96
97 // Inverse spacing to next adjacent collocation point
98 one_over_delta_y_[j] = 1 / b.spacing_collocation_point(j);
99
100 // Eigenvalue estimates using collocation point spacing
101 const real_t inverse_spacing = one_over_delta_y_[j];
102
103 // Estimating wall-normal first derivative eigenvalue magnitudes
104 // Use Clow1 rather than C1 as it is always slightly conservative
105 lambda1_y_[j] = pi * inverse_spacing / Clow1;
106
107 // Estimating wall-normal second derivative eigenvalue magnitudes
108 // Use Clow2 rather than C2 as it is always slightly conservative
109 lambda2_y_[j] = pi * inverse_spacing / Clow2;
110 lambda2_y_[j] *= lambda2_y_[j];
111 }
112}
