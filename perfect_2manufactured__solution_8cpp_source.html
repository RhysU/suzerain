73static void option_adder(
74 boost::program_options::options_description_easy_init easy_init,
75 const char *description,
76 const std::string &name,
77 real_t &value)
78{
79 if ((boost::math::isnan)(value)) {
80 easy_init(name.c_str(),
81 boost::program_options::value(&value),
82 description);
83 } else {
84 easy_init(name.c_str(),
85 boost::program_options::value(&value)->default_value(value),
86 description);
87 }

234void save(const esio_handle h,
235 const std::shared_ptr<manufactured_solution>& msoln,
236 const definition_scenario& scenario,
237 const specification_grid& grid,
238 const char *location)
239{
240 // Only proceed if a manufactured solution is being provided
241 if (!msoln) return;
242
243 const int one = 1;
244 int procid;
245 esio_handle_comm_rank(h, &procid);
246 esio_line_establish(h, 1, 0, (procid == 0 ? 1 : 0));
247 esio_line_write(h, location, &one, 0,
248 "Is a suzerain::perfect::manufactured_solution in use?");
249
250 DEBUG0("Storing manufactured_solution parameters");
251
252 // Non-scenario solution parameters are stored as attributes under location
253 using boost::bind;
254 msoln->rho.foreach_parameter(bind(attribute_storer, h, location, _1, _2));
255 msoln->u .foreach_parameter(bind(attribute_storer, h, location, _1, _2));
256 msoln->v .foreach_parameter(bind(attribute_storer, h, location, _1, _2));
257 msoln->w .foreach_parameter(bind(attribute_storer, h, location, _1, _2));
258 msoln->T .foreach_parameter(bind(attribute_storer, h, location, _1, _2));
259
260#pragma warning(push,disable:1572)
261 // Check parameters stored with the scenario not the manufactured solution
262 // because scenario parameters should be loaded from definition_scenario
263 if (msoln->alpha != scenario.alpha)
264 WARN0("Manufactured solution alpha mismatches with scenario");
265 if (msoln->beta != scenario.beta)
266 WARN0("Manufactured solution beta mismatches with scenario");
267 if (msoln->gamma != scenario.gamma)
268 WARN0("Manufactured solution gamma mismatches with scenario");
269 if (msoln->Ma != scenario.Ma)
270 WARN0("Manufactured solution Ma mismatches with scenario");
271 if (msoln->Re != scenario.Re)
272 WARN0("Manufactured solution Re mismatches with scenario");
273 if (msoln->Pr != scenario.Pr)
274 WARN0("Manufactured solution Pr mismatches with scenario");
275
276 // Check parameters stored with the grid not the manufactured solution
277 // because grid parameters should be loaded from specification_grid
278 if (msoln->Lx != grid.L.x())
279 WARN0("Manufactured solution Lx mismatches with grid");
280 if (msoln->Ly != grid.L.y())
281 WARN0("Manufactured solution Ly mismatches with grid");
282 if (msoln->Lz != grid.L.z())
283 WARN0("Manufactured solution Lz mismatches with grid");
284#pragma warning(pop)
285}

297void load(const esio_handle h,
298 std::shared_ptr<manufactured_solution>& msoln,
299 const definition_scenario& scenario,
300 const specification_grid& grid,
301 const char *location)
302{
303 // Only proceed if a manufactured solution is active in the restart
304 int in_use = 0;
305 esio_line_establish(h, 1, 0, 1); // All ranks load any data
306 if (ESIO_NOTFOUND != esio_line_size(h, location, NULL)) {
307 esio_line_read(h, location, &in_use, 0);
308 }
309 if (!in_use) {
310 msoln.reset();
311 return;
312 }
313
314 DEBUG0("Loading manufactured_solution parameters");
315
316 // Allocate storage which defensively NaNs everything not loaded below.
317 // Protects us against accidentally missing new solution parameters.
318 msoln.reset(new manufactured_solution());
319
320 // Non-scenario solution parameters are stored as attributes under location
321 using boost::bind;
322 msoln->rho.foreach_parameter(bind(attribute_loader, h, location, _1, _2));
323 msoln->u .foreach_parameter(bind(attribute_loader, h, location, _1, _2));
324 msoln->v .foreach_parameter(bind(attribute_loader, h, location, _1, _2));
325 msoln->w .foreach_parameter(bind(attribute_loader, h, location, _1, _2));
326 msoln->T .foreach_parameter(bind(attribute_loader, h, location, _1, _2));
327
328 // Scenario parameters set to match supplied arguments
329 msoln->match(scenario);
330 msoln->match(grid);
331}

333void accumulate_manufactured_solution(
334 const real_t alpha,
335 const manufactured_solution &msoln,
336 const real_t beta,
337 contiguous_state<4,complex_t> &swave,
338 const specification_grid &grid,
339 const pencil_grid &dgrid,
340 const bsplineop &cop,
341 bspline &b,
342 const real_t simulation_time)
343{
344 // We are only prepared to handle a fixed number of fields in this routine
345 enum { state_count = 5 };
346
347 // Ensure state storage meets this routine's assumptions
348 SUZERAIN_ENSURE( swave.shape()[0] == state_count);
349 SUZERAIN_ENSURE(boost::numeric_cast<int>(swave.shape()[1]) == dgrid.local_wave_extent.y());
350 SUZERAIN_ENSURE(boost::numeric_cast<int>(swave.shape()[2]) == dgrid.local_wave_extent.x());
351 SUZERAIN_ENSURE(boost::numeric_cast<int>(swave.shape()[3]) == dgrid.local_wave_extent.z());
352
353 // Initialize operator_base to access decomposition-ready utilities
354 operator_base obase(grid, dgrid, cop, b);
355
356 // Allocate one field of temporary storage for scratch purposes
357 std::unique_ptr<contiguous_state<4,complex_t> > _scratch_ptr( // RAII
358 support::allocate_padded_state<contiguous_state<4,complex_t> >(1,dgrid));
359 contiguous_state<4,complex_t> &scratch = *_scratch_ptr; // Shorthand
360 multi_array::fill(scratch, 0); // Defensive
361
362 // Prepare physical-space view of the wave-space scratch storage
363 physical_view<1> phys(dgrid, scratch);
364
365 // Prepare factored mass matrix for repeated use
366 bsplineop_luz massluz(cop);
367 const complex_t scale_factor = 1 / dgrid.chi();
368 massluz.opform(1, &scale_factor, cop);
369 massluz.factor();
370
371 // For each scalar field...
372 for (size_t f = 0; f < state_count; ++f) {
373
374 // ...compute the manufactured solution in physical space...
375 for (int offset = 0, j = dgrid.local_physical_start.y();
376 j < dgrid.local_physical_end.y();
377 ++j) {
378
379 const real_t y = obase.y(j);
380
381 for (int k = dgrid.local_physical_start.z();
382 k < dgrid.local_physical_end.z();
383 ++k) {
384
385 const real_t z = obase.z(k);
386
387 for (int i = dgrid.local_physical_start.x();
388 i < dgrid.local_physical_end.x();
389 ++i, /* NB */ ++offset) {
390
391 const real_t x = obase.x(i);
392
393 // Ugly, slow switch but performance irrelevant here
394 switch (f) {
395 case ndx::e:
396 phys(0, offset) = msoln.rhoe(x, y, z, simulation_time);
397 break;
398 case ndx::mx:
399 phys(0, offset) = msoln.rhou(x, y, z, simulation_time);
400 break;
401 case ndx::my:
402 phys(0, offset) = msoln.rhov(x, y, z, simulation_time);
403 break;
404 case ndx::mz:
405 phys(0, offset) = msoln.rhow(x, y, z, simulation_time);
406 break;
407 case ndx::rho:
408 phys(0, offset) = msoln.rho (x, y, z, simulation_time);
409 break;
410 default:
411 SUZERAIN_ERROR_REPORT("unknown field",
412 SUZERAIN_ESANITY);
413 }
414
415 } // end X
416
417 } // end Z
418
419 } // end Y
420
421 // ...convert collocation values to non-dealiased coefficients...
422 dgrid.transform_physical_to_wave(&phys.coeffRef(0, 0)); // X, Z
423 obase.zero_dealiasing_modes(scratch, 0);
424 obase.bop_solve(massluz, scratch, 0); // Y
425
426 // ...and accumulate into the corresponding scalar field of swave
427 SUZERAIN_ENSURE(std::equal(scratch.shape() + 1, scratch.shape() + 4,
428 swave.shape() + 1));
429 if (SUZERAIN_UNLIKELY(0U == scratch.shape()[1])) {
430 continue; // Sidestep assertions on trivial data
431 }
432 typedef contiguous_state<4,complex_t>::index index;
433 const index ku = boost::numeric_cast<index>(
434 scratch.index_bases()[2] + scratch.shape()[2]);
435 const index lu = boost::numeric_cast<index>(
436 scratch.index_bases()[3] + scratch.shape()[3]);
437 for (index lx = scratch.index_bases()[3], ly = swave.index_bases()[3];
438 lx < lu;
439 ++lx, ++ly) {
440
441 for (index kx = scratch.index_bases()[2],
442 ky = swave.index_bases()[2];
443 kx < ku;
444 ++kx, ++ky) {
445
446 blas::axpby(scratch.shape()[1], alpha,
447 &scratch[0][scratch.index_bases()[1]][kx][lx],
448 scratch.strides()[1], beta,
449 &swave[f][swave.index_bases()[1]][ky][ly],
450 swave.strides()[1]);
451 }
452 }
453
454 }
455}
