78template<std::size_t NumDims,
79 typename Mutable_RandomAccessIterator,
80 typename RandomAccessIterator,
81 typename InputIterator>
82bool increment(Mutable_RandomAccessIterator indices,
83 RandomAccessIterator max_indices,
84 InputIterator index_order)
85{
86 BOOST_STATIC_ASSERT(NumDims > 0);
87 BOOST_CONCEPT_ASSERT((boost::Mutable_RandomAccessIterator<Mutable_RandomAccessIterator>));
88 BOOST_CONCEPT_ASSERT((boost::RandomAccessIterator<RandomAccessIterator>));
89 BOOST_CONCEPT_ASSERT((boost::InputIterator<InputIterator>));
90
91 typedef typename std::iterator_traits<
92 Mutable_RandomAccessIterator>::value_type index_type;
93 typedef typename std::iterator_traits<
94 RandomAccessIterator>::value_type max_index_type;
95 typedef typename std::iterator_traits<
96 InputIterator>::value_type index_order_type;
97
98 BOOST_CONCEPT_ASSERT((boost::Integer<index_type>));
99 BOOST_CONCEPT_ASSERT((boost::Integer<max_index_type>));
100 BOOST_CONCEPT_ASSERT((boost::Integer<index_order_type>));
101
102 // Because we have 4*NumDims integral operations in the fully unrolled loop
103 // and NumDims is often small, do not break out of loop when overflow is
104 // false. The overflow == false condition causes an effective NOP.
105 bool overflow = true;
106 #pragma unroll
107 for (std::size_t i = 0; i < NumDims; ++i, ++index_order) {
108
109 index_type &index = indices[*index_order];
110 const max_index_type &max_index = max_indices[*index_order];
111
112 assert(1 <= max_index);
113 assert(suzerain::is_nonnegative(index));
114 assert(boost::numeric_cast<max_index_type>(index) < max_index);
115
116 index += overflow; // Handle incoming overflow
117 overflow = index/max_index; // Check outgoing overflow
118 index *= !overflow; // Set to zero on outgoing
119 }
120 return !overflow;
121}

142template<std::size_t NumDims,
143 typename Mutable_RandomAccessIterator,
144 typename RandomAccessIterator,
145 typename InputIterator>
146bool decrement(Mutable_RandomAccessIterator indices,
147 RandomAccessIterator max_indices,
148 InputIterator index_order)
149{
150 BOOST_STATIC_ASSERT(NumDims > 0);
151 BOOST_CONCEPT_ASSERT((boost::Mutable_RandomAccessIterator<Mutable_RandomAccessIterator>));
152 BOOST_CONCEPT_ASSERT((boost::RandomAccessIterator<RandomAccessIterator>));
153 BOOST_CONCEPT_ASSERT((boost::InputIterator<InputIterator>));
154
155 typedef typename std::iterator_traits<
156 Mutable_RandomAccessIterator>::value_type index_type;
157 typedef typename std::iterator_traits<
158 RandomAccessIterator>::value_type max_index_type;
159 typedef typename std::iterator_traits<
160 InputIterator>::value_type index_order_type;
161
162 BOOST_CONCEPT_ASSERT((boost::Integer<index_type>));
163 BOOST_CONCEPT_ASSERT((boost::Integer<max_index_type>));
164 BOOST_CONCEPT_ASSERT((boost::Integer<index_order_type>));
165
166 // Because we have 4*NumDims integral operations in the fully unrolled loop
167 // and NumDims is often small, do not break out of loop when underflow is
168 // false. The underflow == false condition causes an effective NOP.
169 bool underflow = true;
170 #pragma unroll
171 for (std::size_t i = 0; i < NumDims; ++i, ++index_order) {
172
173 index_type &index = indices[*index_order];
174 const max_index_type &max_index = max_indices[*index_order];
175
176 assert(1 <= max_index);
177 assert(suzerain::is_nonnegative(index));
178 assert(boost::numeric_cast<max_index_type>(index) < max_index);
179
180 index -= underflow; // Handle incoming underflow
181 underflow = (index == index_type(-1)); // Check outgoing underflow
182 index += underflow * max_index; // Set max_index-1 on outgoing
183 }
184 return !underflow;
185}

210template<std::size_t NumDims,
211 typename Mutable_RandomAccessIterator,
212 typename RandomAccessIterator1,
213 typename RandomAccessIterator2,
214 typename InputIterator>
215bool crement(Mutable_RandomAccessIterator indices,
216 RandomAccessIterator1 indices_increasing,
217 RandomAccessIterator2 max_indices,
218 InputIterator index_order)
219{
220 BOOST_STATIC_ASSERT(NumDims > 0);
221 BOOST_CONCEPT_ASSERT((boost::Mutable_RandomAccessIterator<Mutable_RandomAccessIterator>));
222 BOOST_CONCEPT_ASSERT((boost::RandomAccessIterator<RandomAccessIterator1>));
223 BOOST_CONCEPT_ASSERT((boost::RandomAccessIterator<RandomAccessIterator2>));
224 BOOST_CONCEPT_ASSERT((boost::InputIterator<InputIterator>));
225
226 typedef typename std::iterator_traits<
227 Mutable_RandomAccessIterator>::value_type index_type;
228 typedef typename std::iterator_traits<
229 RandomAccessIterator2>::value_type max_index_type;
230 typedef typename std::iterator_traits<
231 InputIterator>::value_type index_order_type;
232
233 BOOST_CONCEPT_ASSERT((boost::Integer<index_type>));
234 BOOST_CONCEPT_ASSERT((boost::Integer<max_index_type>));
235 BOOST_CONCEPT_ASSERT((boost::Integer<index_order_type>));
236
237 // Because we have O(NumDims) integral operations in the fully unrolled
238 // loop and NumDims is O(3), do not break out of loop when carry_bit is
239 // false. The carry_bit == false condition causes an effective NOP.
240 bool carry_bit = true;
241 #pragma unroll
242 for (std::size_t i = 0; i < NumDims; ++i, ++index_order) {
243
244 index_type &index = indices[*index_order];
245 const bool is_increasing = indices_increasing[*index_order];
246 const max_index_type &max_index = max_indices[*index_order];
247
248 assert(1 <= max_index);
249 assert(suzerain::is_nonnegative(index));
250 assert(boost::numeric_cast<max_index_type>(index) < max_index);
251
252 index += is_increasing*carry_bit - !is_increasing*carry_bit;
253 const bool overflow = is_increasing && (index/max_index);
254 const bool underflow = !is_increasing && (index == index_type(-1));
255 index *= !overflow; // Set to zero if new overflow
256 index += underflow * max_index; // Set max_index-1 if new underflow
257 carry_bit = overflow + underflow;
258 }
259 return !carry_bit;
260}

771template<class OutputIterator,
772 class InputIterator,
773 typename SizeType,
774 typename StrideType,
775 class C2CBufferProcessor>
776void c2c_fullbuffer_process(OutputIterator out,
777 const SizeType size_out,
778 const StrideType stride_out,
779 InputIterator in,
780 const SizeType size_in,
781 const StrideType stride_in,
782 const C2CBufferProcessor &c2c_buffer_processor)
783{
784 // Highest positive wavenumber
785 const SizeType last_n_pos_out = size_out / 2;
786 const SizeType last_n_pos_in = size_in / 2;
787 const SizeType last_n_pos_copy = std::min(last_n_pos_out, last_n_pos_in);
788 // Lowest negative wavenumber
789 const SizeType first_n_neg_out = -(size_out - 1)/2;
790 const SizeType first_n_neg_in = -(size_in - 1)/2;
791
792 // Code relies on loop conditions being checked prior to any loop entry
793 SizeType n; // Always tracks current wavenumber during iteration
794 for (n = 0; n <= last_n_pos_copy; ++n) {
795 c2c_buffer_processor(*out, *in, n);
796 std::advance(in, stride_in);
797 std::advance(out, stride_out);
798 }
799 if (size_in > size_out) {
800 for (/* init from above */; n <= last_n_pos_in; ++n) {
801 std::advance(in, stride_in);
802 }
803 for (n = first_n_neg_in; n < first_n_neg_out; ++n) {
804 std::advance(in, stride_in);
805 }
806 } else {
807 for (/* init from above */; n <= last_n_pos_out; ++n) {
808 suzerain::complex::assign_complex(*out, 0, 0);
809 std::advance(out, stride_out);
810 }
811 for (n = first_n_neg_out; n < first_n_neg_in; ++n) {
812 suzerain::complex::assign_complex(*out, 0, 0);
813 std::advance(out, stride_out);
814 }
815 }
816 for (/* init from above */; n <= -1; ++n) {
817 c2c_buffer_processor(*out, *in, n);
818 std::advance(in, stride_in);
819 std::advance(out, stride_out);
820 }
821}

913template<class TransformTraits,
914 class ComplexMultiArray1,
915 class ComplexMultiArray2,
916 typename DomainLengthType>
917void transform_c2c(
918 const size_t transform_dim,
919 const ComplexMultiArray1 &in,
920 ComplexMultiArray2 &out,
921 const int fftw_sign,
922 const DomainLengthType domain_length,
923 const int derivative,
924 const unsigned fftw_flags)
925{
926 using suzerain::is_nonnegative;
927
928 // Typedefs fixed separately by ComplexMultiArray template parameters
929 typedef typename ComplexMultiArray1::element element1;
930 typedef typename ComplexMultiArray2::element element2;
931 typedef typename ComplexMultiArray1::index index1;
932 typedef typename ComplexMultiArray2::index index2;
933 typedef typename ComplexMultiArray1::size_type size_type1;
934 typedef typename ComplexMultiArray2::size_type size_type2;
935
936 // Ensure we are operating on complex-valued arrays
937 // If available, C99 _Complex will be typedefed to fftw_complex, etc.
938 BOOST_STATIC_ASSERT(
939 (boost::is_complex<element1>::value)
940 || (boost::is_same<element1, fftwf_complex>::value)
941 || (boost::is_same<element1, fftw_complex>::value)
942 || (boost::is_same<element1, fftwl_complex>::value));
943 BOOST_STATIC_ASSERT(
944 (boost::is_complex<element2>::value)
945 || (boost::is_same<element2, fftwf_complex>::value)
946 || (boost::is_same<element2, fftw_complex>::value)
947 || (boost::is_same<element2, fftwl_complex>::value));
948 // Ensure dimension of both in and out is consistent
949 BOOST_STATIC_ASSERT( ComplexMultiArray1::dimensionality
950 == ComplexMultiArray2::dimensionality);
951 const size_type1 dimensionality = ComplexMultiArray1::dimensionality;
952
953 // Typedefs due to implementation choices
954 typedef array<index1,dimensionality> index_array1;
955 typedef array<index2,dimensionality> index_array2;
956 typedef int shape_type; // Per FFTW
957 typedef array<shape_type,dimensionality> shape_array;
958
959 // Ensure we transform a dimension that exists in the data
960 assert(is_nonnegative(transform_dim) && transform_dim < dimensionality);
961 // Copy all shape information into integers well-suited for FFTW
962 shape_array shape_in, shape_out;
963 {
964 using boost::numeric::converter;
965 typedef converter<shape_type,size_type1> Size1ToShapeType;
966 std::transform(in.shape(), in.shape() + dimensionality,
967 shape_in.begin(), Size1ToShapeType());
968 typedef converter<shape_type,size_type2> Size2ToShapeType;
969 std::transform(out.shape(), out.shape() + dimensionality,
970 shape_out.begin(), Size2ToShapeType());
971 }
972 // Ensure out shape at least as large as in shape for
973 // all non-transformed directions
974 for (size_type1 n = 0; n < dimensionality; ++n) {
975 assert(n == transform_dim || shape_in[n] <= shape_out[n]);
976 }
977 // Ensure transformation direction parameter is sane
978 assert(fftw_sign == FFTW_FORWARD || fftw_sign == FFTW_BACKWARD);
979 // Transform size determined from the physical space dimension
980 const shape_type transform_n = (fftw_sign == FFTW_FORWARD)
981 ? shape_in[transform_dim] : shape_out[transform_dim];
982
983 // Prepare the in-place transform buffer and construct the FFTW plan
984 typedef typename TransformTraits::fftw_complex_type fftw_complex_type;
985 std::shared_ptr<fftw_complex_type[]> buffer(
986 static_cast<fftw_complex_type *>(
987 fftw_malloc(sizeof(fftw_complex_type)*transform_n)),
988 fftw_free);
989 assert(buffer);
990 typedef typename TransformTraits::fftw_plan_type fftw_plan_type;
991 std::shared_ptr<
992 typename boost::remove_pointer<fftw_plan_type>::type
993 > plan(TransformTraits::plan_c2c_1d(transform_n,
994 buffer.get(),
995 buffer.get(),
996 fftw_sign,
997 fftw_flags | FFTW_DESTROY_INPUT),
998 TransformTraits::destroy_plan);
999 assert(plan);
1000
1001 // Dereference all constant parameters outside main processing loop
1002 // Pulls array information into our stack frame
1003 index_array1 index_bases_in;
1004 std::copy_n(in.index_bases(), dimensionality, index_bases_in.begin());
1005 index_array2 index_bases_out;
1006 std::copy_n(out.index_bases(), dimensionality, index_bases_out.begin());
1007 const index1 stride_in_transform_dim = in.strides()[transform_dim];
1008 const index2 stride_out_transform_dim = out.strides()[transform_dim];
1009 const shape_type shape_in_transform_dim = shape_in[transform_dim];
1010 const shape_type shape_out_transform_dim = shape_out[transform_dim];
1011
1012 // Prepare per-pencil outer loop index and loop bounds
1013 shape_array loop_shape(shape_in); // Iterate over all dimensions...
1014 loop_shape[transform_dim] = 1; // ...except the transformed one
1015 index_array1 loop_index = {{ }}; // Initialize to default value
1016 for (size_type1 n = 0; n < dimensionality; ++n) {
1017 assert(loop_index[n] == 0); // Check initialization correct
