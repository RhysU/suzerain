92template<std::size_t D>
93struct for_each_functor {
94
95 BOOST_STATIC_ASSERT(D != 0); // Nonsensical behavior for zero dimensions
96 BOOST_STATIC_ASSERT(D > 1); // Ensure not instantiated for special values
97
98 // See http://groups.google.com/group/boost-list/browse_thread/thread/e16f32c4411dea08
99 // for details about why MultiArray::iterator::reference is used below.
100 template<class MultiArray,
101 class UnaryFunction >
102 void operator()(MultiArray &x, UnaryFunction &f) const {
103 for_each_functor<D-1> functor;
104 for (typename MultiArray::iterator i = x.begin(); i != x.end(); ++i) {
105 typename MultiArray::iterator::reference ri = *i;
106 functor(ri,f);
107 }
108 }
109};

119// Detects storage order to walk memory linearly for 2D MultiArrays
120template<> struct for_each_functor<2> {
121
122 template<class MultiArray,
123 class UnaryFunction >
124 void operator()(MultiArray &x, UnaryFunction &f) const {
125
126 typedef typename MultiArray::index index;
127 typedef typename MultiArray::size_type size_type;
128
129 // Compute lower and upper index bounds
130 const index * const base = x.index_bases();
131 const size_type * const shape = x.shape();
132 const index il = base[0], iu = il + (index) shape[0];
133 const index jl = base[1], ju = jl + (index) shape[1];
134
135 // Obtain absolute strides for storage order determination
136 const index * const stride = x.strides();
137 const index si = std::abs(stride[0]);
138 const index sj = std::abs(stride[1]);
139
140 // Traverse according to runtime-determined storage order
141 if (si >= sj) {
142 for (index i = il; i < iu; ++i)
143 for (index j = jl; j < ju; ++j)
144 f(x[i][j]);
145 } else {
146 for (index j = jl; j < ju; ++j)
147 for (index i = il; i < iu; ++i)
148 f(x[i][j]);
149 }
150
151 }
152
153};

155// Detects storage order to walk memory linearly for 3D MultiArrays
156template<> struct for_each_functor<3> {
157
158 template<class MultiArray,
159 class UnaryFunction >
160 void operator()(MultiArray &x, UnaryFunction &f) const {
161
162 typedef typename MultiArray::index index;
163 typedef typename MultiArray::size_type size_type;
164
165 // Compute lower and upper index bounds
166 const index * const base = x.index_bases();
167 const size_type * const shape = x.shape();
168 const index il = base[0], iu = il + (index) shape[0];
169 const index jl = base[1], ju = jl + (index) shape[1];
170 const index kl = base[2], ku = kl + (index) shape[2];
171
172 // Obtain absolute strides for storage order determination
173 const index * const stride = x.strides();
174 const index si = std::abs(stride[0]);
175 const index sj = std::abs(stride[1]);
176 const index sk = std::abs(stride[2]);
177
178 // Perform all possible stride comparisons
179 const bool oij = si >= sj, oji = !oij;
180 const bool oik = si >= sk, oki = !oik;
181 const bool ojk = sj >= sk, okj = !ojk;
182
183#ifndef SUZERAIN_PARSED_BY_DOXYGEN
184#define FOR(ndx) for (index ndx = ndx##l; ndx < ndx##u; ++ndx)
185#endif
186
187 // Traverse according to runtime-determined storage order
188 if (oij && ojk) // boost::c_storage_order
189 FOR(i) FOR(j) FOR(k) f(x[i][j][k]);
190 else if (okj && oji) // boost::fortran_storage_order
191 FOR(k) FOR(j) FOR(i) f(x[i][j][k]);
192 else if (oki && oij) // suzerain::storage::interleaved<3>
193 FOR(k) FOR(i) FOR(j) f(x[i][j][k]);
194 else if (oik && okj) // suzerain::storage::contiguous<3>
195 FOR(i) FOR(k) FOR(j) f(x[i][j][k]);
196 else if (oji && oik) // lexicographic from here onward...
197 FOR(j) FOR(i) FOR(k) f(x[i][j][k]);
198 else // (ojk && oki)
199 FOR(j) FOR(k) FOR(i) f(x[i][j][k]);
200
201#ifndef SUZERAIN_PARSED_BY_DOXYGEN
202#undef FOR
203#endif
204
205 }
206
207};

404template<class MultiArray>
405bool is_contiguous(const MultiArray &x)
406{
407 // Using ptrdiff_t avoids unsigned/signed warnings when dimensionality == 0.
408 static const std::ptrdiff_t dimensionality = MultiArray::dimensionality;
409
410 typename MultiArray::size_type min_extent = (dimensionality == 0) ? 0 : 1;
411 typename MultiArray::size_type abs_extent = 1;
412
413 // Obtain minimum possible extent and actual extent using abs(strides).
414 // (shape()[i] == 0) overflows abs_extent but min_extent becomes zero.
415 for (std::ptrdiff_t i = 0; i < dimensionality; ++i) {
416 const typename MultiArray::size_type shape = x.shape()[i];
417 min_extent *= shape;
418 abs_extent += (shape - 1)*std::abs(x.strides()[i]);
419 }
420
421 return (min_extent == 0) ? true : (min_extent == abs_extent);
422}
