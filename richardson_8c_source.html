38int
39suzerain_richardson_extrapolation_step(
40 gsl_vector * Ah,
41 const gsl_vector * Aht,
42 const double t,
43 const double ki)
44{
45#pragma warning(push,disable:1572)
46 if (ki == 0.0) {
47 SUZERAIN_ERROR("ki == 0 invalid as it will cause a division-by-zero",
48 SUZERAIN_EDOM);
49 }
50#pragma warning(pop)
51
52 const double tki = pow(t, ki);
53 const double inv_tkim1 = 1.0/(tki-1.0);
54
55 gsl_blas_dscal(-inv_tkim1, Ah);
56
57 int error = gsl_blas_daxpy(tki*inv_tkim1, Aht, Ah);
58 if (error) return error;
59
60 return SUZERAIN_SUCCESS;
61}

63int
64suzerain_richardson_extrapolation(
65 gsl_matrix * const A,
66 const double t,
67 const gsl_vector * k,
68 gsl_matrix * normtable,
69 const gsl_vector * const exact)
70{
71 gsl_vector * scratch = NULL;
72 const size_t k_size = (k) ? k->size : 0;
73
74 if (exact) {
75 if (!normtable) {
76 SUZERAIN_ERROR("exact provided but normtable was not",
77 SUZERAIN_EINVAL);
78 }
79 if (exact->size != A->size1) {
80 SUZERAIN_ERROR("exact->size does not match A->size1",
81 SUZERAIN_EINVAL);
82 }
83 }
84
85 if (normtable) {
86 if (normtable->size1 != A->size2) {
87 SUZERAIN_ERROR("normtable->size1 does not match A->size2",
88 SUZERAIN_EINVAL);
89 }
90 if (normtable->size2 != A->size2) {
91 SUZERAIN_ERROR("normtable->size2 does not match A->size2",
92 SUZERAIN_EINVAL);
93 }
94
95 gsl_matrix_set_all(normtable, GSL_NAN);
96
97 /* Compute the first column in the norm table */
98 if (exact) {
99 scratch = gsl_vector_alloc(A->size1);
100 }
101 for (size_t i = 0; i < normtable->size2; ++i) {
102 double norm;
103 if (exact) {
104 gsl_matrix_get_col(scratch, A, i);
105 gsl_blas_daxpy(-1.0, exact, scratch);
106 norm = gsl_blas_dnrm2(scratch);
107 } else {
108 gsl_vector_view Ai = gsl_matrix_column(A, i);
109 norm = gsl_blas_dnrm2(&Ai.vector);
110 }
111 gsl_matrix_set(normtable, i, 0, norm);
112 }
113 }
114
115 for (size_t i = 0; i < A->size2; ++i) {
116
117 /* Provide automagic around the k parameter */
118 double ki;
119 if (k_size) {
120 if (i < k_size) {
121 ki = gsl_vector_get(k, i);
122 } else if (k_size == 1) {
123 ki = gsl_vector_get(k, 0) + i;
124 } else {
125 const double last_ki = gsl_vector_get(k, k_size-1);
126 const double increment = last_ki - gsl_vector_get(k, k_size-2);
127 ki = last_ki + (i - (k_size-1))*increment;
128 }
129 } else {
130#pragma warning(push,disable:810 2259)
131 ki = i+1;
132#pragma warning(pop)
133 }
134
135 /* Perform the extrapolation using leading term order ki */
136 for (size_t j = 0; j < A->size2 - i - 1; ++j) {
137 gsl_vector_view Aih = gsl_matrix_column(A, j);
138 gsl_vector_view Aiht = gsl_matrix_column(A, j+1);
139
140 const int error = suzerain_richardson_extrapolation_step(
141 &Aih.vector, &Aiht.vector, t, ki);
142 if (error) {
143 if (scratch) {
144 gsl_vector_free(scratch);
145 }
146 return error;
147 }
148
149 if (normtable) {
150 double norm;
151 if (exact) {
152 gsl_vector_memcpy(scratch, &Aih.vector);
153 gsl_blas_daxpy(-1.0, exact, scratch);
154 norm = gsl_blas_dnrm2(scratch);
155 } else {
156 norm = gsl_blas_dnrm2(&Aih.vector);
157 }
158 gsl_matrix_set(normtable, i+j+1, i+1, norm);
159 }
160 }
161 }
162
163 if (scratch) {
164 gsl_vector_free(scratch);
165 }
166
167 return SUZERAIN_SUCCESS;
168}
