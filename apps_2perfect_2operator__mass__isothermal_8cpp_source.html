69real_t operator_mass_isothermal::lower_E(
70 const real_t lower_T,
71 const real_t lower_u,
72 const real_t lower_v,
73 const real_t lower_w,
74 const std::vector<real_t> lower_cs) const
75{
76 // FIXME Gather and use instantaneous averages when isnan(lower_[uvw]).
77 SUZERAIN_ENSURE(lower_cs.size() == 1U && lower_cs[0] == 1.0);
78 const real_t E_internal = lower_T / (scenario.gamma*(scenario.gamma - 1));
79 const real_t E_kinetic = (scenario.Ma*scenario.Ma)
80 * ( lower_u*lower_u
81 + lower_v*lower_v
82 + lower_w*lower_w ) / 2;
83 return E_internal + E_kinetic;
84}
