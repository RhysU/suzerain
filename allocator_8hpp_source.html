81template<typename T>
82class allocator_freestore_policy {
83public:
84 // Typedefs following std::allocator contract
85 typedef T value_type;
86 typedef value_type* pointer;
87 typedef const value_type* const_pointer;
88 typedef value_type& reference;
89 typedef const value_type& const_reference;
90 typedef std::size_t size_type;
91 typedef std::ptrdiff_t difference_type;
92
94 template<typename U>
95 struct rebind {
96 typedef allocator_freestore_policy<U> other;
97 };
98
99 // Make explicit all constructors, including copy constructors
100 explicit allocator_freestore_policy() {}
101 ~allocator_freestore_policy() {}
102 explicit allocator_freestore_policy(allocator_freestore_policy const&) {}
103 template <typename U>
104 explicit allocator_freestore_policy(allocator_freestore_policy<U> const&) {}
105
108 pointer allocate(size_type cnt, const void* = 0)
109 {
110 return reinterpret_cast<pointer>(::operator new(cnt * sizeof (T)));
111 }
112 void deallocate(pointer p, size_type) { ::operator delete(p); }
114
116 size_type max_size() const
117 {
118 return std::numeric_limits<size_type>::max() / sizeof(T);
119 }
120};

151template<typename T,
152 typename Policy = allocator_freestore_policy<T>,
153 typename Traits = allocator_object_traits<T> >
154class allocator : public Policy, public Traits {
155public:
156 // Typedefs required per std::allocator contract
157 typedef typename Policy::size_type size_type;
158 typedef typename Policy::difference_type difference_type;
159 typedef typename Policy::pointer pointer;
160 typedef typename Policy::const_pointer const_pointer;
161 typedef typename Policy::reference reference;
162 typedef typename Policy::const_reference const_reference;
163 typedef typename Policy::value_type value_type;
164
166 template<typename U>
167 struct rebind {
168 typedef allocator<U,typename Policy::template rebind<U>::other> other;
169 };
170
171 // Make explicit all constructors, including copy constructors
172 explicit allocator() {}
173 ~allocator() {}
174 allocator(allocator const& rhs) : Policy(rhs), Traits(rhs) {}
175 template <typename U>
176 explicit allocator(allocator<U> const&) {}
177 template <typename U, typename P, typename T2>
178 allocator(allocator<U, P, T2> const& rhs) : Policy(rhs), Traits(rhs) {}
179
182 pointer allocate(size_type cnt)
183 {
184 return Policy::allocate(cnt, 0);
185 }
186
187 pointer allocate(size_type cnt, const void* hint)
188 {
189 return Policy::allocate(cnt, hint);
190 }
191
192 void deallocate(pointer p, size_type cnt)
193 {
194 Policy::deallocate(p, cnt);
195 }
197};
