189#define SUZERAIN_SHIFTED_SUM_OP(s, state, seq) \
190 (BOOST_PP_SEQ_PUSH_BACK( \
191 BOOST_PP_TUPLE_ELEM(2, 0, state), \
192 (BOOST_PP_TUPLE_ELEM(2, 0, seq), \
193 BOOST_PP_TUPLE_ELEM(2, 1, state))), \
194 BOOST_PP_ADD( \
195 BOOST_PP_TUPLE_ELEM(2, 1, state), \
196 BOOST_PP_TUPLE_ELEM(2, 1, seq)))
197
207#define SUZERAIN_SHIFTED_SUM(seq) \
208 BOOST_PP_TUPLE_ELEM(2, 0, \
209 BOOST_PP_SEQ_FOLD_LEFT( \
210 SUZERAIN_SHIFTED_SUM_OP, \
211 (BOOST_PP_SEQ_NIL, 0), \
212 seq))

214// Provide an operator<<(basic_ostream, boost::array) template in namespace boost
215// http://agentzlerich.blogspot.com/2009/12/small-gotcha-when-combining-boostarray.html
216namespace boost {
217template< typename CharT, typename Traits, typename T, ::std::size_t N >
218::std::basic_ostream<CharT,Traits>& operator<<(
219 ::std::basic_ostream<CharT,Traits> &os,
220 const ::boost::array<T,N> &array)
221{
222 os << '[' << N << "]{ ";
223 ::std::copy(array.begin(), array.end(),
224 ::std::ostream_iterator<T,CharT,Traits>(os, " "));
225 os << '}';
226 return os;
227}
