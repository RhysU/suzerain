140 array<T,N> retval = {{ 1 }};

165{
166 array<T,N> retval;
167 retval[N-1] = 1;
168 for (typename array<T,N>::size_type i = N-1; i > 0; --i) {

266template <typename FPT, typename Stats, std::size_t N>
267std::size_t inconsistent_accumulation_count(
268 const array<boost::accumulators::accumulator_set<FPT, Stats>, N> acc)
269{
270 using boost::accumulators::count;
271 const std::size_t expected = count(acc.front());
272 for (std::size_t i = 1; i < N; ++i) {
273 const std::size_t observed = count(acc[i]);
274 if (expected != observed) {
275 return i;
276 }
277 }
278 return 0;
279}
