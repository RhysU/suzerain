47uniform::uniform(const bsplineop &bop)
48{
49 // Relies on any B-spline basis being a partition of unity.
50 // GCC -O3 misanalyzes Eigen's inlined DenseStorage::resize and emits a
51 // bogus -Wmaybe-uninitialized naming phantom this[1]/this[2] elements
52 // that resize never reads. Silence just this dynamic-vector init.
53#pragma GCC diagnostic push
54#pragma GCC diagnostic ignored "-Wmaybe-uninitialized"
55 shape.setOnes(bop.n());
56#pragma GCC diagnostic pop
57}

59linear::linear(const bspline &b, const real_t lower, const real_t upper)
60{
61 const int n = b.n();
62 shape.resize(n);
63
64 // Pedantically, http://wikipedia.org/wiki/Linear_equation#Two-point_form
65 const real_t x1 = lower, y1 = b.collocation_point(0);
66 const real_t x2 = upper, y2 = b.collocation_point(n-1);
67 const real_t m = (y2 - y1)/(x2 - x1);
68 for (int i = 0; i < n; ++i) {
69 const real_t x = b.collocation_point(i);
70 const real_t y = y1 + m*(x - x1);
71 shape[i] = y;
72 }
73}

75coefficient::coefficient(const bsplineop &bop, const int i)
76{
77 // Apply the mass matrix to express coefficient i via collocation points
78#pragma GCC diagnostic push
79#pragma GCC diagnostic ignored "-Wmaybe-uninitialized" // bogus Eigen -O3 FP; see uniform
80 shape.setZero(bop.n());
81#pragma GCC diagnostic pop
82 shape[i] = 1;
83 bop.apply(0, 1.0, shape.data(), shape.innerStride());
84}

86lower::lower(const bsplineop &bop, const int nderiv)
87{
88 // Reconstructing the first row of D^{(nderiv)} in coeff
89 // requires working with the first column of D_T
90#pragma GCC diagnostic push
91#pragma GCC diagnostic ignored "-Wmaybe-uninitialized" // bogus Eigen -O3 FP; see uniform
92 coeff.setZero(bop.n());
93#pragma GCC diagnostic pop
94 int il, iu;
95 real_t * const col = (real_t *) suzerain_gbmatrix_col(
96 bop.n(), bop.n(), bop.kl(nderiv), bop.ku(nderiv),
97 (void *) bop.D_T(nderiv), bop.ld(), sizeof(real_t),
98 /*first*/0, &il, &iu);
99 for (int i = il; i < iu; ++i) {
100 coeff[i] = col[i];
101 }
102}

117 for (int i = il; i < iu; ++i) {

122bulk::bulk(bspline &b)
123{
124 coeff.resize(b.n());
125 b.integration_coefficients(0, coeff.data());
126 real_t L = b.breakpoint(b.nbreak()-1) - b.breakpoint(0);
127 coeff /= L;
128}
