107#define FIXEDBW_CASE(z,num,prefix) \
108 case num: return BOOST_PP_CAT(prefix,num)( \
109 uplo, n, alpha, a, lda, x, incx, beta, y, incy);

111int
112suzerain_sbmv_s(
113 const char uplo,
114 const int n,
115 const int k,
116 const float alpha,
117 const float *a,
118 const int lda,
119 const float *x,
120 const int incx,
121 const float beta,
122 float *y,
123 const int incy)
124{
125 // Dispatch to fixed bandwidth specialization for small bandwidth...
126 switch (k) {
127 BOOST_PP_REPEAT_FROM_TO(FIXEDBW_LOWER, BOOST_PP_INC(FIXEDBW_UPPER),
128 FIXEDBW_CASE, suzerain_sbmv_internal_s)
129 }
130
131 // ...otherwise employ a general bandwidth implementation
132 return suzerain_sbmv_internal_s(uplo, n, k,
133 alpha, a, lda, x, incx,
134 beta, y, incy);
135}
