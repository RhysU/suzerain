103void operator_hybrid_isothermal::apply_mass_plus_scaled_operator(
104 const complex_t& phi,
105 multi_array::ref<complex_t,4> &state,
106 const std::size_t substep_index) const
107{
108 SUZERAIN_TIMER_SCOPED("apply_mass_plus_scaled_operator");
109 SUZERAIN_ENSURE(common.linearization != linearize::none);
110
111 // Shorthand
112 using inorder::wavenumber;
113 using inorder::wavenumber_absmin;
114 SUZERAIN_UNUSED(substep_index);
115
116 // We are only prepared to handle rho_E, rho_u, rho_v, rho_w, rho!
117 assert(static_cast<int>(ndx::e ) < solver->S);
118 assert(static_cast<int>(ndx::mx ) < solver->S);
119 assert(static_cast<int>(ndx::my ) < solver->S);
120 assert(static_cast<int>(ndx::mz ) < solver->S);
121 assert(static_cast<int>(ndx::rho) < solver->S);
122
123 // Wavenumber traversal modeled after those found in suzerain/diffwave.c
124 const int Ny = dgrid.global_wave_extent.y();
125 const int Nx = grid.N.x();
126 const int dNx = grid.dN.x();
127 const int dkbx = dgrid.local_wave_start.x();
128 const int dkex = dgrid.local_wave_end.x();
129 const int Nz = grid.N.z();
130 const int dNz = grid.dN.z();
131 const int dkbz = dgrid.local_wave_start.z();
132 const int dkez = dgrid.local_wave_end.z();
133 const real_t twopioverLx = twopiover(grid.L.x()); // Weird looking...
134 const real_t twopioverLz = twopiover(grid.L.z()); // ...for FP control
135
136 // Sidesteps assertions when local rank contains no wavespace information
137 if (SUZERAIN_UNLIKELY(0U == state.shape()[1])) return;
138
139 // Incoming state has wall-normal pencils of interleaved state scalars?
140 SUZERAIN_ENSURE(state.shape() [0] == (unsigned) solver->S);
141 SUZERAIN_ENSURE(state.strides()[0] == (unsigned) Ny);
142 SUZERAIN_ENSURE(state.shape() [1] == (unsigned) Ny);
143 SUZERAIN_ENSURE(state.strides()[1] == 1);
144
145 // Scratch for "in-place" suzerain_rholut_imexop_accumulate usage
146 VectorXc tmp(solver->N);
147 suzerain_rholut_imexop_scenario s(this->imexop_s());
148 suzerain_rholut_imexop_ref ref;
149 suzerain_rholut_imexop_refld ld;
150 common.ref.rholut_imexop(ref, ld);
151
152 // Iterate across local wavenumbers and apply operator "in-place"
153 // Short circuit "continues" occur for Nyquist and non-dealiased modes...
154 // ...where the former will be zeroed during the later invert call.
155 switch (common.linearization) {
156
157 case linearize::rhome_xyz:
158 {
159 const real_t *maybe_a = grid.one_sided() ? upper_nrbc_a.data() : NULL;
160 const real_t *maybe_b = grid.one_sided() ? upper_nrbc_b.data() : NULL;
161 const real_t *maybe_c = grid.one_sided() ? upper_nrbc_c.data() : NULL;
162 for (int n = dkbz; n < dkez; ++n) {
163 const int wn = wavenumber(dNz, n);
164 if (std::abs(wn) > wavenumber_absmin(Nz)) continue;
165 const real_t kn = twopioverLz*wn;
166
167 for (int m = dkbx; m < dkex; ++m) {
168 const int wm = wavenumber(dNx, m);
169 if (std::abs(wm) > wavenumber_absmin(Nx)) continue;
170 const real_t km = twopioverLx*wm;
171
172 // Get pointer to (.,m,n)-th state pencil
173 complex_t * const p = &state[0][0][m - dkbx][n - dkbz];
174
175 // Copy pencil into temporary storage
176 blas::copy(solver->N, p, 1, tmp.data(), 1);
177
178 // Accumulate result back into state storage
179 SUZERAIN_TIMER_SCOPED("suzerain_rholut_imexop_accumulate");
180 suzerain_rholut_imexop_accumulate(
181 phi, km, kn, &s, &ref, &ld, cop.get(),
182 tmp.data() + ndx::e * Ny,
183 tmp.data() + ndx::mx * Ny,
184 tmp.data() + ndx::my * Ny,
185 tmp.data() + ndx::mz * Ny,
186 tmp.data() + ndx::rho * Ny,
187 0.0,
188 p + ndx::e * Ny,
189 p + ndx::mx * Ny,
190 p + ndx::my * Ny,
191 p + ndx::mz * Ny,
192 p + ndx::rho * Ny,
193 maybe_a, maybe_b, maybe_c);
194 }
195 }
196 break;
197 }
198
199 case linearize::rhome_y:
200 {
201 const real_t *maybe_c = grid.one_sided() ? upper_nrbc_c.data() : NULL;
202 for (int n = dkbz; n < dkez; ++n) {
203 const int wn = wavenumber(dNz, n);
204 if (std::abs(wn) > wavenumber_absmin(Nz)) continue;
205
206 for (int m = dkbx; m < dkex; ++m) {
207 const int wm = wavenumber(dNx, m);
208 if (std::abs(wm) > wavenumber_absmin(Nx)) continue;
209
210 // Get pointer to (.,m,n)-th state pencil
211 complex_t * const p = &state[0][0][m - dkbx][n - dkbz];
212
213 // Copy pencil into temporary storage
214 blas::copy(solver->N, p, 1, tmp.data(), 1);
215
216 // Accumulate result back into state storage
217 SUZERAIN_TIMER_SCOPED("suzerain_rholut_imexop_accumulate00");
218 suzerain_rholut_imexop_accumulate00(
219 phi, &s, &ref, &ld, cop.get(),
220 tmp.data() + ndx::e * Ny,
221 tmp.data() + ndx::mx * Ny,
222 tmp.data() + ndx::my * Ny,
223 tmp.data() + ndx::mz * Ny,
224 tmp.data() + ndx::rho * Ny,
225 0.0,
226 p + ndx::e * Ny,
227 p + ndx::mx * Ny,
228 p + ndx::my * Ny,
229 p + ndx::mz * Ny,
230 p + ndx::rho * Ny,
231 maybe_c);
232 }
233 }
234 break;
235 }
236
237 default:
238 SUZERAIN_ERROR_VOID_UNIMPLEMENTED();
239
240 }

396class IsothermalPATPTEnforcer
397{
398 // Loop constants to reduce magic number usage
399 enum { nedges = 2, nmomentum = 3 };
400
401 // Indices within PA^TP^T at which to apply boundary conditions
402 // Computed once within constructor and then repeatedly used
403 int rho[nedges], m[nedges][nmomentum], e[nedges];
404
405 // Precomputed coefficients based on the isothermal equation of state
406 real_t E_factor[nedges];
407
408 // Coefficients used to enforce possibly transpiring walls
409 real_t vel_factor[nedges][nmomentum];
410
411 // On which wall indices will constraints be enforced?
412 // These control wall loop extents during enforcement
413 const int wall_begin;
414 const int wall_end;
415
416public:
417
418 IsothermalPATPTEnforcer(const suzerain_bsmbsm& A_T,
419 const suzerain_rholut_imexop_scenario& s,
420 const specification_isothermal& spec,
421 const bool enforce_lower,
422 const bool enforce_upper)
423 : wall_begin(enforce_lower ? 0 : 1)
424 , wall_end (enforce_upper ? 2 : 1)
425 {
426 // Starting offset to named scalars in interleaved_state pencil
427 const int e0 = static_cast<int>(ndx::e ) * A_T.n;
428 const int mx0 = static_cast<int>(ndx::mx ) * A_T.n;
429 const int my0 = static_cast<int>(ndx::my ) * A_T.n;
430 const int mz0 = static_cast<int>(ndx::mz ) * A_T.n;
431 const int rho0 = static_cast<int>(ndx::rho) * A_T.n;
432
433 // Relative to foo0 what is the offset to lower, upper walls
434 const int wall[nedges] = { 0, A_T.n - 1};
435
436 // Prepare indices within PA^TP^T corresponding to the walls.
437 // Uses that {A^T}_{i,j} maps to {PA^TP^T}_{{q^-1}(i),{q^(-1)}(j)}.
438 for (int i = 0; i < nedges; ++i) {
439 e [i] = suzerain_bsmbsm_qinv(A_T.S, A_T.n, e0 + wall[i]);
440 m [i][0] = suzerain_bsmbsm_qinv(A_T.S, A_T.n, mx0 + wall[i]);
441 m [i][1] = suzerain_bsmbsm_qinv(A_T.S, A_T.n, my0 + wall[i]);
442 m [i][2] = suzerain_bsmbsm_qinv(A_T.S, A_T.n, mz0 + wall[i]);
443 rho[i] = suzerain_bsmbsm_qinv(A_T.S, A_T.n, rho0 + wall[i]);
444 }
445
446 // Compute/store the specification_isothermal details for later usage
447 // These are always computed for both walls even if only one enforced
448 E_factor[0] = spec.lower_T / (s.gamma * (s.gamma - 1))
449 + s.Ma * s.Ma / 2 * ( spec.lower_u * spec.lower_u
450 + spec.lower_v * spec.lower_v
451 + spec.lower_w * spec.lower_w );
452 E_factor[1] = spec.upper_T / (s.gamma * (s.gamma - 1))
453 + s.Ma * s.Ma / 2 * ( spec.upper_u * spec.upper_u
454 + spec.upper_v * spec.upper_v
455 + spec.upper_w * spec.upper_w );
456 vel_factor[0][0] = spec.lower_u;
457 vel_factor[0][1] = spec.lower_v;
458 vel_factor[0][2] = spec.lower_w;
459 vel_factor[1][0] = spec.upper_u;
460 vel_factor[1][1] = spec.upper_v;
461 vel_factor[1][2] = spec.upper_w;
462 }

470 template<typename T>
471 void op(const suzerain_bsmbsm& A_T,
472 T * const patpt,
473 int patpt_ld)
474 {
475 // Enforce isothermal conditions at each requested wall
476 for (int wall = wall_begin; wall < wall_end; ++wall) {
477
478 // Set constraint \partial_t rhoE - factor*\partial_t rho = 0
479 // by scanning row and adjusting coefficients for constraint.
480 {
481 int begin, end;
482 T * const col = (T *) suzerain_gbmatrix_col(
483 A_T.N, A_T.N, A_T.KL, A_T.KU, (void *) patpt, patpt_ld,
484 sizeof(T), e[wall], &begin, &end);
485 complex_t& scaling = col[e[wall]]; // Preserve scaling
486 if (scaling == complex_t(0)) scaling = 1; // Avoid degeneracy
487 for (int i = begin; i < end; ++i) {
488 col[i] = i == e [wall] ? +scaling
489 : i == rho[wall] ? -scaling*E_factor[wall]
490 : 0;
491 }
492 }
493
494 // Set constraint \partial_t rho{u,v,w} - factor*\partial_t rho = 0
495 // by scanning row and adjusting coefficients for constraint.
496 for (size_t eqn = 0; eqn < nmomentum; ++eqn) {
497 int begin, end;
498 T * const col = (T *) suzerain_gbmatrix_col(
499 A_T.N, A_T.N, A_T.KL, A_T.KU, (void *) patpt, patpt_ld,
500 sizeof(T), m[wall][eqn], &begin, &end);
501 complex_t& scaling = col[m[wall][eqn]]; // Preserve scaling
502 if (scaling == complex_t(0)) scaling = 1; // Avoid degeneracy
503 for (int i = begin; i < end; ++i) {
504 col[i] = i == m[wall][eqn] ? +scaling
505 : i == rho[wall] ? -scaling*vel_factor[wall][eqn]
506 : 0;
507 }
508 }
509 }
510 }
511
516 template<typename T>
517 void rhs(T * const b)
518 {
519 for (int wall = wall_begin; wall < wall_end; ++wall) {
520 b[e[wall]] = 0;
521 for (int eqn = 0; eqn < nmomentum; ++eqn) {
522 b[m[wall][eqn]] = 0;
523 }
524 }
525 }
526};

568
569 // Sidesteps assertions when local rank contains no wavespace information
570 if (SUZERAIN_UNLIKELY(0U == state.shape()[1])) return;
571
572 // Incoming state has wall-normal pencils of interleaved state scalars?
573 SUZERAIN_ENSURE(state.shape() [1] == (unsigned) Ny);
574 SUZERAIN_ENSURE(state.strides()[1] == 1);
575 SUZERAIN_ENSURE(state.strides()[0] == (unsigned) Ny);
576 SUZERAIN_ENSURE(state.shape() [0] == (unsigned) solver->S);
577
578 // Compute how many additional mean constraints we must solve
579 // Ensure conformant, mean constraints are arriving on the correct rank
580 const std::size_t nconstraints = ic0 ? ic0->shape()[2]*ic0->shape()[3] : 0;
581 if (nconstraints) {
582 SUZERAIN_ENSURE(dgrid.has_zero_zero_modes());
583 SUZERAIN_ENSURE(ic0->shape() [1] == (unsigned) Ny);
584 SUZERAIN_ENSURE(ic0->strides()[1] == 1);
585 SUZERAIN_ENSURE(ic0->strides()[0] == (unsigned) Ny);
586 SUZERAIN_ENSURE(ic0->shape() [0] == (unsigned) solver->S);
587 }
588
589 // treatment_channel step (3) performs the operator solve which for the
590 // implicit treatment must be combined with boundary conditions
591
592 // Pack reference details for suzerain_rholut_imexop routines
593 suzerain_rholut_imexop_scenario s(this->imexop_s());
594 suzerain_rholut_imexop_ref ref;
595 suzerain_rholut_imexop_refld ld;
596 common.ref.rholut_imexop(ref, ld);
597
598 // Prepare an almost functor mutating RHS and PA^TP^T to enforce BCs.
599 IsothermalPATPTEnforcer bc_enforcer(*solver, s, isothermal,
600 true, /* enforce_lower */
601 grid.two_sided() /* enforce_upper */);
602
603 // Prepare a scratch buffer for packc/packf usage per its documentation
604 ArrayX1c buf(std::max(75, solver->ld*solver->n));
605
606 // Iterate across local wavenumbers and "invert" operator "in-place"
607 //
608 // The loop structure changes markedly on rhome_xyz vs rhome_y,
609 // but the basic sequence of steps remains identical in each case.
610 switch (common.linearization) {
611
612 case linearize::rhome_xyz:
613 {
614 const real_t *maybe_a = grid.one_sided() ? upper_nrbc_a.data() : NULL;
615 const real_t *maybe_b = grid.one_sided() ? upper_nrbc_b.data() : NULL;
616 const real_t *maybe_c = grid.one_sided() ? upper_nrbc_c.data() : NULL;
617 for (int n = dkbz; n < dkez; ++n) {
618 const int wn = wavenumber(dNz, n);
619 const real_t kn = twopioverLz*wn;
620
621 // Factorization reuse will not aid us across large jumps in km
622 if (dkex - dkbx > 1) solver->apprx(false);
623
624 for (int m = dkbx; m < dkex; ++m) {
625 const int wm = wavenumber(dNx, m);
626 const real_t km = twopioverLx*wm;
627
628 // Get pointer to (.,m,n)-th state pencil
629 complex_t * const p = &state[0][0][m - dkbx][n - dkbz];
630
631 // Continue didn't yet occur for Nyquist/dealiasing modes...
632 if ( std::abs(wn) > wavenumber_absmin(Nz)
633 || std::abs(wm) > wavenumber_absmin(Nx)) {
634 std::fill_n(p, solver->N, complex_t()); // ...so we can zero,
635 solver->apprx(false); // mark reuse moot,
636 continue; // and short circuit.
637 }
638
639 // Form complex-valued, wavenumber-dependent PA^TP^T
640 static const char trans = 'T';
641 if (solver->spec.in_place()) { // Pack for in-place LU
642 SUZERAIN_TIMER_SCOPED("implicit operator assembly (packf)");
643 suzerain_rholut_imexop_packf(
644 phi, km, kn, &s, &ref, &ld, cop.get(),
645 ndx::e, ndx::mx, ndx::my, ndx::mz, ndx::rho,
646 buf.data(), solver.get(), solver->LU.data(),
647 maybe_a, maybe_b, maybe_c);
648 } else { // Pack for out-of-place LU
649 SUZERAIN_TIMER_SCOPED("implicit operator assembly (packc)");
650 suzerain_rholut_imexop_packc(
651 phi, km, kn, &s, &ref, &ld, cop.get(),
652 ndx::e, ndx::mx, ndx::my, ndx::mz, ndx::rho,
653 buf.data(), solver.get(), solver->PAPT.data(),
654 maybe_a, maybe_b, maybe_c);
655 }
656 // Apply boundary conditions to PA^TP^T
657 {
658 SUZERAIN_TIMER_SCOPED("implicit operator BCs");
659 bc_enforcer.op(*solver,
660 solver->PAPT.data(),
661 solver->PAPT.colStride());
662 }
663 // Inform the solver about the new, unfactorized operator
664 solver->supplied_PAPT();
665
666 // Form right hand side, apply BCs, factorize, and solve.
667 // Beware that much ugly, ugly magic is hidden just below.
668 solver->supply_B(p);
669 {
670 SUZERAIN_TIMER_SCOPED("implicit right hand side BCs");
671 bc_enforcer.rhs(solver->PB.data());
672 }
673 solver->solve(trans);
674 solver->demand_X(p);
675
676 // If necessary, solve any required integral constraints
677 if (SUZERAIN_UNLIKELY(n == 0 && m == 0)) {
678 for (std::size_t i = 0; i < nconstraints; ++i) {
679 SUZERAIN_TIMER_SCOPED("implicit constraint solution");
680 solver->supply_B(ic0->data() + i * solver->N);
681 bc_enforcer.rhs(solver->PB.data());
682 solver->solve(trans);
683 solver->demand_X(ic0->data() + i * solver->N);
684 }
685 }
686 }
687 }
688 break;
689 }
690
691 case linearize::rhome_y:
692 {
693 const real_t *maybe_c = grid.one_sided() ? upper_nrbc_c.data() : NULL;
694
695 // Form complex-valued, wavenumber-independent PA^TP^T
696 static const char trans = 'T';
697 if (solver->spec.in_place()) { // Pack for in-place LU
698 SUZERAIN_TIMER_SCOPED("implicit operator assembly (packf00)");
699 suzerain_rholut_imexop_packf00(
700 phi, &s, &ref, &ld, cop.get(),
701 ndx::e, ndx::mx, ndx::my, ndx::mz, ndx::rho,
702 buf.data(), solver.get(), solver->LU.data(),
703 maybe_c);
704 } else { // Pack for out-of-place LU
705 SUZERAIN_TIMER_SCOPED("implicit operator assembly (packc00)");
706 suzerain_rholut_imexop_packc00(
707 phi, &s, &ref, &ld, cop.get(),
708 ndx::e, ndx::mx, ndx::my, ndx::mz, ndx::rho,
709 buf.data(), solver.get(), solver->PAPT.data(),
710 maybe_c);
711 }
712 // Apply wavenumber-independent boundary conditions to PA^TP^T
713 {
714 SUZERAIN_TIMER_SCOPED("implicit operator BCs");
715 bc_enforcer.op(*solver,
716 solver->PAPT.data(),
717 solver->PAPT.colStride());
718 }
719 // Inform the solver about the new, unfactorized operator
720 solver->supplied_PAPT();
721
722 // Solve using a new right hand side for each wavenumber pair km, kn
723 for (int n = dkbz; n < dkez; ++n) {
724 const int wn = wavenumber(dNz, n);
725
726 for (int m = dkbx; m < dkex; ++m) {
727 const int wm = wavenumber(dNx, m);
728
729 // Get pointer to (.,m,n)-th state pencil
730 complex_t * const p = &state[0][0][m - dkbx][n - dkbz];
731
732 // Continue didn't yet occur for Nyquist/dealiasing modes...
733 if ( std::abs(wn) > wavenumber_absmin(Nz)
734 || std::abs(wm) > wavenumber_absmin(Nx)) {
735 std::fill_n(p, solver->N, complex_t()); // ...so we can zero,
736 continue; // and short circuit.
737 }
738
739 // Form right hand side, apply BCs, factorize, and solve.
740 // Beware that much ugly, ugly magic is hidden just below.
741 solver->supply_B(p);
742 {
743 SUZERAIN_TIMER_SCOPED("implicit right hand side BCs");
744 bc_enforcer.rhs(solver->PB.data());
745 }
746 solver->solve(trans);
747 solver->demand_X(p);
748 }
749 }
750
751 // If necessary, solve any required integral constraints
752 for (std::size_t i = 0; i < nconstraints; ++i) {
753 SUZERAIN_TIMER_SCOPED("implicit constraint solution");
754 solver->supply_B(ic0->data() + i * solver->N);
755 bc_enforcer.rhs(solver->PB.data());
756 solver->solve(trans);
757 solver->demand_X(ic0->data() + i * solver->N);
758 }
759
760 break;
761 }
762
763
764 default:
765 SUZERAIN_ERROR_VOID_UNIMPLEMENTED();
766 }
767
768 // State leaves method as coefficients in X, Y, and Z directions
769}
