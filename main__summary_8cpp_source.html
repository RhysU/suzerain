58 int run(int argc, char **argv)
59 {
60 // Invoke driver-agnostic summarization logic
61 summary_pool_type pool;
62 summary final;
63 support::shared_esio_handle esioh;
64 const int status = summary_run(argc, argv, pool, final, esioh);
65
66 // Perform driver-specific post-processing on nontrivial results
67 if (pool.size()) {
68
69 // Convert final from collocation point values to coefficients
70 bsplineop_lu masslu(*cop);
71 masslu.factor_mass(*cop);
72 masslu.solve(final.nongrid().cols(),
73 final.nongrid().data(),
74 final.nongrid().innerStride(),
75 final.nongrid().outerStride());
76
77 // Compute QoI given final profile information
78 profile prof;
79 prof = final;
80 instantaneous inst;
81 inst = final;
82 log_quantities_of_interest(
83 build_timeprefix_description("summary", "summary"),
84 prof, inst, esioh);
85 }
86
87 return status;
88 }
