97template<class Derived>
98class state_base
99{
100public:
101 typedef typename detail::state_traits<Derived>::element element;
102 typedef typename detail::state_traits<Derived>::index index;
103 typedef typename detail::state_traits<Derived>::size_type size_type;
104 static const size_type dimensionality
105 = detail::state_traits<Derived>::dimensionality;
106
111 Derived& derived()
112 {
113 return *static_cast<Derived*>(this);
114 }
115
117 const Derived& derived() const
118 {
119 return *static_cast<const Derived*>(this);
120 }
121
131 const size_type* shape() const
132 {
133 return derived().shape();
134 }
135
147 void scale(const element& factor)
148 {
149 return derived().scale(factor);
150 }
151
164 template<class OtherType>
165 bool is_isomorphic(const OtherType& other) const
166 {
167 if (dimensionality == OtherType::dimensionality) {
168 return std::equal(shape(), shape() + dimensionality, other.shape());
169 } else {
170 return false;
171 }
172 }
173
183 template<class OtherDerived>
184 void add_scaled(const element& factor,
185 const state_base<OtherDerived>& other)
186 {
187 return derived().add_scaled(factor, other.derived());
188 }
189
197 template<class OtherDerived>
198 void assign_from(const state_base<OtherDerived>& other)
199 {
200 return derived().assign_from(other.derived());
201 }
202
212 template<class OtherDerived>
213 void exchange(state_base<OtherDerived>& other)
214 {
215 return derived().exchange(other.derived());
216 }
217
220};
