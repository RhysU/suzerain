42int pencil_grid::compute_rank_zero_zero_modes_() const
43{
44 // Accumulate the sum of all rank numbers reporting has_zero_zero_modes().
45 // Accumulate the number of ranks where has_zero_zero_modes() == true.
46 // The first is our return value. The second is a sanity check.
47 const bool this_rank_has_it = has_zero_zero_modes();
48 int buf[2] = {
49 this_rank_has_it ? mpi::comm_rank(MPI_COMM_WORLD) : 0,
50 this_rank_has_it
51 };
52 SUZERAIN_MPICHKR(MPI_Allreduce(
53 MPI_IN_PLACE, &buf, sizeof(buf)/sizeof(buf[0]),
54 MPI_INT, MPI_SUM, MPI_COMM_WORLD));
55 SUZERAIN_ENSURE(buf[1] == 1);
56 return buf[0];
57}

67void pencil_grid_p3dfft::construct_(int Nx, int Ny, int Nz, int Pa, int Pb,
68 unsigned rigor_fft, unsigned rigor_mpi)
69#pragma warning(pop)
70{
71 SUZERAIN_UNUSED(rigor_mpi);
72 p3dfft_fftw_rigor(rigor_fft);
73
74 global_physical_extent[0] = Nx;
75 global_physical_extent[1] = Ny;
76 global_physical_extent[2] = Nz;
77 global_wave_extent[0] = Nx / 2 + 1;
78 global_wave_extent[1] = Ny;
79 global_wave_extent[2] = Nz;
80 processor_grid[0] = Pa;
81 processor_grid[1] = Pb;
82
83 // Ensure MPI is in a sane state
84 {
85 int flag;
86 if (MPI_Initialized(&flag) != MPI_SUCCESS || !flag) {
87 throw std::logic_error(
88 std::string("MPI stack not yet initialized"));
89 }
90 if (MPI_Finalized(&flag) != MPI_SUCCESS || flag) {
91 throw std::logic_error(
92 std::string("MPI stack already finalized"));
93 }
94 }
95
96 // Find the global number of processors
97 const int nproc = mpi::comm_size(MPI_COMM_WORLD);
98
99 // Find a suitable 2D Cartesian decomposition if necessary
100 if (processor_grid[0] == 0 || processor_grid[1] == 0) {
101 mpi::dims_create(
102 nproc,
103 processor_grid.data(),
104 processor_grid.data() + processor_grid.size());
105 }
106
107 // If neither Pa nor Pb was specified, enforce P_2 >= P_1
108 if (Pa == 0 && Pb == 0) {
109 std::sort(processor_grid.data(),
110 processor_grid.data() + processor_grid.size());
111 }
112
113 // Sanity check the 2D decomposition
114 if (processor_grid.prod() != nproc) {
115 std::ostringstream what;
116 what << "Processor grid dimensions " << processor_grid
117 << " incompatible with number of processors " << nproc;
118 throw std::runtime_error(what.str());
119 }
120
121 // Initialize P3DFFT using Y as STRIDE1 direction in wave space
122 {
123 p3dfft_setup(processor_grid.data(),
124 global_physical_extent[0],
125 global_physical_extent[2],
126 global_physical_extent[1],
127 1); // nuke btrans
128 p3dfft_setup_called_ = true;
129 }
130
131 // Ensure design assumptions hold for linked P3DFFT library
132 if (!p3dfft_using_stride1())
133 throw std::runtime_error("!p3dfft_using_stride1()");
134 if (p3dfft_get_precision() != 2)
135 throw std::runtime_error("p3dfft_get_precision() != 2");
136
137 // Retrieve information for local input and output pencils
138 // P3DFFT uses int types; defensively ensure we do too
139 p3dfft_get_dims(local_physical_start.data(),
140 local_physical_end.data(),
141 local_physical_extent.data(),
142 1 /* physical */);
143 p3dfft_get_dims(local_wave_start.data(),
144 local_wave_end.data(),
145 local_wave_extent.data(),
146 2 /* wave */);
147 // P3DFFT STRIDE1 physical space get_dims returns in (X, Z, Y) ordering
148 // Suzerain's pencils require (X, Y, Z) ordering; flip Z and Y data
149 std::swap(local_physical_start [1], local_physical_start [2]);
150 std::swap(local_physical_end [1], local_physical_end [2]);
151 std::swap(local_physical_extent[1], local_physical_extent[2]);
152 // P3DFFT STRIDE1 wave space get_dims returns in (Y, X, Z) ordering
153 // Suzerain's pencils require (X, Y, Z) ordering; flip Y and X data
154 std::swap(local_wave_start [0], local_wave_start [1]);
155 std::swap(local_wave_end [0], local_wave_end [1]);
156 std::swap(local_wave_extent[0], local_wave_extent[1]);
157 // Transform indices for C conventions; want ranges like [istart, iend)
158 local_physical_start -= 1;
159 local_wave_start -= 1;
160
161 // Compute which rank possesses the zero zero mode
162 rank_zero_zero_modes = compute_rank_zero_zero_modes_();
163}

172std::size_t pencil_grid_p3dfft::local_physical_storage() const
173{
174 // Wave space scalars are twice as big as physical space scalars
175 return std::max(2*local_wave_extent.prod(), local_physical_extent.prod());
176}
177
178std::size_t pencil_grid_p3dfft::local_wave_storage() const
179{
180 // Physical space scalars are half as big, but we must round up
181 const std::size_t prodphys = local_physical_extent.prod();
182 const std::size_t prodwave = local_wave_extent.prod();
183 return std::max(prodwave, prodphys/2 + (prodphys % 2));
184}

241void
242pencil_grid_underling::construct_(int Nx, int Ny, int Nz, int Pa, int Pb,
243 unsigned rigor_fft, unsigned rigor_mpi)
244{
245 using std::bad_alloc;
246 using std::runtime_error;
247 using std::invalid_argument;
248
249 if (Nx > 1 && Nx % 2)
250 throw invalid_argument("pencil_grid_underling requires even X extent");
251
252 // Assumes underling (and all dependencies) have been initialized
253
254 // Construct grid and problem
255 // Map from pencil_grid to underling's directional expectations
256 // long_n0 will be wave space and long_n2 will be physical space
257 grid.reset(new underling::grid(MPI_COMM_WORLD, Ny, Nz, Nx/2+1, Pa, Pb));
258 if (!grid)
259 throw runtime_error("underling::grid creation error");
260
261 problem.reset(new underling::problem(*grid, 2,
262 underling::transposed::long_n2 & underling::transposed::long_n0));
263 if (!problem)
264 throw runtime_error("underling::problem creation error");
265
266 // Allocate zeroed buffers buf and tmp. Clearing necessary to avoid NaNs.
267 // buf is maintained for execution while tmp is used for planning only.
268 buf.reset(blas::calloc_as<underling::real>(problem->local_memory()),
269 blas::free);
270 if (!buf) throw bad_alloc();
271 std::shared_ptr<underling::real[]> tmp(
272 blas::calloc_as<underling::real>(problem->local_memory()),
273 blas::free);
274 if (!tmp) throw bad_alloc();
275
276 // Prepare execution plans
277 transpose.reset(new underling::plan(
278 *problem, buf.get(), tmp.get(), 0, rigor_mpi));
279 if (!transpose)
280 throw runtime_error("underling::plan creation failed");
281
282 n1_c2c_backward.reset(new underling::fftw::plan(
283 underling::fftw::plan::c2c_backward(),
284 *problem, 1, buf.get(), tmp.get(), rigor_fft,
285 underling::fftw::packed::all));
286 if (!n1_c2c_backward)
287 throw runtime_error("n1_c2c_backward creation failed");
288
289 n2_c2r_backward.reset(new underling::fftw::plan(
290 underling::fftw::plan::c2r_backward(),
291 *problem, 2, buf.get(), tmp.get(), rigor_fft,
292 underling::fftw::packed::all));
293 if (!n2_c2r_backward)
294 throw runtime_error("n2_c2r_backward creation failed");
295
296 n2_r2c_forward.reset(new underling::fftw::plan(
297 *n2_c2r_backward, buf.get(), tmp.get(), rigor_fft));
298 if (!n2_r2c_forward)
299 throw runtime_error("n2_r2c_forward creation failed");
300
301 n1_c2c_forward.reset(new underling::fftw::plan(
302 *n1_c2c_backward, buf.get(), tmp.get(), rigor_fft));
303 if (!n1_c2c_forward)
304 throw runtime_error("n1_c2c_forward creation failed");
305
306 // Copy underling information into format pencil_grid expects
307 global_physical_extent << Nx, Ny, Nz;
308 global_wave_extent << (Nx / 2 + 1), Ny, Nz;
309 processor_grid << grid->pA_size(), grid->pB_size();
310
311 // Prepare information on wave space storage
312 {
313 const underling::extents e = problem->local_extents(0);
314 local_wave_start[0] = e.start[2];
315 local_wave_start[1] = e.start[0];
316 local_wave_start[2] = e.start[1];
317 local_wave_end[0] = e.start[2] + e.size[2];
318 local_wave_end[1] = e.start[0] + e.size[0];
319 local_wave_end[2] = e.start[1] + e.size[1];
320 local_wave_extent = local_wave_end - local_wave_start;
321
322 // Ensure wave space storage is column-major, contiguous Y X/2 Z
323 if (e.stride[3] != 1) throw runtime_error(
324 "pencil_grid_underling: wave space scalars not contiguous");
325 if (e.stride[0] != e.size[3] * e.stride[3]) throw runtime_error(
326 "pencil_grid_underling: wave space Y not contiguous");
327 if (e.stride[2] != e.size[0] * e.stride[0]) throw runtime_error(
328 "pencil_grid_underling: wave space X not contiguous");
329 if (e.stride[1] != e.size[2] * e.stride[2]) throw runtime_error(
330 "pencil_grid_underling: wave space Z not contiguous");
331 }
332
333 // Prepare information on physical space storage
334 {
335 const underling::fftw::extents e
336 = n2_c2r_backward->local_extents_output();
337 local_physical_start[0] = e.start[2];
338 local_physical_start[1] = e.start[0];
339 local_physical_start[2] = e.start[1];
340 local_physical_end[0] = e.start[2] + e.size[2];
341 local_physical_end[1] = e.start[0] + e.size[0];
342 local_physical_end[2] = e.start[1] + e.size[1];
343 local_physical_extent = local_physical_end - local_physical_start;
344
345 // Ensure physical space storage is column-major, contiguous X Z Y
346 if (e.stride[4] != 1) throw runtime_error(
347 "pencil_grid_underling: physical space components not contiguous");
348 if (e.stride[3] != e.size[4] * e.stride[4]) throw runtime_error(
349 "pencil_grid_underling: physical space scalars not contiguous");
350 if (e.stride[2] != e.size[3] * e.stride[3]) throw runtime_error(
351 "pencil_grid_underling: physical space Y not contiguous");
352 if (e.stride[1] != e.size[2] * e.stride[2]) throw runtime_error(
353 "pencil_grid_underling: physical space X not contiguous");
354 if (e.stride[0] != e.size[1] * e.stride[1]) throw runtime_error(
355 "pencil_grid_underling: physical space Z not contiguous");
356 }
357
358 // Compute which rank possesses the zero zero mode
359 rank_zero_zero_modes = compute_rank_zero_zero_modes_();
360}
