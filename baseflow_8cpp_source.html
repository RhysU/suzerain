103void
104baseflow_polynomial::conserved(
105 const real_t y,
106 real_t* base) const
107{
108 assert(x.cols() > 0);
109 const int nstate = x.cols() - 1;
110
111 // Compute baseflow
112 if (x.rows()) {
113
114 for (int j = 0; j < nstate; ++j) {
115 base[j] = gsl_poly_eval(x.col(j).data(), x.rows(), y);
116 }
117
118 } else {
119
120 using namespace std;
121 memset(base, 0, nstate*sizeof(real_t));
122
123 }

126void
127baseflow_polynomial::conserved(
128 const real_t y,
129 real_t* base,
130 real_t* dybase,
131 real_t* dxbase) const
132{
133 // Consistent number of state variables + pressure must be provided.
134 assert(x.cols() > 0);
135 assert(x.cols() == dx.cols());
136 const int nstate = x.cols() - 1;
137
138 // Compute baseflow and y-derivative
139 if (x.rows()) {
140
141 double res[2];
142 for (int j = 0; j < nstate; ++j) {
143 gsl_poly_eval_derivs(x.col(j).data(), x.rows(), y,
144 &res[0], sizeof(res) / sizeof(res[0]));
145 base [j] = res[0];
146 dybase[j] = res[1];
147 }
148
149 } else {
150
151 using namespace std;
152 memset(base, 0, nstate*sizeof(real_t));
153 memset(dybase, 0, nstate*sizeof(real_t));
154
155 }
156
157 // Compute x-derivative of baseflow
158 if (dx.rows()) {
159
160 for (int j = 0; j < nstate; j++) {
161 dxbase[j] = gsl_poly_eval(dx.col(j).data(), dx.rows(), y);
162 }
163
164 } else {
165
166 using namespace std;
167 memset(dxbase, 0, nstate*sizeof(real_t));
168
169 }
170}

178

187void
188baseflow_polynomial::pressure(
189 const real_t y,
190 real_t& P,
191 real_t& dyP,
192 real_t& dxP) const
193{
194 // Ensure consistent number of state variables + pressure
195 assert(x.cols() > 0);
196 assert(x.cols() == dx.cols());
197
198 // Compute pressure and y-derivative
199 double res[2] = { 0, 0 };
200 if (x.rows()) {
201 gsl_poly_eval_derivs(x.rightCols<1>().data(), x.rows(), y,
202 &res[0], sizeof(res) / sizeof(res[0]));
203 }
204 P = res[0];
205 dyP = res[1];
206
207 // Compute x-derivative of baseflow
208 dxP = dx.rows()
209 ? gsl_poly_eval(dx.rightCols<1>().data(), dx.rows(), y)
210 : 0;
211}

228// Beware this makes strong assumptions about baseflow_map::row
229// Structural changes here likely require adjusting baseflow_map::pressure
230// Apologies for the awful bracketing search logic
231void
232baseflow_map::conserved(
233 const real_t y,
234 real_t* base,
235 real_t* dybase,
236 real_t* dxbase) const
237{
238 using namespace std;
239 table_type::const_iterator uit = table.lower_bound(y);
240 if (SUZERAIN_UNLIKELY(uit == table.end())) { // Wildly wrong y?
241 ostringstream os;
242 os << "baseflow_map::conserved("
243 << fullprec<>(y)
244 << ",...) encountered table.lower_bound(y) == table.end()";
245 throw out_of_range(os.str());
246#ifdef __INTEL_COMPILER
247#pragma warning(push,disable:1572)
248#endif
249 } else if (SUZERAIN_LIKELY(uit->first == y)) { // Precomputed y
250#ifdef __INTEL_COMPILER
251#pragma warning(pop)
252#endif
253 const row& v = uit->second;
254 memcpy(base, &v.base, sizeof(v.base ) - sizeof(v.base .p));
255 memcpy(dybase, &v.dybase, sizeof(v.dybase) - sizeof(v.dybase.p));
256 memcpy(dxbase, &v.dxbase, sizeof(v.dxbase) - sizeof(v.dxbase.p));
257 } else {
258 table_type::const_iterator lit = uit;
259 if (SUZERAIN_UNLIKELY(lit == table.begin())) {
260 if (SUZERAIN_UNLIKELY(lit->first < y)) {
261 ostringstream os;
262 os << "baseflow_map::conserved("
263 << fullprec<>(y)
264 << ",...) encountered y < table.begin()->first";
265 throw out_of_range(os.str());
266 }
267 ++uit;
268 } else {
269 --lit;
270 }
271 enum { N = sizeof(lit->second.base)/sizeof(lit->second.base.p) };
272 for (size_t i = 0; i < N; ++i) {
273 // Employs local smoothness to get y, dy but does not for dx
274 // Computation of dx first to aid subexpression elimination
275 using namespace suzerain::math::interpolate;
276 value( /* x1*/ lit->first,
277 /* y1*/ lit->second.dxbase.as_is()[i],
278 /* x2*/ uit->first,
279 /* y2*/ uit->second.dxbase.as_is()[i],
280 /* x3*/ y,
281 /* y3*/ dxbase[i]);
282 value_deriv(/* x1*/ lit->first,
283 /* y1*/ lit->second. base.as_is()[i],
284 /*yp1*/ lit->second.dybase.as_is()[i],
285 /* x2*/ uit->first,
286 /* y2*/ uit->second. base.as_is()[i],
287 /*yp2*/ uit->second.dybase.as_is()[i],
288 /* x3*/ y,
289 /* y3*/ base [i],
290 /*yp3*/ dybase[i]);
291 }
292 }
