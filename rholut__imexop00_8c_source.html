77 // Prepare shorthand for some useful derived values
78 const double gm1 = s->gamma - 1;
79 const double gm3 = s->gamma - 3;
80 const double ap43 = s->alpha + 4.0/3.0;
81 const double ap13 = s->alpha + 1.0/3.0;
82 const double Ma2 = s->Ma*s->Ma;
83 const double invRe = 1 / s->Re;
84 const double invMa2 = 1 / Ma2;
85 const double ginvPr = s->gamma / s->Pr;
86 const double ginvRePr = s->gamma / (s->Re * s->Pr);

88 // Accumulate the requested portions of the M + \phi L operator. Scale
89 // output by beta, accumulate non-mass contributions, and finally accumulate
90 // the mass contributions. Mass contributions come last as they are
91 // expected to have magnitudes much larger than phi-- this helps to ensure
92 // better rounding of \phi L contributions.
93
94 // Readable shorthand for the common code patterns appearing below
95 //
96 // Notice we need to account for suzerain_bsplineop_workspace storing the
97 // transpose of the operators when we invoke suzerain_blasext_* routines.
98 // "Enumerated" constants i and j track the active NRBC matrix element.
99 enum { M = 0, D1 = 1, D2 = 2 };
100# define IN(quantity) in_##quantity, 1
101# define OUT(quantity) out_##quantity, 1
102# define REF(quantity) r->quantity, ld->quantity
103# define PREAMBLE_N(op) 'T', w->n, w->kl[op], w->ku[op]
104# define PREAMBLE_NN(op) 'T', w->n, w->n, w->kl[op], w->ku[op]
105
106 // Tracks operator action on the final "upper" coefficient w->n-1
107 complex_double upper_phi_L_hatV[5] = { 0 };
108
109 if (in_rho_E) { enum { i = 0 }; // Accumulate terms into out_rho_E
110
111 suzerain_blas_zscal(w->n, beta, OUT(rho_E));
112
113 swap(upper_phi_L_hatV + i, out_rho_E + w->n - 1);
114
115 /* in_rho_E */ { enum { j = 0 };
116 suzerain_blasext_zgbdmv_d_z(PREAMBLE_N(D1),
117 -phi*s->gamma, REF(uy),
118 w->D_T[D1], w->ld, IN(rho_E), 1.0, OUT(rho_E));
119
120 suzerain_blasext_zgbdmv_d_z(PREAMBLE_N(D2),
121 phi*ginvRePr, REF(nu),
122 w->D_T[D2], w->ld, IN(rho_E), 1.0, OUT(rho_E));
123 }
124
125 if (in_rho_u) { enum { j = 1 };
126 suzerain_blasext_zgbdmv_d_z(PREAMBLE_N(D2),
127 phi*Ma2*invRe*(1-ginvPr), REF(nuux),
128 w->D_T[D2], w->ld, IN(rho_u), 1.0, OUT(rho_E));
129 }
130
131 if (in_rho_v) { enum { j = 2 };
132 suzerain_blasext_zgbdmv_d_z(PREAMBLE_N(D1),
133 -phi, REF(e_divm),
134 w->D_T[D1], w->ld, IN(rho_v), 1.0, OUT(rho_E));
135
136 suzerain_blasext_zgbdmv_d_z(PREAMBLE_N(D2),
137 phi*Ma2*invRe*(ap43-ginvPr), REF(nuuy),
138 w->D_T[D2], w->ld, IN(rho_v), 1.0, OUT(rho_E));
139 }
140
141 if (in_rho_w) { enum { j = 3 };
142 suzerain_blasext_zgbdmv_d_z(PREAMBLE_N(D2),
143 phi*Ma2*invRe*(1-ginvPr), REF(nuuz),
144 w->D_T[D2], w->ld, IN(rho_w), 1.0, OUT(rho_E));
145 }
146
147 if (in_rho ) { enum { j = 4 };
148 suzerain_blasext_zgbdmv_d_z(PREAMBLE_N(D1),
149 -phi, REF(ey_gradrho),
150 w->D_T[D1], w->ld, IN(rho), 1.0, OUT(rho_E));
151
152 suzerain_blasext_zgbdddmv_d_z(PREAMBLE_N(D2),
153 -phi*Ma2*invRe, REF(nuu2),
154 -phi*Ma2*invRe*ap13, REF(nuuyuy),
155 phi*ginvRePr/gm1, REF(e_deltarho),
156 w->D_T[D2], w->ld, IN(rho), 1.0, OUT(rho_E));
157 }
158
159 swap(upper_phi_L_hatV + i, out_rho_E + w->n - 1);
160 out_rho_E[w->n - 1] += upper_phi_L_hatV[i];
161
162 suzerain_blas_zgbmv_d_z(PREAMBLE_NN(M),
163 1.0, w->D_T[M], w->ld, IN(rho_E), 1.0, OUT(rho_E));
164 }

168 suzerain_blas_zscal(w->n, beta, OUT(rho_u));

343 1.0, w->D_T[M], w->ld, IN(rho), 1.0, OUT(rho));

352 // When necessary, adjust upper boundary using provided NRBC matrix.

355 if (out_rho_E) {

369 }

372 out_rho_v[w->n-1] -= c[ 2]*upper_phi_L_hatV[0]
373 + c[ 7]*upper_phi_L_hatV[1]
374 + c[12]*upper_phi_L_hatV[2]
375 + c[17]*upper_phi_L_hatV[3]
376 + c[22]*upper_phi_L_hatV[4];
377 }
378

381 + c[ 8]*upper_phi_L_hatV[1]
