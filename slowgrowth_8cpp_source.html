58void
59slowgrowth::calculate_baseflow(
60 const real_t y,
61 largo_state &base,
62 largo_state &dy,
63 largo_state &dt,
64 largo_state &dx,
65 largo_state &src) const
66{
67 // Prepare any necessary baseflow information at the wall
68 // assuming that baseflow state already matched code_Ma.
69 if (sg.baseflow) {
70 sg.baseflow->conserved(y, base.as_is(), dy.as_is(), dx.as_is());
71 sg.baseflow->pressure (y, base.p, dy.p, dx.p);
72 } else {
73 base.zero();
74 dy.zero();
75 dx.zero();
76 }
77
78 // When a nontrivial inviscid base flow is present, compute
79 // model-appropriate residual so that it might be eradicated. The base
80 // flow is intended to be stationary so any residual is a numerical
81 // artifact and not part of the intended problem.
82 if (base.trivial()) {
83
84 dt.zero();
85 src.zero();
86
87 } else {
88
89 // TODO Reference relevant equations from Largo model document
90 //
91 // The largo_prestep_baseflow(..., srcbase) parameter
92 // requires semi-trivial, model-specific computations.
93 // The differences lie in the treatment of slow derivatives.
94 Vector3r grad_rho, grad_e, grad_p;
95 Matrix3r grad_m;
96 if (sg.formulation.is_strictly_temporal()) {
97
98 // Notice &this->basewall == &base works because base was populated
99 const real_t scale_u = basewall.u();
100
101 dt.rho = scale_u * dx.rho; // Modeled slow time
102 dt.mx = scale_u * dx.mx; // derivative uses
103 dt.my = scale_u * dx.my; // wall base flow
104 dt.mz = scale_u * dx.mz;
105 dt.e = scale_u * dx.e;
106
107 grad_rho << 0, dy.rho, 0; // Only account for
108 grad_m << 0, dy.mx, 0, // wall-normal variation
109 0, dy.my, 0, // Euler residual below.
110 0, dy.mz, 0; // Notice spanwise trivial.
111 grad_e << 0, dy.e, 0;
112 grad_p << 0, dy.p, 0;
113
114 } else {
115
116 dt.zero(); // Trivial slow time derivative inherent to formulation
117
118 grad_rho << dx.rho, dy.rho, 0; // Account for wall-normal
119 grad_m << dx.mx, dy.mx, 0, // and slow streamwise
120 dx.my, dy.my, 0, // variation in Euler
121 dx.mz, dy.mz, 0; // residual below.
122 grad_e << dx.e, dy.e, 0; // Notice spanwise trivial.
123 grad_p << dx.p, dy.p, 0;
124
125 }
126
127 // Compute spatial right hand side for inviscid Euler equations
128 // agnostic of model-appropriate derivatives supplied above.
129 // Base flow state from local variable "base" inlined below.
130 const Vector3r m (base.mx, base.my, base.mz);
131 const real_t div_m = grad_m.trace();
132 const real_t rhs_rho = - div_m;
133 const Vector3r u = rholut::u(base.rho, m);
134 const real_t div_u = rholut::div_u(base.rho, grad_rho,
135 m, div_m);
136 const Vector3r rhs_m = - rholut::div_u_outer_m(m, grad_m,
137 u, div_u)
138 - grad_p * inv_codeMa2;
139 const real_t rhs_e = - rholut::div_e_u(base.e, grad_e,
140 u, div_u)
141 - rholut::div_p_u(base.p, grad_p,
142 u, div_u);
143
144 // Compute largo_prestep_baseflow..., srcbase) for Largo
145 src.rho = dt.rho - rhs_rho;
146 src.mx = dt.mx - rhs_m.x();
147 src.my = dt.my - rhs_m.y();
148 src.mz = dt.mz - rhs_m.z();
149 src.e = dt.e - rhs_e;
150
151 }

154void
155slowgrowth::initialize(
156 const std::size_t substep_index)
157{
158 // If necessary, perform initialization calls for Largo
159 if (sg.formulation.enabled() && substep_index == 0) {
160
161 SUZERAIN_TIMER_SCOPED("slowgrowth::initialize");
162
163 // If necessary, allocate the Largo workspace.
164 // My apologies for breaking const-ness so disgustingly.
165 static const char who[] = "slowgrowth";
166 if (!sg.workspace) {
167 const std::string& model = sg.formulation.name();
168 enum { neq = 5, ns = 0, ntvar = 0 };
169 static const char ransmodel[] = "dns";
170 INFO0(who, "Allocating Largo model \"" << model
171 << "\" with neq=" << neq
172 << ", ns=" << ns << ", ntvar=" << ntvar
173 << ", ransmodel=" << ransmodel);
174 largo_allocate(const_cast<largo_workspace**>(&sg.workspace),
175 model.c_str(), neq,
176 ns, ntvar, ransmodel);
177 if (!sg.workspace) {
178 WARN0(who, "Largo could not allocate the requested model");
179 }
180 if ((boost::math::isnan)(sg.grdelta)) {
181 WARN0(who, "Slow growth rate grdelta is NaN");
182 }
183 if (sg.gramp_mean.empty()) {
184 WARN0(who, "Assuming trivial defect mean slow growth rates");
185 const_cast<std::vector<real_t>&>(sg.gramp_mean)
186 .assign(neq + /*pressure*/1, 0.0);
187 }
188 if (sg.gramp_rms.empty()) {
189 WARN0(who, "Assuming trivial defect RMS slow growth rates");
190 const_cast<std::vector<real_t>&>(sg.gramp_rms)
191 .assign(neq + /*pressure*/1, 0.0);
192 }
193 if (sg.ignore_fluctuations) {
194 WARN0(who, "Debugging option 'ignore_fluctuations' is active");
195 }
196 }
197
198 // Initialize the slow growth workspace
199 // Avoids debugging-related memory allocation if at all possible
200 SUZERAIN_ENSURE(sg.workspace);
201 SUZERAIN_ENSURE(sg.gramp_mean.size());
202 SUZERAIN_ENSURE(sg.gramp_mean.size() == sg.gramp_rms.size());
203 if (SUZERAIN_UNLIKELY(sg.ignore_gramp_mean || sg.ignore_gramp_rms)) {
204 std::vector<real_t> z(sg.gramp_mean.size(), 0);
205 largo_init(sg.workspace,
206 sg.grdelta,
207 sg.ignore_gramp_mean ? &z[0] : &sg.gramp_mean[0],
208 sg.ignore_gramp_rms ? &z[0] : &sg.gramp_rms [0]);
209 } else {
210 largo_init(sg.workspace,
211 sg.grdelta,
212 &sg.gramp_mean[0],
213 &sg.gramp_rms[0]);
214 }
215
216 // Prepare any necessary baseflow information at the wall
217 // As a side-effect, populates this->basewall appropriately
218 largo_state dy, dt, dx, src;
219 calculate_baseflow(0.0, this->basewall, dy, dt, dx, src);
220
221 // Present the baseflow information to Largo adjusting for code_Ma
222 largo_init_wall_baseflow(sg.workspace,
223 basewall.rescale(inv_codeMa2),
224 dy .rescale(inv_codeMa2),
225 dt .rescale(inv_codeMa2),
226 dx .rescale(inv_codeMa2),
227 src .rescale(inv_codeMa2));
228 }

231void
232slowgrowth::gather_wavexz(
233 const operator_tools &otool,
234 const contiguous_state<4,complex_t> &swave)
235{
236 if (!sg.formulation.enabled()) return; // Quick return when possible
237
238 SUZERAIN_TIMER_SCOPED("slowgrowth::gather_wavexz");
239
240 // Only models consuming conserved growth rates need RMS data,
241 // so we can save global communication in many circumstances.
242 if (sg.formulation.expects_conserved_growth_rates()) {
243
244 // With state being Fourier in X and Z but collocation in Y...
245 // ...compute L^2_{xz} of state at each collocation point
246 meanrms = compute_field_L2xz(swave, otool.grid, otool.dgrid);
247
248 // ...and rescale to convert to root-mean-square (RMS) fluctuations
249 // (mean L2 values are unused so also defensively NaN that storage)
250 const real_t rms_adjust = 1 / sqrt(otool.grid.L.x()*otool.grid.L.z());
251 for (size_t i = 0; i < meanrms.size(); ++i) {
252 meanrms[i].mean.setConstant(
253 std::numeric_limits<real_t>::quiet_NaN());
254 meanrms[i].fluctuating *= rms_adjust;
255 }
256
257 } else if (sg.formulation.expects_specific_growth_rates()) {
258
259 // While no information is saved, the meanrms member
260 // must be resized to match that produced above so
261 // gather_physical_cons can populate meanrms.mean[i].
262 meanrms.resize(swave.shape()[0]);
263 for (size_t i = 0; i < meanrms.size(); ++i) {
264 meanrms[i].mean.setConstant(
265 otool.dgrid.global_physical_extent.y(),
266 std::numeric_limits<real_t>::quiet_NaN());
267 meanrms[i].fluctuating.setConstant(
268 otool.dgrid.global_physical_extent.y(),
269 std::numeric_limits<real_t>::quiet_NaN());
270 }
271
272 } else {
273
274 SUZERAIN_ERROR_VOID_UNIMPLEMENTED();
275
276 }

284void
285slowgrowth::gather_physical_cons(
286 const operator_tools &otool,
287 const slowgrowth::physical_cons &data)
288{
289 if (!sg.formulation.enabled()) return; // Quick return when possible
290
291 SUZERAIN_TIMER_SCOPED("slowgrowth::gather_physical_cons");
292
293 // Slow growth requires mean conserved state at collocation points.
294 // Abuse unused pieces within 'meanrms' to avoid more allocations.
295 SUZERAIN_ENSURE(meanrms.size() == 5);
296 meanrms[ndx::e ].mean = data.rhoE();
297 meanrms[ndx::mx ].mean = data.rhou();
298 meanrms[ndx::my ].mean = data.rhov();
299 meanrms[ndx::mz ].mean = data.rhow();
300 meanrms[ndx::rho].mean = data.rho();
301
302 // Slow growth requires mean pressure and its fluctuation profiles.
303 // Abuse 'meanrms' to store the information after conserved state.
304 // The RMS of fluctuating pressure computation can be numerically
305 // noisy hence require a non-negative result prior to the sqrt.
306 meanrms.resize(meanrms.size() + 1);
307 meanrms.back().mean = data.p();
308 meanrms.back().fluctuating = (data.p2() - data.p().square())
309 .max(0).sqrt();
310
311 // Differing wall-normal derivatives are needed depending on the model.
312 meanrms_y.resize(meanrms.size());
313 if (sg.formulation.expects_conserved_growth_rates()) {
314
315 // Wall-normal derivatives of every mean/RMS quantity are required
316 ArrayX2r tmp(otool.dgrid.global_physical_extent.y(), 2);
317 std::vector<field_L2xz>::const_iterator src = meanrms.begin();
318 std::vector<field_L2xz>::const_iterator end = meanrms.end();
319 std::vector<field_L2xz>:: iterator dst = meanrms_y.begin();
320 for (/*just above*/; src != end; ++src, ++dst) {
321 tmp.col(0) = (*src).mean;
322 tmp.col(1) = (*src).fluctuating;
323 otool.masslu()->solve(tmp.cols(), tmp.data(),
324 tmp.innerStride(), tmp.outerStride());
325 (*dst).mean .resizeLike(tmp.col(0));
326 (*dst).fluctuating.resizeLike(tmp.col(1));
327 otool.cop.accumulate(1, 1.0, tmp.col(0).data(), tmp.innerStride(),
328 0.0, (*dst).mean.data(), 1);
329 otool.cop.accumulate(1, 1.0, tmp.col(1).data(), tmp.innerStride(),
330 0.0, (*dst).fluctuating.data(), 1);
331 }
332
333 } else if (sg.formulation.expects_specific_growth_rates()) {
334
335 // Wall-normal derivatives of only the mean quantities are needed
336 // and consequently we can use fluctuating storage as scratch space.
337 // (fluctuating derivatives unused so defensively NaN that storage)
338 std::vector<field_L2xz>::const_iterator src = meanrms.begin();
339 std::vector<field_L2xz>::const_iterator end = meanrms.end();
340 std::vector<field_L2xz>:: iterator dst = meanrms_y.begin();
341 for (/*just above*/; src != end; ++src, ++dst) {
342 (*dst).fluctuating = (*src).mean;
343 otool.masslu()->solve((*dst).fluctuating.cols(),
344 (*dst).fluctuating.data(),
345 (*dst).fluctuating.innerStride(),
346 (*dst).fluctuating.outerStride());
347 (*dst).mean.resizeLike((*dst).fluctuating);
348 otool.cop.accumulate(1, 1.0, (*dst).fluctuating.col(0).data(),
349 (*dst).fluctuating.innerStride(),
350 0.0, (*dst).mean.data(), 1);
351 (*dst).fluctuating.setConstant(
352 std::numeric_limits<real_t>::quiet_NaN());
353 }
354
355 } else {
356
357 SUZERAIN_ERROR_VOID_UNIMPLEMENTED();
358
359 }

366void
367slowgrowth::gather_physical_rqq(
368 const operator_tools &otool,
369 const physical_rqq &data)
370{
371 if (!sg.formulation.enabled()) return; // Quick return when possible
372
373 SUZERAIN_TIMER_SCOPED("slowgrowth::gather_physical_rqq");
374
375 // Obtain "rqq" values for tensorially-consistent homogenization
376 rqq.resize(otool.dgrid.global_physical_extent.y(), NoChange);
377 rqq.col(ndx::rho) = data.rho ();
378 rqq.col(ndx::mx ) = data.rhouu();
379 rqq.col(ndx::my ) = data.rhovv();
380 rqq.col(ndx::mz ) = data.rhoww();
381 rqq.col(ndx::e ) = data.rhoEE();
382
383 // Compute derivatives of these "rqq" values
384 rqq_y = rqq;
385 ArrayXr tmp;
386 for (int i = 0; i < rqq_y.cols(); ++i) {
387 tmp = rqq_y.col(i);
388 otool.masslu()->solve(tmp.cols(), tmp.data(),
389 tmp.innerStride(), tmp.outerStride());
390 otool.cop.accumulate(1, 1.0, tmp.data(), tmp.innerStride(),
391 0.0, rqq_y.col(i).data(), 1);
392 }

395void
396slowgrowth::inner_y(
397 const int j,
398 const real_t y_j)
399{
400 if (!sg.formulation.enabled()) return; // Quick return when possible
401
402 SUZERAIN_TIMER_SCOPED("slowgrowth::inner_y");
403 assert(sg.workspace);
404
405 // Provide any baseflow-dependent information to Largo
406 if (sg.baseflow) {
407 largo_state base, dy, dt, dx, src;
408 calculate_baseflow(y_j, base, dy, dt, dx, src);
409 largo_prestep_baseflow(sg.workspace,
410 base.rescale(inv_codeMa2),
411 dy .rescale(inv_codeMa2),
412 dt .rescale(inv_codeMa2),
413 dx .rescale(inv_codeMa2),
414 src .rescale(inv_codeMa2));
415 }
416
417 // Repack Y-dependent profiles into a form consumable by Largo
418 assert(meanrms.size() == 5 + 1); // State plus pressure
419 largo_state mean (meanrms [ndx::e ].mean[j],
420 meanrms [ndx::mx ].mean[j],
421 meanrms [ndx::my ].mean[j],
422 meanrms [ndx::mz ].mean[j],
423 meanrms [ndx::rho].mean[j],
424 meanrms .back() .mean[j]);
425 largo_state mean_y(meanrms_y[ndx::e ].mean[j],
426 meanrms_y[ndx::mx ].mean[j],
427 meanrms_y[ndx::my ].mean[j],
428 meanrms_y[ndx::mz ].mean[j],
429 meanrms_y[ndx::rho].mean[j],
430 meanrms_y.back() .mean[j]);
431 largo_state rms (meanrms [ndx::e ].fluctuating[j],
432 meanrms [ndx::mx ].fluctuating[j],
433 meanrms [ndx::my ].fluctuating[j],
434 meanrms [ndx::mz ].fluctuating[j],
435 meanrms [ndx::rho].fluctuating[j],
436 meanrms .back() .fluctuating[j]);
437 largo_state rms_y (meanrms_y[ndx::e ].fluctuating[j],
438 meanrms_y[ndx::mx ].fluctuating[j],
439 meanrms_y[ndx::my ].fluctuating[j],
440 meanrms_y[ndx::mz ].fluctuating[j],
441 meanrms_y[ndx::rho].fluctuating[j],
442 meanrms_y.back() .fluctuating[j]);
443 largo_state mean_rqq (rqq (j, ndx::e ), // Notice pressure
444 rqq (j, ndx::mx ), // entry is NaN as
445 rqq (j, ndx::my ), // it is allegedly
446 rqq (j, ndx::mz ), // unused. This
447 rqq (j, ndx::rho), // NaN makes sure.
448 std::numeric_limits<real_t>::quiet_NaN());
449 largo_state mean_rqq_y(rqq_y(j, ndx::e ), // Ditto re: NaN
450 rqq_y(j, ndx::mx ),
451 rqq_y(j, ndx::my ),
452 rqq_y(j, ndx::mz ),
453 rqq_y(j, ndx::rho),
454 std::numeric_limits<real_t>::quiet_NaN());
455
456 // If requested, have Largo ignore all fluctuations (for debugging)
457 if (SUZERAIN_UNLIKELY(sg.ignore_fluctuations)) {
458 // Hide RMS fluctuations from Largo
459 rms.zero();
460 rms_y.zero();
461
462 // Have "rqq" quantities reflect only the mean profile
463 mean_rqq.rho = mean.rho;
464 mean_rqq.mx = mean.mx * mean.u();
465 mean_rqq.my = mean.my * mean.v();
466 mean_rqq.mz = mean.mz * mean.w();
467 mean_rqq.e = mean.e * mean.E();
468
469 // Have "rqq" derivatives reflect only the mean profile
470 mean_rqq_y.rho = mean_y.rho;
471 mean_rqq_y.mx = mean.u()*(2*mean_y.mx - mean.u()*mean_y.rho);
472 mean_rqq_y.mx = mean.v()*(2*mean_y.my - mean.v()*mean_y.rho);
473 mean_rqq_y.mx = mean.w()*(2*mean_y.mz - mean.w()*mean_y.rho);
474 mean_rqq_y.e = mean.E()*(2*mean_y.e - mean.E()*mean_y.rho);
475 }
476
477 // Present the baseflow information to Largo
478 largo_prestep_seta_innery(
479 sg.workspace,
480 y_j,
481 mean .rescale(inv_codeMa2 ),
482 rms .rescale(inv_codeMa2 ),
483 mean_rqq .rescale(inv_codeMa2*inv_codeMa2),
484 mean_y .rescale(inv_codeMa2 ),
485 rms_y .rescale(inv_codeMa2 ),
486 mean_rqq_y.rescale(inv_codeMa2*inv_codeMa2));
487}
