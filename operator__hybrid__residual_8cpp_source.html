40std::vector<real_t> operator_hybrid_residual::apply_operator(
41 const real_t time,
42 contiguous_state<4,complex_t> &state,
43 const lowstorage::method_interface<element>& method,
44 const std::size_t substep_index) const
45{
46 // Allocate (potentially large) extra working storage
47 state_linear_type extra(shape_array(state));
48
49 // The following steps are taken
50 // (1) extra <- state
51 // (2) state <- R(state)
52 // (3) state <- (M + 0 * L) extra - 1 * state
53 // (4) state <- (M - (1/chi) * L) extra - 1 * state
54 // and they ultimately result in
55 // state <- R(u) - (1/chi) Lu = N(u)
56 // There are lots of ways to make this process less computationally
57 // expensive if it turns out to be problematic, but this procedure
58 // works in the context of the current APIs.
59
60 extra.assign_from(state);
61 std::vector<real_t> retval = R->apply_operator(
62 time, state, method, substep_index);
63 L->accumulate_mass_plus_scaled_operator(
64 0, extra, -1, state, substep_index);
65 L->accumulate_mass_plus_scaled_operator(
66 -1/chi, extra, -1, state, substep_index);
67
68 return retval;
69}
