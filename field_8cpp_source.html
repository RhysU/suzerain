82void save_coefficients(
83 const esio_handle h,
84 const std::vector<field> &fields,
85 const contiguous_state<4,complex_t> &swave,
86 const specification_grid& grid,
87 const pencil_grid& dgrid)
88{
89 // Ensure swave meets this routine's assumptions
90 SUZERAIN_ENSURE( swave.shape()[0] == fields.size() );
91 SUZERAIN_ENSURE(numeric_cast<int>(swave.shape()[1]) == dgrid.local_wave_extent.y());
92 SUZERAIN_ENSURE(numeric_cast<int>(swave.shape()[2]) == dgrid.local_wave_extent.x());
93 SUZERAIN_ENSURE(numeric_cast<int>(swave.shape()[3]) == dgrid.local_wave_extent.z());
94
95 // Compute wavenumber translation logistics for X direction
96 int fxb[2], fxe[2], mxb[2], mxe[2];
97 inorder::wavenumber_translate(grid.N.x(),
98 grid.dN.x(),
99 dgrid.local_wave_start.x(),
100 dgrid.local_wave_end.x(),
101 fxb[0], fxe[0], fxb[1], fxe[1],
102 mxb[0], mxe[0], mxb[1], mxe[1]);
103 // X contains only positive wavenumbers => second range must be empty
104 SUZERAIN_ENSURE(fxb[1] == fxe[1]);
105 SUZERAIN_ENSURE(mxb[1] == mxe[1]);
106
107 // Compute wavenumber translation logistics for Z direction
108 // One or both ranges may be empty
109 int fzb[2], fze[2], mzb[2], mze[2];
110 inorder::wavenumber_translate(grid.N.z(),
111 grid.dN.z(),
112 dgrid.local_wave_start.z(),
113 dgrid.local_wave_end.z(),
114 fzb[0], fze[0], fzb[1], fze[1],
115 mzb[0], mze[0], mzb[1], mze[1]);
116
117 // Save each scalar field in turn...
118 for (size_t i = 0; i < fields.size(); ++i) {
119
120 // ...first generate a metadata comment...
121 std::string comment = fields[i].description;
122 comment += " stored row-major ZXY using a Fourier basis in Z stored"
123 " in-order per /kz; a Fourier basis in X stored using"
124 " Hermitian symmetry per /kx; and a B-spline basis in Y"
125 " defined by /k, /breakpoints_y, and /knots";
126
127 // ...followed by two collective writes per field (once per Z range)
128 for (int j = 0; j < 2; ++j) {
129
130 // Collectively establish size of write across all ranks
131 esio_field_establish(h,
132 grid.N.z(), fzb[j], (fze[j] - fzb[j]),
133 grid.N.x()/2+1, fxb[0], (fxe[0] - fxb[0]),
134 grid.N.y(), 0, ( grid.N.y()));
135
136 // Source of write is NULL for empty WRITE operations
137 // Required since MultiArray triggers asserts on invalid indices
138 const complex_t * src = NULL;
139 if (mxb[0] != mxe[0] && mzb[j] != mze[j]) {
140 src = &swave[i][0][mxb[0] - dgrid.local_wave_start.x()]
141 [mzb[j] - dgrid.local_wave_start.z()];
142 }
143
144 // Perform collective write operation
145 complex_field_write(h, fields[i].location.c_str(), src,
146 swave.strides()[3],
147 swave.strides()[2],
148 swave.strides()[1],
149 comment.c_str());
150 }
151 }
152}

154void load_coefficients(const esio_handle h,
155 const std::vector<field> &fields,
156 contiguous_state<4,complex_t> &state,
157 const specification_grid& grid,
158 const pencil_grid& dgrid,
159 const bsplineop& cop,
160 const bspline& b)
161{
162 typedef contiguous_state<4,complex_t> load_type;
163
164 // Ensure local state storage meets this routine's assumptions
165 SUZERAIN_ENSURE( state.shape()[0] == fields.size());
166 SUZERAIN_ENSURE(numeric_cast<int>(state.shape()[1]) == dgrid.global_wave_extent.y());
167 SUZERAIN_ENSURE(numeric_cast<int>(state.shape()[2]) == dgrid.local_wave_extent.x());
168 SUZERAIN_ENSURE(numeric_cast<int>(state.shape()[3]) == dgrid.local_wave_extent.z());
169
170 // Obtain details on the restart field's global sizes
171 int Fz, Fx, Fy, ncomponents;
172 esio_field_sizev(h, fields[0].location.c_str(), &Fz, &Fx, &Fy, &ncomponents);
173 SUZERAIN_ENSURE(ncomponents == 2);
174
175 // Prepare a file-specific B-spline basis
176 std::shared_ptr<bspline> Fb;
177 load_bsplines(h, Fb);
178 SUZERAIN_ENSURE(Fy == Fb->n());
179
180 // Check if the B-spline basis in the file differs from ours.
181 const double bsp_dist = b.distance_to(*Fb);
182 const bool bsp_same = bsp_dist < suzerain_bspline_distance_distinct;
183
184 // Compute wavenumber translation logistics for X direction.
185 // Requires turning a C2R FFT complex-valued coefficient count into a
186 // real-valued coefficient count. Further, need to preserve even- or
187 // odd-ness of the coefficient count to handle, for example, Fx == 1.
188 int fxb[2], fxe[2], mxb[2], mxe[2];
189 inorder::wavenumber_translate(2 * (Fx - 1) + (Fx & 1),
190 grid.dN.x(),
191 dgrid.local_wave_start.x(),
192 dgrid.local_wave_end.x(),
193 fxb[0], fxe[0], fxb[1], fxe[1],
194 mxb[0], mxe[0], mxb[1], mxe[1]);
195 // X contains only positive wavenumbers => second range must be empty
196 SUZERAIN_ENSURE(fxb[1] == fxe[1]);
197 SUZERAIN_ENSURE(mxb[1] == mxe[1]);
198
199 // Compute wavenumber translation logistics for Z direction
200 // One or both ranges may be empty
201 int fzb[2], fze[2], mzb[2], mze[2];
202 inorder::wavenumber_translate(Fz,
203 grid.dN.z(),
204 dgrid.local_wave_start.z(),
205 dgrid.local_wave_end.z(),
206 fzb[0], fze[0], fzb[1], fze[1],
207 mzb[0], mze[0], mzb[1], mze[1]);
208
209 // Possibly prepare a tmp buffer into which to read each scalar field and a
210 // factorization of b's mass matrix. Used only when !bsp_same.
211 typedef boost::multi_array<
212 complex_t, 3, blas::allocator<complex_t>::type
213 > tmp_type;
214 std::unique_ptr<tmp_type> tmp;
215 std::unique_ptr<bsplineop_luz> mass;
216 if (!bsp_same) {
217 INFO0("Differences in B-spline basis require restart projection ("
218 << bsp_dist << " >= "
219 << suzerain_bspline_distance_distinct << ")");
220 const array<tmp_type::size_type,3> extent = {{
221 static_cast<tmp_type::size_type>(Fy),
222 state.shape()[2],
223 state.shape()[3]
224 }};
225 tmp.reset(new tmp_type(extent, boost::fortran_storage_order()));
226 mass.reset(new bsplineop_luz(cop));
227 mass->factor_mass(cop);
228 }
229
230 // Load each scalar field in turn
231 for (size_t i = 0; i < fields.size(); ++i) {
232
233 // Create a view of the state for just the i-th scalar
234 boost::multi_array_types::index_range all;
235 boost::array_view_gen<load_type, 3>::type field
236 = state[boost::indices[i][all][all][all]];
237
238 // Clear storage prior to load to zero not-loaded coefficients
239 if (bsp_same) {
240 multi_array::fill(field, 0);
241 } else {
242 multi_array::fill(*tmp, 0);
243 }
244
245 // Two ESIO read operations per field (once per Z range)
246 for (int j = 0; j < 2; ++j) {
247
248 // Collectively establish size of read across all ranks
249 esio_field_establish(h, Fz, fzb[j], (fze[j] - fzb[j]),
250 Fx, fxb[0], (fxe[0] - fxb[0]),
251 Fy, 0, ( Fy));
252
253 // Destination of read is NULL for empty READ operations.
254 // Otherwise find starting point for coefficient loading.
255 complex_t * dst = NULL;
256 array<load_type::index,3> dst_strides = {{ 0, 0, 0 }};
257 if (mxb[0] != mxe[0] && mzb[j] != mze[j]) {
258 const array<load_type::index,3> index_list = {{
259 0,
260 mxb[0] - dgrid.local_wave_start.x(),
261 mzb[j] - dgrid.local_wave_start.z()
262 }};
263 if (bsp_same) {
264 dst = &field(index_list);
265 std::copy(field.strides(), field.strides() + 3,
266 dst_strides.begin());
267 } else {
268 dst = &(*tmp)(index_list);
269 std::copy(tmp->strides(), tmp->strides() + 3,
270 dst_strides.begin());
271 }
272 }
273
274 // Perform collective read operation into dst
275 complex_field_read(h, fields[i].location.c_str(), dst,
276 dst_strides[2], dst_strides[1], dst_strides[0]);
277 }
278
279 // If necessary, interpolate between B-spline bases.
280 // Relies heavily on both bases being collocation-based.
281 // This will change dramatically if we ever go the L_2 route.
282 if (!bsp_same) {
283
284 // Step 0: Obtain collocation points for new basis
285 ArrayXr points(b.n());
286 for (int k = 0; k < b.n(); ++k) points[k] = b.collocation_point(k);
287
288 // Step 1: Affine transformation of points into old basis domain
289 // Allows stretching of old range onto new range.
290 const double newmin = b.collocation_point(0);
291 const double newmax = b.collocation_point(b.n() - 1);
292 const double oldmin = Fb->collocation_point(0);
293 const double oldmax = Fb->collocation_point(Fb->n() - 1);
294 // Only pay for floating point loss if strictly necessary
295#pragma warning(push,disable:1572)
296 if (oldmin != newmin || oldmax != newmax) {
297#pragma warning(pop)
298 points = (points - newmin)
299 * ((oldmax - oldmin) / (newmax - newmin)) + oldmin;
300 }
301
302 // Step 2: Evaluate old basis + coefficients at points into new
303 const int jmax = numeric_cast<int>(field.shape()[1]);
304 const int kmax = numeric_cast<int>(field.shape()[2]);
305 for (int k = 0; k < kmax; ++k) {
306 for (int j = 0; j < jmax; ++j) {
307 Fb->linear_combination(0, &(*tmp)[0][j][k], b.n(),
308 points.data(), &field[0][j][k], 0);
309 }
310 }
311
312 // Step 3: Invert for new coefficients using factored mass matrix
313 for (int k = 0; k < kmax; ++k) {
314 mass->solve(jmax, &field[0][0][k],
315 numeric_cast<int>(field.strides()[0]),
316 numeric_cast<int>(field.strides()[1]));
317 }
318 }
319 }

322void save_collocation_values(
323 const esio_handle h,
324 const std::vector<field> &fields,
325 contiguous_state<4,complex_t>& swave,
326 const specification_grid& grid,
327 const pencil_grid& dgrid,
328 const bsplineop& cop)
329{
330 // Ensure state storage meets this routine's assumptions
331 SUZERAIN_ENSURE( swave.shape()[0] == fields.size());
332 SUZERAIN_ENSURE(numeric_cast<int>(swave.shape()[1]) == dgrid.local_wave_extent.y());
333 SUZERAIN_ENSURE(numeric_cast<int>(swave.shape()[2]) == dgrid.local_wave_extent.x());
334 SUZERAIN_ENSURE(numeric_cast<int>(swave.shape()[3]) == dgrid.local_wave_extent.z());
335
336 // Convert coefficients into collocation point values
337 // Transforms state from full-wave coefficients to full-physical points
338 operator_tools otool(grid, dgrid, cop);
339 for (size_t i = 0; i < swave.shape()[0]; ++i) {
340 otool.bop_apply(0, 1, swave, i);
341 otool.zero_dealiasing_modes(swave, i);
342 dgrid.transform_wave_to_physical(
343 reinterpret_cast<real_t *>(swave[i].origin()));
344 }
345
346 // Establish size of collective writes across all ranks
347 esio_field_establish(h, grid.dN.y(), dgrid.local_physical_start.y(),
348 dgrid.local_physical_extent.y(),
349 grid.dN.z(), dgrid.local_physical_start.z(),
350 dgrid.local_physical_extent.z(),
351 grid.dN.x(), dgrid.local_physical_start.x(),
352 dgrid.local_physical_extent.x());
353
354 // Write each field in turn
355 for (size_t i = 0; i < fields.size(); ++i) {
356 std::string comment = fields[i].description;
357 comment += " stored row-major YZX on the 3D rectilinear grid defined"
358 " by taking the outer product of arrays"
359 " /collocation_points_y, /collocation_points_z, and"
360 " /collocation_points_z";
361
362 esio_field_write(h, fields[i].location.c_str(),
363 reinterpret_cast<real_t *>(swave[i].origin()),
364 0, 0, 0, comment.c_str());
365 }
366}

368void load_collocation_values(
369 const esio_handle h,
370 const std::vector<field> &fields,
371 contiguous_state<4,complex_t>& state,
372 const specification_grid& grid,
373 const pencil_grid& dgrid,
374 const bsplineop& cop,
375 const bspline& b)
376{
377 // Ensure state storage meets this routine's assumptions
378 SUZERAIN_ENSURE( state.shape()[0] == fields.size());
379 SUZERAIN_ENSURE(numeric_cast<int>(state.shape()[1]) == dgrid.local_wave_extent.y());
380 SUZERAIN_ENSURE(numeric_cast<int>(state.shape()[2]) == dgrid.local_wave_extent.x());
381 SUZERAIN_ENSURE(numeric_cast<int>(state.shape()[3]) == dgrid.local_wave_extent.z());
382
383 // This routine does no grid interpolation. Yell loudly if necessary
384 {
385 // Check that restart file size matches runtime dealiased extents
386 int cg, bg, ag;
387 if ( ESIO_SUCCESS
388 != esio_field_size(h, fields[0].location.c_str(), &cg, &bg, &ag)) {
389 SUZERAIN_ERROR_VOID("Unable to find fields[0] size from restart",
390 SUZERAIN_EFAILED);
391 }
392 if (cg != grid.dN.y() || bg != grid.dN.z() || ag != grid.dN.x()) {
393 ERROR0("Physical space restart fields have row-major YZX extents "
394 << "(" << cg << "," << bg << "," << ag << ")" << " but "
395 << "(" << grid.dN.y() << "," << grid.dN.z() << ","
396 << grid.dN.x() << ") are required");
397 SUZERAIN_ERROR_VOID(
398 "Cannot interpolate during physical space restart",
399 SUZERAIN_EFAILED);
400 }
401
402 // Check that restart file specifies the same B-spline basis.
403 // TODO Too restrictive? Any floating point differences kill us.
404 std::shared_ptr<bspline> Fb;
405 load_bsplines(h, Fb);
406 const double bsp_dist = b.distance_to(*Fb);
407 const bool bsp_same = bsp_dist < suzerain_bspline_distance_distinct;
408 if (!bsp_same) {
409 ERROR0("Physical restart has different wall-normal bases ("
410 << bsp_dist << ")");
411 SUZERAIN_ERROR_VOID(
412 "Cannot interpolate during physical space restart",
413 SUZERAIN_EFAILED);
414 }
415 }
416
417 // Prepare physical space view of the state storage
418 physical_view<> sphys(dgrid, state);
419
420 // Establish size of collective reads across all ranks
421 esio_field_establish(h, grid.dN.y(), dgrid.local_physical_start.y(),
422 dgrid.local_physical_extent.y(),
423 grid.dN.z(), dgrid.local_physical_start.z(),
424 dgrid.local_physical_extent.z(),
425 grid.dN.x(), dgrid.local_physical_start.x(),
426 dgrid.local_physical_extent.x());
427
428 // Read each field from collocation values on disk into the same in memory
429 for (size_t i = 0; i < fields.size(); ++i) {
430 esio_field_read(h, fields[i].location.c_str(), &sphys(i,0), 0, 0, 0);
431 }
432
433 // Collectively convert physical state to wave space coefficients
434 // Build FFT normalization constant into Y direction's mass matrix
435 bsplineop_luz massluz(cop);
436 const complex_t scale_factor = 1 / dgrid.chi();
437 massluz.opform(1, &scale_factor, cop);
438 massluz.factor();
439 operator_tools otool(grid, dgrid, cop);
440 for (size_t i = 0; i < state.shape()[0]; ++i) {
441 dgrid.transform_physical_to_wave(&sphys.coeffRef(i, 0)); // X, Z
442 otool.zero_dealiasing_modes(state, i);
443 otool.bop_solve(massluz, state, i); // Y
444 }
445}
