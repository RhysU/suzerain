112static void processfile(const char *name, int failsilent, int ignore_eol,
113 int *max_build, int *accum_build,
114 int *max_year, int *max_month, int *max_day,
115 int *ismodified)
116
117{
118 char str[MAX_LINELENGTH], str_base[MAX_LINELENGTH];
119 char name_base[MAX_LINELENGTH];
120 char *p1;
121 FILE *fp, *fp_base;
122 int build, maj_build;
123 int year, month, day;
124 int cnt;
125 char modchar;
126
127 /* since we also want to verify whether the file is modified in version
128 * control, get the path to the working copy name
129 * for every source file "<path><filename>, the "working copy" base can
130 * be found in "<path>\.svn\text-base<filename>.svn-base"
131 */
132 if ((p1 = strrchr(name, DIRSEP)) != NULL) {
133 ++p1; /* skip directory separator character ('\' in Windows, '/' in Linux) */
134 strncpy(name_base, name, (int)(p1 - name));
135 name_base[(int)(p1 - name)] = '\0';
136 } else {
137 name_base[0] = '\0';
138 p1 = (char*)name;
139 } /* if */
140 sprintf(name_base + strlen(name_base), ".svn%ctext-base%c%s.svn-base",
141 DIRSEP, DIRSEP, p1);
142
143 /* first extract the revision keywords */
144 fp = fopen(name, "r");
145 if (fp == NULL) {
146 if (!failsilent)
147 fprintf(stderr, "Failed to open input file '%s'\n", name);
148 return;
149 } /* if */
150 fp_base = fopen(name_base, "r"); /* fail silently */
151 build = 0;
152 maj_build = 0; /* RCS / CVS */
153 year = month = day = 0;
154
155 while (fgets(str, sizeof str, fp) != NULL) {
156 if (fp_base == NULL || fgets(str_base, sizeof str_base, fp_base) == NULL)
157 str_base[0] = '\0';
158 if ((p1 = strstr(str, "$Id:")) != NULL && strchr(p1+1, '$') != NULL) {
159 if (sscanf(p1, "$Id: %*s %d %d-%d-%d", &build, &year, &month, &day) < 4
160 && sscanf(p1, "$Id: %*s %d %d/%d/%d", &build, &year, &month, &day) < 4)
161 if (sscanf(p1, "$Id: %*s %d.%d %d-%d-%d", &maj_build, &build, &year, &month, &day) < 5)
162 sscanf(p1, "$Id: %*s %d.%d %d/%d/%d", &maj_build, &build, &year, &month, &day);
163 } else if ((p1 = strstr(str, "$Rev:")) != NULL && strchr(p1+1, '$') != NULL) {
164 if (sscanf(p1, "$Rev: %d.%d", &maj_build, &build) < 2) {
165 sscanf(p1, "$Rev: %d", &build);
166 maj_build = 0;
167 } /* if */
168 } else if ((p1 = strstr(str, "$Revision:")) != NULL && strchr(p1+1, '$') != NULL) {
169 if (sscanf(p1, "$Revision: %d.%d", &maj_build, &build) < 2) {
170 /* SvnRev also writes this keyword in its own generated file; read it
171 * back for partial updates
172 */
173 cnt = sscanf(p1, "$Revision: %d%c", &build, &modchar);
174 if (cnt == 2 && modchar == 'M' && ismodified != NULL)
175 *ismodified = 1;
176 maj_build = 0;
177 } /* if */
178 } else if ((p1 = strstr(str, "$Date:")) != NULL && strchr(p1+1, '$') != NULL) {
179 if (sscanf(p1, "$Date: %d-%d-%d", &year, &month, &day) < 3)
180 sscanf(p1, "$Date: %d/%d/%d", &year, &month, &day);
181 } else if (ismodified != NULL && *ismodified == 0 && fp_base != NULL) {
182 /* no keyword present, compare the lines for equivalence,
183 * but (optionally) first cut off the strings at the first CR/LF character
184 * (for cross-platform compatibility)
185 */
186 if (ignore_eol) {
187 for (p1 = strchr(str, '\0'); p1 != NULL && p1 > str && (*(p1 - 1) == '\r' || *(p1 - 1) == '\n'); p1--)
188 *--p1 = '\0';
189 for (p1 = strchr(str_base, '\0'); p1 != NULL && p1 > str_base && (*(p1 - 1) == '\r' || *(p1 - 1) == '\n'); p1--)
190 *--p1 = '\0';
191 } /* if */
192 *ismodified = strcmp(str, str_base) != 0;
193 } /* if */
194
195 if (maj_build)
196 *accum_build += build; /* RCS / CVS */
197 else if (build > *max_build)
198 *max_build = build; /* Subversion */
199 if (year > *max_year
200 || (year == *max_year && month > *max_month)
201 || (year == *max_year && month == *max_month && day > *max_day))
202 {
203 *max_year = year;
204 *max_month = month;
205 *max_day = day;
206 } /* if */
207 if (build > 0 && year > 0 && (fp_base == NULL || ismodified == NULL || *ismodified != 0))
208 break; /* both build # and date found, not comparing or modification
209 * already found => no need to search further */
210
211 } /* while */
212 fclose(fp);
213 if (fp_base != NULL)
214 fclose(fp_base);
215}

217static int scanmodifications(const char *filename, const char *basename)
218{
219 FILE *fp;
220 char str[MAX_LINELENGTH];
221 char constant[128];
222 char *ptr;
223 size_t constlen;
224 int modifcount = 0;
225
226 if ((fp = fopen(filename, "r")) == NULL)
227 return 0;
228
229 sprintf(constant, "%sMODIFIED", basename);
230 constlen = strlen(constant);
231 while (fgets(str, sizeof str, fp) != NULL) {
232 if ((ptr = strstr(str, constant)) != NULL && ptr > str && *(ptr - 1) <= ' ' && *(ptr + constlen) <= ' ')
233 modifcount = strtol(ptr + constlen, NULL, 10);
234 } /* while */
235
236 fclose(fp);
237 return modifcount;
238}

240static void get_buildno_ptr(char **start, char **stop)
241{
242 int rem_points = 3;
243 char *stop_old = *stop;
244
245 while(rem_points > 0 && *start < stop_old)
246 {
247 while(**start != '.' && *start < stop_old)
248 {
249 (*start)++;
250 }
251 (*start)++;
252 rem_points--;
253 }
254 *stop = *start;
255 while(**stop != '"' && *stop < stop_old) (*stop)++;
256}

258static void update_prop_file(char *filename, int rev)
259{
260 char line[512];
261 char fileout[128];
262 char *out;
263 char *outComment;
264 char *outVersion;
265 char *outFileVersion;
266 FILE *fsource;
267 FILE *fdest;
268
269 if(filename == NULL || strlen(filename) == 0)
270 {
271 return;
272 }
273
274 if((fsource = fopen(filename, "r+")) == NULL)
275 {
276 fprintf(stderr, "Failed to open output file '%s'\n", filename);
277 return;
278 }
279
280 sprintf(fileout, "%s.tmp", filename);
281 if((fdest = fopen(fileout, "w")) == NULL)
282 {
283 fprintf(stderr, "Failed to create output file '%s'\n", fileout);
284 fclose(fsource);
285 return;
286 }
287
288 /*
289 look for these two lines:
290 [assembly: AssemblyVersion("1.2.3.4")]
291 [assembly: AssemblyFileVersion("1.2.3.4")]
292 */
293
294 while((out = fgets(line, 512, fsource)) != NULL)
295 {
296 outComment = strstr(line, "//");
297 outVersion = strstr(line, "AssemblyVersion");
298 outFileVersion = strstr(line, "AssemblyFileVersion");
299 if(outVersion == NULL && outFileVersion == NULL)
300 {
301 fputs(line, fdest);
302 continue;
303 }
304
305 if(outVersion != NULL)
306 {
307 if(outComment == NULL || outComment > outVersion)
308 {
309 char *start, *stop;
310 char buffer[512];
311 start = outVersion + strlen("AssemblyVersion") + 2;
312 stop = start + strlen(start) - 1;
313 get_buildno_ptr(&start, &stop);
314 strcpy(buffer, stop);
315 sprintf(start, "%i%s", rev, buffer);
316 fputs(line, fdest);
317 }
318 else
319 {
320 fputs(line, fdest);
321 continue;
322 }
323 }
324
325 if(outFileVersion != NULL)
326 {
327 if(outComment == NULL || outComment > outFileVersion)
328 {
329 char *start, *stop;
330 char buffer[512];
331 start = outFileVersion + strlen("AssemblyFileVersion") + 2;
332 stop = start + strlen(start) - 1;
333 get_buildno_ptr(&start, &stop);
334 strcpy(buffer, stop);
335 sprintf(start, "%i%s", rev, buffer);
336 fputs(line, fdest);
337 }
338 else
339 {
340 fputs(line, fdest);
341 continue;
342 }
343 }
344 }
345
346 fclose(fsource);
347 fclose(fdest);
348
349 remove(filename);
350 rename(fileout, filename);
351
352 (void)out; /* to avoid a compiler warning */
353}

355int main(int argc, char *argv[])
356{
357 char *outname = NULL;
358 FILE *fp;
359 int index;
360 int process_self = 0;
361 int verbose = 0;
362 int ignore_eol = 0;
363 int max_build, accum_build;
364 int max_year, max_month, max_day;
365 int ismodified, filemodified, modificationcount;
366 char prefix[MAX_SYMBOLLENGTH], suffix[MAX_SYMBOLLENGTH];
367 char *const_basename = NULL;
368 char modified_suffix[2];
369 int output_language = LANG_C_CPP; /* flag for C/C++, Java or C# output */
370 char *namespace_name = NULL; /* java package or C# namespace to put revision info in */
371 char *propname = NULL;
372 char *startComment = NULL;
373 char *endComment = NULL;
374 char *continueComment = NULL;
375
376 if (argc <= 1)
377 about();
378
379 /* collect the options */
380 prefix[0] = '\0';
381 suffix[0] = '\0';
382
383 for (index = 1; index < argc; index++) {
384 /* check for options */
385 if (argv[index][0] == '-'
386#if defined __WIN32__ || defined _Win32 || defined _WIN32
387 || argv[index][0] == '/'
388#endif
389 )
390 {
391 switch (argv[index][1]) {
392 case 'f': {
393 size_t len;
394 char *ptr = strchr(&argv[index][2], '#');
395 len = (ptr != NULL) ? (int)(ptr - &argv[index][2]) : (int)strlen(&argv[index][2]);
396 if (len >= MAX_SYMBOLLENGTH)
397 len = MAX_SYMBOLLENGTH - 1;
398 strncpy(prefix, &argv[index][2], len);
399 prefix[len] = '\0';
400 ptr = (ptr != NULL) ? ptr + 1 : strchr(argv[index], '\0');
401 len = strlen(ptr);
402 if (len >= MAX_SYMBOLLENGTH)
403 len = MAX_SYMBOLLENGTH - 1;
404 strncpy(suffix, ptr, len);
405 suffix[len] = '\0';
406 break;
407 } /* case */
408 case 'i':
409 process_self = 1;
410 break;
411 case 'j':
412 output_language = LANG_JAVA;
413 namespace_name = &argv[index][2];
414 break;
415 case 'c':
416 output_language = LANG_CS;
417 namespace_name = &argv[index][2];
418 break;
419 case 'P':
420 output_language = LANG_PROP;
421 break;
422 case 'O':
423 output_language = LANG_ORACLE;
424 namespace_name = &argv[index][2];
425 break;
426 case 'p':
427 propname = &argv[index][2];
428 break;
429 case 'm':
430 const_basename = &argv[index][2];
431 break;
432 case 'o':
433 outname = &argv[index][2];
434 break;
435 case 'v':
436 verbose = 1;
437 break;
438 case 'n':
439 ignore_eol = 1;
440 break;
441 default:
442 fprintf(stderr, "Invalid option '%s'\n", argv[index]);
443 about();
444 } /* switch */
445 } /* if */
446 } /* for */
447
448 if (outname == NULL) {
449 switch (output_language) {
450 case LANG_C_CPP:
451 outname = "svnrev.h";
452 break;
453 case LANG_JAVA:
454 outname = "SvnRevision.java";
455 break;
456 case LANG_CS:
457 outname = "SvnRevision.cs";
458 break;
459 case LANG_PROP:
460 outname = "svnrev.property";
461 break;
462 case LANG_ORACLE:
463 outname = "svnrev.sql";
464 break;
465 } /* switch */
466 } /* if */
467
468 /* scan through the existing svnrev file to find the "build" count */
469 if (const_basename == NULL || *const_basename == '\0')
470 const_basename = "SVN_REV";
471 assert(outname != NULL);
472 modificationcount = scanmodifications(outname, const_basename);
473 if (!process_self && *outname != '\0')
474 remove(outname);
475
476 /* phase 1: scan through all files and get the highest build number */
477 max_build = 0;
478 accum_build = 0; /* for RCS / CVS */
479 max_year = max_month = max_day = 0;
480 ismodified = 0;
481 for (index = 1; index < argc; index++) {
482 /* skip the options (already handled) */
483 if (argv[index][0] == '-'
484#if defined __WIN32__ || defined _Win32 || defined _WIN32
485 || argv[index][0] == '/'
486#endif
487 )
488 continue;
489
490 filemodified = 0;
491 if (strcasecmp(argv[index], outname)!=0)
492 processfile(argv[index], 0, ignore_eol, &max_build, &accum_build, &max_year, &max_month, &max_day, &filemodified);
493 if (filemodified && verbose)
494 fprintf(stderr, "\tNotice: modified file '%s'\n", argv[index]);
495 ismodified = ismodified || filemodified;
496 } /* for */
497
498 /* also run over the existing header file, if any */
499 if (process_self && *outname != '\0')
500 processfile(outname, 1, ignore_eol, &max_build, &accum_build, &max_year, &max_month, &max_day, NULL/*&ismodified*/);
501
502 if (accum_build > max_build)
503 max_build = accum_build;
504 modified_suffix[0] = (char)(ismodified ? 'M' : '\0');
505 modified_suffix[1] = '\0';
506
507 /* phase 2: write a file with this highest build number */
508 if (*outname == '\0') {
509 fp = stdout;
510 } else if ((fp = fopen(outname, "w")) == NULL) {
511 fprintf(stderr, "Failed to create output file '%s'\n", outname);
512 return 2;
513 } /* if */
514 if (*outname != '\0') {
515 switch (output_language) {
516 case LANG_C_CPP:
517 case LANG_JAVA:
518 case LANG_CS:
519 startComment = "/* ";
520 endComment = " */";
521 continueComment = " * ";
522 break;
523 case LANG_ORACLE:
524 startComment = "-- ";
525 endComment = "-- ";
526 continueComment = "-- ";
527 break;
528 case LANG_PROP:
529 startComment = "# ";
530 endComment = "# ";
531 continueComment = "# ";
532 break;
533 } /* switch */
534 /* don't print the comments to stdout */
535 fprintf(fp, "%sThis file was generated by the \"svnrev\" utility\n"
536 "%s(http://www.compuphase.com/svnrev.htm).\n"
537 "%sYou should not modify it manually, as it may be re-generated.\n"
538 "%s\n"
539 "%s$Revision: %d%s$\n"
540 "%s$Date: %04d-%02d-%02d$\n"
541 "%s\n\n", startComment, continueComment, continueComment, continueComment,
542 continueComment, max_build,
543 modified_suffix, continueComment, max_year, max_month, max_day,endComment);
544 } /* if */
545
546 switch (output_language) {
547 case LANG_C_CPP:
548 fprintf(fp, "#ifndef %s_H\n", const_basename);
549 fprintf(fp, "#define %s_H\n\n", const_basename);
550 fprintf(fp, "#define %s\t\t%d\n", const_basename, max_build);
551 fprintf(fp, "#define %sSTR\t\"%s%d%s%s\"\n", const_basename, prefix, max_build, modified_suffix, suffix);
552 fprintf(fp, "#define %sDATE\t\"%04d-%02d-%02d\"\n", const_basename, max_year, max_month, max_day);
553 fprintf(fp, "#define %sSTAMP\t%04d%02d%02dL\n", const_basename, max_year, max_month, max_day);
554 fprintf(fp, "#define %sMODIFIED\t%d\n", const_basename, ismodified ? modificationcount + 1 : 0);
555 fprintf(fp, "\n#endif /* %s_H */\n", const_basename);
556 break;
557 case LANG_JAVA:
558 if (namespace_name == NULL || *namespace_name == '\0')
559 namespace_name = "com.compuphase";
560 fprintf(fp, "package %s;\n\n", namespace_name);
561 fprintf(fp, "public interface SvnRevision\n");
562 fprintf(fp, "{\n");
563 fprintf(fp, " public final static int %s = %d;\n", const_basename, max_build);
564 fprintf(fp, " public final static String %sSTR = \"%s%d%s%s\";\n", const_basename, prefix, max_build, modified_suffix, suffix);
565 fprintf(fp, " public final static String %sDATE = \"%04d-%02d-%02d\";\n", const_basename, max_year, max_month, max_day);
566 fprintf(fp, " public final static long %sSTAMP = %04d%02d%02dL;\n", const_basename, max_year, max_month, max_day);
567 fprintf(fp, " public final static int %sMODIFIED = %d;\n", const_basename, ismodified ? modificationcount + 1 : 0);
568 fprintf(fp, "}\n\n");
569 break;
570 case LANG_CS: {
571 char *cls;
572 if (namespace_name == NULL || (cls = strrchr(namespace_name, '.')) == NULL) {
573 namespace_name = "compuphase";
574 cls = "svnrev";
575 } else {
576 *cls++ = '\0';
577 } /* if */
578 fprintf(fp, "using System;\n\n");
579 fprintf(fp, "namespace %s\n{\n", namespace_name);
580 fprintf(fp, "\tpublic static partial class %s\n", cls);
581 fprintf(fp, "\t{\n");
582 fprintf(fp, "\t\tpublic const Int32 %s = %d;\n", const_basename, max_build);
583 fprintf(fp, "\t\tpublic const String %sSTR = \"%s%d%s%s\";\n", const_basename, prefix, max_build, modified_suffix, suffix);
584 fprintf(fp, "\t\tpublic const String %sDATE = \"%04d-%02d-%02d\";\n", const_basename, max_year, max_month, max_day);
585 fprintf(fp, "\t\tpublic const Int64 %sSTAMP = %04d%02d%02d;\n", const_basename, max_year, max_month, max_day);
586 fprintf(fp, "\t\tpublic const Int32 %sMODIFIED = %d;\n", const_basename, ismodified ? modificationcount + 1 : 0);
587 fprintf(fp, "\t}\n}\n");
588 update_prop_file(propname, max_build);
589 break;
590 } /* case */
591 case LANG_PROP:
592 fprintf(fp, "%s = %d\n", const_basename, max_build);
593 fprintf(fp, "%sSTR = %s%d%s%s\n", const_basename, prefix, max_build, modified_suffix, suffix);
594 fprintf(fp, "%sDATE = %04d-%02d-%02d\n", const_basename, max_year, max_month, max_day);
595 fprintf(fp, "%sSTAMP = %04d%02d%02dL\n", const_basename, max_year, max_month, max_day);
596 fprintf(fp, "%sMODIFIED = %d\n", const_basename, ismodified ? modificationcount + 1 : 0);
597 break;
598 case LANG_ORACLE:
599 if (namespace_name == NULL || *namespace_name == '\0')
600 namespace_name = "compuphase";
