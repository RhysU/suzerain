75int
76suzerain::reacting::driver_init::run(int argc, char **argv)
77{
78 using boost::math::constants::pi;
79 using boost::math::tgamma;
80 using std::pow;
81 using std::sin;
82
83 // Establish default grid and domain extents
84 grid.reset(new support::definition_grid( 4 * pi<real_t>() // Lx
85 , 1 // Nx
86 , real_t(3) / 2 // DAFx
87 , 2 // Ly
88 , 48 // Ny
89 , 8 // k
90 , 3 // htdelta
91 , 4 * pi<real_t>() / 3 // Lz
92 , 1 // Nz
93 , real_t(3) / 2 // DAFz
94 ));
95
96 // Establish default scenario parameters
97 cmods->Le = 0.9;
98 cmods->alpha = 0.0;
99 chdef.reset(new definition_channel(
100 1 // bulk_rho
101 , 1 // bulk_rho_u
102 , std::numeric_limits<real_t>::quiet_NaN() // bulk_rho_E
103 ));
104
105 // Establish default isothermal boundary conditions
106 // TODO Check the correctness of this default behavior
107 isothermal.reset(new support::definition_isothermal(/* wall_T */ 273));
108
109 // Establish default time step aggressiveness
110 timedef = std::make_shared<support::definition_time>(/* per Venugopal */ 0.72);
111
112 // Establish default MMS parameters and plug into program options
113 msoln = std::make_shared<manufactured_solution>(
114 manufactured_solution::default_caption
115 + " (active only when --mms supplied)");
116 options.add_definition(msoln->isothermal_channel());
117
118 // Establish default filter parameters
119 fsdef->filter_phi = 0.0;
120
121 // FIXME: Establish default parameters for slow growth
122
123 // Establish binary-specific options
124 real_t mms = -1;
125 real_t npower = 1;
126 options.add_options()
127 ("clobber", "Overwrite an existing restart file?")
128 ("npower",
129 boost::program_options::value(&npower)->default_value(npower),
130 "Power n in (0, 1] used to control the flatness of the"
131 " \"parabolic\" streamwise velocity profile (y*(L-y))^n.")
132 ("mms",
133 boost::program_options::value(&mms)
134 ->notifier([](const real_t& v){
135 validation::ensure_nonnegative(v, "mms"); }),
136 "If given, prepare a manufactured solution at the specified time.")
137 ;
138
139 // TODO: Add maybe tanh option through options.add_option
140 // as it's done in the perfect initialization.
141 // I think we'd like to have the parabolic profile by default
142 // for a channel grid (htdelta>=0), and the tanh profile
143 // by default for a plate case.
144
145 // Initialize application and then process binary-specific options
146 // (henceforth suzerain::support::logging macros becomes usable)
147 std::vector<std::string> positional = initialize(argc, argv);
148
149 if (positional.size() != 1) {
150 FATAL0("Exactly one restart file name must be specified");
151 return EXIT_FAILURE;
152 }
153 const std::string restart_file = positional[0];
154 const bool clobber = options.variables().count("clobber");
155 if (npower <= 0 || npower > 1) {
156 FATAL0("npower in (0,1] required");
157 return EXIT_FAILURE;
158 }
159 if (grid->k < 4) {
160 FATAL0("k >= 4 required for two non-trivial wall-normal derivatives");
161 return EXIT_FAILURE;
162 }
163 if (grid->one_sided() && options.variables().count("mms")) {
164 FATAL0("Manufactured solution not enabled for one-sided grid");
165 return EXIT_FAILURE;
166 }
167
168 DEBUG0(who, "Initializing antioch_constitutive");
169 cmods->init_antioch();
170
171 // If msoln was provided, match its contents to other members. Do
172 // here b/c cmods doesn't make sense until after init_antioch.
173 // Must protect against calling match when not use mms b/c match
174 // will throw exception when multispecies case detected.
175 if (mms >=0 && msoln) {
176 if (cmods) msoln->match(*cmods);
177 if (grid) msoln->match(*grid);
178 }
179
180 DEBUG0(who, "Add species to fields");
181 add_species_fields(cmods->species_names, this->fields);
182
183 DEBUG0(who, "Establishing runtime parallel infrastructure and resources");
184 establish_ieee_mode();
185 load_grid_and_operators(NULL);
186 establish_decomposition();
187 establish_state_storage(fields.size(), fields.size());
188
189 DEBUG0(who, "Initializing state_linear to contain the data in wave space");
190 if (mms >= 0) {
191
192 INFO0(who, "Manufactured solution will be initialized at t = " << mms);
193 INFO0(who, "Disabling bulk_rho, bulk_rho_u, and bulk_rho_E constraints"
194 " due to manufactured solution use");
195 chdef->bulk_rho = std::numeric_limits<real_t>::quiet_NaN();
196 chdef->bulk_rho_u = std::numeric_limits<real_t>::quiet_NaN();
197 chdef->bulk_rho_E = std::numeric_limits<real_t>::quiet_NaN();
198
199 accumulate_manufactured_solution(
200 1, *msoln, 0, *state_nonlinear, *grid, *dgrid, *cop, *b, mms);
201 state_linear->assign_from(*state_nonlinear);
202
203 } else if (!dgrid->has_zero_zero_modes()) {
204
205 DEBUG("Saving no non-zero information on non-zero-zero rank");
206 msoln.reset();
207 fill(*state_linear, 0);
208
209 } else {
210
211 real_t rho;
212 real_t v ;
213 real_t w ;
214 real_t T ;
215 std::vector<real_t> rho_s;
216 ArrayXr u(grid->N.y());
217 ArrayXr E(grid->N.y());
218 MatrixXXr cs(grid->N.y(),cmods->Ns());
219
220 if (!grid->one_sided()) { // Channel profile
221 // TODO: Account for bulk_rho_E in initialization
222 INFO("Parabolic profile will be initialized with npower = " << npower);
223 msoln.reset();
224 fill(*state_linear, 0);
225
226 INFO("Initialization uses constant rho, v, w, and T");
227 rho = chdef->bulk_rho;
228 v = 0;
229 w = 0;
230 const real_t T = isothermal->lower_T;
231 SUZERAIN_ENSURE(isothermal->lower_T == isothermal->upper_T);
232
233 if (cmods->Ns()>1) {
234 // TODO: Assert that number of lower_cs is correct
235
236 INFO("Initialization uses constant rho_s equal to bulk_rho*lower_cs");
237 rho_s.resize(cmods->Ns()-1);
238
239 // ensure for diluter
240 SUZERAIN_ENSURE(isothermal->lower_cs[0] == isothermal->upper_cs[0]);
241 for (size_t s=0; s<cmods->Ns()-1; ++s) {
242 // yes, +1 b/c first is the diluter
243 rho_s[s] = chdef->bulk_rho*isothermal->lower_cs[s+1];
244 SUZERAIN_ENSURE(isothermal->lower_cs[s+1] == isothermal->upper_cs[s+1]);
245 }
246 }
247
248 INFO("Finding normalization so u = (y*(L-y))^npower integrates to 1");
249 real_t normalization;
250 if (npower == 1) {
251 // Mathematica: (Integrate[(x (L-x)),{x,0,L}]/L)^(-1)
252 normalization = 6 / pow(grid->L.y(), 2);
253 } else {
254 // Mathematica: (Integrate[(x (L - x))^n, {x, 0, L}]/L)^(-1)
255 // - (Gamma[-n] Gamma[3/2+n])
256 // / (2^(-1-2 n) L^(1+2 n) \[Pi]^(3/2) Csc[n \[Pi]])
257 const real_t num1 = sin(npower * pi<real_t>());
258 const real_t num2 = tgamma(-npower);
259 const real_t num3 = tgamma(real_t(3)/2 + npower);
260 const real_t denom1 = pow( 2, -1-2*npower);
261 const real_t denom2 = pow( grid->L.y(), 1+2*npower);
262 const real_t denom3 = pow(pi<real_t>(), real_t(3)/2);
263 normalization = - (num1 * num2 * num3 * grid->L.y())
264 / (denom1 * denom2 * denom3);
265 }
266
267 INFO("Preparing the wall-normal streamwise velocity profile");
268 // ArrayXr u(grid->N.y());
269 for (int j = 0; j < u.size(); ++j) {
270 const real_t y_j = b->collocation_point(j);
271 u[j] = ( chdef->bulk_rho_u / chdef->bulk_rho )
272 * normalization
273 * pow(y_j * (grid->L.y() - y_j), npower);
274 }
275
276 INFO("Preparing specific internal energy using the equation of state");
277 E = cmods->e_from_T(T, isothermal->lower_cs)
278 + 0.5*(u*u + v*v + w*w);
279
280 } else { // the grid is one sided, construct a tanh profile
281
282 INFO("Hyperbolic tangent profile will be initialized");
283 msoln.reset();
284 fill(*state_linear, 0);
285
286 INFO("Initialization uses constant rho, v, w, and cs");
287 // NOTE: Maybe add profile for species concentrations,
288 // to avoid a jump in the cs concentration at the wall
289 // in the first time step
290 rho = isothermal->upper_rho;
291 v = isothermal->lower_v;
292 w = 0;
293 // using isothermal->upper_cs as species concentrations values
294
295 if (cmods->Ns()>1) {
296 // TODO: Assert that number of lower_cs is correct
297
298 INFO("Initialization uses constant rho_s equal to bulk_rho*lower_cs");
299 rho_s.resize(cmods->Ns()-1);
300
301 // ensure for diluter
302 for (size_t s=0; s<cmods->Ns()-1; ++s) {
303 // yes, +1 b/c first is the diluter
304 rho_s[s] = isothermal->upper_rho*isothermal->upper_cs[s+1];
305 }
306 }
307
308 INFO("Preparing the wall-normal streamwise velocity profile");
309 std::vector<real_t> csj;
310 csj.resize(cmods->Ns());
311 // TODO: Storing species concentrations at each j index in a
312 // separate container, csj, to be passed to the e_from_T method.
313 // Can this be avoided?
314 for (int j = 0; j < u.size(); ++j) {
315 const real_t y_j = b->collocation_point(j);
316 u[j] = (isothermal->lower_u)
317 + (isothermal->upper_u - isothermal->lower_u)
318 * tanh(10/grid->L.y() * y_j);
319 T = (isothermal->lower_T)
320 + (isothermal->upper_T - isothermal->lower_T)
321 * tanh(10/grid->L.y() * y_j);
322 for (size_t s=0; s<cmods->Ns(); ++s) {
323 csj[s] = isothermal->lower_cs[s]
324 + (isothermal->upper_cs[s] - isothermal->lower_cs[s])
325 * tanh(10/grid->L.y() * y_j);
326 cs(j,s) = csj[s];
327 }
328 E[j] = cmods->e_from_T(T, csj)
329 + 0.5*(u[j]*u[j] + v*v + w*w);
330 }
331 } // end "if (!grid->one_sided())"
332
333 INFO("Converting the u and E profiles to B-spline coefficients");
334 // (By partition of unity property rho, v, and w are so already)
335 suzerain::bsplineop_lu masslu(*cop);
336 masslu.factor_mass(*cop);
337 masslu.solve(1, u.data(), 1, u.size());
338 masslu.solve(1, E.data(), 1, E.size());
339
340 INFO("Copying the coefficients directly into the zero-zero modes");
341 Map<VectorXc>((*state_linear)[ndx::e ].origin(), grid->N.y())
342 = (rho * E).cast<complex_t>();
343 Map<VectorXc>((*state_linear)[ndx::mx ].origin(), grid->N.y())
344 = (rho * u).cast<complex_t>();
345 Map<VectorXc>((*state_linear)[ndx::my ].origin(), grid->N.y())
346 .setConstant(rho * v);
347 Map<VectorXc>((*state_linear)[ndx::mz ].origin(), grid->N.y())
348 .setConstant(rho * w);
349 Map<VectorXc>((*state_linear)[ndx::rho].origin(), grid->N.y())
350 .setConstant(rho );
351
352 if (!grid->one_sided()) { // Uniform rho_s
353 if (cmods->Ns()>1) {
354 for (size_t s=1; s<cmods->Ns(); ++s) {
355 Map<VectorXc>((*state_linear)[ndx::rho+s].origin(), grid->N.y())
356 .setConstant(rho_s[s-1]);
357 }
358 }
359 } else { // tanh profile for rho_s
360 if (cmods->Ns()>1) {
361 INFO("Converting species profiles to B-spline coefficients"
362 " and copying the coefficients directly into the"
363 " zero-zero modes");
364 for (size_t s=1; s<cmods->Ns(); ++s) {
365 ArrayXr css(grid->N.y());
366 css = cs.col(s);
367 masslu.solve(1, css.data(), 1, css.size());
368 Map<VectorXc>((*state_linear)[ndx::rho+s].origin(), grid->N.y())
369 = (rho * css).cast<complex_t>();
370 }
371 }
372 }
373
374
375 // Compute and print the wall viscosity (known b/c only
376 // depends on T and mass fractions)
377 const real_t wall_visc =
378 this->cmods->wilke_evaluator->mu(isothermal->lower_T,
379 isothermal->lower_cs);
380 INFO("The wall (dynamic) viscosity is " << wall_visc);
381
382 // Note: 2 in Re below takes channel height to half-height
383 INFO("So that Re_bulk = " <<
384 chdef->bulk_rho_u * grid->L.y() / (2.0*wall_visc) );
385
386 // Compute and print the speed of sound (frozen)
387 const real_t R_mix = this->cmods->mixture->R(isothermal->lower_cs);
388 const real_t Cv = this->cmods->sm_thermo->cv(isothermal->lower_T,
389 isothermal->lower_T,
390 isothermal->lower_cs);
391
392
393 const real_t a = std::sqrt( (1.0 + R_mix/Cv)*R_mix*isothermal->lower_T);
394
395 INFO("The speed of sound at the wall is " << a);
396
397 }
398
399 INFO0(who, "Saving the newly initialized state to disk");
400 if (msoln) {
401 save_restart(mms, restart_file, clobber);
402 } else {
403 save_restart(0, restart_file, clobber);
404 }
405
406 return EXIT_SUCCESS;
407}
