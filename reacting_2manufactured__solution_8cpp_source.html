118manufactured_solution&
119manufactured_solution::match(antioch_constitutive& cmods)
120{
121 // Check incoming cmods for consistency with available
122 // manufactured solution
123 if (cmods.Ns()!=1)
124 throw std::invalid_argument("MMS only available for single species!");
125
126 if (cmods.species_names[0]!="CPAir" && cmods.species_names[0]!="CPN2")
127 throw std::invalid_argument(
128 "MMS only available for calorically perfect gas!");
129
130 // Tell user we're going to mess with their incoming cmods
131 INFO0("Manipulating constitutive model parameters to "
132 "match input manufactured solution.");
133
134 // Since only allow single species, mass fractions are trivial
135 std::vector<real_t> trivial_mass_fractions(1, 1.0);
136
137 // Thermodynamics
138 // Note: T, Tv, and mass_fractions inputs
139 // Temperatures are irrelevant (calorically perfect) but mass
140 // fraction input is necessary
141 real_t Cv = cmods.sm_thermo->cv(1.0,1.0,trivial_mass_fractions);
142 real_t Cp = cmods.sm_thermo->cp(1.0,1.0,trivial_mass_fractions);
143
144 real_t Rmix = cmods.mixture->R(trivial_mass_fractions);
145
146 this->gamma = Cp/Cv;
147 this->R = Rmix;
148
149 // Transport
150 // Note: Parameters of transport models are stored as part of MMS
151 // class. Here we update the constitutive laws class to be
152 // consistent with what is stored by MMS.
153
154 // Viscosity: Make Blottner match power law
155 const real_t mu0 = this->mu_r;
156 const real_t beta = this->beta;
157 const real_t Tref = this->T_r;
158
159 std::vector<real_t> blottner_coeffs(3);
160 // Blottner says: mu = 0.1*exp( a*log(T)^2 + b*log(T) + c). Thus,
161
162 // for a = 0 reduces blottner to power law
163 blottner_coeffs[0] = 0.0;
164
165 // with b = power law exponent
166 blottner_coeffs[1] = beta;
167
168 // and 0.1*exp(c) = mu0/(Tref^beta)
169 blottner_coeffs[2] = std::log(mu0/(0.1*std::pow(Tref, beta)));
170
171 // Set blottner parameters
172 cmods.mixture_mu->reset_coeffs(0, blottner_coeffs);
173
174 // Thermal conductivity: Weird factor of 19/4 makes
175 // man. soln. match Eucken for CPAir and CPN2
176 this->kappa_r = (real_t(19)/real_t(4))*this->mu_r*this->R;
177
178 // Bulk viscosity
179 this->lambda_r = (cmods.alpha-(real_t(2)/real_t(3)))*this->mu_r;
180
181 return *this;
182}

193manufactured_solution&
194manufactured_solution::isothermal_channel()
195{
196 //nsctpl_rholut::isothermal_channel(*this);
197 nsctpl::isothermal_channel(*this);
198
199 // Rescale time frequencies and density coefficients
200 const real_t rho_scale = real_t(355)/real_t(10000000);
201 const real_t itime_scale = 200;
202
203 // density coefficients
204 this->rho.a_0 = rho_scale * real_t(1);
205 this->rho.a_xy = rho_scale * real_t(1) / real_t(11);
206 this->rho.a_y = rho_scale * real_t(1) / real_t(7);
207 this->rho.a_yz = rho_scale * real_t(1) / real_t(31);
208
209 // time frequency coefficients
210 this->rho.f_xy = itime_scale * real_t(3);
211 this->rho.f_y = itime_scale * real_t(1);
212 this->u.f_xy = itime_scale * real_t(3);
213 this->u.f_y = itime_scale * real_t(1);
214 this->u.f_yz = itime_scale * real_t(2);
215 this->v.f_y = itime_scale * real_t(1);
216 this->v.f_yz = itime_scale * real_t(2);
217 this->w.f_xy = itime_scale * real_t(3);
218 this->w.f_y = itime_scale * real_t(1);
219 this->w.f_yz = itime_scale * real_t(2);
220 this->T.f_xy = itime_scale * real_t(3);
221 this->T.f_y = itime_scale * real_t(1);
222 this->T.f_yz = itime_scale * real_t(2);
223
224 return *this;
225}

271void save(const esio_handle h,
272 const std::shared_ptr<manufactured_solution>& msoln,
273 const antioch_constitutive& cmods,
274 const specification_grid& grid,
275 const char *location)
276{
277 SUZERAIN_UNUSED(cmods); // TODO?
278
279 // Only proceed if a manufactured solution is being provided
280 if (!msoln) return;
281
282 const int one = 1;
283 int procid;
284 esio_handle_comm_rank(h, &procid);
285 esio_line_establish(h, 1, 0, (procid == 0 ? 1 : 0));
286 esio_line_write(h, location, &one, 0,
287 "Is a suzerain::reacting::manufactured_solution in use?");
288
289 DEBUG0("Storing manufactured_solution parameters");
290
291 // Non-scenario solution parameters are stored as attributes under location
292 using boost::bind;
293 msoln->rho.foreach_parameter(bind(attribute_storer, h, location, _1, _2));
294 msoln->u .foreach_parameter(bind(attribute_storer, h, location, _1, _2));
295 msoln->v .foreach_parameter(bind(attribute_storer, h, location, _1, _2));
296 msoln->w .foreach_parameter(bind(attribute_storer, h, location, _1, _2));
297 msoln->T .foreach_parameter(bind(attribute_storer, h, location, _1, _2));
298
299 // Scenario params
300 esio_line_write(h, name_T_r, &msoln->T_r, 0, desc_T_r );
301 esio_line_write(h, name_mu_r, &msoln->mu_r, 0, desc_mu_r);
302 esio_line_write(h, name_beta, &msoln->beta, 0, desc_beta);
303
304#pragma warning(push,disable:1572)
305 // Check parameters stored with the scenario not the manufactured solution
306 // because scenario parameters should be loaded from definition_scenario
307
308 // FIXME: Add this check back!!!
309
310 // Check parameters stored with the grid not the manufactured solution
311 // because grid parameters should be loaded from specification_grid
312 if (msoln->Lx != grid.L.x())
313 WARN0("Manufactured solution Lx mismatches with grid");
314 if (msoln->Ly != grid.L.y())
315 WARN0("Manufactured solution Ly mismatches with grid");
316 if (msoln->Lz != grid.L.z())
317 WARN0("Manufactured solution Lz mismatches with grid");
318#pragma warning(pop)
319}

337void load(const esio_handle h,
338 std::shared_ptr<manufactured_solution>& msoln,
339 const antioch_constitutive& cmods,
340 const specification_grid& grid,
341 const char *location)
342{
343 SUZERAIN_UNUSED(cmods); // TODO?
344
345 // Only proceed if a manufactured solution is active in the restart
346 int in_use = 0;
347 esio_line_establish(h, 1, 0, 1); // All ranks load any data
348 if (ESIO_NOTFOUND != esio_line_size(h, location, NULL)) {
349 esio_line_read(h, location, &in_use, 0);
350 }
351 if (!in_use) {
352 msoln.reset();
353 return;
354 }
355
356 DEBUG0("Loading manufactured_solution parameters");
357
358 // Allocate storage and defensively NaN every parameter not explicitly
359 // loaded below. Protects us against accidentally missing new solution
360 // parameters.
361 msoln.reset(new manufactured_solution());
362 msoln->foreach_parameter(&NaNer);
363
364 // Non-scenario solution parameters are stored as attributes under location
365 using boost::bind;
366 msoln->rho.foreach_parameter(bind(attribute_loader, h, location, _1, _2));
367 msoln->u .foreach_parameter(bind(attribute_loader, h, location, _1, _2));
368 msoln->v .foreach_parameter(bind(attribute_loader, h, location, _1, _2));
369 msoln->w .foreach_parameter(bind(attribute_loader, h, location, _1, _2));
370 msoln->T .foreach_parameter(bind(attribute_loader, h, location, _1, _2));
371
372 // Scenario params
373 esio_line_read(h, name_T_r, &msoln->T_r, 0);
374 esio_line_read(h, name_mu_r, &msoln->mu_r, 0);
375 esio_line_read(h, name_beta, &msoln->beta, 0);
376
377 // Parameters set to match supplied arguments
378 // Match must happen *after* call to antioch_constitutive::init_antioch
379 //msoln->match(cmods);
380 msoln->match(grid);
381}

383void accumulate_manufactured_solution(
384 const real_t alpha,
385 const manufactured_solution &msoln,
386 const real_t beta,
387 contiguous_state<4,complex_t> &swave,
388 const specification_grid &grid,
389 const pencil_grid &dgrid,
390 const bsplineop &cop,
391 bspline &b,
392 const real_t simulation_time)
393{
394 // We are only prepared to handle a fixed number of fields in this routine
395 enum { state_count = 5 };
396
397 // Ensure state storage meets this routine's assumptions
398 SUZERAIN_ENSURE( swave.shape()[0] == state_count);
399 SUZERAIN_ENSURE(boost::numeric_cast<int>(swave.shape()[1]) == dgrid.local_wave_extent.y());
400 SUZERAIN_ENSURE(boost::numeric_cast<int>(swave.shape()[2]) == dgrid.local_wave_extent.x());
401 SUZERAIN_ENSURE(boost::numeric_cast<int>(swave.shape()[3]) == dgrid.local_wave_extent.z());
402
403 // Initialize operator_base to access decomposition-ready utilities
404 operator_base obase(grid, dgrid, cop, b);
405
406 // Allocate one field of temporary storage for scratch purposes
407 std::unique_ptr<contiguous_state<4,complex_t> > _scratch_ptr( // RAII
408 support::allocate_padded_state<contiguous_state<4,complex_t> >(1,dgrid));
409 contiguous_state<4,complex_t> &scratch = *_scratch_ptr; // Shorthand
410 multi_array::fill(scratch, 0); // Defensive
411
412 // Prepare physical-space view of the wave-space scratch storage
413 physical_view<1> phys(dgrid, scratch);
414
415 // Prepare factored mass matrix for repeated use
416 bsplineop_luz massluz(cop);
417 const complex_t scale_factor = 1 / dgrid.chi();
418 massluz.opform(1, &scale_factor, cop);
419 massluz.factor();
420
421 // For each scalar field...
422 for (size_t f = 0; f < state_count; ++f) {
423
424 // ...compute the manufactured solution in physical space...
425 for (int offset = 0, j = dgrid.local_physical_start.y();
426 j < dgrid.local_physical_end.y();
427 ++j) {
428
429 const real_t y = obase.y(j);
430
431 for (int k = dgrid.local_physical_start.z();
432 k < dgrid.local_physical_end.z();
433 ++k) {
434
435 const real_t z = obase.z(k);
436
437 for (int i = dgrid.local_physical_start.x();
438 i < dgrid.local_physical_end.x();
439 ++i, /* NB */ ++offset) {
440
441 const real_t x = obase.x(i);
442
443 // Ugly, slow switch but performance irrelevant here
444 switch (f) {
445 case ndx::e:
446 phys(0, offset) = msoln.rhoe(x, y, z, simulation_time);
447 break;
448 case ndx::mx:
449 phys(0, offset) = msoln.rhou(x, y, z, simulation_time);
450 break;
451 case ndx::my:
452 phys(0, offset) = msoln.rhov(x, y, z, simulation_time);
453 break;
454 case ndx::mz:
455 phys(0, offset) = msoln.rhow(x, y, z, simulation_time);
456 break;
457 case ndx::rho:
458 phys(0, offset) = msoln.rho (x, y, z, simulation_time);
459 break;
460 default:
461 SUZERAIN_ERROR_REPORT("unknown field",
462 SUZERAIN_ESANITY);
463 }
464
465 } // end X
466
467 } // end Z
468
469 } // end Y
470
471 // ...convert collocation values to non-dealiased coefficients...
472 dgrid.transform_physical_to_wave(&phys.coeffRef(0, 0)); // X, Z
473 obase.zero_dealiasing_modes(scratch, 0);
474 obase.bop_solve(massluz, scratch, 0); // Y
475
476 // ...and accumulate into the corresponding scalar field of swave
477 SUZERAIN_ENSURE(std::equal(scratch.shape() + 1, scratch.shape() + 4,
478 swave.shape() + 1));
479 if (SUZERAIN_UNLIKELY(0U == scratch.shape()[1])) {
480 continue; // Sidestep assertions on trivial data
481 }
482 typedef contiguous_state<4,complex_t>::index index;
483 const index ku = boost::numeric_cast<index>(
484 scratch.index_bases()[2] + scratch.shape()[2]);
485 const index lu = boost::numeric_cast<index>(
486 scratch.index_bases()[3] + scratch.shape()[3]);
487 for (index lx = scratch.index_bases()[3], ly = swave.index_bases()[3];
488 lx < lu;
489 ++lx, ++ly) {
490
491 for (index kx = scratch.index_bases()[2],
492 ky = swave.index_bases()[2];
493 kx < ku;
494 ++kx, ++ky) {
495
496 blas::axpby(scratch.shape()[1], alpha,
497 &scratch[0][scratch.index_bases()[1]][kx][lx],
498 scratch.strides()[1], beta,
499 &swave[f][swave.index_bases()[1]][ky][ly],
500 swave.strides()[1]);
501 }
502 }
503
504 }
505}
