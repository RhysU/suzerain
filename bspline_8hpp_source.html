309 template< typename Complex >
310 typename boost::enable_if<
311 suzerain::complex::traits::is_complex_t<Complex>, int
312 >::type integration_coefficients(
313 const std::size_t nderiv,
314 Complex *coeffs,
315 const real_t lo = -std::numeric_limits<real_t>::max(),
316 const real_t hi = std::numeric_limits<real_t>::max())
317 {
318 // Zero real and imaginary components
319 for (std::size_t i = 0; i < gsl_bspline_ncontrol(bw); ++i) {
320 suzerain::complex::assign_components(coeffs[i], 0, 0);
321 }
322
323 // Populate the real components
324 return suzerain_bspline_integration_coefficients(
325 nderiv, reinterpret_cast<real_t *>(coeffs),
326 sizeof(complex_t)/sizeof(real_t), lo, hi, db_, bw);
327 }
