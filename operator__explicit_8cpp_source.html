84std::vector<real_t> explicit_operator_nonlinear::apply_operator(
85 const real_t time,
86 contiguous_state<4,complex_t> &swave,
87 const lowstorage::method_interface<complex_t> &method,
88 const std::size_t substep_index) const
89{
90
91#define ARGUMENTS *this, common, fsdef, sgdef, msoln, cmods, *massluz(), \
92 time, swave, method
93
94 // Dispatch to an optimized implementation depending on case:
95 switch (common.filter_treatment) {
96
97 case filter::none:
98
99 switch (common.linearization) {
100 case linearize::none:
101 return (substep_index == 0)
102 ? apply_navier_stokes_spatial_operator<true,
103 linearize::none, filter::none>(ARGUMENTS)
104 : apply_navier_stokes_spatial_operator<false,
105 linearize::none, filter::none>(ARGUMENTS);
106
107 case linearize::rhome_y:
108 return (substep_index == 0)
109 ? apply_navier_stokes_spatial_operator<true,
110 linearize::rhome_y, filter::none>(ARGUMENTS)
111 : apply_navier_stokes_spatial_operator<false,
112 linearize::rhome_y, filter::none>(ARGUMENTS);
113
114 default:
115 SUZERAIN_ERROR_VAL_UNIMPLEMENTED(std::vector<real_t>());
116 break;
117 }
118 break;
119
120 case filter::cook:
121
122 switch (common.linearization) {
123 case linearize::none:
124 return (substep_index == 0)
125 ? apply_navier_stokes_spatial_operator<true,
126 linearize::none, filter::cook>(ARGUMENTS)
127 : apply_navier_stokes_spatial_operator<false,
128 linearize::none, filter::cook>(ARGUMENTS);
129
130 case linearize::rhome_y:
131 return (substep_index == 0)
132 ? apply_navier_stokes_spatial_operator<true,
133 linearize::rhome_y, filter::cook>(ARGUMENTS)
134 : apply_navier_stokes_spatial_operator<false,
135 linearize::rhome_y, filter::cook>(ARGUMENTS);
136
137 default:
138 SUZERAIN_ERROR_VAL_UNIMPLEMENTED(std::vector<real_t>());
139 break;
140 }
141 break;
142
143 case filter::viscous:
144
145 switch (common.linearization) {
146 case linearize::none:
147 return (substep_index == 0)
148 ? apply_navier_stokes_spatial_operator<true,
149 linearize::none, filter::viscous>(ARGUMENTS)
150 : apply_navier_stokes_spatial_operator<false,
151 linearize::none, filter::viscous>(ARGUMENTS);
152
153 case linearize::rhome_y:
154 return (substep_index == 0)
155 ? apply_navier_stokes_spatial_operator<true,
156 linearize::rhome_y, filter::viscous>(ARGUMENTS)
157 : apply_navier_stokes_spatial_operator<false,
158 linearize::rhome_y, filter::viscous>(ARGUMENTS);
159
160 default:
161 SUZERAIN_ERROR_VAL_UNIMPLEMENTED(std::vector<real_t>());
162 break;
163 }
164 break;
165
166 default:
167 SUZERAIN_ERROR_VAL_UNIMPLEMENTED(std::vector<real_t>());
168 break;
169
170 }
171
172
173#undef ARGUMENTS
174}

176real_t operator_mass_isothermal::lower_E(
177 const real_t lower_T,
178 const real_t lower_u,
179 const real_t lower_v,
180 const real_t lower_w,
181 const std::vector<real_t> lower_cs) const
182{
183 // FIXME Gather and use instantaneous averages when isnan(lower_[uvw]).
184
185 // compute internal and kinetic energies
186 const real_t E_internal = cmods.e_from_T(lower_T, lower_cs);
187 const real_t E_kinetic = 0.5 * ( lower_u*lower_u
188 + lower_v*lower_v
189 + lower_w*lower_w);
190
191 return E_internal + E_kinetic;
192}
