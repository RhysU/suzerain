39// Parse Unix-like verbosity flags (http://stackoverflow.com/questions/5486753)
40static std::pair<std::string, std::string> verbosity(const std::string& s)
41{
42 std::pair<std::string, std::string> retval;
43
44 if (s.find("-v") == 0) {
45
46 size_t value = 1;
47 try {
48 value = boost::lexical_cast<size_t>(s.substr(2));
49 } catch(...) {
50 while (s[1+value] == 'v') ++value;
51 }
52 retval.first = "verbose";
53 retval.second = boost::lexical_cast<std::string>(value);
54
55 } else if (s.find("-V") == 0) {
56
57 size_t value = 1;
58 try {
59 value = boost::lexical_cast<size_t>(s.substr(2));
60 } catch(...) {
61 while (s[1+value] == 'V') ++value;
62 }
63 retval.first = "verbose-all";
64 retval.second = boost::lexical_cast<std::string>(value);
65
66 }
67
68 return retval;
69}

145std::vector<std::string> program_options::process_internal(
146 int argc, char **argv, MPI_Comm comm,
147 std::ostream &debug, std::ostream &info,
148 std::ostream &warn, std::ostream &error)
149{
150 SUZERAIN_UNUSED(debug); // Possibly unused
151 SUZERAIN_UNUSED(info); // Possibly unused
152 SUZERAIN_UNUSED(warn); // Possibly unused
153 SUZERAIN_UNUSED(error); // Possibly unused
154 using std::endl;
155 using std::istream;
156 using std::streambuf;
157 using std::string;
158 using std::vector;
159 namespace po = boost::program_options;
160 const int rank = mpi::comm_rank(comm);
161
162 // Obtain the program name using basename(argv[0])
163 std::string program_name;
164 {
165 std::shared_ptr<char> name_copy(strdup(argv[0]), &free);
166 program_name = basename(name_copy.get());
167 }
168
169 // Prepare options allowed only on command line
170 po::options_description desc_clionly("Program information");
171 desc_clionly.add_options()
172 ("help,h",
173 "Show usage information")
174 ("version",
175 "Print version string")
176 ("verbose,v", po::value<std::vector<std::string> >(),
177 "Increase verbosity of messages from rank zero")
178 ("verbose-all,V", po::value<std::vector<std::string> >(),
179 "Increase verbosity of all rank-specific messages")
180 ;
181
182 // Prepare a response-file option iff non-trivial options available
183 if (options_.options().size() > 0) {
184 desc_clionly.add_options()
185 ("response-file", po::value< vector<string> >()->composing(),
186 "File to additionally read for options")
187 ;
188 }
189
190 // Prepare options allowed on command line and in configuration file
191 // These are never shown to the user.
192 // TODO Do we want __positional__ available in a response file? Eh...
193 po::options_description desc_hidden("Hidden options");
194 desc_hidden.add_options()
195 ("__positional__", po::value< vector<string> >(), "positional args")
196 ;
197
198 // Guess at the option width to use (krufty and bash-ish, currently).
199 // FIXME Make terminal-width detection work as expected.
200 // Problem is not in the code below, but in how linewidth is used
201 // by Boost. See http://lists.boost.org/boost-users/2014/01/81007.php.
202 unsigned linewidth = po::options_description::m_default_line_length;
203 {
204 using namespace std;
205 if (getenv("COLUMNS")) {
206 errno = 0;
207 const long t = strtol(getenv("COLUMNS"), (char **) NULL, 10);
208 if ( !errno
209 && t >= numeric_limits<unsigned>::min()
210 && t <= numeric_limits<unsigned>::max()) {
211 linewidth = t;
212 }
213 }
214 }
215
216 // Build the options acceptable on the CLI, in a file, and in help message
217 po::options_description opts_cli (linewidth);
218 po::options_description opts_file (linewidth);
219 po::options_description opts_visible(linewidth);
220 opts_cli .add(options_).add(desc_hidden).add(desc_clionly);
221 opts_file .add(options_).add(desc_hidden) ;
222 opts_visible.add(options_) .add(desc_clionly);
223
224 // Collect positional arguments into "__positional__"
225 po::positional_options_description opts_positional;
226 opts_positional.add("__positional__", -1);
227
228 // Parse all the command line options
229 po::parsed_options parsed_cli = po::command_line_parser(argc, argv)
230 .options(opts_cli)
231 .positional(opts_positional)
232 .allow_unregistered()
233 .extra_parser(verbosity)
234 .run();
235 po::store(parsed_cli, variables_);
236
237 // Warn whenever an unrecognized option is encountered on the CLI
238 // Warn instead of balk because MPI stacks may utilize argc/argv
239 {
240 vector<string> unrecognized = po::collect_unrecognized(
241 parsed_cli.options, po::exclude_positional);
242 for (vector<string>::const_iterator it = unrecognized.begin();
243 it != unrecognized.end();
244 ++it) {
245 warn << "Unrecognized option '" << *it << "'"
246 << " on command line"
247 << endl;
248 }
249 }
250
251 // Process --help
252 if (variables_.count("help")) {
253 info << "Usage: " << program_name
254 << " [OPTION]... " << argument_synopsis_
255 << '\n';
256
257 if (!application_synopsis_.empty()) {
258 info << application_synopsis_ << '\n';
259 }
260
261 if (opts_visible.options().size() > 0) {
262 if (options_.options().size() > 0) {
263 info << '\n' << "Options:" << '\n';
264 }
265 info << opts_visible << '\n';
266 }
267
268 if (!application_description_.empty()) {
269 info << '\n' << application_description_ << '\n';
270 }
271
272 exit(0);
273 }
274
275 // Process --version using suzerain::version(...)
276 if (variables_.count("version")) {
277 info << version(program_name, application_version_) << std::endl;
278 exit(0);
279 }
280
281 // Parse any input files provided on the command line
282 // Earlier files shadow/override settings found in later files
283 // TODO This open-file-and-broadcast would be a nice library routine
284 const vector<string>& responsefiles
285 = variables_.count("response-file")
286 ? variables_["response-file"].as< vector<string> >()
287 : vector<string>();
288 for (vector<string>::const_iterator it = responsefiles.begin();
289 it != responsefiles.end();
290 ++it) {
291
292 const string &filename = *it;
293 debug << "Reading additional options from file '"
294 << filename << "'" << endl;
295
296 // Specialized istream::failure subclass for this routine
297 struct failure : public istream::failure {
298 failure(const string &op, const string& file, int errnum)
299 : istream::failure("Failure during " + op
300 + " for response-file '" + file
301 + "': " + strerror(errnum) ) {}
302
303 failure(const string& file) : istream::failure(
304 "Failure processing response-file '" + file + "'") {}
305 };
306
307 long len;
308 std::unique_ptr<char[]> buf;
309
310 // Rank zero slurps the file into appropriately-sized buf
311 if (rank == 0) {
312 errno = 0;
313 FILE * fp = NULL;
314 try {
315 // Open file and obtain length
316 fp = fopen(filename.c_str(), "rb");
317 if (!fp || errno)
318 throw failure("fopen", filename, errno);
319 if (fseek(fp, 0, SEEK_END))
320 throw failure("fseek", filename, errno);
321 len = ftell(fp);
322 if (len == -1)
323 throw failure("ftell", filename, errno);
324 if (fseek(fp, 0, SEEK_SET))
325 throw failure("fseek", filename, errno);
326
327 // Allocate buffer and slurp contents
328 buf.reset(new char[len]);
329 if (fread(buf.get(), 1, len, fp) != (size_t) len) {
330 throw failure("fread", filename, errno);
331 }
332
333 // Close appears after catch
334 } catch (istream::failure &f) {
335 error << f.what() << endl;
336 len = -1;
337 }
338 if (fp) fclose(fp);
339 }
340
341 // Buffer size and then buf contents are broadcast to other ranks
342 MPI_Bcast(&len, 1, MPI_LONG, 0, comm);
343 if (len == -1)
344 throw failure(filename);
345 if (rank > 0)
346 buf.reset(new char[len]);
347 MPI_Bcast(buf.get(), len, MPI_CHAR, 0, MPI_COMM_WORLD);
348
349 // Create an istream from the buffer contents a la
350 // bytes.com/topic/c/answers/582365-making-istream-char-array
351 struct membuf : public streambuf
352 {
353 membuf(streambuf::char_type *b, size_t n)
354 {
355 this->setg(b, b, b + n);
356 }
357 };
358 membuf mb(buf.get(), len);
359 istream is(&mb);
360
361 // Parse and store from the buffer-based istream
362 po::parsed_options parsed_file
363 = po::parse_config_file(is, opts_file, true);
364 po::store(parsed_file, variables_);
365
366 // TODO Display appropriate warning on unrecognized options
367 // collect_unrecognized/parse_config_file broken in Boost 1.40
368 // Refer to https://svn.boost.org/trac/boost/ticket/3775
369 {
370 vector< string > unrecognized = po::collect_unrecognized(
371 parsed_file.options, po::exclude_positional);
372 BOOST_FOREACH( string option, unrecognized ) {
373 warn << "Unrecognized option '" << option << "'"
374 << " in response-file '" << filename << "'"
375 << endl;
376 }
377 }
378
379 // TODO Display appropriate debug message on shadowed options
380 }
381
382 // Compute --verbose and --verbose-all levels by summation
383 // following http://stackoverflow.com/questions/5486753
384 verbose_ = 0;
385 if (variables_.count("verbose")) {
386 const std::vector<std::string> &values
387 = variables_["verbose"].as<std::vector<std::string> >();
388 for (std::size_t i = 0; i < values.size(); ++i) {
389 verbose_ += boost::lexical_cast<int>(values[i]);
390 }
391 }
392 verbose_all_ = 0;
393 if (variables_.count("verbose-all")) {
394 const std::vector<std::string> &values
395 = variables_["verbose-all"].as<std::vector<std::string> >();
396 for (std::size_t i = 0; i < values.size(); ++i) {
397 verbose_all_ += boost::lexical_cast<int>(values[i]);
398 }
399 }
400
401 // Perform all notification callbacks because all processing is done
402 po::notify(variables_);
403
404 // Return all position arguments to the caller for his/her use. Run into
405 // any_cast issues if returning non-existent values, so be a bit careful
406 // about what exactly we return.
407 if (variables_.count("__positional__")) {
408 return variables_["__positional__"].as< vector<string> >();
409 } else {
410 return vector<string>();
411 }
412}
