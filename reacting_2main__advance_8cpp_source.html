85int
86suzerain::reacting::driver_advance::run(int argc, char **argv)
87{
88 using namespace std;
89 using boost::math::isnan;
90
91 // Storage for binary-specific options
92 const support::definition_noise noisedef;
93 string solver_spec(static_cast<string>(suzerain::specification_zgbsv()));
94 string filter_spec("viscous");
95 string undriven;
96
97 // Register binary-specific options
98 options.add_definition(const_cast<support::definition_noise&>(noisedef));
99 options.add_options()
100 ("explicit", boost::program_options::bool_switch(),
101 "Use purely explicit operators")
102 ("implicit", boost::program_options::bool_switch(),
103 "Use hybrid implicit/explicit operators")
104 ("solver", boost::program_options::value(&solver_spec)
105 ->default_value(solver_spec),
106 "Use the specified algorithm for any implicit solves")
107 ("filter", boost::program_options::value(&filter_spec)
108 ->default_value(filter_spec),
109 "Use the specified type to construct filter source")
110 ("undriven", boost::program_options::value(&undriven)
111 ->implicit_value("all"),
112 "Disable all or some physics-related driving forces"
113 " by specifying 'all' or 'rho', 'rho_u', etc.")
114 ("preserve_upper_input", boost::program_options::bool_switch(),
115 "Preserve upper input or file values (applies to"
116 " slow growth baseflow scenarios)")
117 ("preserve_largo_gramp_input", boost::program_options::bool_switch(),
118 "Preserve gramp file values (applies to"
119 " slow growth bl_spatiotemporal_consistent model)")
120 ;
121
122 // Initialize application and then process binary-specific options
123 // (henceforth suzerain::support::logging macros becomes usable)
124 vector<string> positional = initialize(argc, argv);
125
126 // Select type of timestepping operators to use (default implicit)
127 options.conflicting_options("explicit", "implicit");
128 const bool use_explicit = options.variables()["explicit"].as<bool>();
129 const bool use_implicit = options.variables()["implicit"].as<bool>()
130 || !use_explicit;
131
132 // Only one implicit option now, so just use if we got '--implicit'
133 if (use_implicit) {
134 common_block.linearization = linearize::rhome_y;
135 } else {
136 common_block.linearization = linearize::none;
137 }
138
139 // Select type of filtering to use (default viscous)
140 const bool use_filter = options.variables().count("filter");
141
142 if (use_filter) {
143 boost::algorithm::trim(filter_spec);
144 if (filter_spec == "none") {
145 common_block.filter_treatment = filter::none;
146 } else if (filter_spec == "cook") {
147 common_block.filter_treatment = filter::cook;
148 } else if (filter_spec == "viscous") {
149 common_block.filter_treatment = filter::viscous;
150 } else {
151 FATAL0("Unknown --filter argument: " << filter_spec);
152 return EXIT_FAILURE;
153 }
154 } else {
155 common_block.filter_treatment = filter::viscous;
156 }
157
158 if (positional.size() != 1) {
159 FATAL0("Exactly one restart file name must be specified");
160 return EXIT_FAILURE;
161 }
162 const string restart_file = positional[0];
163
164 // Reset or preserve the file or input upper values
165 const bool preserve_upper_input =
166 options.variables()["preserve_upper_input"].as<bool>();
167
168 // Reset or preserve the file amplitude growth rate values
169 const bool preserve_largo_gramp_input =
170 options.variables()["preserve_largo_gramp_input"].as<bool>();
171
172 INFO0(who, "Loading restart file: " << restart_file);
173 real_t initial_t = numeric_limits<real_t>::quiet_NaN();
174 {
175 // Preserve exact restart file details via Push/Pop/Merge below
176 std::shared_ptr<definition_channel> restart_chdef
177 = std::make_shared<definition_channel>();
178
179 // Load the restart details with state going into state_linear
180 support::shared_esio_handle h(MPI_COMM_WORLD);
181 esio_file_open(h.get(), restart_file.c_str(), 0);
182 restart_chdef.swap(chdef);
183 load_restart(h.get(), initial_t);
184 restart_chdef.swap(chdef);
185 chdef->populate(*restart_chdef, true);
186
187 // FIXME: filter source
188 // initialize with info from restart here (?)
189
190 // Adjust total energy as necessary to account for any scenario change
191 state_nonlinear->assign_from(*state_linear);
192 // FIXME: Currently no functionality to adjust scenario
193 //adjust_scenario(*state_nonlinear, *scenario, *grid, *dgrid, *cop, restart_scenario->Ma, restart_scenario->gamma);
194 state_linear->assign_from(*state_nonlinear);
195 }
196
197 INFO0(who, "Initializing antioch_constitutive");
198 cmods->init_antioch();
199
200 common_block.Ns = cmods->Ns();
201
202 // However, if msoln was provided, match its contents to other members
203 // Do here b/c cmods doesn't make sense until after init_antioch
204 if (msoln) {
205 if (cmods) msoln->match(*cmods);
206 if (grid) msoln->match(*grid);
207 }
208
209 if (msoln) {
210 INFO0(who, "Restart file prescribes a manufactured solution");
211 if (!(isnan)(chdef->bulk_rho)) {
212 WARN0(who, "Manufactured solution incompatible with bulk_rho = "
213 << chdef->bulk_rho);
214 }
215 if (!(isnan)(chdef->bulk_rho_u)) {
216 WARN0(who, "Manufactured solution incompatible with bulk_rho_u = "
217 << chdef->bulk_rho_u);
218 }
219 }
220
221 // If requested, add noise to the momentum fields at startup (expensive).
222 if (noisedef.percent > 0) {
223 state_nonlinear->assign_from(*state_linear);
224 add_noise(*state_nonlinear, noisedef,
225 *grid, *dgrid, *cop, *b);
226 state_linear->assign_from(*state_nonlinear);
227 }
228
229
230 // Prepare any necessary, problem-specific constraints
231 std::shared_ptr<constraint::treatment> constrainer(new constraint::treatment(
232 1.0, *dgrid, *b, common_block, common_block));
233 if (grid->two_sided()) { // Channel per treatment_channel.tex
234
235 INFO0(who, "Establishing driving, channel-like state constraints");
236 constrainer->physical[ndx::rho].reset(
237 new constraint::reference_bulk(chdef->bulk_rho , *b));
238 constrainer->physical[ndx::mx ].reset(
239 new constraint::reference_bulk(chdef->bulk_rho_u, *b));
240 constrainer->physical[ndx::e ].reset(
241 new constraint::reference_bulk(chdef->bulk_rho_E, *b));
242
243 } else if (grid->one_sided()) { // Flat plate
244
245 real_t rho_inf;
246 real_t mx_inf;
247 real_t e_inf;
248
249 // Compute baseflow from coefficients
250 if (sgdef->baseflow && !preserve_upper_input) {
251 INFO0(who, "Setting mean freestream reference state per baseflow");
252 WARN0(who, "... there is no correction to freestream values for"
253 " displacement effects ");
254 if (isothermal->lower_v != 0.) {
255 WARN0(who, "... there is no correction to freestream energy"
256 " for wall-normal velocity");
257 }
258
259 const size_t Ns = cmods->Ns();
260 if (Ns > 1) {
261 WARN0(who, "... setting mean freestream reference state"
262 " per baseflow NOT tested for multispecies,"
263 " proceed at your own risk");
264 }
265 real_t base [Ns+4+1];
266 real_t dybase [Ns+4+1];
267 real_t dxbase [Ns+4+1];
268 sgdef->baseflow->conserved(
269 grid->L.y(), &base[0], &dybase[0], &dxbase[0]);
270 sgdef->baseflow->pressure(
271 grid->L.y(), base[Ns+4], dybase[Ns+4], dxbase[Ns+4]);
272
273 // Clobber upper values except upper verocity
274 // FIXME: Account for the correction to upper_v for the
275 // modified characteristics from slow growth.
276 // NOTE: Maybe the proper place to implement the upper_v
277 // correction is in the nonreflecting implementation,
278 // instead of setting a corrected upper_v here.
279 // Working with a modified upper velocity here might be
280 // confusing and lead to wrong results, when computing
281 // derived quantities (eg, temperature), that depend on
282 // the physical upper_v (not the effective one from
283 // characteristics)
284 isothermal->upper_rho = base[0];
285 isothermal->upper_u = base[1]/base[0];
286 isothermal->upper_w = base[3]/base[0];
287
288 for (unsigned int s=1; s<Ns; ++s) {
289 isothermal->upper_cs[s] = base[5+s]/base[0];
290 }
291
292 // Get upper temperature from antioch
293 const real_t irho = 1.0/isothermal->upper_rho ;
294
295 // ... momentum
296 // ... use the actual freestream rho_v value at the upper boundary
297 // to compute temperature properly
298 const real_t upper_rho_v (base[2]);
299 const Vector3r m (isothermal->upper_rho * isothermal->upper_u,
300 upper_rho_v,
301 isothermal->upper_rho * isothermal->upper_w);
302
303 // ... total energy
304 const real_t e (base[4]);
305
306 // ... species densities
307 VectorXr species(Ns); // species densities
308 VectorXr cs (Ns); // species mass fractions
309
310 // NOTE: In species vector, idx 0 is the dilluter (the species
311 // that is not explicitly part of the state vector)
312 species(0) = isothermal->upper_rho;
313 for (unsigned int s=1; s<Ns; ++s) {
314 species(s) = isothermal->upper_cs[s];
315
316 // dilluter density = rho_0 = rho - sum_{s=1}^{Ns-1} rho_s
317 species(0) -= species(s);
318 }
319
320 // ... compute mass fractions
321 for (unsigned int s=0; s<Ns; ++s) {
322 cs(s) = irho * species(s);
323 }
324
325 // ... temperature and pressure-related quantities
326 real_t T, pr;
327 cmods->evaluate_pressure(
328 e, m, isothermal->upper_rho, species, cs, isothermal->lower_T,
329 T, pr);
330
331 // ... finally we got what we wanted...
332 isothermal->upper_T = T;
333
334 INFO0(who, "... upper_rho set to " << isothermal->upper_rho );
335 INFO0(who, "... upper_u set to " << isothermal->upper_u );
336 INFO0(who, "... upper_v set to " << isothermal->upper_v );
337 INFO0(who, "... upper_w set to " << isothermal->upper_w );
338 INFO0(who, "... upper_T set to " << isothermal->upper_T );
339
340 // ... set inf values for freestream constraints
341 rho_inf = isothermal->upper_rho;
342 mx_inf = isothermal->upper_u * rho_inf;
343 e_inf = e;
344
345 } else {
346 INFO0(who, "Computing mean freestream behavior per plate scenario");
347 if (sgdef->baseflow && preserve_upper_input) {
348 WARN0(who, "... preserving upper input or file values as per"
349 " 'preserve_upper_input' option");
350 }
351 const real_t T_inf = isothermal->upper_T;
352 const real_t u_inf = isothermal->upper_u;
353 rho_inf = isothermal->upper_rho;
354 mx_inf = u_inf * rho_inf;
355
356 // Compute total energy at infinity based on input upper values
357 // Neglect contribution of wall-normal velocity to kinetic energy,
358 // and assume that species concentrations are at equilibrium
359 // FIXME: adjust the value of target rho_E dynamically
360 // to account for wall-normal velocity and variations in
361 // species mass fractions
362 const size_t Ns = cmods->Ns();
363 vector<real_t> mass_fractions(Ns);
364 for (unsigned int s=0; s<Ns; ++s) {
365 mass_fractions[s] = isothermal->upper_cs[s];
366 }
367 e_inf = (cmods->e_from_T(T_inf, mass_fractions)
368 + 0.5 * (u_inf * u_inf)) * rho_inf;
369 }
370
371 INFO0(who, "Setting constraints using freestream reference state"
372 " on upper boundary");
373
374 if (chdef->bulk_rho) {
375 WARN0(who, "Removing channel-like bulk_rho setting");
376 chdef->bulk_rho = numeric_limits<real_t>::quiet_NaN();
377 }
378 if (chdef->bulk_rho_u) {
379 WARN0(who, "Removing channel-like bulk_rho_u setting");
380 chdef->bulk_rho_u = numeric_limits<real_t>::quiet_NaN();
381 }
382 if (chdef->bulk_rho_E) {
383 WARN0(who, "Removing channel-like bulk_rho_E setting");
384 chdef->bulk_rho_E = numeric_limits<real_t>::quiet_NaN();
385 }
386
387 INFO0(who, "Establishing driving, freestream-like state constraints");
388 constrainer->physical[ndx::e ].reset(
389 new constraint::constant_upper(e_inf, *cop, 0));
390 constrainer->physical[ndx::mx ].reset(
391 new constraint::constant_upper(mx_inf, *cop, 0));
392 constrainer->physical[ndx::rho].reset(
393 new constraint::constant_upper(rho_inf, *cop, 0));
394
395 } else {
396
397 FATAL0(who, "Sanity error in constraint selection");
398 return EXIT_FAILURE;
399
400 }
401
402
403 // Nonreflecting must mutate chdef/isothermal before instantiating
404 // linear operator
405 if (grid->one_sided()) {
406 // FIXME: Plate implementation in progress.
407 // Need to set reference value for rho (rho_ref)
408
409 INFO0(who, "Preparing nonreflecting upper boundary treatment");
410
411 WARN0(who, "Non-reflecting boundary treatment enabling boundary"
412 " layer simulations is still under development.");
413
414
415 // Assign some reference values
416 const size_t Ns = cmods->Ns();
417
418 // SUZERAIN_ENSURE(!(isnan)(isothermal->upper_rho));
419 if ((isnan)(isothermal->upper_rho)) {
420 FATAL0(who, "upper_rho for nonreflecting boundary"
421 " not specified and not in restart file");
422 return EXIT_FAILURE;
423 }
424
425 common_block.rho_ref = isothermal->upper_rho;
426 common_block.T_ref = isothermal->upper_T;
427 common_block.u_ref = isothermal->upper_u;
428 common_block.v_ref = isothermal->upper_v;
429 common_block.w_ref = isothermal->upper_w;
430
431 common_block.cs_ref.resize(Ns);
432 vector<real_t> mass_fractions(Ns);
433 for (unsigned int s=0; s<Ns; ++s) {
434 common_block.cs_ref(s) = isothermal->upper_cs[s];
435 mass_fractions[s] = isothermal->upper_cs[s];
436 }
437
438 common_block.etots_ref.resize(Ns);
439 cmods->evaluate_for_nonreflecting(common_block.T_ref,
440 common_block.cs_ref,
441 common_block.a_ref,
442 common_block.gamma_ref,
443 common_block.R_ref,
444 common_block.etots_ref);
445
446 common_block.Cv_ref = common_block.R_ref
447 / (common_block.gamma_ref - 1);
448
449 // FIXME: Remove this info or make it a debug output option
450 INFO0(who, "rho_ref = " << common_block.rho_ref );
451 INFO0(who, "T_ref = " << common_block.T_ref );
452 INFO0(who, "a_ref = " << common_block.a_ref );
453 INFO0(who, "gamma_ref = " << common_block.gamma_ref);
454 INFO0(who, "R_ref = " << common_block.R_ref );
455 INFO0(who, "E_ref = " << cmods->e_from_T(common_block.T_ref,
456 mass_fractions));
457 }
458
459 // Reset amplitude growth rate parameters for spatiotemporal formulation
460 if (sgdef->formulation == largo_formulation::spatiotemporal_consistent
461 && !preserve_largo_gramp_input) {
462 INFO0(who, "Setting largo growth rate amplitude values");
463
464 const size_t Ns = cmods->Ns();
465
466 // resize to 5 flow vars + (Ns-1) species + pressure
467 sgdef->gramp_mean.resize(5+(Ns-1)+1);
468
469 // set all values to zero
470 for (unsigned int i=0; i<sgdef->gramp_mean.size(); i++) {
471 sgdef->gramp_mean[i] = 0.;
472 }
473
474 // pre-computations to get wall baseflow values
475 const size_t iu = 1;
476 const size_t ie = 4;
477 const size_t is0 = 5;
478 const size_t ip = Ns + 4;
479 real_t wall [Ns+4+1];
480 real_t dywall [Ns+4+1];
481 real_t dxwall [Ns+4+1];
482 sgdef->baseflow->conserved(
483 0., &wall[0], &dywall[0], &dxwall[0]);
484 sgdef->baseflow->pressure(
485 0., wall[ip], dywall[ip], dxwall[ip]);
486
487
488 // thermo quantities to estimate wall flow values
489 VectorXr wall_cs(Ns);
490 for (unsigned int s=0; s<Ns; ++s) {
491 wall_cs(s) = isothermal->lower_cs[s];
492 }
493
494 real_t wall_a; // not needed
495 real_t wall_gamma;
496 real_t wall_Rmix;
497 VectorXr wall_etots(Ns); // not needed
498 cmods->evaluate_for_nonreflecting(isothermal->lower_T,
499 wall_cs,
500 wall_a,
501 wall_gamma,
502 wall_Rmix,
503 wall_etots);
504
505 // pre-compute these for convenience
506 const real_t wall_Cv = wall_Rmix / (wall_gamma - 1.);
507 const real_t wall_P = wall[ip];
508 const real_t wall_ddx_P = dxwall[ip];
509 const real_t wall_rhoI = wall[0];
510 const real_t wall_ddx_rhoI = dxwall[0];
511 const real_t wall_rho = wall_P / wall_Rmix / isothermal->lower_T;
512 const real_t wall_ddx_rho = wall_ddx_P / wall_Rmix / isothermal->lower_T;
513 const real_t wall_ruI = wall[iu];
514 const real_t wall_ddx_ruI = dxwall[iu];
515 const real_t wall_EI = wall[ie] / wall[0];
516 const real_t wall_rEI = wall[ie];
517 const real_t wall_ddx_rEI = dxwall[ie];
518 const real_t wall_E = wall_Cv * isothermal->lower_T;
519
520 // growth rate amplitudes for density, streamwise velocity, energy
521 sgdef->gramp_mean[0] = 1./(wall_rho - wall_rhoI) * (wall_ddx_rho - wall_ddx_rhoI);
522 sgdef->gramp_mean[1] = wall_ddx_ruI/wall_ruI - wall_ddx_rhoI/wall_rhoI;
523 sgdef->gramp_mean[ie] = - (wall_EI/(wall_E - wall_EI))
524 * (wall_ddx_rEI/wall_rEI - wall_ddx_rhoI/wall_rhoI);
525
526 // growth rate amplitude for species must be set all equal for consistency
527 // setting them all equal to zero
528 for (size_t s=0; s<Ns-1; s++) {
529 sgdef->gramp_mean[is0+s] = sgdef->gramp_mean[0];
530 }
531
532 // rms amplitude growth rates
533 sgdef->gramp_rms.resize(5+(Ns-1)+1);
534
535
536 // output computed values
537 DEBUG0(who, "... wall rho is " << wall_rho);
538 DEBUG0(who, "... wall Cv is " << wall_Cv);
539 DEBUG0(who, "... wall Rmix is " << wall_Rmix);
540 DEBUG0(who, "... wall E is " << wall_E);
541
542 INFO0(who, "... gramp_mean for rho set to " << sgdef->gramp_mean[0]);
543 INFO0(who, "... gramp_mean for u set to " << sgdef->gramp_mean[iu]);
544 INFO0(who, "... gramp_mean for E set to " << sgdef->gramp_mean[ie]);
545 WARN0(who, "... gramp_mean for cs set all to 0. ");
546
547 // set all values to zero
548 for (unsigned int i=0; i<sgdef->gramp_rms.size(); i++) {
549 sgdef->gramp_rms[i] = 0.;
550 }
551 } else if (sgdef->formulation == largo_formulation::spatiotemporal_consistent
552 && preserve_largo_gramp_input) {
553 WARN0(who, "Preserving largo_gramp values from file"
554 " per 'preserve_largo_gramp_input' option");
555 }
556
557 // Prepare spatial operators depending on requested advance type
558 // Notice constrainer always wraps the implicit operator
559 // and that the same operator_nonlinear is used pervasively.
560 L = constrainer;
561 if (use_explicit) {
562 INFO0(who, "Initializing fully explicit spatial operators");
563
564 N.reset(new explicit_operator_nonlinear(
565 *cmods, *grid, *dgrid, *cop, *b, common_block, *fsdef,
566 *sgdef, msoln));
567
568 if (grid->one_sided()) {
569 INFO0(who, "Preparing nonreflecting upper boundary treatment");
570 std::shared_ptr<treatment_nonreflecting> nonreflecting(
571 new treatment_nonreflecting(
572 *grid, *dgrid, *cop, *b, common_block));
573 nonreflecting->N = N;
574 N = nonreflecting;
575 }
576
577 constrainer->L.reset(new operator_mass_isothermal(
578 *cmods, *isothermal,
579 *chdef, *grid, *dgrid, *cop, *b, common_block));
580
581 } else if (use_implicit) {
582 INFO0(who, "Initializing hybrid implicit/explicit spatial operators");
583
584 constrainer->L.reset(new isothermal_hybrid_linear_operator(
585 solver_spec, *cmods, *isothermal, *chdef, *grid, *dgrid,
586 *cop, *b, common_block));
587
588 std::shared_ptr<suzerain::operator_hybrid_residual>
589 tmp_hybrid( new operator_hybrid_residual(dgrid->chi()) );
590
591 tmp_hybrid->R.reset(new explicit_operator_nonlinear(
592 *cmods, *grid, *dgrid, *cop, *b,
593 common_block, *fsdef, *sgdef, msoln));
594
595 tmp_hybrid->L = this->L;
596
597 this->N = tmp_hybrid;
598
599 if (grid->one_sided()) {
600 WARN0(who, "Non-reflecting boundary treatment with hybrid"
601 " implicit/explicit time marching is experimental.");
602
603 std::shared_ptr<treatment_nonreflecting> nonreflecting(
604 new treatment_nonreflecting(
605 *grid, *dgrid, *cop, *b, common_block));
606 nonreflecting->N = this->N;
607 this->N = nonreflecting;
608 }
609
610
611 } else {
612 FATAL0(who, "Sanity error in operator selection");
613 return EXIT_FAILURE;
614 }
615
616 // Allocate slow growth workspace
617 if (sgdef->formulation.enabled()) {
618 const std::string& model = sgdef->formulation.name();
619 const int neq = static_cast<int>(cmods->Ns()) + 4;
620 const int ns = static_cast<int>(cmods->Ns()) - 1;
621 enum { ntvar = 0 };
622 static const char ransmodel[] = "dns";
623 INFO0(who, "Allocating Largo model \"" << model << "\" with neq="
624 << neq << ", ns=" << ns << ", ntvar=" << ntvar
625 << ", ransmodel=" << ransmodel);
626 largo_allocate(&sgdef->workspace, model.c_str(), neq,
627 ns, ntvar, ransmodel);
628 if (!sgdef->workspace) {
629 FATAL0(who, "Largo could not allocate requested model");
630 return EXIT_FAILURE;
631 }
632 if ((isnan)(sgdef->grdelta)) {
633 WARN0(who, "Slow growth rate grdelta is NaN");
634 }
635 if (sgdef->ignore_fluctuations) {
636 // TODO Permit option by modifying operator_nonlinear.hpp.
637 FATAL0(who, "Debugging option 'ignore_fluctuations' not supported");
638 return EXIT_FAILURE;
639 }
640 if (sgdef->ignore_gramp_mean) {
