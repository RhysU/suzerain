1739template<typename Scalar,
1740 typename Vector,
1741 typename Tensor >
1742inline
1743void p_T(
1744 const Scalar &alpha,
1745 const Scalar &beta,
1746 const Scalar &gamma,
1747 const Scalar &rho,
1748 const Vector &grad_rho,
1749 const Vector &m,
1750 const Tensor &grad_m,
1751 const Scalar &e,
1752 const Vector &grad_e,
1753 Scalar &p,
1754 Vector &grad_p,
1755 Scalar &T,
1756 Vector &grad_T)
1757{
1758 SUZERAIN_UNUSED(alpha); // Present for API consistency
1759 SUZERAIN_UNUSED(beta); // Present for API consistency
1760
1761 const Scalar rho_inverse = 1/rho;
1762 const Scalar half_rho_inverse = rho_inverse/2;
1763 const Scalar gamma1 = gamma - 1;
1764
1765 // Compute scalar quantities
1766 p = gamma1*(e - half_rho_inverse*m.squaredNorm());
1767 T = gamma * p * rho_inverse;
1768
1769 // Compute vector quantities
1770 grad_p = gamma1*(
1771 grad_e + rho_inverse*(
1772 (half_rho_inverse*m.squaredNorm())*grad_rho
1773 - grad_m.transpose()*m
1774 )
1775 );
1776 grad_T = gamma*rho_inverse*(grad_p - rho_inverse*p*grad_rho);
1777}
