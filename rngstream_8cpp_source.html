95//-------------------------------------------------------------------------
96// Return (a*s + c) MOD m; a, s, c and m must be < 2^35
97//
98double MultModM (double a, double s, double c, double m)
99{
100 double v;
101 long a1;
102
103 v = a * s + c;
104
105 if (v >= two53 || v <= -two53) {
106 a1 = static_cast<long> (a / two17); a -= a1 * two17;
107 v = a1 * s;
108 a1 = static_cast<long> (v / m); v -= a1 * m;
109 v = v * two17 + a * s + c;
110 }
111
112 a1 = static_cast<long> (v / m);
113 /* in case v < 0)*/
114 if ((v -= a1 * m) < 0.0) return v += m; else return v;
115}

118//-------------------------------------------------------------------------
119// Compute the vector v = A*s MOD m. Assume that -m < s[i] < m.
120// Works also when v = s.
121//
122void MatVecModM (const double A[3][3], const double s[3], double v[3],
123 double m)
124{
125 int i;
126 double x[3]; // Necessary if v = s
127
128 for (i = 0; i < 3; ++i) {
129 x[i] = MultModM (A[i][0], s[0], 0.0, m);
130 x[i] = MultModM (A[i][1], s[1], x[i], m);
131 x[i] = MultModM (A[i][2], s[2], x[i], m);
132 }
133 for (i = 0; i < 3; ++i)
134 v[i] = x[i];
135}

138//-------------------------------------------------------------------------
139// Compute the matrix C = A*B MOD m. Assume that -m < s[i] < m.
140// Note: works also if A = C or B = C or A = B = C.
141//
142void MatMatModM (const double A[3][3], const double B[3][3],
143 double C[3][3], double m)
144{
145 int i, j;
146 double V[3], W[3][3];
147
148 for (i = 0; i < 3; ++i) {
149 for (j = 0; j < 3; ++j)
150 V[j] = B[j][i];
151 MatVecModM (A, V, V, m);
152 for (j = 0; j < 3; ++j)
153 W[j][i] = V[j];
154 }
155 for (i = 0; i < 3; ++i)
156 for (j = 0; j < 3; ++j)
157 C[i][j] = W[i][j];
158}

161//-------------------------------------------------------------------------
162// Compute the matrix B = (A^(2^e) Mod m); works also if A = B.
163//
164void MatTwoPowModM (const double A[3][3], double B[3][3], double m, long e)
165{
166 int i, j;
167
168 /* initialize: B = A */
169 if (A != B) {
170 for (i = 0; i < 3; ++i)
171 for (j = 0; j < 3; ++j)
172 B[i][j] = A[i][j];
173 }
174 /* Compute B = A^(2^e) mod m */
175 for (i = 0; i < e; i++)
176 MatMatModM (B, B, B, m);
177}

180//-------------------------------------------------------------------------
181// Compute the matrix B = (A^n Mod m); works even if A = B.
182//
183void MatPowModM (const double A[3][3], double B[3][3], double m, long n)
184{
185 int i, j;
186 double W[3][3];
187
188 /* initialize: W = A; B = I */
189 for (i = 0; i < 3; ++i)
190 for (j = 0; j < 3; ++j) {
191 W[i][j] = A[i][j];
192 B[i][j] = 0.0;
193 }
194 for (j = 0; j < 3; ++j)
195 B[j][j] = 1.0;
196
197 /* Compute B = A^n mod m using the binary decomposition of n */
198 while (n > 0) {
199 if (n % 2) MatMatModM (W, B, B, m);
200 MatMatModM (W, W, W, m);
201 n /= 2;
202 }
203}

249//-------------------------------------------------------------------------
250// Generate the next random number.
251//
252double rngstream::U01 ()
253{
254 long k;
255 double p1, p2, u;
256
257 /* Component 1 */
258 p1 = a12 * Cg[1] - a13n * Cg[0];
259 k = static_cast<long> (p1 / m1);
260 p1 -= k * m1;
261 if (p1 < 0.0) p1 += m1;
262 Cg[0] = Cg[1]; Cg[1] = Cg[2]; Cg[2] = p1;
263
264 /* Component 2 */
265 p2 = a21 * Cg[5] - a23n * Cg[3];
266 k = static_cast<long> (p2 / m2);
267 p2 -= k * m2;
268 if (p2 < 0.0) p2 += m2;
269 Cg[3] = Cg[4]; Cg[4] = Cg[5]; Cg[5] = p2;
270
271 /* Combination */
272 u = ((p1 > p2) ? (p1 - p2) * norm : (p1 - p2 + m1) * norm);
273
274 return (anti == false) ? u : (1 - u);
275}

278//-------------------------------------------------------------------------
279// Generate the next random number with extended (53 bits) precision.
280//
281double rngstream::U01d ()
282{
283 double u;
284 u = U01();
285 if (anti) {
286 // Don't forget that U01() returns 1 - u in the antithetic case
287 u += (U01() - 1.0) * fact;
288 return (u < 0.0) ? u + 1.0 : u;
289 } else {
290 u += U01() * fact;
291 return (u < 1.0) ? u : (u - 1.0);
292 }
293}

392//-------------------------------------------------------------------------
393// if e > 0, let n = 2^e + c;
394// if e < 0, let n = -2^(-e) + c;
395// if e = 0, let n = c.
396// Jump n steps forward if n > 0, backwards if n < 0.
397//
398void rngstream::AdvanceState (long e, long c)
399{
400 double B1[3][3], C1[3][3], B2[3][3], C2[3][3];
401
402 if (e > 0) {
403 MatTwoPowModM (A1p0, B1, m1, e);
404 MatTwoPowModM (A2p0, B2, m2, e);
405 } else if (e < 0) {
406 MatTwoPowModM (InvA1, B1, m1, -e);
407 MatTwoPowModM (InvA2, B2, m2, -e);
408 }
409
410 if (c >= 0) {
411 MatPowModM (A1p0, C1, m1, c);
412 MatPowModM (A2p0, C2, m2, c);
413 } else {
414 MatPowModM (InvA1, C1, m1, -c);
415 MatPowModM (InvA2, C2, m2, -c);
416 }
417
418 if (e) {
419 MatMatModM (B1, C1, C1, m1);
420 MatMatModM (B2, C2, C2, m2);
421 }
422
423 MatVecModM (C1, Cg, Cg, m1);
424 MatVecModM (C2, &Cg[3], &Cg[3], m2);
425}

518//-------------------------------------------------------------------------
519// Approximate the inverse of the standard normal per Peter John Acklam
520// (http://home.online.no/~pjacklam/notes/invnorm/) with error handling
521// following Chad Sprouse's implementation given at
522// (http://home.online.no/~pjacklam/notes/invnorm/impl/sprouse/ltqnorm.c).
523//
524static double acklam(const double p)
525{
526 // Coefficients in rational approximations.
527 static const double a1 = -3.969683028665376e+01;
528 static const double a2 = 2.209460984245205e+02;
529 static const double a3 = -2.759285104469687e+02;
530 static const double a4 = 1.383577518672690e+02;
531 static const double a5 = -3.066479806614716e+01;
532 static const double a6 = 2.506628277459239e+00;
533
534 static const double b1 = -5.447609879822406e+01;
535 static const double b2 = 1.615858368580409e+02;
536 static const double b3 = -1.556989798598866e+02;
537 static const double b4 = 6.680131188771972e+01;
538 static const double b5 = -1.328068155288572e+01;
539
540 static const double c1 = -7.784894002430293e-03;
541 static const double c2 = -3.223964580411365e-01;
542 static const double c3 = -2.400758277161838e+00;
543 static const double c4 = -2.549732539343734e+00;
544 static const double c5 = 4.374664141464968e+00;
545 static const double c6 = 2.938163982698783e+00;
546
547 static const double d1 = 7.784695709041462e-03;
548 static const double d2 = 3.224671290700398e-01;
549 static const double d3 = 2.445134137142996e+00;
550 static const double d4 = 3.754408661907416e+00;
551
552 // Define break-points.
553 const double p_low = 0.02425;
554 const double p_high = 1 - p_low;
555
556 // Compute rational approximation for...
557 double x, q, r;
558 if (p < 0) { // ...domain error.
559 errno = EDOM;
560 return 0;
561#pragma warning(push,disable:1572)
562 } else if (p == 0) { // ...improbable point.
563#pragma warning(pop)
564 errno = ERANGE;
565#ifdef HUGE_VAL
566 return -HUGE_VAL;
567#else
568 return -DBL_MAX;
569#endif
570 } else if (p < p_low) { // ...lower region.
571 q = sqrt(-2*log(p));
572 x = (((((c1*q + c2)*q + c3)*q + c4)*q + c5)*q + c6) /
573 ((((d1*q + d2)*q + d3)*q + d4)*q + 1);
574 } else if (p <= p_high) { // ...central region
575 q = p - 0.5;
576 r = q*q;
577 x = (((((a1*r + a2)*r + a3)*r + a4)*r + a5)*r + a6)*q /
578 (((((b1*r + b2)*r + b3)*r + b4)*r + b5)*r + 1);
579 } else if (p < 1) { // ...upper region.
580 q = sqrt(-2*log(1 - p));
581 x = -(((((c1*q + c2)*q + c3)*q + c4)*q + c5)*q + c6) /
582 ((((d1*q + d2)*q + d3)*q + d4)*q + 1);
583#pragma warning(push,disable:1572)
584 } else if (p == 1) { // ...improbable point.
585#pragma warning(pop)
586 errno = ERANGE;
587#ifdef HUGE_VAL
588 return HUGE_VAL;
589#else
590 return DBL_MAX;
591#endif
592 } else { // ...domain error.
593 errno = EDOM;
594 return 0;
595 }
596
597 // One iteration of Halley's rational method improves to machine precision.
598 static const double neg_1_sqrt2 = - 1 / M_SQRT2;
599 static const double sqrt_2pi = sqrt(2 * M_PI);
600 double e, u;
601 e = 0.5 * erfc(x * neg_1_sqrt2) - p;
602 u = e * sqrt_2pi * exp(x*x/2);
603 x = x - u/(1 + x*u/2);
604
605 return x;
606}
