112 void operator()(complex_t& lower_rho) const
113 {
114 // Locate the upper density relative to the provided lower density
115 complex_t& upper_rho = (&lower_rho)[Ny - 1];
116
117#define LOWER(var) ((&lower_rho)[((var) - ndx::rho)*incf])
118#define UPPER(var) ((&upper_rho)[((var) - ndx::rho)*incf])
119
120 // Set total energy to be density times specific total energy
121 if (enforce_lower_e) LOWER(ndx::e) = lower_rho * lower_E;
122 if (enforce_upper_e) UPPER(ndx::e) = upper_rho * upper_E;
123
124 // Set streamwise momentum to be density times prescribed velocity
125 if (enforce_lower_u) LOWER(ndx::mx) = lower_rho * spec.lower_u;
126 if (enforce_upper_u) UPPER(ndx::mx) = upper_rho * spec.upper_u;
127
128 // Set wall-normal momentum to be density times prescribed velocity
129 if (enforce_lower_v) LOWER(ndx::my) = lower_rho * spec.lower_v;
130 if (enforce_upper_v) UPPER(ndx::my) = upper_rho * spec.upper_v;
131
132 // Set spanwise momentum to be density times prescribed velocity
133 if (enforce_lower_w) LOWER(ndx::mz) = lower_rho * spec.lower_w;
134 if (enforce_upper_w) UPPER(ndx::mz) = upper_rho * spec.upper_w;
135
136 // Do nothing to the density equation
137
138 // Set species partial densities to be density times mixture fraction.
139 if (enforce_lower_cs & enforce_upper_cs) {
140 assert(spec.lower_cs.size() == spec.upper_cs.size());
141 const std::size_t num_species = spec.lower_cs.size();
142 for (std::size_t s = 1; s < num_species; ++s) {
143 LOWER(ndx::rho_(s)) = lower_rho * spec.lower_cs[s];
144 UPPER(ndx::rho_(s)) = upper_rho * spec.upper_cs[s];
145 }
146 } else if (enforce_lower_cs) {
147 const std::size_t num_species = spec.lower_cs.size();
148 for (std::size_t s = 1; s < num_species; ++s) {
149 LOWER(ndx::rho_(s)) = lower_rho * spec.lower_cs[s];
150 }
151 } else if (enforce_upper_cs) {
152 const std::size_t num_species = spec.upper_cs.size();
153 for (std::size_t s = 1; s < num_species; ++s) {
154 UPPER(ndx::rho_(s)) = upper_rho * spec.upper_cs[s];
155 }
156 } else {
157 // NOP
158 }
159
160#undef LOWER
161#undef UPPER
162
163 }

167void operator_mass_isothermal::invert_mass_plus_scaled_operator(
168 const complex_t &phi,
169 multi_array::ref<complex_t,4> &state,
170 const lowstorage::method_interface<complex_t> &method,
171 const component delta_t,
172 const std::size_t substep_index,
173 multi_array::ref<complex_t,4> *ic0) const
174{
175 // State enters method as coefficients in X and Z directions
176 // State enters method as collocation point values in Y direction
177
178 // Prepare functor setting pointwise BCs given lower density locations.
179 // Applies these to BOTH lower and upper boundaries given only lower one!
180 SUZERAIN_ENSURE(state.shape()[1] == (unsigned) grid.N.y());
181 const isothermal_enforcer enforcer(this->grid,
182 state.strides()[0], // field_stride
183 this->spec,
184 this->lower_E(spec.lower_T,
185 spec.lower_u,
186 spec.lower_v,
187 spec.lower_w,
188 spec.lower_cs),
189 this->upper_E(spec.upper_T,
190 spec.upper_u,
191 spec.upper_v,
192 spec.upper_w,
193 spec.upper_cs));
194
195 // Apply the enforcer to the lower boundary densities.
196 typedef boost::multi_array_types::index_range range;
197 multi_array::ref<complex_t,4>::array_view<2>::type state_view
198 = state[boost::indices[ndx::rho][0][range()][range()]];
199 multi_array::for_each(state_view,
200 boost::bind<void>(boost::cref(enforcer), _1));
201
202 // Apply zero-zero mode boundary conditions to any requested constraints.
203 if (ic0) {
204 SUZERAIN_ENSURE(ic0->shape()[0] == state.shape()[0]);
205 SUZERAIN_ENSURE(ic0->shape()[1] == state.shape()[1]);
206 multi_array::ref<complex_t,4>::array_view<2>::type ic0_view
207 = (*ic0)[boost::indices[ndx::rho][0][range()][range()]];
208 multi_array::for_each(ic0_view,
209 boost::bind<void>(boost::cref(enforcer), _1));
210 }
211
212 // Perform the usual operator_mass solve across all equations
213 base::invert_mass_plus_scaled_operator(
214 phi, state, method, delta_t, substep_index, ic0);
215
216 // State leaves method as coefficients in X, Y, and Z directions
217}
