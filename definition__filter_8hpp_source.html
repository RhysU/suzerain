189 template<typename MultiArrayX, typename MultiArrayY>
190 int source_accumulate(const specification_grid &grid,
191 const pencil_grid &dgrid,
192 const typename MultiArrayX::element& alpha,
193 const MultiArrayX &x,
194 int ndx_x,
195 const typename MultiArrayY::element& beta,
196 MultiArrayY &y,
197 int ndx_y) const
198 {
199 SUZERAIN_TIMER_SCOPED("definition_filter::source_accumulate");
200
201 assert(std::equal(x.shape() + 1, x.shape() + 4, y.shape() + 1));
202 assert(std::equal(x.strides() + 1, x.strides() + 4, y.strides() + 1));
203
204 if (SUZERAIN_UNLIKELY(!z || z->n != dgrid.global_wave_extent.y())) {
205 SUZERAIN_TIMER_SCOPED("definition_filter::prepare_complex");
206 const int err = prepare_complex(dgrid.global_wave_extent.y());
207 if (SUZERAIN_UNLIKELY(err)) return err;
208 }
209
210 return suzerain_filteropz_source_accumulate(
211 alpha, x[ndx_x].origin(),
212 beta, y[ndx_y].origin(),
213 dgrid.global_wave_extent.y(),
214 grid.N.x(),
215 grid.dN.x(),
216 dgrid.local_wave_start.x(),
217 dgrid.local_wave_end.x(),
218 grid.N.z(),
219 grid.dN.z(),
220 dgrid.local_wave_start.z(),
221 dgrid.local_wave_end.z(),
222 z.get());
223 }
