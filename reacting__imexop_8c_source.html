35void
36suzerain_reacting_flow_imexop_accumulate(
37 const complex_double phi,
38 const suzerain_reacting_imexop_scenario * const s,
39 const suzerain_reacting_imexop_ref * const r,
40 const suzerain_reacting_imexop_refld * const ld,
41 const suzerain_bsplineop_workspace * const w,
42 const int imagzero,
43 const complex_double *in_rho_E,
44 const complex_double *in_rho_u,
45 const complex_double *in_rho_v,
46 const complex_double *in_rho_w,
47 const complex_double *in_rho ,
48 const complex_double beta,
49 complex_double *out_rho_E,
50 complex_double *out_rho_u,
51 complex_double *out_rho_v,
52 complex_double *out_rho_w,
53 complex_double *out_rho )
54{
55 // When you modify this routine, you must also modify reacting_imexop.def so
56 // that operator accumulation-without-assembly and assembly match. The
57 // test cases in tests/test_reacting_imexop.cpp are invaluable in checking
58 // the coherence of these two pieces of logic.
59
60 // Sanity checks
61 assert(w->nderiv >= 2); // Adequate workspace?
62 assert(!(!in_rho_E ^ !out_rho_E)); // Both non-NULL or NULL?
63 assert(!(!in_rho_u ^ !out_rho_u)); // ditto
64 assert(!(!in_rho_v ^ !out_rho_v)); // ditto
65 assert(!(!in_rho_w ^ !out_rho_w)); // ditto
66 assert(!(!in_rho ^ !out_rho )); // ditto
67
68 // Prepare shorthand for some useful derived values
69 const double ap43 = s->alpha + 4.0/3.0;
70
71 // Accumulate the requested portions of the M + \varphi L operator. Scale
72 // output by beta, accumulate non-mass contributions, and finally
73 // accumulate the mass contributions. Mass contributions come last as they
74 // are expected to have magnitudes much larger than phi-- this helps to
75 // ensure better rounding of \varphi L contributions.
76
77 // Two use cases are accommodated. The first (and vastly more common) case
78 // is applying the operator to in_rho{,u,v,w,e} as provided. The second
79 // case applies the operator for Im(in_rho{,u,v,w,e}) == 0. Imaginary
80 // parts are entirely ignored. This is useful to handle the 0th and
81 // Nyquist modes in an even-length DFT without resorting to auxiliary
82 // buffers, different memory access patterns, or duplicated code.
83 //
84 // For brevity, believe it or not, typedef some gbmv-like function
85 // signatures operations but without a (const void *) x vector type:
86 typedef int gbmv_t(const char, const int, const int, const int, const int,
87 const complex_double, const double *, const int,
88 /* NB */ const void *, const int,
89 const complex_double, complex_double *, const int);
90 typedef int gbdmv_t (const char, const int, const int, const int,
91 const complex_double, const double *, const int,
92 const double *, const int,
93 /* NB */ const void *, const int,
94 const complex_double, complex_double *, const int);
95 typedef int gbddmv_t (const char, const int, const int, const int,
96 const complex_double, const double *, const int,
97 const complex_double, const double *, const int,
98 const double *, const int,
99 /* NB */ const void *, const int,
100 const complex_double, complex_double *, const int);
101
102 // In the first case, treat x arguments as complex with stride one...
103 int inc_in = 1;
104 gbmv_t *p_gbmv = (gbmv_t *) & suzerain_blas_zgbmv_d_z;
105 gbdmv_t *p_gbdmv = (gbdmv_t *) & suzerain_blasext_zgbdmv_d_z;
106 gbddmv_t *p_gbddmv = (gbddmv_t *) & suzerain_blasext_zgbddmv_d_z;
107
108 // ...but in the second case, treat x arguments as real with stride two.
109 if (SUZERAIN_UNLIKELY(imagzero)) {
110 inc_in = 2;
111 p_gbmv = (gbmv_t *) & suzerain_blas_zgbmv_d_d;
112 p_gbdmv = (gbdmv_t *) & suzerain_blasext_zgbdmv_d_d;
113 p_gbddmv = (gbddmv_t *) & suzerain_blasext_zgbddmv_d_d;
114 }
115
116 // Shorthand for the common pattern of providing "in_foo, inc" pairs.
117# define IN(quantity) in_##quantity, inc_in
118# define OUT(quantity) out_##quantity, 1
119
120 // Shorthand for the common pattern of providing a "r->foo, ld->foo" pair.
121# define REF(quantity) r->quantity, ld->quantity
122
123 // Just plain shorthand
124# define LIKELY(expr) SUZERAIN_LIKELY(expr)
125
126 // We need to account for suzerain_bsplineop_workspace storing the
127 // transpose of the operators when we invoke suzerain_blaseext_* routines.
128 static const char trans = 'T';
129
130 // Prepare several oft-used constants to aid readability
131 static const int M = 0;
132 static const int D1 = 1;
133 static const int D2 = 2;
134 const int n = w->n;
135
136
137 if (LIKELY(in_rho_E)) { // Accumulate total energy terms into out_rho_E
138
139 suzerain_blas_zscal(n, beta, OUT(rho_E));
140
141 /* in_rho_E */ {
142 (*p_gbdmv)(trans, n, w->kl[D1], w->ku[D1],
143 -phi, REF(vp_rE),
144 w->D_T[D1], w->ld, IN(rho_E), 1.0, OUT(rho_E));
145
146 (*p_gbdmv)(trans, n, w->kl[D2], w->ku[D2],
147 phi, REF(korCv),
148 w->D_T[D2], w->ld, IN(rho_E), 1.0, OUT(rho_E));
149 }
150
151 if (LIKELY(in_rho_u)) {
152
153 (*p_gbdmv)(trans, n, w->kl[D1], w->ku[D1],
154 -phi, REF(vp_ru),
155 w->D_T[D1], w->ld, IN(rho_u), 1.0, OUT(rho_E));
156
157 }
158
159 if (LIKELY(in_rho_v)) {
160
161 (*p_gbdmv)(trans, n, w->kl[D1], w->ku[D1],
162 phi, REF(Ce_rv),
163 w->D_T[D1], w->ld, IN(rho_v), 1.0, OUT(rho_E));
164
165 }
166
167 if (LIKELY(in_rho_w)) {
168
169 (*p_gbdmv)(trans, n, w->kl[D1], w->ku[D1],
170 -phi, REF(vp_rw),
171 w->D_T[D1], w->ld, IN(rho_w), 1.0, OUT(rho_E));
172
173 }
174
175 if (LIKELY(in_rho)) {
176
177 (*p_gbdmv)(trans, n, w->kl[D1], w->ku[D1],
178 phi, REF(Ce_rho),
179 w->D_T[D1], w->ld, IN(rho), 1.0, OUT(rho_E));
180
181 }
182
183 (*p_gbmv)(trans, n, n, w->kl[M], w->ku[M],
184 1.0, w->D_T[M], w->ld, IN(rho_E), 1.0, OUT(rho_E));
185 }
186
187 if (LIKELY(in_rho_u)) { // Accumulate X momentum terms into out_rho_u
188
189 suzerain_blas_zscal(n, beta, OUT(rho_u));
190
191 if (LIKELY(in_rho_E)) {/* NOP */};
192
193 /* in_rho_u */ {
194
195 (*p_gbdmv)(trans, n, w->kl[D2], w->ku[D2],
196 phi, REF(nu),
197 w->D_T[D2], w->ld, IN(rho_u), 1.0, OUT(rho_u));
198 }
199
200 if (LIKELY(in_rho_v)) {
201
202 (*p_gbdmv)(trans, n, w->kl[D1], w->ku[D1],
203 -phi, REF(ux),
204 w->D_T[D1], w->ld, IN(rho_v), 1.0, OUT(rho_u));
205 }
206
207 if (LIKELY(in_rho_w)) {/* NOP */};
208
209 if (LIKELY(in_rho)) {
210
211 (*p_gbdmv)(trans, n, w->kl[D1], w->ku[D1],
212 phi, REF(uxuy),
213 w->D_T[D1], w->ld, IN(rho), 1.0, OUT(rho_u));
214
215 }
216
217
218 (*p_gbmv)(trans, n, n, w->kl[M], w->ku[M],
219 1.0, w->D_T[M], w->ld, IN(rho_u), 1.0, OUT(rho_u));
220 }
221
222 if (LIKELY(in_rho_v)) { // Accumulate Y momentum terms into out_rho_v
223
224 suzerain_blas_zscal(n, beta, OUT(rho_v));
225
226 if (LIKELY(in_rho_E)) {
227
228 (*p_gbdmv)(trans, n, w->kl[D1], w->ku[D1],
229 -phi, REF(p_rE),
230 w->D_T[D1], w->ld, IN(rho_E), 1.0, OUT(rho_v));
231 }
232
233
234 if (LIKELY(in_rho_u)) {
235
236 (*p_gbdmv)(trans, n, w->kl[D1], w->ku[D1],
237 -phi, REF(p_ru),
238 w->D_T[D1], w->ld, IN(rho_u), 1.0, OUT(rho_v));
239 }
240
241 /* in_rho_v */ {
242
243 (*p_gbddmv)(trans, n, w->kl[D1], w->ku[D1],
244 -phi, REF(uy),
245 phi, REF(vp_rE), // FIXME: correct but misleading (since vp_rE = v*(gam-1))
246 w->D_T[D1], w->ld, IN(rho_v), 1.0, OUT(rho_v));
247
248 (*p_gbdmv)(trans, n, w->kl[D2], w->ku[D2],
249 phi*ap43, REF(nu),
250 w->D_T[D2], w->ld, IN(rho_v), 1.0, OUT(rho_v));
251 }
252
253 if (LIKELY(in_rho_w)) {
254
255 (*p_gbdmv)(trans, n, w->kl[D1], w->ku[D1],
256 -phi, REF(p_rw),
257 w->D_T[D1], w->ld, IN(rho_w), 1.0, OUT(rho_v));
258 }
259
260 if (LIKELY(in_rho)) {
261
262 (*p_gbdmv)(trans, n, w->kl[D1], w->ku[D1],
263 phi, REF(Cmy_rho),
264 w->D_T[D1], w->ld, IN(rho), 1.0, OUT(rho_v));
265
266 }
267
268 (*p_gbmv)(trans, n, n, w->kl[M], w->ku[M],
269 1.0, w->D_T[M], w->ld, IN(rho_v), 1.0, OUT(rho_v));
270 }
271
272 if (LIKELY(in_rho_w)) { // Accumulate Z momentum terms into out_rho_w
273
274 suzerain_blas_zscal(n, beta, OUT(rho_w));
275
276 if (LIKELY(in_rho_E)) {/* NOP */};
277
278 if (LIKELY(in_rho_u)) {/* NOP */};
279
280 if (LIKELY(in_rho_v)) {
281
282 (*p_gbdmv)(trans, n, w->kl[D1], w->ku[D1],
283 -phi, REF(uz),
284 w->D_T[D1], w->ld, IN(rho_v), 1.0, OUT(rho_w));
285 }
286
287 /* in_rho_w */ {
288
289 (*p_gbdmv)(trans, n, w->kl[D2], w->ku[D2],
290 phi, REF(nu),
291 w->D_T[D2], w->ld, IN(rho_w), 1.0, OUT(rho_w));
292 }
293
294 if (LIKELY(in_rho)) {
295
296 (*p_gbdmv)(trans, n, w->kl[D1], w->ku[D1],
297 phi, REF(uyuz),
298 w->D_T[D1], w->ld, IN(rho), 1.0, OUT(rho_w));
299
300 }
301
302 (*p_gbmv)(trans, n, n, w->kl[M], w->ku[M],
303 1.0, w->D_T[M], w->ld, IN(rho_w), 1.0, OUT(rho_w));
304 }
305
306 if (LIKELY(in_rho )) { // Accumulate density terms into out_rho
307
308 suzerain_blas_zscal(n, beta, OUT(rho));
309
310 if (LIKELY(in_rho_E)) {/* NOP */};
311
312 if (LIKELY(in_rho_u)) {/* NOP */};
313
314 if (LIKELY(in_rho_v)) {/* NOP */};
315
316 if (LIKELY(in_rho_w)) {/* NOP */};
317
318 (*p_gbmv)(trans, n, n, w->kl[M], w->ku[M],
319 1.0, w->D_T[M], w->ld, IN(rho), 1.0, OUT(rho));
320 }
321
322# undef LIKELY
323# undef REF
324# undef IN
325# undef OUT
326
327}
