54definition_noise::options_description()
55{
56 using boost::bind;
57 using boost::program_options::options_description;
58 using boost::program_options::value;
59 using std::string;
60 using validation::ensure_nonnegative;
61 using validation::ensure_positive;
62
63 options_description retval(
64 "Additive random velocity perturbations on startup");
65
66 retval.add_options()
67 ("fluct_percent", value(&percent)->default_value(percent)
68 ->notifier(bind(&ensure_nonnegative<real_t>, _1,
69 "fluct_percent")),
70 "Maximum fluctuation magnitude to add as a percentage of"
71 " centerline mean streamwise velocity")
72 ("fluct_kxfrac", value<string>()->default_value("0:1")
73 ->notifier(bind(&exprparse_range<const string&,real_t>, _1,
74 &kxfrac_min, &kxfrac_max,
75 0, 1, 0, 1, "fluct_kxfrac")),
76 "Range of X wavenumbers in which to generate fluctuations")
77 ("fluct_kzfrac", value<string>()->default_value("0:1")
78 ->notifier(bind(&exprparse_range<const string&,real_t>, _1,
79 &kzfrac_min, &kzfrac_max,
80 0, 1, 0, 1, "fluct_kzfrac")),
81 "Range of Z wavenumbers in which to generate fluctuations")
82 ("fluct_seed", value(&seed)->default_value(seed)
83 ->notifier(bind(&ensure_positive<unsigned long>, _1,
84 "fluct_seed")),
85 "rngstream generator seed (L'Ecuyer et al. 2002)")
86 ;
87
88 return retval;
