73std::vector<real_t> treatment_nonreflecting::apply_operator(
74 const real_t time,
75 contiguous_state<4,complex_t> &swave,
76 const lowstorage::method_interface<complex_t> &method,
77 const std::size_t substep_index) const
78{
79 // Implementation approach:
80 // 1) Preserve the wavenumber-dependent state at the upper boundary
81 // 2) Invoke the wrapped nonlinear operator in the usual fashion
82 // 3) Build the various matrices we need from the reference state.
83 // 4) Prepare pre-computable products of the various matrices
84 // 5) Modify the right hand side in a wave-number dependent fashion.
85
86 // State enters method as coefficients in X, Y, and Z directions
87
88 // Make a local, stride-1 copy of I times upper boundary state point values.
89 // Notice that boundary coefficients are 1-1 with boundary point values.
90 // That is, applying the mass matrix to the boundary is an ignorable NOP.
91
92 // Number of state variables
93 const size_t state_count = swave.shape()[0];
94 const size_t Ns = state_count - 4;
95
96 // Name indices to make (hopefully) the code more legible
97 // Local indices for primitive variables U=(rho_s(0,...,Ns-1),u,v,w,p)
98 // indices of species densities start at 0 and run through Ns-1
99 // index 0 corresponds to the diluter
100 const int irhos0 = 0;
101 const int iu = Ns;
102 const int iv = Ns+1;
103 const int iw = Ns+2;
104 const int ip = Ns+3;
105
106 // Local indices for conserved variables V=(rho,rho_s(1,...,Ns-1),rhoU,rhoV,rhoW,rhoE)
107 // indices of species densities start at irho+1 and run through irho+Ns-1
108 const int irho = 0;
109 const int irhoU = Ns;
110 const int irhoV = Ns+1;
111 const int irhoW = Ns+2;
112 const int irhoE = Ns+3;
113
114 // Indices for characteristic (ic*) variables
115 const int icrhos0 = 0;
116 const int icv = Ns;
117 const int icw = Ns+1;
118 const int icu_p = Ns+2; // u+a characteristic
119 const int icu_m = Ns+3; // u-a characteristic
120
121 MatrixXXc i_stash(state_count, swave.shape()[2] * swave.shape()[3]);
122 {
123 const complex_t imag_unit(0, 1);
124 const int ku = boost::numeric_cast<int>(swave.shape()[0]);
125 const int l = boost::numeric_cast<int>(swave.shape()[1]) - 1; // Upper
126 const int mu = boost::numeric_cast<int>(swave.shape()[2]);
127 const int nu = boost::numeric_cast<int>(swave.shape()[3]);
128 for (int k = 0; k < ku; ++k) {
129 for (int n = 0; n < nu; ++n) {
130 for (int m = 0; m < mu; ++m) {
131 i_stash(k, m + n * mu) = imag_unit * swave[k][l][m][n];
132 }
133 }
134 }
135 }
136
137 // Prepare storage for chemistry sources to get added later
138 common.chemsrcw.resize(Ns-1, swave.shape()[2] * swave.shape()[3]);
139 common.chemsrcw.setZero();
140
141 // Invoke the wrapped nonlinear operator (which may compute references!)
142 const std::vector<real_t> retval = N->apply_operator(
143 time, swave, method, substep_index);
144
145 // State is now coefficients in X and Z directions
146 // State is now collocation point values in Y direction
147
148 // Prepare the rotation and its inverse that reorders from
149 // ndx::{e, mx, my, mz, rho, rho_s} to {rho_s, rho, my, mz, mx, e}.
150 // All remaining matrices/logic within the routine uses the latter order!
151 // FIXME
152 MatrixXXr RY(MatrixXXr::Zero(state_count, state_count));
153 {
154 for (unsigned int is_local = 1; is_local < Ns; ++is_local) {
155 RY(irho+is_local, ndx::rho+is_local) = 1;
156 }
157 RY(irho, ndx::rho) = 1;
158 RY(irhoU, ndx::my ) = 1;
159 RY(irhoV, ndx::mz ) = 1;
160 RY(irhoW, ndx::mx ) = 1;
161 RY(irhoE, ndx::e ) = 1;
162 }
163 // FIXME
164 MatrixXXr inv_RY(MatrixXXr::Zero(state_count, state_count));
165 {
166 inv_RY(ndx::e , irhoE) = 1;
167 inv_RY(ndx::mx , irhoW) = 1;
168 inv_RY(ndx::my , irhoU) = 1;
169 inv_RY(ndx::mz , irhoV) = 1;
170 inv_RY(ndx::rho, irho ) = 1;
171 for (unsigned int is_local = 1; is_local < Ns; ++is_local) {
172 inv_RY(ndx::rho+is_local, irho+is_local) = 1;
173 }
174 }
175
176 // Ny from grid
177 const int Ny = dgrid.global_wave_extent.y();
178
179 // Retrieve upper boundary reference state from the common block.
180 // These assignments are odd looking to permit others as documented.
181 const real_t rho = common.rho_ref; // swave[ndx::rho][Ny-1][0][0].real(); //FIXME: Check if I get mean(rho) at the upper boundary
182 const real_t u = common.v_ref; // common.ref_uy ().tail<1>()[0]; // Ref u = v' per RY
183 const real_t v = common.w_ref; // common.ref_uz ().tail<1>()[0]; // Ref v = w' per RY
184 const real_t w = common.u_ref; // common.ref_ux ().tail<1>()[0]; // Ref w = u' per RY
185 const real_t a = common.a_ref; // common.ref_a ().tail<1>()[0];
186 const real_t gamma = common.gamma_ref; // common.ref_gamma().tail<1>()[0];
187 const real_t T = common.T_ref;
188 const real_t Cv = common.Cv_ref;
189
190 VectorXr etots;
191 etots.resize(Ns);
192 etots = common.etots_ref;
193
194 // Update mean species densities
195 // ... only at initial substep
196 if (substep_index == 0 ) {
197 rhos.resize(Ns);
198 // Initialize diluter density to total density
199 rhos(0) = rho;
200 DEBUG0(who, "rho = " << rho );
201 for (unsigned int is_local = 1; is_local < Ns; ++is_local) {
202 rhos(is_local) = rho * common.cs_ref(is_local);
203 DEBUG0(who, "rhos(" << is_local << ") = " << rhos(is_local));
204
205 // subtract to compute diluter density
206 rhos(0) -= rhos(is_local);
207 }
208 DEBUG0(who, "rhos(0) = " << rhos(0) );
209 }
210
211
212 // Prepare oft-used quantities derived from the reference state
213 const real_t inv_rho = 1 / rho;
214 const real_t u2 = u * u;
215 const real_t v2 = v * v;
216 const real_t w2 = w * w;
217 real_t inv_a = 1 / a;
218 real_t a2 = a * a;
219 real_t inv_a2 = 1 / a2;
220
221 // Prepare oft-used scenario-related constants
222 const real_t half = static_cast<real_t>(1) / 2;
223 const real_t gamma1 = gamma - 1;
224 const real_t inv_gamma1 = 1 / gamma1;
225
226 // Build the conserved -> primitive transformation and its inverse
227 MatrixXXr S(MatrixXXr::Zero(state_count, state_count));
228 {
229 S(irhos0, irho) = 1;
230 S(iu, irho) = - u * inv_rho;
231 S(iv, irho) = - v * inv_rho;
232 S(iw, irho) = - w * inv_rho;
233 S(ip, irho) = (half*(u2+v2+w2)-etots(0)+Cv*T)*gamma1;
234
235 for (unsigned int is_local = 1; is_local < Ns; ++is_local) {
236 S(irhos0+is_local, irho+is_local) = 1;
237 S(irhos0, irho+is_local) = -1;
238 S(ip, irho+is_local) = (-etots(is_local)
239 +etots(0))*gamma1;
240 }
241
242 S(iu, irhoU) = inv_rho;
243 S(ip, irhoU) = - gamma1 * u;
244
245 S(iv, irhoV) = inv_rho;
246 S(ip, irhoV) = - gamma1 * v;
247
248 S(iw, irhoW) = inv_rho;
249 S(ip, irhoW) = - gamma1 * w;
250
251 S(ip, irhoE) = gamma1;
252 }
253 MatrixXXr inv_S(MatrixXXr::Zero(state_count, state_count));
254 {
255 for (unsigned int is_local = 0; is_local < Ns; ++is_local) {
256 inv_S(irho, irhos0+is_local) = 1;
257 inv_S(irho+is_local, irhos0+is_local) = 1;
258 inv_S(irhoU, irhos0+is_local) = u ;
259 inv_S(irhoV, irhos0+is_local) = v ;
260 inv_S(irhoW, irhos0+is_local) = w ;
261 inv_S(irhoE, irhos0+is_local) = half * (u2+v2+w2)
262 + etots(is_local)
263 - Cv * T;
264 }
265
266 inv_S(irhoU, iu) = rho;
267 inv_S(irhoE, iu) = rho * u;
268
269 inv_S(irhoV, iv) = rho;
270 inv_S(irhoE, iv) = rho * v;
271
272 inv_S(irhoW, iw) = rho;
273 inv_S(irhoE, iw) = rho * w;
274
275 inv_S(irhoE, ip) = inv_gamma1;
276 }
277
278 // Build the primitive -> characteristic transformation and its inverse
279 MatrixXXr VL(MatrixXXr::Zero(state_count, state_count));
280 {
281 for (unsigned int is_local = 0; is_local < Ns; ++is_local) {
282 VL(icrhos0+is_local, irhos0+is_local) = -a2;
283 }
284
285 VL(icu_p, iu) = rho * a;
286 VL(icu_m, iu) = - rho * a;
287
288 VL(icv, iv) = rho * a;
289
290 VL(icw, iw) = rho * a;
291
292 for (unsigned int is_local = 0; is_local < Ns; ++is_local) {
293 VL(icrhos0+is_local, ip) = rhos(is_local) * inv_rho;
294 }
295 VL(icu_p, ip) = 1;
296 VL(icu_m, ip) = 1;
297 }
298 MatrixXXr inv_VL(MatrixXXr::Zero(state_count, state_count));
299 {
300 // First Ns rows
301 for (unsigned int is_local = 0; is_local < Ns; ++is_local) {
302 inv_VL(irhos0+is_local, icrhos0+is_local) = - inv_a2;
303 inv_VL(irhos0+is_local, icu_p ) =
304 half * rhos(is_local) * inv_rho * inv_a2;
305 inv_VL(irhos0+is_local, icu_m ) =
306 half * rhos(is_local) * inv_rho * inv_a2;
307 }
308
309 // Last 4 rows
310 inv_VL(iu, icu_p) = half * inv_rho * inv_a;
311 inv_VL(iu, icu_m) = - half * inv_rho * inv_a;
312
313 inv_VL(iv, icv ) = inv_rho * inv_a;
314
315 inv_VL(iw, icw ) = inv_rho * inv_a;
316
317 inv_VL(ip, icu_p) = half;
318 inv_VL(ip, icu_m) = half;
319 }
320
321 // The upper boundary is an inflow when the reference velocity is negative.
322 // In the event of u == 0, treat the boundary like an outflow.
323 const bool inflow = u < 0;
324
325 // Build the in-vs-outflow characteristic-preserving projection
326 // FIXME: Incoming waves for inflow or outflow
327 // correspond to SUBSONIC UPPER BOUNDARY
328 // (Generalize implementation for arbitrary application to
329 // lower or upper boundary and supersonic flow?)
330 MatrixXXr PG(MatrixXXr::Zero(state_count, state_count));
331 if (inflow) {
332 for (unsigned int is_local = 0; is_local < Ns; ++is_local) {
333 PG(irhos0+is_local, irhos0+is_local) = 1;
334 }
335 PG(Ns, Ns ) = 1;
336 PG(Ns+1, Ns+1) = 1;
337 PG(Ns+3, Ns+3) = 1;
338 } else {
339 PG(Ns+3, Ns+3) = 1;
340 }
341
342// MatrixXXr BG(MatrixXXr::Zero(state_count, state_count)); // Medida's B^G_1
343 // Declare this matrix for transverse wave approximation,
344 // see perfect implementation/documentation, and the original
345 // work of Giles and Medida
346
347// MatrixXXr CG(MatrixXXr::Zero(state_count, state_count)); // Medida's C^G_1
348 // Declare this matrix for transverse wave approximation,
349 // see perfect implementation/documentation, and the original
350 // work of Giles and Medida
351
352 // Prepare all necessary real-valued products of the above matrices
353 const MatrixXXr VL_S_RY = VL * S * RY;
354 // FIXME: Uncomment if BG, CG are declared
355// const Matrix5r BG_VL_S_RY_by_chi = BG * VL_S_RY / chi;
356// const Matrix5r CG_VL_S_RY_by_chi = CG * VL_S_RY / chi;
357 const MatrixXXr ImPG_VL_S_RY = (MatrixXXr::Identity(state_count, state_count) - PG) * VL_S_RY;
358 const MatrixXXr inv_VL_S_RY = inv_RY * inv_S * inv_VL;
359
360 // Wavenumber traversal modeled after those found in suzerain/diffwave.c
361 // Ny previously declared
362 const int dNx = grid.dN.x();
363 const int dkbx = dgrid.local_wave_start.x();
364 const int dkex = dgrid.local_wave_end.x();
365 const int dNz = grid.dN.z();
366 const int dkbz = dgrid.local_wave_start.z();
367 const int dkez = dgrid.local_wave_end.z();
368 const real_t twopioverLx = twopiover(grid.L.x()); // Weird looking...
369 const real_t twopioverLz = twopiover(grid.L.z()); // ...for FP control
370
371 // Traverse wavenumbers updating the RHS with the Giles boundary condition
372 using suzerain::inorder::wavenumber;
373 const int mu = boost::numeric_cast<int>(swave.shape()[2]);
374 VectorXc N(state_count);
375 VectorXc tmp(state_count);
376 for (int n = dkbz; n < dkez; ++n) {
377 const int wn = wavenumber(dNz, n);
378 const real_t kn = twopioverLz*wn;
379 SUZERAIN_UNUSED(kn);
380
381 for (int m = dkbx; m < dkex; ++m) {
382 const int wm = wavenumber(dNx, m);
383 const real_t km = twopioverLx*wm;
384 SUZERAIN_UNUSED(km);
385
386 // Unpack upper boundary RHS into a contiguous buffer
387 for (int f = 0; f < N.size(); ++f) {
388 N(f) = swave[f][Ny - 1][m - dkbx][n - dkbz];
389 }
390
391 // Modify the packed RHS per
392 // "Implementation primarily within the nonlinear explicit operator"
393 // The imaginary unit has already been included within i_stash.
394 tmp.noalias() = ImPG_VL_S_RY.cast<complex_t>() * N;
395 // FIXME: Uncomment if BG, CG are declared
396// tmp.noalias() -= kn
397// * BG_VL_S_RY_by_chi.cast<complex_t>()
398// * i_stash.col((m - dkbx) + mu*(n - dkbz));
399// tmp.noalias() -= km
400// * CG_VL_S_RY_by_chi.cast<complex_t>()
401// * i_stash.col((m - dkbx) + mu*(n - dkbz));
402 N.noalias() = inv_VL_S_RY.cast<complex_t>() * tmp;
403
404 // Pack new upper boundary RHS from the contiguous buffer
405 for (int f = 0; f < N.size(); ++f) {
406 swave[f][Ny - 1][m - dkbx][n - dkbz] = N(f);
407 }
408
409 // Add chemistry sources if the bc is an inflow
410 if (inflow) {
411 assert(Ns >= 1);
412 for (unsigned int is = 0; is<Ns-1; ++is) {
413 int f = is + (N.size()-Ns+1);
414 swave[f][Ny - 1][m - dkbx][n - dkbz] +=
415 common.chemsrcw(is, (m - dkbx) + mu*(n - dkbz));
416 }
417 }
418 }
419 }
420
421 return retval;
422}
