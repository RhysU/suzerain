331 const Scalar &gamma,
332 const Scalar &Ma,
333 const Scalar &rho,
334 const Vector &grad_rho,
335 const Vector &m,
336 const Scalar &div_m,

342 const VectorCoefficient1 &refcoeff_grad_rho,

534template<typename Scalar,
535 typename Vector,
536 typename Tensor,
537 typename ScalarCoefficient1,
538 typename ScalarCoefficient2,
539 typename VectorCoefficient >
540inline
541Scalar explicit_mu_div_grad_T(
542 const Scalar &gamma,
543 const Scalar &Ma,
544 const Scalar &mu,
545 const Scalar &rho,
546 const Vector &grad_rho,
547 const Scalar &div_grad_rho,
548 const Vector &m,
549 const Tensor &grad_m,
550 const Vector &div_grad_m,
551 const Scalar &e,
552 const Scalar &div_grad_e,
553 const Scalar &p,
554 const Vector &grad_p,
555 const ScalarCoefficient1 &refcoeff_div_grad_e,
556 const VectorCoefficient &refcoeff_div_grad_m,
557 const ScalarCoefficient2 &refcoeff_div_grad_rho)
558{
559 const Scalar rho_inverse = 1/rho;
560 const Scalar rho_inverse2 = rho_inverse*rho_inverse;
561 const Scalar coeff_div_grad_e(
562 explicit_mu_div_grad_T_refcoeff_div_grad_e(mu, rho)
563 - refcoeff_div_grad_e);
564 const Vector coeff_div_grad_m(
565 explicit_mu_div_grad_T_refcoeff_div_grad_m(mu, rho, m)
566 - refcoeff_div_grad_m);
567 const Scalar coeff_div_grad_rho(
568 explicit_mu_div_grad_T_refcoeff_div_grad_rho(gamma, mu, rho, e, p)
569 - refcoeff_div_grad_rho);
570
571 return gamma*(
572 coeff_div_grad_rho*div_grad_rho
573 - 2*mu*rho_inverse2*grad_rho.dot(
574 grad_p - rho_inverse*p*grad_rho
575 )
576 + (gamma-1)*(
577 coeff_div_grad_e*div_grad_e
578 - Ma * Ma * (
579 mu*rho_inverse2*(
580 grad_m.squaredNorm() // Frobenius norm
581 - rho_inverse*(
582 2*grad_rho.dot(grad_m.transpose()*m)
583 - rho_inverse*m.squaredNorm()*grad_rho.squaredNorm()
584 )
585 )
586 + div_grad_m.dot(coeff_div_grad_m)
587 )
588 )
589 );
590}

711template<typename Scalar,
712 typename Vector,
713 typename Tensor >
714inline
715void p_T(
716 const Scalar &alpha,
717 const Scalar &beta,
718 const Scalar &gamma,
719 const Scalar &Ma,
720 const Scalar &rho,
721 const Vector &grad_rho,
722 const Vector &m,
723 const Tensor &grad_m,
724 const Scalar &e,
725 const Vector &grad_e,
726 Scalar &p,
727 Vector &grad_p,
728 Scalar &T,
729 Vector &grad_T)
730{
731 SUZERAIN_UNUSED(alpha); // Present for API consistency
732 SUZERAIN_UNUSED(beta); // Present for API consistency
733 const Scalar rho_inverse = 1/rho;
734 const Scalar half_rho_inverse = rho_inverse/2;
735 const Scalar gamma1 = gamma - 1;
736 const Scalar squared_Ma = Ma * Ma;
737
738 // Compute scalar quantities
739 p = gamma1*(e - squared_Ma*half_rho_inverse*m.squaredNorm());
740 T = gamma * p * rho_inverse;
741
742 // Compute vector quantities
743 grad_p = gamma1*(
744 grad_e + squared_Ma*rho_inverse*(
745 (half_rho_inverse*m.squaredNorm())*grad_rho
746 - grad_m.transpose()*m
747 )
748 );
749 grad_T = rho_inverse*(gamma*grad_p - T*grad_rho);
750}

768template<typename Scalar,
769 typename Vector >
770inline
771void p_T_mu_lambda(
772 const Scalar &alpha,
773 const Scalar &beta,
774 const Scalar &gamma,
775 const Scalar &Ma,
776 const Scalar &rho,
777 const Vector &m,
778 const Scalar &e,
779 Scalar &p,

781 Scalar &mu,
782 Scalar &lambda)
783{
784 using std::pow;
785 const Scalar rho_inverse = 1/rho;
786
787 // Compute scalar quantities
788 p = (gamma - 1)*(e - Ma*Ma*rho_inverse*m.squaredNorm()/2);

790 mu = pow(T, beta);
791 lambda = (alpha - Scalar(2)/3)*mu;

838template<typename Scalar,
839 typename Vector,
840 typename Tensor >
841inline
842void p_T_mu_lambda(
843 const Scalar &alpha,
844 const Scalar &beta,
845 const Scalar &gamma,
846 const Scalar &Ma,
847 const Scalar &rho,
848 const Vector &grad_rho,
849 const Vector &m,
850 const Tensor &grad_m,
851 const Scalar &e,
852 const Vector &grad_e,
853 Scalar &p,
854 Vector &grad_p,
855 Scalar &T,
856 Vector &grad_T,
857 Scalar &mu,
858 Vector &grad_mu,
859 Scalar &lambda,
860 Vector &grad_lambda)
861{
862 using std::pow;
863 const Scalar rho_inverse = 1/rho;
864 const Scalar half_rho_inverse = rho_inverse/2;
865 const Scalar gamma1 = gamma - 1;
866 const Scalar alpha23 = alpha - Scalar(2)/3;
867 const Scalar squared_Ma = Ma * Ma;
868
869 // Compute scalar quantities
870 p = gamma1*(e - squared_Ma*half_rho_inverse*m.squaredNorm());
871 T = gamma * p * rho_inverse;
872
873 const Scalar T_to_beta1 = pow(T, beta - 1); // Avoid div for grad_mu
874
875 mu = T_to_beta1 * T;
876 lambda = alpha23*mu;
877
878 // Compute vector quantities
879 grad_p = gamma1*(
880 grad_e + squared_Ma*rho_inverse*(
881 (half_rho_inverse*m.squaredNorm())*grad_rho
882 - grad_m.transpose()*m
883 )
884 );
885 grad_T = rho_inverse*(gamma*grad_p - T*grad_rho);
886 grad_mu = (beta * T_to_beta1)*grad_T;
887 grad_lambda = alpha23*grad_mu;
888}

1025template<typename Scalar,
1026 typename Vector,
1027 typename Tensor >
1028inline
1029Scalar div_grad_p(
1030 const Scalar &gamma,
1031 const Scalar &Ma,
1032 const Scalar &rho,
1033 const Vector &grad_rho,
1034 const Scalar &div_grad_rho,
1035 const Vector &m,
1036 const Tensor &grad_m,
1037 const Vector &div_grad_m,
1038 const Scalar &e,
1039 const Vector &grad_e,
1040 const Scalar &div_grad_e)
1041{
1042 SUZERAIN_UNUSED(e);
1043 SUZERAIN_UNUSED(grad_e);
1044
1045 const Scalar rho_inverse = 1/rho;
1046 const Scalar m_dot_m = m.squaredNorm();
1047
1048 return (gamma - 1)*(
1049 div_grad_e
1050 - Ma*Ma*rho_inverse*(
1051 grad_m.squaredNorm() // Frobenius norm
1052 + div_grad_m.dot(m)
1053 - rho_inverse*(
1054 2*grad_rho.dot(grad_m.transpose()*m)
1055 + m_dot_m*div_grad_rho/2
1056 - rho_inverse*m_dot_m*grad_rho.squaredNorm()
1057 )
1058 )
1059 );
1060}
