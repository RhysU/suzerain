122// http://www.xemacs.org/Documentation/sources/xemacs/src/compiler.h.
123// Seemingly requires using typeof for both GCC/Intel correctness.
124#ifdef SUZERAIN_typeof
126#define SUZERAIN_CONST_CAST_VOIDP(l) \
127 (((union { const void *p; SUZERAIN_typeof(l) v; })(l)).p)

146# define SUZERAIN_GCC_DIAG_PRAGMA(x) SUZERAIN_GCC_DIAG_DO_PRAGMA(GCC diagnostic x)
147# define SUZERAIN_GCC_DIAG_OFF(x) SUZERAIN_GCC_DIAG_PRAGMA(push) \
148 SUZERAIN_GCC_DIAG_PRAGMA(ignored SUZERAIN_GCC_DIAG_JOINSTR(-W,x))
149# define SUZERAIN_GCC_DIAG_ON(x) SUZERAIN_GCC_DIAG_PRAGMA(pop)
