42static void parse_size_t(const std::string& s,
43 std::size_t* value,
44 const char* name)
45{
46 using std::floor;
47#pragma warning(push,disable:2259)
48 const real_t t = floor(exprparse<real_t>(s, name) + real_t(1)/2);
49#pragma warning(pop)
50 validation::ensure_nonnegative(t, name);
51 *value = t;
52}

104boost::program_options::options_description
105definition_arsel::options_description()
106{
107 using boost::bind;
108 using boost::lexical_cast;
109 using boost::program_options::bool_switch;
110 using boost::program_options::value;
111 using std::string;
112 using validation::ensure_positive;
113
114 boost::program_options::options_description retval(desc_location);
115
116 retval.add_options()
117 (name_minorder, value<string>(NULL)
118 ->value_name("MIN")
119 ->notifier(bind(&parse_size_t, _1, &minorder, name_minorder))
120 ->default_value(lexical_cast<string>(minorder)),
121 desc_minorder)
122 (name_maxorder, value<string>(NULL)
123 ->value_name("MAX")
124 ->notifier(bind(&parse_size_t, _1, &maxorder, name_maxorder))
125 ->default_value(lexical_cast<string>(maxorder)),
126 desc_maxorder)
127 (name_criterion,
128 value<std::string>()
129 ->notifier(bind(&definition_arsel::criterion, this, _1))
130 ->default_value(criterion()),
131 desc_criterion)
132 (name_absrho,
133 bool_switch(&absrho)->default_value(absrho),
134 desc_absrho)
135 (name_wlenT0, value<string>(NULL)
136 ->value_name("WLEN")
137 ->notifier(bind(&parse_option<real_t>, _1, &wlenT0,
138 &ensure_positive<real_t>, name_wlenT0))
139 ->default_value(lexical_cast<string>(wlenT0)),
140 desc_wlenT0)
141 ;
142
143 return retval;
144}

146void
147definition_arsel::save(
148 const esio_handle h) const
149{
150 DEBUG0("Storing definition_arsel parameters");
151
152 // Only root writes the containing location
153 const int one = 1;
154 int procid;
155 esio_handle_comm_rank(h, &procid);
156 esio_line_establish(h, 1, 0, (procid == 0 ? 1 : 0));
157 esio_line_write(h, location, &one, 0, desc_location);
158
159 // Everyone writes the metadata
160 const int absrho = this->absrho;
161 const int minorder = this->minorder;
162 const int maxorder = this->maxorder;
163 esio_attribute_write(h, location, attr_absrho, &absrho);
164 esio_attribute_write(h, location, attr_minorder, &minorder);
165 esio_attribute_write(h, location, attr_maxorder, &maxorder);
166 esio_attribute_write(h, location, attr_wlenT0, &wlenT0);
167 esio_string_set (h, location, attr_criterion, criterion().c_str());
168
169}
