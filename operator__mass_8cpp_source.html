56void operator_mass::apply_mass_plus_scaled_operator(
57 const complex_t &phi,
58 multi_array::ref<complex_t,4> &state,
59 const std::size_t substep_index) const
60{
61 SUZERAIN_UNUSED(phi);
62 SUZERAIN_UNUSED(substep_index);
63
64 // Verify required assumptions
65 SUZERAIN_ENSURE(state.strides()[1] == 1);
66 SUZERAIN_ENSURE(state.shape()[1] == static_cast<unsigned>(cop.n()));
67 SUZERAIN_ENSURE(multi_array::is_contiguous(state));
68
69 // Those assumptions holding, apply operator to each wall-normal pencil.
70 // Speedup may be possible by not applying to dealiased modes
71 const int nrhs = state.shape()[0]*state.shape()[2]*state.shape()[3];
72 cop.apply(0, nrhs, 1, state.data(), 1, state.shape()[1]);
73}

75void operator_mass::accumulate_mass_plus_scaled_operator(
76 const complex_t &phi,
77 const multi_array::ref<complex_t,4> &input,
78 const complex_t &beta,
79 contiguous_state<4,complex_t> &output,
80 const std::size_t substep_index) const
81{
82 SUZERAIN_UNUSED(phi);
83 SUZERAIN_UNUSED(substep_index);
84
85 SUZERAIN_ENSURE(output.is_isomorphic(input));
86
87 const multi_array::ref<complex_t,4> &x = input; // Shorthand
88 contiguous_state<4,complex_t> &y = output; // Shorthand
89 const complex_t c_one = 1;
90
91 // Sidesteps assertions triggered by dereferencing trivial input and output
92 if (SUZERAIN_UNLIKELY(0U == x.shape()[1] * x.shape()[2])) return;
93
94 // Loops go from slower to faster indices for contiguous_state<4,complex_t>
95 // Speedup may be possible by not accumulating dealiased modes
96 typedef contiguous_state<4,complex_t>::index index;
97 for (index ix = x.index_bases()[0], iy = y.index_bases()[0];
98 ix < static_cast<index>(x.index_bases()[0] + x.shape()[0]);
99 ++ix, ++iy) {
100
101 for (index lx = x.index_bases()[3], ly = y.index_bases()[3];
102 lx < static_cast<index>(x.index_bases()[3] + x.shape()[3]);
103 ++lx, ++ly) {
104
105 cop.accumulate(0, x.shape()[2],
106 c_one,
107 &x[ix][x.index_bases()[1]][x.index_bases()[2]][lx],
108 x.strides()[1], x.strides()[2],
109 beta,
110 &y[iy][y.index_bases()[1]][y.index_bases()[2]][ly],
111 y.strides()[1], y.strides()[2]);
112 }
113 }
114}
