366template<typename FPT>
367FPT convective_stability_criterion(
368 const FPT u_x,
369 const FPT a_x,
370 const FPT lambda1_x,
371 const FPT u_y,
372 const FPT a_y,
373 const FPT lambda1_y,
374 const FPT u_z,
375 const FPT a_z,
376 const FPT lambda1_z,
377 const FPT evmaxmag_imag)
378{
379 using ::std::abs;
380 return evmaxmag_imag
381 / ( (abs(u_x) + a_x)*lambda1_x
382 + (abs(u_y) + a_y)*lambda1_y
383 + (abs(u_z) + a_z)*lambda1_z);
384}

487template<typename FPT>
488FPT diffusive_stability_criterion(
489 const FPT lambda2_x,
490 const FPT lambda2_y,
491 const FPT lambda2_z,
492 const FPT Re,
493 const FPT Pr,
494 const FPT gamma,
495 const FPT evmaxmag_real,
496 const FPT nu,
497 const FPT nu0,
498 const FPT nuB,
499 const FPT nuB0)
500{
501 // Find maximum diffusive coefficient amongst all possible criteria
502 using std::abs;
503 using std::max;
504 const FPT maxcoeff = max(max(gamma/Pr,FPT(1))*abs(nu - nu0 )/Re,
505 abs(nuB - nuB0)/Re);
506
507 return evmaxmag_real
508 / (maxcoeff * (lambda2_x + lambda2_y + lambda2_z));
509}

695template< typename StateA, typename StateB = StateA >
696class multiplicative_operator
697 : public linear_operator<StateA,StateB>,
698 public operator_nonlinear<StateB>
699{
700public:
701
703 typedef typename StateA::element element;
704 BOOST_STATIC_ASSERT(
705 (boost::is_same<element, typename StateB::element>::value));
706
708 typedef typename traits::component<element>::type component;
709
718 template< typename FactorType, typename DeltaTType >
719 multiplicative_operator(const FactorType& factor, const DeltaTType& delta_t)
720 : factor(factor)
721 , delta_t(delta_t)
722 {}
723
731 template< typename FactorType >
732 multiplicative_operator(const FactorType& factor)
733 : factor(factor)
734 , delta_t(std::numeric_limits<component>::infinity())
735 {}
736
747 virtual std::vector<component> apply_operator(
748 const component time,
749 StateB& state,
750 const method_interface<element>& method,
751 const std::size_t substep_index = 0) const
752 {
753 SUZERAIN_UNUSED(time);
754 SUZERAIN_UNUSED(method);
755 SUZERAIN_UNUSED(substep_index);
756 state.scale(factor);
757 return std::vector<component>(1, delta_t);
758 }
759
769 virtual void apply_mass_plus_scaled_operator(
770 const element& phi,
771 StateA& state,
772 const std::size_t substep_index = 0) const
773 {
774 SUZERAIN_UNUSED(substep_index);
775 state.scale(phi*factor + element(1));
776 }
777
789 virtual void accumulate_mass_plus_scaled_operator(
790 const element& phi,
791 const StateA& input,
792 const element& beta,
793 StateB& output,
794 const std::size_t substep_index = 0) const
795 {
796 SUZERAIN_UNUSED(substep_index);
797 output.scale(beta);
798 output.add_scaled(phi*factor + element(1), input);
799 }
800
813 virtual void invert_mass_plus_scaled_operator(
814 const element& phi,
815 StateA& state,
816 const method_interface<element>& method,
817 const component delta_t = 0,
818 const std::size_t substep_index = 0,
819 StateA *ic0 = NULL) const
820 {
821 SUZERAIN_UNUSED(method);
822 SUZERAIN_UNUSED(delta_t);
823 SUZERAIN_UNUSED(substep_index);
824 state.scale((element(1))/(phi*factor + element(1)));
825
826 if (ic0) ic0->scale((element(1))/(phi*factor + element(1)));
827 }
828
829private:
830
832 const element factor;
833
835 const component delta_t;
836};

1400template< typename Element,
1401 typename LinearState,
1402 typename NonlinearState,
1403 typename State >
1404const typename traits::component<Element>::type substep(
1405 const method_interface<Element>& m,
1406 const linear_operator<LinearState>& L,
1407 const typename traits::component<Element>::type chi,
1408 const operator_nonlinear<NonlinearState>& N,
1409 const typename traits::component<Element>::type time,
1410 State& a,
1411 State& b,
1412 const typename traits::component<Element>::type delta_t,
1413 const std::size_t substep_index)
1414{
1415 BOOST_STATIC_ASSERT(
1416 (boost::is_same<Element,typename LinearState::element>::value));
1417 BOOST_STATIC_ASSERT(
1418 (boost::is_same<Element,typename NonlinearState::element>::value));
1419 BOOST_STATIC_ASSERT(
1420 (boost::is_same<Element,typename State::element>::value));
1421
1422 if (SUZERAIN_UNLIKELY(substep_index >= m.substeps()))
1423 throw std::invalid_argument("Requested substep too large");
1424
1425 L.accumulate_mass_plus_scaled_operator(
1426 delta_t * m.alpha(substep_index), a,
1427 chi * delta_t * m.zeta(substep_index), b,
1428 substep_index);
1429 N.apply_operator(time + delta_t * m.eta(substep_index), a,
1430 m, substep_index);
1431 b.add_scaled(chi * delta_t * m.gamma(substep_index), a);
1432 L.invert_mass_plus_scaled_operator(-delta_t * m.beta(substep_index), b,
1433 m, delta_t, substep_index);
1434
1435 return delta_t;
1436}

1469template< typename Element, typename Reducer,
1470 typename LinearA, typename LinearB, typename NonlinearB,
1471 typename StateA, typename StateB >
1472const typename traits::component<Element>::type step(
1473 const method_interface<Element>& m,
1474 Reducer& reducer,
1475 const linear_operator<LinearA,LinearB>& L,
1476 const typename traits::component<Element>::type chi,
1477 const operator_nonlinear<NonlinearB>& N,
1478 const typename traits::component<Element>::type time,
1479 StateA& a,
1480 StateB& b,
1481 const typename traits::component<Element>::type max_delta_t = 0)
1482{
1483 SUZERAIN_TIMER_SCOPED("lowstorage::step");
1484
1485 using boost::is_same;
1486 BOOST_STATIC_ASSERT((is_same<Element,typename LinearA::element>::value));
1487 BOOST_STATIC_ASSERT((is_same<Element,typename LinearB::element>::value));
1488 BOOST_STATIC_ASSERT((is_same<Element,typename NonlinearB::element>::value));
1489 BOOST_STATIC_ASSERT((is_same<Element,typename StateA::element>::value));
1490 BOOST_STATIC_ASSERT((is_same<Element,typename StateB::element>::value));
1491 typedef typename traits::component<Element>::type component_type;
1492
1493 // First substep handling is special since we need to determine delta_t
1494 b.assign_from(a);
1495 const std::vector<component_type> delta_t_candidates
1496 = N.apply_operator(time, b, m, 0);
1497 component_type delta_t = reducer(delta_t_candidates);
1498
1499 if (max_delta_t > 0) {
1500 delta_t = math::minnan(delta_t, max_delta_t);
1501 }
1502 L.apply_mass_plus_scaled_operator(delta_t * m.alpha(0), a, 0);
1503 a.add_scaled(chi * delta_t * m.gamma(0), b);
1504 L.invert_mass_plus_scaled_operator(-delta_t * m.beta(0), a, m, delta_t, 0);
1505
1506 // Second and subsequent substeps are identical
1507 for (std::size_t i = 1; i < m.substeps(); ++i) {
1508 L.accumulate_mass_plus_scaled_operator(
1509 delta_t * m.alpha(i), a,
1510 chi * delta_t * m.zeta(i), b,
1511 i);
1512 b.exchange(a); // Note nonlinear storage controls exchange operation
1513 N.apply_operator(time + delta_t * m.eta(i), b,
1514 m, i);
1515 a.add_scaled(chi * delta_t * m.gamma(i), b);
1516 L.invert_mass_plus_scaled_operator(-delta_t * m.beta(i),
1517 a, m, delta_t, i);
1518 }
1519
1520 return delta_t;
1521}
