76template<typename FPT, typename Integer>
77typename boost::enable_if< boost::is_signed<Integer>, FPT >::type
78integer_power(FPT x, Integer n)
79{
80 using std::numeric_limits;
81 BOOST_STATIC_ASSERT(numeric_limits<Integer>::is_integer);
82
83 // Prevent accidentally requesting a negative power for integer x.
84 BOOST_STATIC_ASSERT(!numeric_limits<FPT>::is_integer);
85
86 FPT retval = 1;
87 // Convert all requests into one involving a positive power
88 if (n < 0) {
89 x = ((FPT) 1)/x;
90 n = -n;
91 }
92 // Repeated squaring method. Returns 0.0^0 = 1; continuous in x
93 do {
94 if (n & 1) retval *= x; /* for n odd */
95 n >>= 1;
96 x *= x;
97 } while (n);
98
99 return retval;
100}

193template<typename FPT, typename Integer, typename OutputIterator>
194OutputIterator linspace(const FPT xbegin,
195 const FPT xend,
196 const Integer n,
197 OutputIterator x)
198{
199 BOOST_STATIC_ASSERT(std::numeric_limits<Integer>::is_integer);
200 if (SUZERAIN_UNLIKELY(n <= 0)) throw std::invalid_argument("n <= 0");
201
202 if (SUZERAIN_UNLIKELY(n == 1)) {
203 if (SUZERAIN_UNLIKELY(xbegin != xend))
204 throw std::invalid_argument("n == 1 && xbegin != xend");
205 *x++ = xbegin;
206 } else {
207 const FPT xh = (xend - xbegin)/(n-1);
208 *x++ = xbegin;
209 for (Integer i = 1; i < n-1; ++i) {
210 *x++ = xh*i + xbegin;
211 }
212 *x++ = xend;
213 }
214
215 return x;
216}

230template<typename FPT, typename Integer, typename OutputIterator>
231OutputIterator logspace(const FPT xbegin,
232 const FPT xend,
233 const Integer n,
234 OutputIterator x,
235 const FPT base = 10)
236{
237 BOOST_STATIC_ASSERT(std::numeric_limits<Integer>::is_integer);
238 if (SUZERAIN_UNLIKELY(n <= 0)) throw std::invalid_argument("n <= 0");
239
240 if (SUZERAIN_UNLIKELY(n == 1)) {
241 if (SUZERAIN_UNLIKELY(xbegin != xend))
242 throw std::invalid_argument("n == 1 && xbegin != xend");
243 *x++ = std::pow(base, xbegin);
244 } else {
245 const FPT xh = (xend - xbegin)/(n-1);
246 *x++ = pow(base, xbegin);
247 for (Integer i = 1; i < n-1; ++i) {
248 *x++ = std::pow(base, xh*i + xbegin);
249 }
250 *x++ = std::pow(base, xend);
251 }
252
253 return x;
254}

271template<typename FPT, typename Integer, typename OutputIterator>
272OutputIterator stretchspace(const FPT xbegin,
273 const FPT xend,
274 const Integer n,
275 const FPT alpha,
276 OutputIterator x)
277{
278 BOOST_STATIC_ASSERT(std::numeric_limits<Integer>::is_integer);
279 if (SUZERAIN_UNLIKELY(n < 3))
280 throw std::invalid_argument("n < 3");
281 if (SUZERAIN_UNLIKELY(alpha <= 0))
282 throw std::invalid_argument("alpha <= 0");
283
284 // Compute parameters; see notes dated 6 April 2010. Expressions arise
285 // from setting dx_{n-1} = alpha * dx_{0} = beta*(n-1) + dx_{0}, expressing
286 // beta in terms of dx_{0}, and solving b-a = \sum_{i=0}^{ninterval} dx_{i}
287 // for dx0.
288 const Integer ninterval = n - 1;
289 const FPT dx0 = (2*(xend - xbegin)) / (ninterval*(1 + alpha));
290 const FPT beta = (alpha - 1) / (ninterval - 1) * dx0;
291
292 // Use parameters to compute the output sequence
293 FPT xlast = xbegin;
294 for (Integer i = 0; i < ninterval; ++i) {
295 *x++ = xlast;
296 xlast += beta*i + dx0;
297 }
298 *x++ = xend;
299
300 return x;
301}

531template<typename FPT>
532void value_deriv(const FPT& x1,
533 const FPT& y1,
534 const FPT& yp1,
535 const FPT& x2,
536 const FPT& y2,
537 const FPT& yp2,
538 const FPT& x3,
539 FPT& y3,
540 FPT& yp3)
541{
542 const FPT x2m1 = x2 - x1;
543 const FPT x2m1p2 = x2m1 * x2m1;
544 const FPT x2m1p3 = x2m1p2 * x2m1;
545 const FPT invx2m1p3 = 1 / x2m1p3;
546 const FPT x3m1 = x3 - x1;
547 const FPT x3m1p2 = x3m1 * x3m1;
548 const FPT x3m2 = x3 - x2;
549 const FPT y2m1 = y2 - y1;
550
551 y3 = invx2m1p3*( x2m1p3*(y1 + x3m1*yp1)
552 + x3m1p2*(- 2*x3m2*y2m1 - x2m1p2*yp1
553 + x2m1*(y2m1 + x3m2*(yp1 + yp2))));
554 yp3 = invx2m1p3*( x2m1p3*yp1
555 - 2*x3m1*(x3m1*y2m1 + 2*x3m2*y2m1 + x2m1p2*yp1)
556 + x2m1*x3m1*(2*y2m1 + (x3m1 + 2*x3m2)*(yp1 + yp2))) ;
557}
