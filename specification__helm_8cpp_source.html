74double
75specification_helm::steady(
76 const double t,
77 const double u,
78 const double y) const
79{
80 assert(this->enabled);
81 assert(!(boost::math::isnan)(this->t));
82 assert(!(boost::math::isnan)(this->v));
83 assert(!(boost::math::isnan)(this->r));
84
85 // Invoke control logic
86 const double dt = t - this->t;
87 const double dv = helm_steady(&this->h, dt, r, u, v, y);
88
89 // Update absolute state
90 this->t = t;
91 this->v += dv;
92
93 return dv;
94}
