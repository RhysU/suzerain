34running_statistics<double,1>
35arsel(const std::size_t Ny,
36 const std::vector<double>& t,
37 const double * data,
38 const std::size_t ld,
39 const specification_arsel& spec,
40 std::vector<double>& eff_N,
41 std::vector<double>& eff_var,
42 std::vector<double>& mu,
43 std::vector<double>& mu_sigma,
44 std::vector<double>& p,
45 std::vector<double>& T)
46{
47 const std::size_t Nt = t.size();
48 if (Ny == 0) throw std::invalid_argument("arsel received Ny == 0");
49 if (Nt == 0) throw std::invalid_argument("arsel received Nt == 0");
50 if (ld < Ny) throw std::invalid_argument("arsel received ld < Ny");
51
52 // Prepare output vectors to gather burg_method()
53 eff_N .resize(Ny);
54 eff_var .resize(Ny);
55 mu .resize(Ny);
56 mu_sigma.resize(Ny);
57 p .resize(Ny);
58 T .resize(Ny);
59
60 // Compute sampling rate statistics
61 running_statistics<double,1> dt;
62 for (std::size_t i = 1; i < t.size(); ++i) {
63 const double diff = t[i] - t[i-1];
64 dt(&diff);
65 }
66
67 // Prepare vectors to capture burg_method() output
68 std::vector<double> params, sigma2e, gain, autocor;
69 params .reserve(spec.maxorder*(spec.maxorder + 1)/2);
70 sigma2e.reserve(spec.maxorder + 1);
71 gain .reserve(spec.maxorder + 1);
72 autocor.reserve(spec.maxorder + 1);
73
74 // Prepare repeatedly-used working storage for burg_method().
75 std::vector<double> f, b, Ak, ac;
76
77 for (std::size_t j = 0; j < Ny; ++j) {
78
79 std::size_t maxorder = spec.maxorder;
80 params .clear();
81 sigma2e.clear();
82 gain .clear();
83 autocor.clear();
84 ar::burg_method(ar::strided_adaptor<const double*>(&data[j+ 0*ld], ld),
85 ar::strided_adaptor<const double*>(&data[j+Nt*ld], ld),
86 mu[j],
87 maxorder,
88 back_inserter(params),
89 back_inserter(sigma2e),
90 back_inserter(gain),
91 back_inserter(autocor),
92 spec.submean,
93 true /* output hierarchy? */,
94 f, b, Ak, ac);
95
96 spec.best_model(Nt, params, sigma2e, gain, autocor);
97
98 const double T0 = ar::decorrelation_time(
99 static_cast<std::size_t>(spec.wlenT0*Nt),
100 ar::autocorrelation(params.begin(), params.end(),
101 gain[0], autocor.begin()),
102 spec.absrho);
103
104 eff_var[j] = (Nt*gain[0]*sigma2e[0]) / (Nt - T0);
105 eff_N[j] = Nt / T0;
106 mu_sigma[j] = std::sqrt(eff_var[j] / eff_N[j]);
107 p[j] = params.size();
108 T[j] = T0 * dt.avg(0);
109
110 }
111
112 return dt;
113}
