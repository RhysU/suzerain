175template <typename InputIterator,
176 typename OutputType1,
177 typename OutputType2>
178std::size_t welford_nvariance(InputIterator first,
179 InputIterator last,
180 OutputType1& mean,
181 OutputType2& nvar)
182{
183 using std::iterator_traits;
184 using std::size_t;
185 typedef typename iterator_traits<InputIterator>::value_type value;
186
187 size_t N = 1; // Running next sample number
188 value m = 0; // Running mean of data thus far
189 value nv = 0; // Running variance times the number of samples
190
191 while (first != last)
192 {
193 value x = *first++;
194 value d = x - m;
195 m += d / N++;
196 nv += d*(x - m);
197 }
198
199 mean = m;
200 nvar = nv;
201 return N-1;
202}

266template <typename InputIterator1,
267 typename InputIterator2,
268 typename OutputType1,

271std::size_t welford_ncovariance(InputIterator1 first1,
272 InputIterator1 last1,
273 InputIterator2 first2,
274 OutputType1& mean1,
275 OutputType2& mean2,
276 OutputType3& ncovar)
277{
278 using std::iterator_traits;
279 using std::size_t;
280 typedef typename iterator_traits<InputIterator1>::value_type value1;
281 typedef typename iterator_traits<InputIterator2>::value_type value2;
282
283 size_t N = 1; // Running next sample number
284 value1 m1 = 0; // Running mean of first data set thus far
285 value2 m2 = 0; // Running mean of second data set thus far
286 OutputType3 nc = 0; // Running covariance times the number of samples
287
288 while (first1 != last1)
289 {
290 value1 x1 = *first1++;
291 value1 d1 = x1 - m1;
292 m1 += d1 / N;
293
294 value2 x2 = *first2++;
295 value2 d2 = x2 - m2;
296 m2 += d2 / N;
297
298 nc += d1*(x2 - m2);
299
300 ++N;
301 }
302
303 mean1 = m1;
304 mean2 = m2;
305 ncovar = nc;
306 return N-1;
307}

463template <typename ValueType,
464 typename InputIterator1,
465 typename InputIterator2>
466ValueType
467#if (AR_GCC_VERSION > 40305)
468 __attribute__((__optimize__("no-associative-math")))
469#endif
470negative_half_reflection_coefficient(InputIterator1 a_first,
471 InputIterator1 a_last,
472 InputIterator2 b_first)
473#if (AR_GCC_VERSION > 40305) || (_MSC_VER > 1400)
474{
475 ValueType ns = 0, nt, nc = 0, ny; // Kahan numerator accumulation
476 ValueType ds = 0, dt, dc = 0, dy; // Kahan denominator accumulation
477
478 while (a_first != a_last)
479 {
480 ValueType xa = *a_first++; // Denominator: \vec{a}\cdot\vec{a}
481 dy = (xa * xa) - dc;
482 dt = ds + dy;
483 dc = (dt - ds) - dy;
484 ds = dt;
485
486 ValueType xb = *b_first++; // Denominator: \vec{b}\cdot\vec{b}
487 dy = (xb * xb) - dc;
488 dt = ds + dy;
489 dc = (dt - ds) - dy;
490 ds = dt;
491
492 ny = (xa * xb) - nc; // Numerator: \vec{a}\cdot\vec{b}
493 nt = ns + ny;
494 nc = (nt - ns) - ny;
495 ns = nt;
496 }
497
498 return ns + nc == 0 // Does special zero case apply?
499 ? 0 // Yes, to avoid NaN from 0 / 0
500 : (ns + nc) / (ds + dc); // No, correct final sums and form ratio
501}
502#else
503#warning Using Non-Kahan version of ar::negative_half_reflection_coefficient.
504{
505 ValueType ns = 0;
506 ValueType ds = 0;
507
508 while (a_first != a_last)
509 {
510 ValueType xa = *a_first++;
511 ValueType xb = *b_first++;
512 ns += xa * xb; // Numerator
513 ds += xa * xa + xb * xb; // Denominator
514 }
515
516 return ns == 0 // Does special zero case apply?
517 ? 0 // Yes, to avoid NaN from 0 / 0
518 : ns / ds; // No, form ratio
519}
520#endif

535template <class InputIterator,
536 class Value,
537 class OutputIterator1,
538 class OutputIterator2,
539 class OutputIterator3,
540 class OutputIterator4,
541 class Vector>
542std::size_t burg_method(InputIterator data_first,
543 InputIterator data_last,
544 Value& mean,
545 std::size_t& maxorder,
546 OutputIterator1 params_first,
547 OutputIterator2 sigma2e_first,
548 OutputIterator3 gain_first,
549 OutputIterator4 autocor_first,
550 const bool subtract_mean,
551 const bool hierarchy,
552 Vector& f,
553 Vector& b,
554 Vector& Ak,
555 Vector& ac)
556{
557 using std::copy;
558 using std::distance;
559 using std::fill;
560 using std::inner_product;
561 using std::min;
562 using std::size_t;
563
564 // Initialize f from [data_first, data_last) and fix number of samples
565 f.assign(data_first, data_last);
566 const size_t N = f.size();
567
568 // Stably compute the incoming data's mean and population variance
569 mean = 0;
570 Value sigma2e = 0;
571 welford_variance_population(f.begin(), f.end(), mean, sigma2e);
572
573 // When requested, subtract the just-computed mean from the data.
574 // Adjust, if necessary, to make sigma2e the second moment.
575 if (subtract_mean)
576 {
577 for (typename Vector::iterator it = f.begin(); it != f.end(); ++it) {
578 *it -= mean;
579 }
580 }
581 else
582 {
583 sigma2e += mean*mean;
584 }
585
586 // At most maxorder N-1 can be fit from N samples. Beware N is unsigned.
587 maxorder = (N == 0) ? 0 : min(static_cast<size_t>(maxorder), N-1);
588
589 // Output sigma2e and gain for a zeroth order model, if requested.
590 Value gain = 1;
591 if (hierarchy || maxorder == 0)
592 {
593 *sigma2e_first++ = sigma2e;
594 *gain_first++ = gain;
595 }
596
597 // Initialize and perform Burg recursion
598 if (maxorder) b = f; // Copy iff non-trivial work required
599 Ak.assign(maxorder + 1, Value(0));
600 Ak[0] = 1;
601 ac.clear();
602 ac.reserve(maxorder);
603 for (size_t kp1 = 1; kp1 <= maxorder; ++kp1)
604 {
605 // Compute mu from f, b, and Dk and then update sigma2e and Ak using mu
606 // Afterwards, Ak[1:kp1] contains AR(k) coefficients by the recurrence
607 // Must treat mu result of 0 / 0 as 0 to avoid NaNs on constant signals
608 // By the recurrence, Ak[kp1] will also be the reflection coefficient
609 Value mu = -2 * negative_half_reflection_coefficient<Value>(
610 f.begin() + kp1, f.end(), b.begin());
611
612 sigma2e *= (1 - mu*mu);
613 for (size_t n = 0; n <= kp1/2; ++n)
614 {
615 Value t1 = Ak[n] + mu*Ak[kp1 - n];
616 Value t2 = Ak[kp1 - n] + mu*Ak[n];
617 Ak[n] = t1;
618 Ak[kp1 - n] = t2;
619 }
620
621 // Update the gain per Broersen 2006 equation (5.25)
622 gain *= 1 / (1 - Ak[kp1]*Ak[kp1]);
623
624 // Compute and output the next autocorrelation coefficient
625 // See Broersen 2006 equations (5.28) and (5.31) for details
626 ac.push_back(-inner_product(ac.rbegin(), ac.rend(),
627 Ak.begin() + 1, Ak[kp1]));
628
629 // Output parameters and the input and output variances when requested
630 if (hierarchy || kp1 == maxorder)
631 {
632 params_first = copy(Ak.begin() + 1, Ak.begin() + kp1 + 1,
633 params_first);
634 *sigma2e_first++ = sigma2e;
635 *gain_first++ = gain;
636 }
637
638 // Update f and b for the next iteration if another remains
639 if (kp1 < maxorder)
640 {
641 for (size_t n = 0; n < N - kp1; ++n)
642 {
643 Value t1 = f[n + kp1] + mu*b[n];
644 Value t2 = b[n] + mu*f[n + kp1];
645 f[n + kp1] = t1;
646 b[n] = t2;
647 }
648 }
649 }
650
651 // Output the lag [0,maxorder] autocorrelation coefficients in single pass
652 *autocor_first++ = 1;
653 copy(ac.begin(), ac.end(), autocor_first);
654
655 // Return the number of values processed in [data_first, data_last)
656 return N;
657}

952 // Zero the simulation time.

954

1236template<class RandomAccessIterator,
1237 class InputIterator,
1238 class OutputIterator>
1239void zohar_linear_solve(RandomAccessIterator a_first,
1240 RandomAccessIterator a_last,
1241 RandomAccessIterator r_first,
1242 InputIterator d_first,
1243 OutputIterator s_first)
1244{
1245 using std::copy;
1246 using std::distance;
1247 using std::inner_product;
1248 using std::invalid_argument;
1249 using std::iterator_traits;
1250 using std::reverse_iterator;
1251 using std::vector;
1252
1253 // Tildes indicate transposes while hats indicate reversed vectors.
1254
1255 // InputIterator::value_type determines the working precision
1256 typedef typename iterator_traits<InputIterator>::value_type value_type;
1257 typedef vector<value_type> vector_type;
1258 typedef typename vector_type::size_type size_type;
1259
1260 // Determine problem size using [a_first,a_last) and ensure nontrivial
1261 typename iterator_traits<RandomAccessIterator>::difference_type dist
1262 = distance(a_first, a_last);
1263 if (dist < 1) throw invalid_argument("distance(a_first, a_last) < 1");
1264 const size_type n = static_cast<size_type>(dist);
1265
1266 // Allocate working storage and set initial values for recursion:
1267 vector_type s; s .reserve(n+1); s .push_back( *d_first);
1268 vector_type ehat; ehat.reserve(n ); ehat.push_back(-a_first[0]);
1269 vector_type g; g .reserve(n ); g .push_back(-r_first[0]);
1270 value_type lambda = 1 - a_first[0]*r_first[0];
1271
1272 // Though recursive updates to s and g can be done in-place, updates to
1273 // ehat seemingly require one additional vector for storage:
1274 //
1275 // "This sequence of computations is economical of storage. It is only
1276 // necessary to retain quantities computed at level m - 1 until the
1277 // computations at level m are complete." [Trench1967, page 1504]
1278 vector_type next_ehat; next_ehat.reserve(n);
1279
1280 // Recursion for i = {1, 2, ..., n - 1}:
1281 for (size_type i = 1; i < n; ++i)
1282 {
1283
1284 reverse_iterator<RandomAccessIterator> rhat_first(r_first + i);
1285
1286 // \theta_i = \delta_{i+1} - \tilde{s}_i \hat{r}_i
1287 const value_type neg_theta = inner_product(
1288 s.begin(), s.end(), rhat_first, value_type(-(*++d_first)));
1289
1290 // \eta_i = -\rho_{-(i+1)} - \tilde{a}_i \hat{e}_i
1291 const value_type neg_eta = inner_product(
1292 ehat.begin(), ehat.end(), a_first, value_type(a_first[i]));
1293
1294 // \gamma_i = -\rho_{i+1} - \tilde{g}_i \hat{r}_i
1295 const value_type neg_gamma = inner_product(
1296 g.begin(), g.end(), rhat_first, value_type(r_first[i]));
1297
1298 /*
1299 * s_{i+1} = \bigl(\begin{smallmatrix}
1300 * s_i + (\theta_i/\lambda_i) \hat{e}_i \\
1301 * \theta_i/\lambda_i
1302 * \end{smallmatrix}\bigr)
1303 *
1304 * \hat{e}_{i+1} = \bigl(\begin{smallmatrix}
1305 * \eta_i/\lambda_i \\
1306 * \hat{e}_i + (\eta_i/\lambda_i) g_i
1307 * \end{smallmatrix}\bigr)
1308 *
1309 * g_{i+1} = \bigl(\begin{smallmatrix}
1310 * g_i + (\gamma_i/\lambda_i) \hat{e}_i \\
1311 * \gamma_i/\lambda_i
1312 * \end{smallmatrix}\bigr)
1313 */
1314 const value_type theta_by_lambda = -neg_theta/lambda;
1315 const value_type eta_by_lambda = -neg_eta /lambda;
1316 const value_type gamma_by_lambda = -neg_gamma/lambda;
1317 next_ehat.clear();
1318 next_ehat.push_back(eta_by_lambda);
1319 for (size_type j = 0; j < i; ++j)
1320 {
1321 s[j] += theta_by_lambda*ehat[j];
1322 next_ehat.push_back(ehat[j] + eta_by_lambda*g[j]);
1323 g[j] += gamma_by_lambda*ehat[j];
1324 }
1325 s.push_back(theta_by_lambda);
1326 g.push_back(gamma_by_lambda);
1327 ehat.swap(next_ehat);
1328
1329 // \lambda_{i+1} = \lambda_i - \eta_i \gamma_i / \lambda_i
1330 lambda -= neg_eta*neg_gamma/lambda;
1331 }
1332
1333 // Recursion for i = n differs slightly per Zohar's "Last Computed Values"
1334 // Computing g_n above was unnecessary but the incremental expense is small
1335 {
1336 reverse_iterator<RandomAccessIterator> rhat_first(r_first + n);
1337
1338 // \theta_n = \delta_{n+1} - \tilde{s}_n \hat{r}_n
1339 const value_type neg_theta = inner_product(
1340 s.begin(), s.end(), rhat_first, value_type(-(*++d_first)));
1341
1342 /*
1343 * s_{n+1} = \bigl(\begin{smallmatrix}
1344 * s_n + (\theta_n/\lambda_n) \hat{e}_n \\
1345 * \theta_n/\lambda_n
1346 * \end{smallmatrix}\bigr)
1347 */
1348 const value_type theta_by_lambda = -neg_theta/lambda;
1349 for (size_type j = 0; j < n; ++j)
1350 {
1351 s[j] += theta_by_lambda*ehat[j];
1352 }
1353 s.push_back(theta_by_lambda);
1354 }
1355
1356 // Output solution
1357 copy(s.begin(), s.end(), s_first);
1358}
