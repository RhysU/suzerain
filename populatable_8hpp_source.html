113// Compare and contrast maybe_override_impl in overridable.hpp
114template<typename T>
115static bool
116maybe_populate_impl(
117 const char* name,
118 const char* description,
119 T& destination,
120 const T& source,
121 const bool verbose,
122 bool (* const default_value)(const T& v))
123{
124 if (default_value(destination)) {
125 if (verbose) {
126 // Populating a default isn't interesting, hence DEBUG0
127 if (description) {
128 DEBUG0("Populating " << name
129 << " (" << description << ") to be " << source);
130 } else {
131 DEBUG0("Populating " << name << " to be " << source);
132 }
133 }
134 destination = source;
135 return true;
136 }
137
138 if (verbose) {
139 if (description) {
140 // Retaining an existing setting is interesting, hence INFO0
141 INFO0("Clutching onto " << name
142 << " (" << description << ") of " << destination);
143 } else {
144 INFO0("Clutching onto " << name << " of " << destination);
145 }
146 }
147 return false;
148}
