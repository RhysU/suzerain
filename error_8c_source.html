44void
45suzerain_error(const char * reason,
46 const char * file,
47 int line,
48 int suzerain_errno)
49{
50 if (suzerain_error_handler) {
51 (*suzerain_error_handler) (reason, file, line, suzerain_errno);
52 return ;
53 }
54
55 suzerain_stream_printf ("ERROR", file, line, reason);
56
57 fflush (stdout);
58 fprintf (stderr, "Default suzerain error handler invoked.\n");
59 fflush (stderr);
60
61 abort ();
62}

99void
100suzerain_stream_printf(const char *label,
101 const char *file,
102 int line,
103 const char *reason)
104{
105 if (suzerain_stream == NULL) {
106 suzerain_stream = stderr;
107 }
108 if (suzerain_stream_handler) {
109 (*suzerain_stream_handler) (label, file, line, reason);
110 return;
111 }
112 fprintf(suzerain_stream,
113 "suzerain: %s:%d: %s: %s\n", file, line, label, reason);
114
115}

137const char *
138suzerain_strerror(const int suzerain_errno)
139{
140 switch (suzerain_errno) {
141 case SUZERAIN_FAILURE:
142 return "failure" ;
143 case SUZERAIN_CONTINUE:
144 return "iteration has not converged";
145 case SUZERAIN_SUCCESS:
146 return "success" ;
147 case SUZERAIN_EDOM:
148 return "input domain error" ;
149 case SUZERAIN_ERANGE:
150 return "output range error" ;
151 case SUZERAIN_EFAULT:
152 return "invalid pointer" ;
153 case SUZERAIN_EINVAL:
154 return "invalid argument supplied by user" ;
155 case SUZERAIN_EFAILED:
156 return "generic failure" ;
157 case SUZERAIN_ESANITY:
158 return "sanity check failed - shouldn't happen" ;
159 case SUZERAIN_ENOMEM:
160 return "malloc failed" ;
161 case SUZERAIN_EBADFUNC:
162 return "problem with user-supplied function";
163 case SUZERAIN_EMAXITER:
164 return "exceeded max number of iterations";
165 case SUZERAIN_EZERODIV:
166 return "tried to divide by zero" ;
167 case SUZERAIN_EROUND:
168 return "failed because of roundoff error";
169 case SUZERAIN_EBADLEN:
170 return "matrix or vector lengths are not conformant";
171 case SUZERAIN_ESING:
172 return "apparent singularity detected";
173 case SUZERAIN_EDIVERGE:
174 return "integral or series is divergent";
175 case SUZERAIN_EUNIMPL:
176 return "requested feature not (yet) implemented";
177 default:
178 return "unknown error code" ;
179 }
180}
