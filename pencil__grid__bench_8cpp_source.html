37static real_t sample_field(const real_t x, const real_t y, const real_t z)
38{
39 using std::sin;
40 return 1* 1
41 + 2*y
42 + 2* 3*sin(x)
43 + 2* 5*sin(z)
44 + 2* 7*sin(x)*y
45 + 2*11*y*sin(z)
46 + 4*13*sin(x)*sin(z)
47 + 4*17*sin(x)*y*sin(z)
48 ;
49}

51int main(int argc, char **argv)
52{
53 // Instantiate application_base instance and set desired defaults
54 support::application_base app(
55 "Benchmark parallel FFTs implemented using suzerain::pencil_grid",
56 /* No arguments */ "", /* No further description*/ "", REVISIONSTR);
57 app.grid->htdelta = 0;
58 app.grid->k = 4;
59 app.grid->L.x() = 8 * boost::math::constants::pi<real_t>();
60 app.grid->L.y() = 2;
61 app.grid->L.z() = 4 * boost::math::constants::pi<real_t>();
62 app.grid->Nx(16);
63 app.grid->Ny(16);
64 app.grid->Nz(16);
65 app.grid->DAFx(1);
66 app.grid->DAFz(1);
67
68 // Add additional command line options
69 std::size_t repeat = 1;
70 std::size_t nfields = 1;
71 bool check = false;
72 app.options.add_options()
73 ("repeat,r", boost::program_options::value(&repeat)
74 ->default_value(repeat),
75 "Number of repetitions to perform")
76 ("nfields,n", boost::program_options::value(&nfields)
77 ->default_value(nfields),
78 "Number of independent fields")
79 ("check,c", boost::program_options::value(&check)
80 ->default_value(check)->zero_tokens(),
81 "Check results against expected values")
82 ;
83
84 // Initialize the parallel decomposition
85 app.initialize(argc, argv);
86 app.establish_ieee_mode();
87 app.load_grid_and_operators(NULL);
88 app.establish_decomposition();
89
90 // Allocate necessary storage and prepare physical space view
91 app.establish_state_storage(/* linear state is wave-only */ 0,
92 /* nonlinear state is transformable */ nfields);
93 physical_view<> p(*app.dgrid, *app.state_nonlinear);
94
95 // Initialize manufactured field using operator_base utilities
96 // p is a 2D, real-valued view of (NFIELDS, X*Z*Y) leftmost fastest
97 // We need x_i, y_j, z_k positions making structure below more complicated
98 suzerain::operator_base o(*app.grid, *app.dgrid, *app.cop, *app.b);
99 for (int offset = 0, j = app.dgrid->local_physical_start.y(); // Y
100 j < app.dgrid->local_physical_end.y();
101 ++j) {
102 const real_t y = o.y(j);
103
104 for (int k = app.dgrid->local_physical_start.z(); // Z
105 k < app.dgrid->local_physical_end.z();
106 ++k) {
107 const real_t z = o.z(k);
108
109 for (int i = app.dgrid->local_physical_start.x(); // X
110 i < app.dgrid->local_physical_end.x();
111 ++i, /* NB */ ++offset) {
112 const real_t x = o.x(i);
113
114 const real_t data = sample_field(x, y, z);
115 for (std::size_t f = 0; f < nfields; ++f) { // Fields
116 p(f, offset) = f + data;
117 }
118
119 }
120
121 }
122
123 }
124
125 SUZERAIN_MPICHKQ(MPI_Barrier(MPI_COMM_WORLD));
126
127 // Perform transformations accumulating parallel FFT timing in time
128 // Barriers are used to ensure each timing starts from synchronized state
129 double time = 0;
130 for (std::size_t m = 0; m < repeat; m++) {
131 INFO0("Iteration " << (m + 1));
132
133 // Physical to wave
134 SUZERAIN_MPICHKQ(MPI_Barrier(MPI_COMM_WORLD));
135 time -= MPI_Wtime();
136 for (std::size_t l = 0; l < nfields; ++l) {
137 app.dgrid->transform_physical_to_wave(p.data());
138 }
139 time += MPI_Wtime();
140
141 // Wave space normalization uses only dealiased Fourier X and Z extents
142 app.state_nonlinear->scale(app.dgrid->chi());
143
144 // Wave to physical
145 SUZERAIN_MPICHKQ(MPI_Barrier(MPI_COMM_WORLD));
146 time -= MPI_Wtime();
147 for (std::size_t l = 0; l < nfields; ++l) {
148 app.dgrid->transform_wave_to_physical(p.data());
149 }
150 time += MPI_Wtime();
151
152 }
153
154 // Report the worst time observed on any rank
155 const int rank = mpi::comm_rank(MPI_COMM_WORLD);
156 struct { double time; int rank; } worst = { time, rank };
157 SUZERAIN_MPICHKQ(MPI_Allreduce(MPI_IN_PLACE, &worst, 1, MPI_DOUBLE_INT,
158 MPI_MAXLOC, MPI_COMM_WORLD));
159 if (worst.rank == rank) {
160 INFO("Seconds per iteration " << time / (double) repeat);
161 }
162
163 // Stop processing if correctness checking not requested
164 if (!check) return EXIT_SUCCESS;
165
166 // Quantify discrepancies between source data and computed results
167 namespace acc = boost::accumulators;
168 acc::accumulator_set<
169 real_t, acc::stats< acc::tag::min, acc::tag::max >
170 > track;
171 for (int offset = 0, j = app.dgrid->local_physical_start.y(); // Y
172 j < app.dgrid->local_physical_end.y();
173 ++j) {
174 const real_t y = o.y(j);
175
176 for (int k = app.dgrid->local_physical_start.z(); // Z
177 k < app.dgrid->local_physical_end.z();
178 ++k) {
179 const real_t z = o.z(k);
180
181 for (int i = app.dgrid->local_physical_start.x(); // X
182 i < app.dgrid->local_physical_end.x();
183 ++i, /* NB */ ++offset) {
184 const real_t x = o.x(i);
185
186 const real_t data = sample_field(x, y, z);
187 for (std::size_t f = 0; f < nfields; ++f) { // Fields
188 track((data + f) - p(f,offset));
189 }
190
191 }
192
193 }
194
195 }
196
197 // Report the minimum discrepancy observed on any rank
198 struct { double discrepancy; int rank; } min = { acc::min(track), rank };
199 SUZERAIN_MPICHKQ(MPI_Allreduce(MPI_IN_PLACE, &min, 1, MPI_DOUBLE_INT,
200 MPI_MINLOC, MPI_COMM_WORLD));
201 if (min.rank == rank) {
202 INFO("Minimum discrepancy " << std::scientific << min.discrepancy);
203 }
204
205 // Report the maximum discrepancy observed on any rank
206 struct { double discrepancy; int rank; } max = { acc::max(track), rank };
207 SUZERAIN_MPICHKQ(MPI_Allreduce(MPI_IN_PLACE, &max, 1, MPI_DOUBLE_INT,
208 MPI_MAXLOC, MPI_COMM_WORLD));
209 if (max.rank == rank) {
210 INFO("Maximum discrepancy " << std::scientific << max.discrepancy);
211 }
212
213 return EXIT_SUCCESS;
214}
