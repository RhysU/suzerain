45int
46suzerain_channel_compute_viscous(
47 const double code_Re,
48 const suzerain_channel_local * wall,
49 suzerain_channel_viscous * viscous)
50{
51 FILL_WITH_NANS(viscous);
52
53 // Compute quantities of interest in "code units" divided by [units ]
54 viscous->tau_w = wall->mu * wall->u__y; // [\mu u / l]
55 viscous->u_tau = sqrt(viscous->tau_w / wall->rho); // [u ]
56 viscous->delta_nu = wall->mu / wall->rho / viscous->u_tau; // [l ]
57
58 // Adjust for code Reynolds number per writeups/viscous_nondim.pdf which
59 // details why and how how these scaling factors appear. Provided that
60 // they are addressed here, they don't "leak" into other computations.
61 const double sqrt_code_Re = sqrt(code_Re);
62 viscous->tau_w /= code_Re; // Now nondim by [ rho0 u0^2 ]
63 viscous->u_tau /= sqrt_code_Re; // Now nondim by [ u0 ]
64 viscous->delta_nu /= sqrt_code_Re; // Now nondim by [ l0 ]
65
66 return SUZERAIN_SUCCESS;
67}

69int
70suzerain_channel_compute_qoi(
71 const double code_Ma,
72 const double code_Re,
73 const suzerain_channel_local * const wall,
74 const suzerain_channel_viscous * const viscous,
75 const suzerain_channel_local * const center,
76 suzerain_channel_qoi * const qoi)
77{
78 FILL_WITH_NANS(qoi);
79
80 // Nondimensional quantities are computed with the first line being the
81 // quantity and the final line being any needed "code unit" correction.
82 // Notice viscous->tau_w and viscous->u_tau already account for code_Re;
83 // see the suzerain_channel_viscous struct declaration to check their scaling.
84 qoi->cf = 2 * viscous->tau_w / center->rho / square(center->u)
85 * 1;
86 qoi->gamma_c = center->gamma
87 * 1;
88 qoi->Ma_c = center->u / center->a
89 * code_Ma;
90 qoi->Ma_tau = viscous->u_tau / wall->a
91 * code_Ma;
92 qoi->Pr_w = wall->Pr;
93 qoi->Bq = - (wall->mu * wall->T__y)
94 / (qoi->Pr_w * wall->rho * viscous->u_tau * wall->T)
95 / code_Re;
96 qoi->ratio_nu = (center->mu / center->rho) / (wall->mu / wall->rho)
97 * 1;
98 qoi->ratio_rho = center->rho / wall->rho
99 * 1;
100 qoi->ratio_T = center->T / wall->T
101 * 1;
102 qoi->Re_c = center->rho * center->u * center->y / center->mu
103 * code_Re;
104 qoi->Re_tau = center->y / viscous->delta_nu
105 * 1;
106 qoi->v_wallplus = wall->v / viscous->u_tau
107 * 1;
108
109 return SUZERAIN_SUCCESS;
110}
