46suzerain_lapackext_sgbsv(
47 char * const fact,
48 const char trans,
49 const int n,
50 const int kl,
51 const int ku,
52 const int nrhs,
53 float *ab,
54 const int ldab,
55 int *ipiv,
56 float *b,
57 const int ldb)

59 int info = 0;
60 if (toupper(*fact) == 'N') {
61 info = suzerain_lapack_sgbtrf(n, n, kl, ku, ab, ldab, ipiv);
