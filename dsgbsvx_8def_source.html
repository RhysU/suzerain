64// Shorthand for reporting internal sanity mistakes;
65#define MAYBE_XERBLA(cond, code) \
66 if (UNLIKELY(cond)) { \
67 return suzerain_blas_xerbla(__func__, code); \
68 }
69
70int
71FUNCTION(
72 char * const fact,
73 int * const apprx,
74 const int aiter,
75 char trans,
76 const int n,
77 const int kl,
78 const int ku,
79 DOUBLE * const ab,
80 NORM * const afrob,
81 DOUBLE * const afb,
82 int * const ipiv,
83 DOUBLE * const b,
84 DOUBLE * const x,
85 int * const siter,
86 int * const diter,
87 NORM * const tolsc,
88 DOUBLE * const r,
89 NORM * const res)
90{
91#pragma warning(push,disable:2259)
92 *fact = toupper(*fact);
93 trans = toupper(trans);
94#pragma warning(pop)
95
96 int info = 0;
97 if (UNLIKELY(*fact != 'N' && *fact != 'S' && *fact != 'D')) {
98 info = -1;
99 } else if (UNLIKELY(*apprx < 0)) {
100 info = -2;
101 } else if (UNLIKELY(aiter < 0)) {
102 info = -3;
103 } else if (UNLIKELY(trans != 'N' && trans != 'T' && trans != 'C')) {
104 info = -4;
105 } else if (UNLIKELY(n < 0)) {
106 info = -5;
107 } else if (UNLIKELY(kl < 0)) {
108 info = -6;
109 } else if (UNLIKELY(ku < 0)) {
110 info = -7;
111 } else if (UNLIKELY(*tolsc < 0)) {
112 info = -16;
113 } else if (UNLIKELY(*siter < 0 && *diter < 0)) {
114 info = -99;
115 }
116 if (info) return suzerain_blas_xerbla(__func__, info);
117
118 // Error handling from here onward deserves a word. Internal errors are
119 // marked by calls to suzerain_blas_xerbla using MAYBE_XERBLA(...,
120 // -__LINE__). Usage errors by a user (e.g. providing a singular matrix)
121 // simply return the appropriate info value.
122
123 // Lookup machine-specific floating point information
124 const NORM eps = suzerain_lapack_dlamch('E');
125
126 // Incoming vectors and matrices must be contiguous in memory
127 static const int inc = 1;
128 const int ldab = kl + 1 + ku;
129 const int ldafb = 2*kl + 1 + ku;
130
131 // Compute (a usually awful) solution estimate starting from x = 0
132 memset(x, 0, n*sizeof(x[0])); // x = 0 for inc == 1
133 DCOPY(n, b, inc, r, inc); // r = b - A*x = b - A*0 = b
134 *res = DNRM2(n, r, inc); // res = |r|_2
135
136 // Fake that this initial solution is better by more than a factor of 2
137 const NORM resdecay = 2;
138 NORM lastres = (resdecay + 1) * (*res + 1);
139
140 // The stopping residual may be either 'eps' or that from Langou et al
141 // Compare the former against criteria within LAPACK {D,Z}GBRFS
142 // The latter requires ||A||_{fro} and running ||x||_2 information
143#pragma warning(push,disable:1572)
144 const int use_eps_tol = (*tolsc == 0.0);
145#pragma warning(pop)
146 NORM tolconst;
147 NORM tol;
148 if (use_eps_tol) {
149
150 tolconst = 1;
151 tol = eps;
152
153 } else {
154
155 // Compute Frobenius norm of A if it was not supplied
156 if (*afrob < 0) {
157 *afrob = DLANGB('F', n, kl, ku, ab, ldab, NULL);
158 }
159
160 // Compute const part of stopping tolerance per Langou et al:
161 // ||r||_2 = ||b - OP(A) x||_2 <= ||x||_2 ||A||_{fro} eps sqrt(n)
162 tolconst = *afrob * eps * sqrt(n) * (*tolsc);
163
164 // Compute the complete RHS ||x||_2 * tolconst for x == 0
165 tol = 0;
166 }
167
168 // Save the maximum iteration count prior to entering compute loops
169 const int smax = *siter; *siter = -1;
170 const int dmax = *diter; *diter = -1;
171
172 if (smax >= 0 && *fact != 'D' && *res > tol) { // Attempt single?
173
174 if (*fact != 'S') { // Ensure single precision factorization available
175 *fact = 'S';
176 *apprx = 0;
177
178 DLACPY('F', ldab, n, ab, ldab, afb + kl, ldafb);
179 info = DDEMOTE(n*ldafb, afb);
180 MAYBE_XERBLA(info, -__LINE__)
181 info = SGBTRF(n, n, kl, ku, (SINGLE*)afb, ldafb, ipiv);
182 if (UNLIKELY(info > 0 && dmax >= 0)) {
183 // Single precision factorization failed where
184 // double precision might succeed. Bail on single.
185 goto double_precision_attempt;
186 } else MAYBE_XERBLA(info < 0, -__LINE__)
187 }
188
189 while (*siter < smax && *res > tol) {
190
191 // Perform one step of mixed precision iterative refinement
192 // updating norm computations r = b - op(A) x and possibly tol
193 *siter += 1;
194 info = DDEMOTE(n, r);
195 MAYBE_XERBLA(info, -__LINE__)
196 info = SGBTRS(trans, n, kl, ku, 1,
197 (SINGLE*)afb, ldafb, ipiv,
198 (SINGLE*)r, n);
199 MAYBE_XERBLA(info, -__LINE__)
200 info = DPROMOTE(n, r);
201 MAYBE_XERBLA(info, -__LINE__)
202 DAXPY(n, 1, r, inc, x, inc);
203 tol = (use_eps_tol ? eps : DNRM2(n, x, inc)) * tolconst;
204 DCOPY(n, b, inc, r, inc);
205 info = DGBMV(trans, n, n, kl, ku, -1, ab, ldab, x, inc, 1, r, inc);
206 MAYBE_XERBLA(info, -__LINE__)
207 *res = DNRM2(n, r, inc);
208
209 // Getting here required *res > tol so isnan(*res) now is badness
210 // likely stemming from NaNs within matrix ab not caught because the
211 // matrix was not factorized within this invocation. Or our bug.
212 MAYBE_XERBLA(isnan(*res), -__LINE__)
213
214 // Are we experiencing slow convergence after aiter-th step?
215 if (*siter >= aiter && lastres < *res * resdecay) {
216 if (*apprx) { // Blame the approximate factorization
217 *fact = 'N';
218 *siter = smax;
219 *diter = dmax;
220 return FUNCTION(fact, apprx, aiter, trans, n, kl,
221 ku, ab, afrob, afb, ipiv, b, x,
222 siter, diter, tolsc, r, res);
223
224 } else { // Blame working in single precision
225 break;
226 }
227 }
228
229 // Save residual for next iteration
230 // Notice conservative location when failing to double precision
231 lastres = *res;
232
233 }
234
235 } else { // Single not attempted
236 *siter = 0; // Report zero refinements
237 }
238
239double_precision_attempt:
240
241 if (dmax >= 0 && *res > tol) { // Attempt double?
242
243 if (*fact != 'D') { // Ensure double precision factorization available
244 *fact = 'D';
245 *apprx = 0;
246 DLACPY('F', ldab, n, ab, ldab, afb + kl, ldafb);
247 info = DGBTRF(n, n, kl, ku, afb, ldafb, ipiv);
248 MAYBE_XERBLA(info < 0, -__LINE__) else if (UNLIKELY(info > 0)) { return info; }
249 }
250
251 while (*diter < dmax && *res > tol) {
252
253 // Perform one step of double precision iterative refinement
254 // updating norm computations r = b - op(A) x and possibly tol
255 *diter += 1;
256 info = DGBTRS(trans, n, kl, ku, 1, afb, ldafb, ipiv, r, n);
257 MAYBE_XERBLA(info, -__LINE__)
258 DAXPY(n, 1.0, r, inc, x, inc);
259 tol = (use_eps_tol ? eps : DNRM2(n, x, inc)) * tolconst;
260 DCOPY(n, b, inc, r, inc);
261 info = DGBMV(trans, n, n, kl, ku, -1, ab, ldab, x, inc, 1, r, inc);
262 MAYBE_XERBLA(info, -__LINE__)
263 *res = DNRM2(n, r, inc);
264
265 // Getting here required *res > tol so isnan(*res) now is badness
266 // likely stemming from NaNs within matrix ab not caught because the
267 // matrix was not factorized within this invocation. Or our bug.
268 MAYBE_XERBLA(isnan(*res), -__LINE__)
269
270 // Are we experiencing slow convergence after aiter-th step?
271 if (*diter >= aiter && lastres < *res * resdecay) {
272 if (*apprx) { // Blame the approximate factorization
273 *fact = 'N';
274 *siter = smax;
275 *diter = dmax;
276 return FUNCTION(fact, apprx, aiter, trans, n, kl, ku, ab,
277 afrob, afb, ipiv, b, x,
278 siter, diter, tolsc, r, res);
279
280 } else { // Blame the problem itself
281 break;
282 }
283 }
284
285 lastres = *res; // Save residual for next iteration
286
287 }
288
289 } else { // Double not attempted
290 *diter = 0; // Report zero refinements
291 }
292
293 // Make tolsc the fraction of the tolerance represented by the solution
294#pragma warning(push,disable:1572)
295 if (*res == 0) {
296#pragma warning(pop)
297 *tolsc = 0;
298 } else {
299 *tolsc = *res / tol;
300 }
301
302 // As we originally presumed an all-zero solution vector x and performing
303 // any backsubstitions above hinged on (*res > tol), isnan(*res) stemming
304 // from a NaN in input vector b just produced an all-zero output x. That
305 // atrocity needs to be corrected though things like 'fact' should be okay.
306 if (UNLIKELY(isnan(*res))) {
307 for (int i = 0; i < n; ++i) {
308 x[i] = NAN;
309 }
310 }
311
312 return info;
313}
