66std::vector<real_t> operator_nonlinear::apply_operator(
67 const real_t time,
68 contiguous_state<4,complex_t> &swave,
69 const lowstorage::method_interface<complex_t> &method,
70 const std::size_t substep_index) const
71{
72#define ARGUMENTS scenario, *this, common, sg, msoln, time, swave, \
73 method, substep_index
74
75 // Dispatch to an optimized implementation depending on runtime settings.
76 // Done since the compiler can theoretically hammer out much savings when
77 // many small, localized jumps can be hoisted up to this level. In
78 // practice, with ZerothSubstep, explicitly this saved 1-2% of wall time.
79 if (sg.formulation.enabled()) {
80
81 switch (common.linearization) {
82 case linearize::rhome_xyz:
83 return (substep_index == 0)
84 ? apply_navier_stokes_spatial_operator<
85 true, linearize::rhome_xyz, true>(ARGUMENTS)
86 : apply_navier_stokes_spatial_operator<
87 false, linearize::rhome_xyz, true>(ARGUMENTS);
88
89 case linearize::rhome_y:
90 return (substep_index == 0)
91 ? apply_navier_stokes_spatial_operator<
92 true, linearize::rhome_y, true>(ARGUMENTS)
93 : apply_navier_stokes_spatial_operator<
94 false, linearize::rhome_y, true>(ARGUMENTS);
95
96 case linearize::none:
97 return (substep_index == 0)
98 ? apply_navier_stokes_spatial_operator<
99 true, linearize::none, true>(ARGUMENTS)
100 : apply_navier_stokes_spatial_operator<
101 false, linearize::none, true>(ARGUMENTS);
102
103 default:
104 SUZERAIN_ERROR_VAL_UNIMPLEMENTED(std::vector<real_t>());
105 break;
106 }
107
108 } else {
109
110 switch (common.linearization) {
111 case linearize::rhome_xyz:
112 return (substep_index == 0)
113 ? apply_navier_stokes_spatial_operator<
114 true, linearize::rhome_xyz, false>(ARGUMENTS)
115 : apply_navier_stokes_spatial_operator<
116 false, linearize::rhome_xyz, false>(ARGUMENTS);
117
118 case linearize::rhome_y:
119 return (substep_index == 0)
120 ? apply_navier_stokes_spatial_operator<
121 true, linearize::rhome_y, false>(ARGUMENTS)
122 : apply_navier_stokes_spatial_operator<
123 false, linearize::rhome_y, false>(ARGUMENTS);
124
125 case linearize::none:
126 return (substep_index == 0)
127 ? apply_navier_stokes_spatial_operator<
128 true, linearize::none, false>(ARGUMENTS)
129 : apply_navier_stokes_spatial_operator<
130 false, linearize::none, false>(ARGUMENTS);
131
132 default:
133 SUZERAIN_ERROR_VAL_UNIMPLEMENTED(std::vector<real_t>());
134 break;
135 }
136
137 }
138
139#undef ARGUMENTS
140}
