42static void parse_size_t(const std::string& s,
43 std::size_t* value,
44 const char* name)
45{
46 using std::floor;
47#pragma warning(push,disable:2259)
48 const real_t t = floor(exprparse<real_t>(s, name) + real_t(1)/2);
49#pragma warning(pop)
50 validation::ensure_nonnegative(t, name);
51 *value = t;
52}

69static void parse_walltime(const std::string& s,
70 real_t* value,
71 const char* name)
72{
73 // Parses dd:hh:mm:ss.ss, ..., mm:ss.ss, or ss.ss into seconds.
74 // Additionally allows any of dd, hh, mm, or ss.ss to be expressions.
75
76 using boost::algorithm::is_any_of;
77 using boost::algorithm::split;
78 using boost::algorithm::trim;
79 using boost::bind;
80 using std::back_inserter;
81 using std::for_each;
82 using std::invalid_argument;
83 using std::locale;
84 using std::remove;
85 using std::string;
86 using std::transform;
87 using std::vector;
88
89 // Split colon-separated spec into whitespace-trimmed, non-empty tokens
90 vector<string> tokens;
91 split(tokens, s, is_any_of(":"));
92 for_each(tokens.begin(), tokens.end(), bind(trim<string>, _1, locale()));
93 tokens.erase(remove(tokens.begin(), tokens.end(), ""), tokens.end());
94
95 // Check incoming string lengths
96 if (tokens.size() < 1) throw invalid_argument(
97 string(name) + " did not specify a wall time");
98 if (tokens.size() > 4) throw invalid_argument(
99 string(name) + " not of format [dd:[hh:[mm:]]]ss.ss");
100
101 // Parse dd, hh, mm, and ss into components[0], [1], [2], and [3]
102 vector<real_t> components(4 - tokens.size(), 0); // Zero missing values
103 components.reserve(4); // Preallocate storage
104 real_t (*f)(const std::string&, const char*) = &exprparse<real_t>;
105 transform(tokens.begin(), tokens.end(), back_inserter(components),
106 bind(f, _1, name));
107
108 // Convert components to floating seconds
109 real_t t = components[0]; // days
110 t = 24 * t + components[1]; // hours
111 t = 60 * t + components[2]; // minutes
112 t = 60 * t + components[3]; // seconds
113
114 // Validate and store result
115 validation::ensure_nonnegative(t, name);
116 *value = t;
117}

182boost::program_options::options_description
183definition_time::options_description()
184{
185 using boost::bind;
186 using boost::lexical_cast;
187 using boost::program_options::options_description;
188 using boost::program_options::typed_value;
189 using boost::program_options::value;
190 using std::unique_ptr;
191 using std::numeric_limits;
192 using std::string;
193 using validation::ensure_nonnegative;
194 using validation::ensure_positive;
195
196 boost::program_options::options_description retval(
197 "Time advancement parameters");
198
199 // See constructor definition for rational behind conditional
200 if (this->constructor != constructor2) {
201 retval.add_options()
202 ("advance_dt", value<string>(NULL)
203 ->notifier(bind(&parse_option<real_t>, _1, &advance_dt,
204 &ensure_nonnegative<real_t>, "advance_dt"))
205 ->default_value(lexical_cast<string>(advance_dt)),
206 description_advance_dt)
207 ("advance_nt", value<string>(NULL)
208 ->notifier(bind(&parse_size_t, _1, &advance_nt, "advance_nt"))
209 ->default_value(lexical_cast<string>(advance_nt)),
210 description_advance_nt)
211 ("advance_wt", value<string>(NULL)
212 ->notifier(bind(&parse_walltime, _1, &advance_wt, "advance_wt"))
213 ->default_value(lexical_cast<string>(advance_wt)),
214 description_advance_wt)
215 ("status_dt", value<string>(NULL)
216 ->notifier(bind(&parse_option<real_t>, _1, &status_dt,
217 &ensure_nonnegative<real_t>, "status_dt"))
218 ->default_value(lexical_cast<string>(status_dt)),
219 description_status_dt)
220 ("status_nt", value<string>(NULL)
221 ->notifier(bind(&parse_size_t, _1, &status_nt, "status_nt"))
222 ->default_value(lexical_cast<string>(status_nt)),
223 description_status_nt)
224 ("status_final", value<bool>(&status_final)
225 ->default_value(status_final),
226 description_status_final)
227 ("min_dt", value<string>(NULL)
228 ->notifier(bind(&parse_option<real_t>, _1, &min_dt,
229 &ensure_nonnegative<real_t>, "min_dt"))
230 ->default_value(lexical_cast<string>(min_dt)),
231 description_min_dt)
232 ("max_dt", value<string>(NULL)
233 ->notifier(bind(&parse_option<real_t>, _1, &max_dt,
234 &ensure_nonnegative<real_t>, "max_dt"))
235 ->default_value(lexical_cast<string>(max_dt)),
236 description_max_dt)
237 ;
238 }
239
240 // Complicated add_options() calls allow changing the default value of
241 // evmagfactor displayed depending on the constructor. NaN used as a NOP
242 // value by clients. Validation routines used below silently allow NaNs.
243 unique_ptr<typed_value<string> > p(value<string>());
244 p->notifier(bind(&parse_option<real_t>, _1, &evmagfactor,
245 &ensure_positive<real_t>, "evmagfactor"));
246 if (constructor == constructor2)
247 {
248 p->default_value(lexical_cast<string>(evmagfactor));
249 }
250 retval.add_options()
251 ("evmagfactor", p.release(), description_evmagfactor)
252 ;
253
254 return retval;
255}

257void definition_time::save(
258 const esio_handle h) const
259{
260 DEBUG0("Saving some, but not all, definition_time parameters");
261
262 // Only root writes data
263 int procid;
264 esio_handle_comm_rank(h, &procid);
265
266 esio_line_establish(h, 1, 0, (procid == 0 ? 1 : 0));
267
268 esio_line_write(h, "evmagfactor", &evmagfactor, 0, description_evmagfactor);
269}

271void definition_time::load(
272 const esio_handle h,
273 const bool verbose)
274{
275 DEBUG0("Loading some, but not all, definition_time parameters");
276
277 esio_line_establish(h, 1, 0, 1); // All ranks load
278
279 // evmagfactor not present in pre-revision 24043 restart files
280 if (!(boost::math::isnan)(evmagfactor)) {
281 if (verbose)
282 INFO0("Overriding timedef using evmagfactor = " << evmagfactor);
283 } else if (ESIO_NOTFOUND == esio_line_size(h, "evmagfactor", NULL)) {
284 evmagfactor = 0.72;
285 if (verbose)
286 INFO0("Employing default evmagfactor = " << evmagfactor);
287 } else {
288 esio_line_read(h, "evmagfactor", &evmagfactor, 0);
289 }
290}
