104template <typename Real, std::size_t N>
105running_statistics<Real,N>& running_statistics<Real,N>::operator()(
106 const Real x[N])
107{
108 // Algorithm from Knuth TAOCP vol 2, 3rd edition, page 232.
109 // Knuth shows better behavior than Welford 1962 on test data.
110
111 using std::max;
112 using std::min;
113
114 ++n_;
115
116 if (n_ > 1) { // Second and subsequent invocation
117
118 const Real inv_n = Real(1) / n_;
119 for (std::size_t i = 0; i < N; ++i) {
120 const Real d = x[i] - M_[i];
121 M_[i] += d * inv_n;
122 S_[i] += d * (x[i] - M_[i]);
123 min_[i] = min(min_[i], x[i]);
124 max_[i] = max(max_[i], x[i]);
125 }
126
127 } else { // First invocation requires special treatment
128
129 for (std::size_t i = 0; i < N; ++i) {
130 M_[i] = x[i];
131 S_[i] = 0;
132 min_[i] = x[i];
133 max_[i] = x[i];
134 }
135 }
136
137 return *this;
138}

140template <typename Real, std::size_t N>
141running_statistics<Real,N>& running_statistics<Real,N>::operator()(
142 const running_statistics& o)
143{
144 using std::max;
145 using std::min;
146
147 const std::size_t total = n_ + o.n_; // How many samples in combined data?
148 if (o.n_ == 0) return *this; // Other contains no data; run away
149 if (n_ == 0) return *this = o; // We contain no data; simply copy
150 assert(total >= 1); // Voila, degeneracy sidestepped
151
152 // Combine statistics from both instances into this
153 for (std::size_t i = 0; i < N; ++i) {
154 const Real dM = M_[i] - o.M_[i]; // Cancellation issues?
155 M_[i] = (n_ * avg(i) + o.n_ * o.avg(i)) / total;
156 S_[i] = (n_ == 1 ? 0 : S_[i])
157 + (o.n_ == 1 ? 0 : o.S_[i])
158 + ((dM * dM) * (n_ * o.n_)) / total;
159 min_[i] = min(min_[i], o.min_[i]);
160 max_[i] = max(max_[i], o.max_[i]);
161 }
162 n_ = total;
163
164 return *this;
165}
