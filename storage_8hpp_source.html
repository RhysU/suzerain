150template< typename StorageOrderSequence >
151template< typename RandomAccessIterator1,
152 typename RandomAccessIterator2,
153 typename OutputIterator >
154std::size_t general<StorageOrderSequence>::compute_strides(
155 RandomAccessIterator1 sizes,
156 RandomAccessIterator2 minstrides,
157 OutputIterator strides)
158{
159 typedef typename std::iterator_traits<OutputIterator>::value_type
160 output_value_type;
161 array<output_value_type, dimensionality> stride_array;
162
163 // Compute stride details using known storage order
164 const sequence_array_type seq;
165 stride_array[seq[0]] = std::max<output_value_type>(1, minstrides[seq[0]]);
166 for (std::size_t i = 1; i < dimensionality; ++i) {
167 stride_array[seq[i]] = std::max<output_value_type>(
168 sizes[seq[i-1]] * stride_array[seq[i-1]],
169 minstrides[seq[i]]);
170 }
171
172 // Copy stride information to output parameter
173 std::copy(stride_array.begin(), stride_array.end(), strides);
174
175 // Return storage requirements from the slowest stride and size
176 return sizes[seq[dimensionality-1]] * stride_array[seq[dimensionality-1]];
177}

181std::size_t general<StorageOrderSequence>::compute_strides(
182 RandomAccessIterator sizes,
183 OutputIterator strides)
184{
185 array<
186 typename std::iterator_traits<OutputIterator>::value_type,
187 dimensionality
188 > scratch;
189 std::fill(scratch.begin(), scratch.end(), 1);
190 return compute_strides(sizes, scratch.begin(), strides);
191}
