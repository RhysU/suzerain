36template <typename Scalar>
37const Scalar primitive<Scalar>::twopi = 8 * std::atan(Scalar(1));

39template <typename Scalar>
40template <typename T1, typename T2, typename T3, typename T4>
41Scalar primitive<Scalar>::operator()(T1 x, T2 y, T3 z, T4 t) const
42{
43 using ::std::cos;
44 using ::std::sin;
45 const Scalar twopi_invLx = twopi / *Lx;
46 const Scalar twopi_invLy = twopi / *Ly;
47 const Scalar twopi_invLz = twopi / *Lz;
48
49 return a_0*cos(g_0 + f_0*t) + a_x*cos(c_x + b_x*twopi_invLx*x)*cos(g_x + f_x*t) + a_y*cos(g_y + f_y*t)*cos(c_y + b_y*twopi_invLy*y) + a_z*cos(g_z + f_z*t)*cos(c_z + b_z*twopi_invLz*z) + a_xy*cos(g_xy + f_xy*t)*cos(c_xy + b_xy*twopi_invLx*x)*cos(e_xy + d_xy*twopi_invLy*y) + a_xz*cos(g_xz + f_xz*t)*cos(c_xz + b_xz*twopi_invLx*x)*cos(e_xz + d_xz*twopi_invLz*z) + a_yz*cos(c_yz + b_yz*twopi_invLy*y)*cos(e_yz + d_yz*twopi_invLz*z)*cos(g_yz + f_yz*t);
50}

52template <typename Scalar>
53template <typename T1, typename T2, typename T3, typename T4>
54Scalar primitive<Scalar>::_t(T1 x, T2 y, T3 z, T4 t) const
55{
56 using ::std::cos;
57 using ::std::sin;
58 const Scalar twopi_invLx = twopi / *Lx;
59 const Scalar twopi_invLy = twopi / *Ly;
60 const Scalar twopi_invLz = twopi / *Lz;
61
62 return -a_0*f_0*sin(g_0 + f_0*t) - a_x*f_x*cos(c_x + b_x*twopi_invLx*x)*sin(g_x + f_x*t) - a_y*f_y*cos(c_y + b_y*twopi_invLy*y)*sin(g_y + f_y*t) - a_z*f_z*cos(c_z + b_z*twopi_invLz*z)*sin(g_z + f_z*t) - a_xy*f_xy*cos(c_xy + b_xy*twopi_invLx*x)*cos(e_xy + d_xy*twopi_invLy*y)*sin(g_xy + f_xy*t) - a_xz*f_xz*cos(c_xz + b_xz*twopi_invLx*x)*cos(e_xz + d_xz*twopi_invLz*z)*sin(g_xz + f_xz*t) - a_yz*f_yz*cos(c_yz + b_yz*twopi_invLy*y)*cos(e_yz + d_yz*twopi_invLz*z)*sin(g_yz + f_yz*t);
63}

273NSCTPL_MANUFACTURED_SOLUTION_TPDECLARATION
274template <typename T1, typename T2, typename T3, typename T4>
275Scalar manufactured_solution<NSCTPL_MANUFACTURED_SOLUTION_TPNAMES>::e(
276 T1 x, T2 y, T3 z, T4 t) const
277{
278 const Scalar T = this->T(x, y, z, t);
279 const Scalar u = this->u(x, y, z, t);
280 const Scalar v = this->v(x, y, z, t);
281 const Scalar w = this->w(x, y, z, t);
282 const Scalar e = R * T / (gamma - 1) + (u*u + v*v + w*w ) / 2;
283 return e;
284}

339NSCTPL_MANUFACTURED_SOLUTION_TPDECLARATION
340template <typename T1, typename T2, typename T3, typename T4>
341Scalar manufactured_solution<NSCTPL_MANUFACTURED_SOLUTION_TPNAMES>::grad_e(
342 T1 x, T2 y, T3 z, T4 t, int index) const
343{
344 const Scalar u = this->u(x, y, z, t);
345 const Scalar v = this->v(x, y, z, t);
346 const Scalar w = this->w(x, y, z, t);
347 Scalar dT, du, dv, dw;
348 switch (index) {
349 case (IndexBase + 0):
350 dT = this->T._x(x, y, z, t);
351 du = this->u._x(x, y, z, t);
352 dv = this->v._x(x, y, z, t);
353 dw = this->w._x(x, y, z, t);
354 break;
355 case (IndexBase + 1):
356 dT = this->T._y(x, y, z, t);
357 du = this->u._y(x, y, z, t);
358 dv = this->v._y(x, y, z, t);
359 dw = this->w._y(x, y, z, t);
360 break;
361 case (IndexBase + 2):
362 dT = this->T._z(x, y, z, t);
363 du = this->u._z(x, y, z, t);
364 dv = this->v._z(x, y, z, t);
365 dw = this->w._z(x, y, z, t);
366 break;
367 default: return ::std::numeric_limits<Scalar>::signaling_NaN();
368 }
369 const Scalar de = R * dT / (gamma - 1) + (u*du + v*dv + w*dw);
370 return de;
371}

373NSCTPL_MANUFACTURED_SOLUTION_TPDECLARATION
374template <typename T1, typename T2, typename T3, typename T4>
375Scalar manufactured_solution<NSCTPL_MANUFACTURED_SOLUTION_TPNAMES>::grad_p(
376 T1 x, T2 y, T3 z, T4 t, int index) const
377{
378 const Scalar rho = this->rho(x, y, z, t); // shadow
379 const Scalar T = this->T(x, y, z, t);
380 Scalar drho, dT;
381 switch (index) {
382 case (IndexBase + 0):
383 dT = this->T._x(x, y, z, t);
384 drho = this->rho._x(x, y, z, t);
385 break;
386 case (IndexBase + 1):
387 dT = this->T._y(x, y, z, t);
388 drho = this->rho._y(x, y, z, t);
389 break;
390 case (IndexBase + 2):
391 dT = this->T._z(x, y, z, t);
392 drho = this->rho._z(x, y, z, t);
393 break;
394 default: return ::std::numeric_limits<Scalar>::signaling_NaN();
395 }
396 const Scalar dp = drho * R * T + rho * R * dT;
397 return dp;
398}

402Scalar manufactured_solution<NSCTPL_MANUFACTURED_SOLUTION_TPNAMES>::grad_mu(

423NSCTPL_MANUFACTURED_SOLUTION_TPDECLARATION
424template <typename T1, typename T2, typename T3, typename T4>
425Scalar manufactured_solution<NSCTPL_MANUFACTURED_SOLUTION_TPNAMES>::Q_rho(
426 T1 x, T2 y, T3 z, T4 t) const
427{
428 /* Computations retrieving primitive solution details */
429 const Scalar rho = this->rho (x, y, z, t); // shadow
430 const Scalar rho_t = this->rho._t(x, y, z, t);
431 const Scalar rho_x = this->rho._x(x, y, z, t);
432 const Scalar rho_y = this->rho._y(x, y, z, t);
433 const Scalar rho_z = this->rho._z(x, y, z, t);
434
435 const Scalar u = this->u (x, y, z, t); // shadow
436 const Scalar u_x = this->u._x(x, y, z, t);
437

684 const Scalar v_x = this->v._x (x, y, z, t);
685 const Scalar v_xx = this->v._xx(x, y, z, t);

697 const Scalar w_xz = this->w._xz(x, y, z, t);
698 const Scalar w_y = this->w._y (x, y, z, t);
699 const Scalar w_yy = this->w._yy(x, y, z, t);
700 const Scalar w_yz = this->w._yz(x, y, z, t);
701 const Scalar w_z = this->w._z (x, y, z, t);
702 const Scalar w_zz = this->w._zz(x, y, z, t);
703

705 const Scalar T_t = this->T._t (x, y, z, t);
706 const Scalar T_x = this->T._x (x, y, z, t);

721 const Scalar p_y = rho_y * R * T + rho * R * T_y;

731 const Scalar lambda_x = lambda_r / mu_r * mu_x;
732 const Scalar lambda_y = lambda_r / mu_r * mu_y;
733 const Scalar lambda_z = lambda_r / mu_r * mu_z;
734 const Scalar qx_x = - kappa_r / mu_r * (mu_x * T_x + mu * T_xx);
735 const Scalar qy_y = - kappa_r / mu_r * (mu_y * T_y + mu * T_yy);
736 const Scalar qz_z = - kappa_r / mu_r * (mu_z * T_z + mu * T_zz);
737

740

742 const Scalar rhove_y = (rho_y * v * e) + (rho * v_y * e) + (rho * v * e_y);
743 const Scalar rhowe_z = (rho_z * w * e) + (rho * w_z * e) + (rho * w * e_z);
744
745 const Scalar tauxx = mu * (u_x + u_x) + lambda * (u_x + v_y + w_z);
746 const Scalar tauyy = mu * (v_y + v_y) + lambda * (u_x + v_y + w_z);
747 const Scalar tauzz = mu * (w_z + w_z) + lambda * (u_x + v_y + w_z);
748 const Scalar tauxy = mu * (u_y + v_x);
749 const Scalar tauxz = mu * (u_z + w_x);
750 const Scalar tauyz = mu * (v_z + w_y);
751
752 const Scalar tauxx_x = mu_x * (u_x + u_x )
753 + lambda_x * (u_x + v_y + w_z )
754 + mu * (u_xx + u_xx)
755 + lambda * (u_xx + v_xy + w_xz);
756 const Scalar tauyy_y = mu_y * (v_y + v_y )
757 + lambda_y * (u_x + v_y + w_z )
758 + mu * (v_yy + v_yy)
759 + lambda * (u_xy + v_yy + w_yz);
760 const Scalar tauzz_z = mu_z * (w_z + w_z )
761 + lambda_z * (u_x + v_y + w_z )
762 + mu * (w_zz + w_zz)
763 + lambda * (u_xz + v_yz + w_zz);

768 const Scalar tauxz_z = mu_z * (u_z + w_x) + mu * (u_zz + w_xz);
769 const Scalar tauyz_y = mu_y * (v_z + w_y) + mu * (v_yz + w_yy);
770 const Scalar tauyz_z = mu_z * (v_z + w_y) + mu * (v_zz + w_yz);
771
772 const Scalar pu_x = p_x * u + p * u_x;
773 const Scalar pv_y = p_y * v + p * v_y;

776 const Scalar vtauxy_x = v_x * tauxy + v * tauxy_x;

783 const Scalar wtauzz_z = w_z * tauzz + w * tauzz_z;
784
785 const Scalar Q_rhoe = rhoe_t + rhoue_x + rhove_y + rhowe_z
786 + pu_x + pv_y + pw_z
787 + qx_x + qy_y + qz_z
788 - utauxx_x - vtauxy_x - wtauxz_x
789 - utauxy_y - vtauyy_y - wtauyz_y
790 - utauxz_z - vtauyz_z - wtauzz_z;
791
792 return Q_rhoe;

846 t.rho.b_y = scalar(1) / scalar(2);
847 t.rho.f_y = scalar(1);
