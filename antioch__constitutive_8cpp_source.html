99boost::program_options::options_description
100antioch_constitutive::options_description()
101{
102 using boost::bind;
103 using boost::lexical_cast;
104 using boost::program_options::options_description;
105 using boost::program_options::typed_value;
106 using boost::program_options::value;
107 using std::unique_ptr;
108 using std::string;
109 using std::vector;
110
111 options_description retval("libantioch constitutive law parameters");
112
113 // Complicated add_options() calls done to allow changing the default value
114 // displayed when the default is NaN. NaN is used as a NOP value by client
115 // code. Validation routines used below all silently allow NaNs.
116
117 unique_ptr<typed_value<string> > p;
118
119 // species
120 retval.add_options()(name_species_names,
121 value(&species_names),
122 desc_species_names);
123
124 // chemistry input file
125 retval.add_options()(name_chem_input_file,
126 value(&chem_input_file),
127 desc_chem_input_file);
128 // Le
129 p.reset(value<string>());
130 p->notifier(bind(&parse_positive, _1, &Le, name_Le));
131 if (!(boost::math::isnan)(Le)) {
132 p->default_value(lexical_cast<string>(Le));
133 }
134 retval.add_options()(name_Le, p.release(), desc_Le);
135
136 // alpha
137 p.reset(value<string>());
138 p->notifier(bind(&parse_nonnegative, _1, &alpha, name_alpha));
139 if (!(boost::math::isnan)(alpha)) {
140 p->default_value(lexical_cast<string>(alpha));
141 }
142 retval.add_options()(name_alpha, p.release(), desc_alpha);
143
144
145 return retval;
146}

174void
175antioch_constitutive::save(
176 const esio_handle h) const
177{
178
179 using boost::lexical_cast;
180
181 DEBUG0("Storing antioch_constitutive parameters");
182
183 // Only root writes data
184 int procid;
185 esio_handle_comm_rank(h, &procid);
186 esio_line_establish(h, 1, 0, (procid == 0 ? 1 : 0));
187
188 // NOTE: approach similar to that taken for "metadata_generated"
189 // in driver_base::save_metadata
190 static const char acd[] = "antioch_constitutive_data";
191
192 esio_line_write(h, acd, &antioch_ver, 0,
193 "Antioch version number and antioch_contitutive data.");
194
195 // number of species
196 int Ns = this->species_names.size();
197 esio_attribute_write(h, acd, "Ns", Ns);
198
199 // species names
200 for (size_t i=0; i<species_names.size(); ++i)
201 {
202 std::string speci("Species_" + lexical_cast<std::string>(i));
203 esio_string_set(h, acd, speci.c_str(), this->species_names[i].c_str());
204 }
205
206 // chemistry input file
207 esio_string_set(h, acd, "chem_input_file", this->chem_input_file.c_str());
208
209 // Lewis number
210 esio_attribute_write(h, acd, name_Le , &this->Le);
211
212 // alpha
213 esio_attribute_write(h, acd, name_alpha, &this->alpha);
214}

216void
217antioch_constitutive::load(
218 const esio_handle h,
219 const bool verbose)
220{
221 using boost::lexical_cast;
222
223 DEBUG0("Loading antioch_constitutive parameters");
224
225 // All ranks load
226 esio_line_establish(h, 1, 0, 1);
227
228 antioch_constitutive t;
229
230 static const char acd[] = "antioch_constitutive_data";
231
232 // antioch version
233 esio_line_read(h, acd, &t.antioch_ver, 0);
234
235 // Will never overwrite antioch version info
236 if (t.antioch_ver != this->antioch_ver) {
237 //... but warn if it doesn't match
238 WARN0("Antioch version has changed. Was " << t.antioch_ver
239 << " when written. Is now " << this->antioch_ver);
240 }
241
242 // number of species
243 int Ns;
244 esio_attribute_read(h, acd, "Ns", &Ns);
245
246 t.species_names.reserve(Ns);
247
248 // species names
249 for (int i=0; i<Ns; ++i)
250 {
251 std::string speci("Species_" + lexical_cast<std::string>(i));
252 char* tmp = esio_string_get(h, acd, speci.c_str());
253 t.species_names.push_back(tmp);
254 free(tmp);
255 }
256
257 // chemistry input file
258 char* tmp = esio_string_get(h, acd, "chem_input_file");
259 t.chem_input_file = tmp;
260 free(tmp);
261
262 // Lewis number
263 esio_attribute_read(h, acd, name_Le , &t.Le);
264
265 // alpha
266 esio_attribute_read(h, acd, name_alpha, &t.alpha);
267
268 // Prefer this to incoming (handle explictly for string data)
269 if (this->species_names.size() > 0)
270 INFO0("Clutching onto " << name_species_names << " over incoming");
271 else
272 this->species_names = t.species_names;
273
274 if (this->chem_input_file.size() > 0)
275 INFO0("Clutching onto " << name_chem_input_file << " over incoming");
276 else
277 this->chem_input_file = t.chem_input_file;
278
279
280 // Prefer this to incoming
281 this->populate(t, verbose);
282}

286{
287 WARN0("antioch_constitutive is not fully functional yet!");

289 mixture = std::make_shared<Antioch::ChemicalMixture<real_t> >(species_names);
290 reactions = std::make_shared<Antioch::ReactionSet<real_t> >(*mixture);
291 cea_thermo = std::make_shared<Antioch::CEAThermodynamics<real_t> >(*mixture);
292 sm_thermo = std::make_shared<Antioch::StatMechThermodynamics<real_t> >(*mixture);
293
294
295 if (this->Ns() > 1) {
296 // FIXME: This will do for now, but need to think about what
297 // happens in parallel to avoid all ranks trying to read this
298 // file.
299 Antioch::read_reaction_set_data_xml<real_t>(chem_input_file,
300 false /* verbose */,
301 *reactions);
302 kinetics = std::make_shared<Antioch::KineticsEvaluator<real_t> >(*reactions,0);
303 }
304
305
306 mixture_mu = std::make_shared<Antioch::MixtureViscosity<
307 Antioch::BlottnerViscosity<real_t>, real_t> >
308 (*mixture);
309
310 Antioch::read_blottner_data_ascii_default(*mixture_mu);
311
312 mixture_kappa = std::make_shared<Antioch::EuckenThermalConductivity<
313 Antioch::StatMechThermodynamics<real_t> > >
314 (*sm_thermo);
315
316 wilke_mixture = std::make_shared<Antioch::WilkeMixture<real_t> >(*mixture);
317
318 wilke_evaluator = std::make_shared<
319 Antioch::WilkeEvaluator<Antioch::MixtureViscosity<Antioch::BlottnerViscosity<real_t>, real_t>,
320 Antioch::EuckenThermalConductivity<Antioch::StatMechThermodynamics<real_t> >, real_t> >
321 (*wilke_mixture, *mixture_mu, *mixture_kappa );
322
323 // TODO: Add consistency asserts... everybody has same number of
324 // species, number of reactions is sane, valid curve fits,

328// Evaluate everything
329void
330antioch_constitutive::evaluate (const real_t e,
331 const real_t* m,
332 const real_t rho,
333 const real_t* species,
334 const real_t* cs,
335 const real_t Tinit,
336 real_t& T,
337 real_t& p,
338 real_t* Ds,
339 real_t& mu,
340 real_t& kap,
341 real_t* hs,
342 real_t* om,
343 real_t& a,
344 real_t& Cv,
345 real_t& Cp) const
346{
347 //WARN0("antioch_constitutive::evaluate is not fully functional yet!");
348 SUZERAIN_UNUSED(species); // TODO?
349 SUZERAIN_TIMER_SCOPED("antioch evaluate");
350
351 const real_t irho = 1.0/rho;
352
353 // FIXME: Incoming vectors are currently raw pointers (result of
354 // .data calls on Eigen objects), but Antioch wants std::vector.
355 // I'll kludge it here to get the thing compiling. Soon, roystgnr
356 // should have Antioch taking Eigen::Array objects (see issue
357 // #2799), which we can pass directly from
358 // apply_navier_stokes_spatial_operator. If not, I will refactor
359 // apply_navier_stokes_spatial_operator to use Eigen::Map to wrap
360 // std::vector, allowing easy passing to Antioch.
361 //
362 // Either way, this whole function will be refactored, but the
363 // basic steps are as follows:
364
365 const size_t Ns = this->Ns();
366
367 // Mass fractions
368 std::vector<real_t> Y(cs, cs+Ns);
369
370 // Mixture gas constant
371 const real_t R_mix = this->mixture->R(Y);
372
373 std::vector<real_t> molar_densities(Ns,0.0);
374 this->mixture->molar_densities(rho,Y,molar_densities);
375
376 // Compute temperature from internal energy (assuming thermal equilibrium)
377 const real_t re_internal = e - 0.5*irho*(m[0]*m[0] + m[1]*m[1] + m[2]*m[2]);
378 T = this->sm_thermo->T_from_e_tot(irho*re_internal, Y, Tinit);
379
380 // Compute pressure: ideal gas law with mixture gas constant
381 p = rho*R_mix*T;
382
383 // Use CEA thermo to compute h_RT_minus_s_R for reaction
384 // calculations
385 //
386 // NOTE: This is how FIN-S does it, so we follow for
387 // complete consistency. But, it might make more sense to use
388 // stat mech based thermo to get this info... I'm not sure.
389 std::vector<real_t> h_RT_minus_s_R(Ns);
390 typedef typename Antioch::CEAThermodynamics<real_t>::Cache<real_t> Cache;
391 Cache cea_cache(T);
392 this->cea_thermo->h_RT_minus_s_R(cea_cache,h_RT_minus_s_R);
393
394
395 // protect against calling kinetics when there are no kinetics
396 // TODO: Set up antioch to avoid this if (i.e., make call to kinetics ok)
397 if (Ns>1) {
398 // Species eqn source terms
399 std::vector<real_t> omega_dot(Ns);
400 std::vector<real_t> molar_densities_nonnegative = molar_densities;
401 for (size_t i=0; i<Ns; ++i) {
402 if (molar_densities_nonnegative[i] < 0.) {
403 molar_densities_nonnegative[i] = 0.;
404 }
405 }
406#if ANTIOCH_VERSION_AT_LEAST(0,0,4)
407 this->kinetics->compute_mass_sources(T, molar_densities_nonnegative,
408 h_RT_minus_s_R, omega_dot);
409#else
410 this->kinetics->compute_mass_sources(T, rho, R_mix, Y,
411 molar_densities_nonnegative,
412 h_RT_minus_s_R, omega_dot);
413#endif
414 for (size_t i=0; i<Ns; ++i) om[i] = omega_dot[i];
415 } else {
416 om[0] = 0.0;
417 }
418
419 // Species enthalpies (assuming thermal equilibrium)
420 for (unsigned int i=0; i<Ns; ++i)
421 hs[i] = this->sm_thermo->h_tot(i, T);
422
423 // Transport
424 mu = this->wilke_evaluator->mu(T, Y);
425 kap = this->wilke_evaluator->k (T, Y);
426
427 // Used by transport calcs and output
428 Cp = this->sm_thermo->cp(T, T, Y);
429
430 // Is this right? Copied from FIN-S (and antioch has same) but
431 // looks like inverse of Le to me.
432 real_t D0 = this->Le*kap*irho/Cp;
433
434 for (unsigned int i=0; i<Ns; ++i)
435 Ds[i] = D0;
436
437
438 // Speed of sound (frozen)
439 Cv = this->sm_thermo->cv(T, T, Y);
440
441 real_t af2 = (1.0 + R_mix/Cv)*R_mix*T;
442
443 a = std::sqrt(af2);
444
445 // TODO: assert that om sums to zero
446 // TODO: assert that T and p are positive
447 // TODO: assert transport props are positive
448}

583void
584antioch_constitutive::evaluate_pressure_derivs_and_trans (const real_t e,
585 const Vector3r& m,
586 const real_t rho,
587 const VectorXr& species,
588 const VectorXr& cs,
589 const real_t Tinit,
590 real_t& T,
591 real_t& p,
592 real_t& p_rho,
593 real_t& p_rsum,
594 Vector3r& p_m,
595 real_t& p_e,
596 real_t& mu,
597 real_t& kaporCv,
598 VectorXr& Ds,
599 real_t& gamma,
600 real_t& a) const
601{
602 SUZERAIN_UNUSED(species); // TODO?
603 SUZERAIN_TIMER_SCOPED("antioch ref quantities");
604
605 const real_t irho = 1.0/rho;
606
607 const size_t Ns = this->Ns();
608
609 // Compute temperature from internal energy (assuming thermal equilibrium)
610 const real_t re_kinetic = 0.5*irho*(m[0]*m[0] + m[1]*m[1] + m[2]*m[2]);
611 const real_t re_internal = e - re_kinetic;
612 T = this->sm_thermo->T_from_e_tot(irho*re_internal, cs, Tinit);
613
614 // Pressure
615 const real_t R_mix = this->mixture->R(cs);
616 p = rho*R_mix*T;
617
618 // Ratio of mixture specific heats
619 const real_t Cp = this->sm_thermo->cp(T, T, cs);
620 const real_t Cv = this->sm_thermo->cv(T, T, cs);
621
622 // gamm and gamma-1 for convenience
623 gamma = Cp/Cv;
624 const real_t gmi = gamma-1.0;
625
626 // Gas constant and internal energy of the diluter
627 const real_t R0 = this->mixture->R(0);
628 const real_t e0int = this->sm_thermo->e_tot(0, T, T);
629
630 // dp/drho
631 p_rho = R0*T + gmi*(-e0int + irho*re_kinetic);
632
633 // \sum_{s=2}^{Ns} c_s dp/drho_s
634 p_rsum = 0.0;
635 for (unsigned int i=1; i<Ns; ++i){
636 const real_t dR = this->mixture->R(i) - R0;
637 const real_t deint = e0int - this->sm_thermo->e_tot(i, T, T);
638
639 p_rsum += cs[i]*(dR*T + gmi*deint);
640 }
641
642 // dp/dru
643 p_m = -gmi*irho*m;
644
645 // dp/drE
646 p_e = gmi;
647
648 // Transport
649 mu = this->wilke_evaluator->mu(T, cs);
650 const real_t kap = this->wilke_evaluator->k (T, cs);
651
652 kaporCv = kap*irho/Cv;
653
654 // Constant Lewis number diffusivity
655 real_t D0 = this->Le*kap*irho/Cp;
656
657 for (unsigned int i=0; i<Ns; ++i)
658 Ds[i] = D0;
659
660 // Speed of sound(frozen)
661 // TODO: Unify computation of speed of sound with that from
662 // the evaluate method
