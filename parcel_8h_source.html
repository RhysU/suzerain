100 BOOST_PP_SEQ_FOR_EACH(SUZERAIN_PARCEL_DATASIZE,,decl_seq) \

122#define SUZERAIN_PARCEL_ALIGNMENT(r,data,elem) \
123 SUZERAIN_PARCEL_APPLY(SUZERAIN_PARCEL_ALIGNMENT_,elem)
124#define SUZERAIN_PARCEL_ALIGNMENT_(varname, datatype, varcount, varqual, minalign) \

148#define SUZERAIN_PARCEL_OP(s, state, elem) ( \
149 BOOST_PP_TUPLE_ELEM(5, 0, elem) BOOST_PP_COMMA() \
150 BOOST_PP_TUPLE_ELEM(5, 2, elem) BOOST_PP_COMMA() \
151 BOOST_PP_TUPLE_ELEM(3, 2, state) SUZERAIN_PARCEL_OP_( \
152 BOOST_PP_TUPLE_ELEM(5, 0, elem), \
153 BOOST_PP_TUPLE_ELEM(5, 1, elem), \
154 BOOST_PP_TUPLE_ELEM(5, 2, elem), \
155 BOOST_PP_TUPLE_ELEM(5, 3, elem), \
156 BOOST_PP_TUPLE_ELEM(3, 0, state), \
157 BOOST_PP_TUPLE_ELEM(3, 1, state) \
158 ) \
159 )
160#define SUZERAIN_PARCEL_OP_(varname, datatype, varcount, varqual, prev, prevcount) \
161 ; datatype * varqual varname = (void *) \
162 ((((uintptr_t)(prev + prevcount)+BOOST_PP_CAT(_PARCEL_pad_,varname)) \
163 & BOOST_PP_CAT(_PARCEL_mask_,varname)))
