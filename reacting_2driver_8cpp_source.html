102void
103driver::log_manufactured_solution_absolute_error(
104 const std::string& timeprefix,
105 const real_t t,
106 const std::size_t nt)
107{
108 SUZERAIN_UNUSED(nt);
109
110 // NOP whenever a manufactured_solution is not in use
111 if (!msoln) return;
112
113 // Avoid computational cost when logging is disabled
114 support::logging::logger_type mms_abserr
115 = support::logging::get_logger("mms.abserr");
116 if (!INFO0_ENABLED(mms_abserr)) return;
117
118 // Compute $L^2_{xyz}$ of error of state against manufactured solution
119 state_nonlinear->assign_from(*state_linear);
120 accumulate_manufactured_solution(
121 1, *msoln, -1, *state_nonlinear,
122 *grid, *dgrid, *cop, *b, t);
123 const std::vector<field_L2xyz> L2
124 = compute_field_L2xyz(*state_nonlinear, *grid, *dgrid, *gop);
125
126 // Output absolute global errors for each field
127 std::ostringstream msg;
128 msg << timeprefix;
129 for (size_t k = 0; k < L2.size(); ++k) {
130 msg << ' ' << fullprec<>(L2[k].total());
131 }
132 INFO0(mms_abserr, msg.str());
133}

135void
136driver::compute_statistics(
137 driver::time_type t)
138{
139 SUZERAIN_TIMER_SCOPED("driver::compute_statistics");
140
141 // Obtain mean samples from instantaneous fields stored in state_linear
142 state_nonlinear->assign_from(*state_linear);
143 mean = reacting::sample_quantities(*cmods, *grid, *dgrid,
144 *cop, *state_nonlinear, t);
145
146 // Obtain mean quantities computed via implicit forcing (when possible)
147 if (common_block.means.rows() == mean.storage.rows()) {
148 mean.f().col(0) = common_block.fx();
149 mean.f().col(1) = common_block.fy();
150 mean.f().col(2) = common_block.fz();
151 mean.f_dot_u() = common_block.f_dot_u();
152 mean.qb() = common_block.qb();
153 mean.CrhoE() = common_block.CrhoE();
154 mean.C2rhoE() = common_block.C2rhoE();
155 mean.Crhou() .col(0) = common_block.Crhou();
156 mean.C2rhou().col(0) = common_block.C2rhou();
157 mean.Crhou() .col(1) = common_block.Crhov();
158 mean.C2rhou().col(1) = common_block.C2rhov();
159 mean.Crhou() .col(2) = common_block.Crhow();
160 mean.C2rhou().col(2) = common_block.C2rhow();
161 mean.Crho() = common_block.Crho();
162 mean.C2rho() = common_block.C2rho();
163 mean.Crhou_dot_u() = common_block.Crhou_dot_u();
164 mean.C2rhou_dot_u() = common_block.C2rhou_dot_u();
165
166 // Convert values at collocation points to expansion coefficients
167 bsplineop_lu mass(*cop);
168 mass.opform_mass(*cop);
169 mass.factor();
170 mass.solve(quantities::nscalars::implicit,
171 mean.storage.middleCols<quantities::nscalars::implicit>(
172 quantities::start::implicit).data(),
173 mean.storage.innerStride(), mean.storage.outerStride());
174 } else {
175 WARN0(who, "Could not obtain mean quantities set by implicit forcing");
176 }
177}

179void
180driver::save_spectra_primitive(
181 const esio_handle esioh)
182{
183 SUZERAIN_TIMER_SCOPED("driver::save_spectra_primitive");
184
185 const size_t Ns = cmods->Ns();
186 const size_t state_count = Ns+4;
187
188 // Indices for auxiliary storage to allow the diluter to
189 // be included in the two-point corralation computation
190 struct aux { enum {
191 T = 0,
192 u = 1,
193 v = 2,
194 w = 3,
195 rho = 4,
196 species = 5
197 }; };
198
199 // Get total number of fields in aux storage
200 const size_t aux_count = (aux::species + Ns);
201
202 // Obtain the auxiliary storage (likely from a pool to avoid fragmenting).
203 // We assume no garbage values in the memory will impact us (for speed).
204 // Maybe FIXME: In this way, instead of reusing the memory of state_nonlinear,
205 // we are allocating new memory for all the variables to include the diluter.
206 // There might be a more memory-efficient way to do this...
207 typedef contiguous_state<4,complex_t> state_type;
208 std::unique_ptr<state_type> _auxw_ptr(
209 support::allocate_padded_state<state_type>(
210 aux_count, *dgrid)); // RAII
211 state_type &auxw = *_auxw_ptr; // Shorthand
212
213 // Ensure state and auxw storage meets this routine's assumptions
214 const unsigned int swave_count = state_count;
215 assert(static_cast<int>(ndx::e ) < swave_count);
216 assert(static_cast<int>(ndx::mx ) < swave_count);
217 assert(static_cast<int>(ndx::my ) < swave_count);
218 assert(static_cast<int>(ndx::mz ) < swave_count);
219 assert(static_cast<int>(ndx::rho ) < swave_count);
220 if (Ns > 1) {
221 assert(static_cast<int>(ndx::species) < swave_count);
222 }
223 SUZERAIN_ENSURE(state_nonlinear->shape()[0] == swave_count);
224 SUZERAIN_ENSURE(std::equal(state_nonlinear->shape() + 1,
225 state_nonlinear->shape() + 4,
226 auxw.shape() + 1));
227 SUZERAIN_ENSURE(std::equal(state_nonlinear->strides() + 1,
228 state_nonlinear->strides() + 4,
229 auxw.strides() + 1));
230
231 // Obtain information from instantaneous fields in *state_linear
232 state_nonlinear->assign_from(*state_linear);
233
234 // Convert to pointwise conserved state in physical space
235 physical_view<> sphys(*dgrid, *state_nonlinear);
236 std::shared_ptr<operator_tools> otool = obtain_operator_tools();
237 for (size_t f = 0; f < swave_count; ++f) {
238 otool->zero_dealiasing_modes( *state_nonlinear, f);
239 otool->bop_apply (0, 1, *state_nonlinear, f);
240 dgrid->transform_wave_to_physical(&sphys.coeffRef(f,0));
241 }
242
243 // Pointwise conversion from e, mx, my, mz, rho, rho_s
244 // to T, u, v, w, rho, c_s
245 physical_view<> auxp(*dgrid, auxw);
246 real_t Tguess = -1;
247 VectorXr species(Ns); // species densities
248 VectorXr cs (Ns); // species mass fractions
249 for (int offset = 0; offset < sphys.cols(); ++offset) {
250
251 // Unpack conserved state
252 const real_t e (sphys(ndx::e, offset));
253 const Vector3r m (sphys(ndx::mx, offset),
254 sphys(ndx::my, offset),
255 sphys(ndx::mz, offset));
256 const real_t rho(sphys(ndx::rho, offset));
257
258 // Compute desired primitive state
259 const Vector3r u = m / rho;
260 real_t T = 0;
261
262 // Unpack species variables
263 // NOTE: In species vector, idx 0 is the dilluter (the species
264 // that is not explicitly part of the state vector)
265 // Compute mass fractions
266 const real_t irho = 1.0/rho;
267 species(0) = rho;
268 for (unsigned int s=1; s<Ns; ++s) {
269 species(s) = sphys(ndx::species + s - 1, offset);
270 cs(s) = irho * species(s);
271
272 // dilluter density = rho_0 = rho - sum_{s=1}^{Ns-1} rho_s
273 species(0) -= species(s);
274 }
275 cs(0) = irho * species(0);
276
277 // Compute temperature from internal energy
278 // (assuming thermal equilibrium)
279 const real_t re_internal = e -
280 0.5*irho*(m[0]*m[0] + m[1]*m[1] + m[2]*m[2]);
281 T = cmods->sm_thermo->T_from_e_tot(irho*re_internal, cs, Tguess);
282 Tguess = T;
283
284 // Pack desired primitive state
285 auxp(aux::T , offset) = T;
286 auxp(aux::u , offset) = u.x();
287 auxp(aux::v , offset) = u.y();
288 auxp(aux::w , offset) = u.z();
289 auxp(aux::rho, offset) = rho;
290 for (unsigned int s=0; s<Ns; ++s) {
291 auxp(aux::species + s, offset) = cs(s);
292 }
293 }
294
295 // TODO Zero dealiasing modes? If so, can't recover m = rho * u.
296 // Convert to normalized Fourier coefficients in XZ but points in Y
297 // Keep means, compute raw statistics
298 auxp *= dgrid->chi();
299 for (size_t f = 0; f < aux_count; ++f) {
300 dgrid->transform_physical_to_wave(&auxp.coeffRef(f,0));
301 }
302
303 // Only rank zero will write the results though all ranks must participate
304 const int npairs = (aux_count*(aux_count+1))/2;
305 int procid;
306 esio_handle_comm_rank(esioh, &procid);
307
308 // Compute and save the two-point correlation as (y_j, k_x, ndxpair)
309 // Ordering arises from packing of primitive state in physical space
310 {
311 std::shared_ptr<complex_t[]> twopoint_x = compute_twopoint_x(
312 auxw, aux_count, *grid, *dgrid);
313 esio_field_establish(esioh,
314 npairs, 0, procid == 0 ? npairs : 0,
315 grid->N.x()/2+1, 0, procid == 0 ? grid->N.x()/2+1 : 0,
316 grid->N.y(), 0, procid == 0 ? grid->N.y() : 0);
317 support::complex_field_write(esioh,
318 "twopoint_kx", twopoint_x.get(), 0, 0, 0,
319 "Streamwise two-point correlations stored row-major"
320 " (/collocation_points_y, /kx, scalarpair) for scalarpair"
321 " in { T*T, T*u, T*v, T*w, T*rho, T*{c_j}, u*u, u*v, u*w,"
322 " u*rho, u*{c_j}, v*v, v*w, v*rho, v*{c_j}, w*w, w*rho,"
323 " w*{c_j}, rho*rho, rho*{c_j}, {c_i}*{c_j}}");
324 }
325
326 // Compute and save the two-point correlation as (y_j, k_z, ndxpair)
327 // Ordering arises from packing of primitive state in physical space
328 {
329 std::shared_ptr<complex_t[]> twopoint_z = compute_twopoint_z(
330 auxw, aux_count, *grid, *dgrid);
331 esio_field_establish(esioh,
332 npairs, 0, procid == 0 ? npairs : 0,
333 grid->N.z(), 0, procid == 0 ? grid->N.z() : 0,
334 grid->N.y(), 0, procid == 0 ? grid->N.y() : 0);
335 support::complex_field_write(esioh,
336 "twopoint_kz", twopoint_z.get(), 0, 0, 0,
337 "Spanwise two-point correlations stored row-major"
338 " (/collocation_points_y, /kz, scalarpair) for scalarpair"
339 " in { T*T, T*u, T*v, T*w, T*rho, T*{c_j}, u*u, u*v, u*w,"
340 " u*rho, u*{c_j}, v*v, v*w, v*rho, v*{c_j}, w*w, w*rho,"
341 " w*{c_j}, rho*rho, rho*{c_j}, {c_i}*{c_j}}");
342 }

405bool
406driver::save_statistics_hook(
407 const esio_handle esioh,
408 const driver::time_type t)
409{
410 DEBUG0(who, "In driver::save_statistics_hook");
411
412 // Should we compute fresh statistics or re-use cached values?
413#pragma warning(push,disable:1572)
414 const bool use_cached = controller && (t == mean.t);
415#pragma warning(pop)
416
417 // Compute a prefix for logging purposes, including a trailing space
418 std::string prefix;
419 if (controller) {
420 prefix = build_timeprefix(t, controller->current_nt());
421 prefix.append(1, ' ');
422 }
423
424 // Compute statistics whenever necessary
425 if (use_cached) {
426 DEBUG0(who, prefix << "Cowardly refusing to re-compute statistics");
427 } else {
428 const double starttime = MPI_Wtime();
429 compute_statistics(t);
430 const double elapsed = MPI_Wtime() - starttime;
431 INFO0(who, prefix << "Computed statistics in "
432 << elapsed << " seconds");
433 }
434
435 // Save statistics and invoke superclass hook
436 if (!mean.save(esioh)) {
437 WARN0(who, "Incomplete statistics saved at time " << t);
438 }
439 return super::save_statistics_hook(esioh, t);
440}

443driver::load_statistics_hook(

445{
446 support::load_time(esioh, mean.t);
447
448 if (statsdef && statsdef->stale) {
449 // Notice this forces use_cached == false in save_statistics_hook
450 WARN0(who, "Ignoring incoming statistics explicitly marked as stale");
451 mean.t = std::numeric_limits<real_t>::quiet_NaN();
452 } else if (!mean.load(esioh)) {

454 WARN0(who, "Incomplete statistics loaded from time " << mean.t);
455 mean.t = std::numeric_limits<real_t>::quiet_NaN();
456 }
457 return super::load_statistics_hook(esioh);
458}

460// Heavy-handed, but I seemingly cannot get Intel to cooperator otherwise.
461// I want non-NaNs to be preferred to NaNs within this max operation.
462static
463real_t maxmf(const real_t a, const real_t b)
464{
465 return (boost::math::isnan)(a) ? b
466 : (boost::math::isnan)(b) ? a
467 : std::max(a, b);
468}

470// Please review and possibly synchronize apps/perfect/driver.cpp when changing
471void
472driver::default_restart_interval(
473 time_type& t,
474 step_type&)
475{
476 using std::abs;
477
478 // Look for the largest magnitude, problem-dependent, macro velocity scale
479 real_t velocity = std::numeric_limits<real_t>::quiet_NaN();
480
481 // In a channel...
482 if (grid && grid->two_sided()) {
483
484 // ...any approximate bulk velocity from a driving force...
485 if (chdef) {
486
487 if ((boost::math::isnan)(chdef->bulk_rho)) {
488 WARN0(who, "No bulk density scale available so assuming 1");
489 velocity = maxmf(abs(chdef->bulk_rho_u) / /*rho*/ 1,
490 velocity);
491 } else {
492 velocity = maxmf(abs(chdef->bulk_rho_u) / chdef->bulk_rho,
493 velocity);
494 }
495
496 }
497
498 // ...may be trumped by driving the upper and lower walls.
499 if (isothermal) {
500 velocity = maxmf(abs(isothermal->upper_u), velocity);
501 velocity = maxmf(abs(isothermal->lower_u), velocity);
502 }
503
504 }
505
506 // On a plate...
507 if (grid && grid->one_sided()) {
508
509 if (sgdef->formulation.enabled() && sgdef->baseflow && fields.size()) {
510
511 // ...prefer wall velocity from inviscid baseflow as freestream
512 // surrogate as it is independent of the wall-normal domain size...
513 const size_t nvalues = fields.size() + /*pressure*/1;
514 std::vector<real_t> buf(3 * nvalues);
515 sgdef->baseflow->conserved(0.0, // wall
516 &buf[0*nvalues], // base
517 &buf[1*nvalues], // dybase
518 &buf[2*nvalues]); // dxbase
519 const real_t base_rho = buf[0];
520 const real_t base_rho_u = buf[1];
521 if (base_rho > 0) {
522 velocity = maxmf(abs(base_rho_u) / base_rho, velocity);
523 } else {
524 TRACE0(who, "Trivial baseflow detected; using upper velocity");
525 velocity = maxmf(abs(isothermal->upper_u), velocity);
526 }
527
528 } else if (isothermal) {
529 // ...taking freestream reference if-and-only-if no baseflow...
530 velocity = maxmf(abs(isothermal->upper_u), velocity);
531 }
532
533 // ...and permit a driven lower wall velocity to trump.
534 if (isothermal) {
535 velocity = maxmf(abs(isothermal->lower_u), velocity);
536 }
537
538 }
539
540 // Have a domain size and a velocity?
541 if (boost::math::isnormal(velocity)) {
542 if (grid) {
543
544 // Let's start from the associated time scale.
545 t = grid->L.x() / velocity;
546
547 // Eight restarts per flow through in a channel giving 80 restarts
548 // in 10 flow throughs, a reasonable channel simulation duration.
549 // Across O(25) flow throughs in a boundary layer campaign it
550 // produces O(200) restart files which is slighly excessive but
551 // increases data redundancy for these more expensive simulations.
552 t /= 8;
553
554 } else {
555 DEBUG0(who, "No grid details for default_restart_interval");
556 }
557 } else {
558 DEBUG0(who, "No macro scale velocity for default_restart_interval");
559 }
560}
